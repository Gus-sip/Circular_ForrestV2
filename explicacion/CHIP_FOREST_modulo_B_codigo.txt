/*
 * =============================================================================
 * CHIP FOREST - MODULO B (PASARELA) - FIRMWARE DE PRODUCCION
 * =============================================================================
 *
 * Que hace esta placa:
 *   Recibe por radio LoRa las tramas de los nodos sensores (Modulo C), las
 *   interpreta, les pone hora real y las sube a ThingsBoard por la red movil NB-IoT
 *   usando MQTT. En sentido contrario, recibe ordenes de ThingsBoard (Modulo A) y se
 *   las entrega a cada nodo en su ventana de escucha, ademas de darles la hora de red.
 *   No tiene sensores propios.
 *
 *   Modulo C  --LoRa 868 MHz-->  Modulo B  --NB-IoT / MQTT-->  ThingsBoard
 *             <--CFG,...-------            <--RPC-------------
 *
 * Ademas ofrece, para el banco, una pantalla OLED con el estado y un pequeno panel
 * web por su propio punto de acceso WiFi (con descarga del historico en CSV).
 *
 * Cableado de la radio (ver Config.h):
 *   VDD del modulo -> 3V3          GND del modulo  -> GND
 *   TXD del modulo -> GPIO4 (RX)   RXD del modulo  -> GPIO5 (TX)
 *   NRST del modulo -> al aire
 *
 * Arquitectura: radio/RYLR998 (transporte LoRa), telemetry/TelemetryParser (texto
 * -> SensorSnapshot) y modem_nbiot_mqtt (subida a ThingsBoard) no saben nada unos de
 * otros ni del servidor web; este archivo es el unico sitio donde se juntan. loop()
 * va alternando el servidor web, la radio y el modem, y ninguno bloquea a los demas:
 * servir una pagina nunca hace perder un paquete, y esperar a la radio (que poll()
 * nunca hace) nunca para el servidor.
 */

// -1 = sin pin de reset propio: el RESET de este modulo va unido al del propio
// ESP32, como en todas las placas SSD1306 de este proyecto.
Adafruit_SSD1306 oled(OLED_WIDTH, OLED_HEIGHT, &Wire, -1);

RYLR998 radio(Serial1, LORA_RX_PIN, LORA_TX_PIN, LORA_BAUD);
// UART2 - a proposito no Serial1, que ya es de la radio LoRa de arriba. Ver el
// cableado del NB-IoT en Config.h: los pines son GP44/GP43 (UART0 nativo),
// confirmados en el hardware, no los GP8/GP7 que se supusieron al principio.
ModemNBIoTMqtt modem(Serial2, NBIOT_RX_PIN, NBIOT_TX_PIN, NBIOT_EN_PIN, NBIOT_CHANNEL_PIN, NBIOT_PWRKEY_PIN);
WebServer server(80);

// La ultima lectura recibida (la que muestra el panel web y la OLED).
SensorSnapshot g_latest;

// Anillo con cada lectura LoRa interpretada con exito, solo en RAM (se pierde al
// reiniciar o al quedarse sin corriente) - ver LORA_HISTORY_CAPACITY en Config.h.
// Independiente del anillo propio del modem: aquel existe para agrupar las subidas
// por NB-IoT y se vacia al publicar; este es solo un registro local para /history,
// nunca se vacia y, cuando se llena, sobrescribe empezando por la mas antigua.
SensorSnapshot g_history[LORA_HISTORY_CAPACITY];
uint16_t g_historyHead = 0;
uint16_t g_historyCount = 0;

// ---- Panel web ----

// Pagina principal: el HTML de web/DashboardPage.h.
static void handleRoot() {
  server.send_P(200, "text/html", DASHBOARD_HTML);
}

// /data: la ultima lectura y el estado del modem, en JSON, para el panel.
static void handleData() {
  uint32_t now = millis();
  bool stale = g_latest.hasData && (now - g_latest.lastHeardMs > DATA_STALE_MS);
  const char *status = !g_latest.hasData ? "no_data" : (stale ? "stale" : "ok");
  uint32_t secondsAgo = g_latest.hasData ? (now - g_latest.lastHeardMs) / 1000 : 0;

  char json[800];
  snprintf(json, sizeof(json),
           "{\"status\":\"%s\",\"secondsAgo\":%lu,\"rssi\":%d,\"snr\":%d,"
           "\"temp\":%.2f,\"hum\":%.2f,\"pres\":%.2f,\"gas\":%.2f,"
           "\"pm1\":%.2f,\"pm25\":%.2f,\"pm10\":%.2f,"
           "\"co2\":%.2f,\"co\":%.2f,\"coTemp\":%.2f,"
           "\"windAngle\":%.2f,\"windSpeed\":%.2f,\"windValid\":%s,"
           "\"nbiot\":{\"state\":\"%s\",\"lastError\":\"%s\",\"rssiDbm\":%d,"
           "\"attached\":%s,\"attachedSec\":%lu,\"mqttConnected\":%s,\"mqttReconnects\":%lu,"
           "\"sent\":%lu,\"failed\":%lu,\"dropped\":%lu,"
           "\"batchReadings\":%u,\"batchSeconds\":%lu,\"ringCount\":%u}}",
           status, (unsigned long)secondsAgo, g_latest.rssi, g_latest.snr, g_latest.temp, g_latest.hum,
           g_latest.pres, g_latest.gas, g_latest.pm1, g_latest.pm25, g_latest.pm10, g_latest.co2,
           g_latest.co, g_latest.coTemp, g_latest.windAngle, g_latest.windSpeed,
           g_latest.windValid ? "true" : "false",
           modem.stateName(), modem.lastError(), modem.rssiDbm(), modem.attached() ? "true" : "false",
           (unsigned long)(modem.attachedUptimeMs() / 1000), modem.mqttConnected() ? "true" : "false",
           (unsigned long)modem.mqttReconnects(), (unsigned long)modem.packetsSent(),
           (unsigned long)modem.packetsFailed(), (unsigned long)modem.packetsDropped(),
           modem.batchReadingsTarget(), (unsigned long)modem.batchSecondsTarget(), modem.ringCount());

  server.send(200, "application/json", json);
}

// /history: todas las lecturas guardadas como CSV descargable, de la mas antigua a
// la mas reciente. millisSinceBoot permite calcular la hora real si se sabe cuando
// arranco la placa, y es monotono en todo el historico aunque esta placa no tenga
// reloj. Se construye en un buffer estatico (no en el monton de String de Arduino)
// dimensionado para LORA_HISTORY_CAPACITY filas de este ancho, con margen.
static char g_historyCsvBuf[LORA_HISTORY_CAPACITY * 110 + 256];

static void handleHistory() {
  size_t pos = 0;
  auto append = [&](const char *s) {
    size_t n = strlen(s);
    if (pos + n < sizeof(g_historyCsvBuf)) {
      memcpy(g_historyCsvBuf + pos, s, n);
      pos += n;
    }
  };

  append(
      "millisSinceBoot,senderAddr,rssi,snr,temp,hum,pres,gas,pm1,pm25,pm10,co2,co,coTemp,windAngle,"
      "windSpeed,windValid\n");

  for (uint16_t i = 0; i < g_historyCount; i++) {
    const SensorSnapshot &s = g_history[(g_historyHead + i) % LORA_HISTORY_CAPACITY];

    // UN CAMPO NO MEDIDO SALE VACIO, no como -999.
    //
    // El nodo manda -999 en los campos que esa lectura no midio - el CO2 va cada
    // 30 min y el VOC cada 20 - y volcarlo tal cual al CSV seria peor que inutil:
    // una columna con -999 entre valores reales se promedia, se grafica y
    // arruina la serie sin avisar. La celda vacia es la convencion que Excel y
    // pandas ya entienden como "aqui no hubo dato".
    char f[12][12];
    const float vals[12] = {s.temp, s.hum,  s.pres, s.gas,       s.pm1,       s.pm25,
                            s.pm10, s.co2,  s.co,   s.coTemp,    s.windAngle, s.windSpeed};
    for (int k = 0; k < 12; k++) {
      if (vals[k] < -900.0f) {
        f[k][0] = '\0';
      } else {
        snprintf(f[k], sizeof(f[k]), "%.2f", (double)vals[k]);
      }
    }

    char row[180];
    snprintf(row, sizeof(row), "%lu,%u,%d,%d,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%d\n",
             (unsigned long)s.lastHeardMs, s.senderAddr, s.rssi, s.snr, f[0], f[1], f[2], f[3],
             f[4], f[5], f[6], f[7], f[8], f[9], f[10], f[11], s.windValid ? 1 : 0);
    append(row);
  }

  server.sendHeader("Content-Disposition", "attachment; filename=\"module_b_history.csv\"");
  server.send(200, "text/csv", g_historyCsvBuf);
}

// ---- Pantalla OLED ----

// Dibuja un indicador de cobertura de 4 barras, estilo movil (crecen de izquierda a
// derecha y se rellenan hasta el nivel actual), con la esquina en (x, y), alineado
// abajo dentro de una fila de 8 px de alto. Los umbrales son la convencion habitual
// de barras de movil, sobre el rango -113..-51 dBm de parseCsq() (ver
// NbiotProtocol.cpp). rssiDbm==0 significa "aun no hay lectura de AT+CSQ" (una
// lectura real nunca es 0) y se dibuja con todas las barras vacias, en vez de
// mostrar una cobertura que no se ha medido.
static void drawSignalBars(int x, int y, int rssiDbm) {
  int level;
  if (rssiDbm == 0) {
    level = 0;
  } else if (rssiDbm >= -70) {
    level = 4;
  } else if (rssiDbm >= -85) {
    level = 3;
  } else if (rssiDbm >= -100) {
    level = 2;
  } else {
    level = 1;
  }

  const int barW = 3, gap = 1, maxH = 8;
  for (int i = 0; i < 4; i++) {
    int h = (i + 1) * 2;  // 2, 4, 6, 8 px de alto
    int bx = x + i * (barW + gap);
    int by = y + (maxH - h);
    if (i < level) {
      oled.fillRect(bx, by, barW, h, SSD1306_WHITE);
    } else {
      oled.drawRect(bx, by, barW, h, SSD1306_WHITE);
    }
  }
}

// Refresca la OLED con una vista compacta de la actividad, para saber de un vistazo
// que esta haciendo la placa sin necesitar el panel web: lo reciente de la recepcion
// LoRa y su calidad, y el estado actual de la maquina de estados NB-IoT/MQTT
// (stateName() pone literalmente "PUBLISHING" mientras se esta mandando un lote,
// "ATTACHING"/"MQTT_CONNECT" mientras conecta, "IDLE" mientras espera al siguiente
// lote - ver el enum State de modem_nbiot_mqtt.h).
static void updateOledStatus() {
  uint32_t now = millis();

  oled.clearDisplay();
  oled.setTextSize(1);
  oled.setTextColor(SSD1306_WHITE);
  oled.setCursor(0, 0);

  oled.println("Modulo B");

  if (g_latest.hasData) {
    uint32_t ageS = (now - g_latest.lastHeardMs) / 1000;
    // La marca "RX!" dura poco (un paquete nuevo es un instante, no un flujo): lo
    // justo para que una pantalla que se refresca cada 500 ms lo llegue a mostrar.
    oled.printf("LoRa: %s hace %lus\n", ageS < 3 ? "RX!" : "inactivo", (unsigned long)ageS);
    oled.printf(" rssi %d snr %d\n", g_latest.rssi, g_latest.snr);
  } else {
    oled.println("LoRa: sin datos aun");
    oled.println("");
  }

  oled.printf("NBIoT: %s\n", modem.stateName());
  oled.printf("MQTT: %s\n", modem.mqttConnected() ? "activo" : "caido");
  oled.printf("Env %lu Err %lu Prd %lu\n", (unsigned long)modem.packetsSent(),
              (unsigned long)modem.packetsFailed(), (unsigned long)modem.packetsDropped());

  drawSignalBars(112, 0, modem.rssiDbm());  // esquina superior derecha, en la fila de "Modulo B"

  oled.display();
}

// =============================================================================
// setup(): WiFi + panel web, OLED, radio LoRa y arranque (no bloqueante) del modem
// =============================================================================
void setup() {
  Serial.begin(115200);
  uint32_t waitStart = millis();
  while (!Serial && millis() - waitStart < 3000) delay(10);
  delay(500);

  Serial.println();
  Serial.println("=== CHIP FOREST LoRa ground-station receiver ===");

  // Punto de acceso WiFi propio, para el panel web local.
  WiFi.softAP(WIFI_AP_SSID, WIFI_AP_PASSWORD);
  IPAddress apIP = WiFi.softAPIP();
  Serial.printf("WiFi AP: SSID=\"%s\"  password=\"%s\"\n", WIFI_AP_SSID, WIFI_AP_PASSWORD);
  Serial.printf("Dashboard: http://%s/\n", apIP.toString().c_str());

  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.on("/history", handleHistory);
  server.begin();
  Serial.printf("Web server started - reading history CSV at http://%s/history\n", apIP.toString().c_str());

  Wire.begin(OLED_SDA_PIN, OLED_SCL_PIN);
  if (oled.begin(SSD1306_SWITCHCAPVCC, OLED_I2C_ADDR)) {
    oled.clearDisplay();
    oled.display();
    Serial.println("OLED: init OK");
  } else {
    Serial.println("OLED: init FAILED (check wiring - SDA=GPIO1, SCL=GPIO2)");
  }

  pinMode(LORA_EN_PIN, OUTPUT);
  digitalWrite(LORA_EN_PIN, LOW);  // encender el rail de LoRa (puerta Q4) - LOW = encendido, confirmado probando pines
  delay(300);

  // Los parametros del enlace TIENEN que coincidir con los de los nodos.
  Serial.println("Configuring RYLR998 (link parameters must match the sensor node):");
  bool ok = radio.begin(LORA_MY_ADDR, LORA_NETWORK_ID, LORA_BAND_HZ,
                         {LORA_PARAM_SF, LORA_PARAM_BW, LORA_PARAM_CR, LORA_PARAM_PREAMBLE}, &Serial);
  Serial.println(ok ? "Radio init OK - listening for packets."
                     : "Radio init FAILED - check wiring/baud before assuming a parameter is wrong.");
  Serial.println();

  // El modem arranca aqui pero trabaja poco a poco en cada vuelta de loop() (tick()).
  Serial.println("Starting NB-IoT/MQTT uplink to ThingsBoard - non-blocking, see modem_nbiot_mqtt.h:");
  modem.begin();
  Serial.println();
}

// ---- Ultimo estado conocido de cada nodo ----

// Ultimo estado de carga de cada nodo, por direccion LoRa.
//
// Se guarda aparte del anillo de telemetria porque STAT y telemetria llegan en
// paquetes distintos en momentos distintos: un nodo transmite sus lecturas y su
// salud por separado, y cualquiera de los dos puede ser el que llegue.
struct NodeCharge {
  uint16_t addr;
  int16_t pct;
  int16_t mv;
};
static NodeCharge g_nodeCharge[8];
static uint8_t g_nodeChargeCount = 0;

static void nodeChargeRemember(uint16_t addr, int pct, int mv) {
  for (uint8_t i = 0; i < g_nodeChargeCount; i++) {
    if (g_nodeCharge[i].addr == addr) {
      g_nodeCharge[i].pct = (int16_t)pct;
      g_nodeCharge[i].mv = (int16_t)mv;
      return;
    }
  }
  if (g_nodeChargeCount < (uint8_t)(sizeof(g_nodeCharge) / sizeof(g_nodeCharge[0]))) {
    g_nodeCharge[g_nodeChargeCount].addr = addr;
    g_nodeCharge[g_nodeChargeCount].pct = (int16_t)pct;
    g_nodeCharge[g_nodeChargeCount].mv = (int16_t)mv;
    g_nodeChargeCount++;
  }
}

// Ultimo estado de incendio de cada nodo, por direccion LoRa. Misma forma que el de
// carga y por el mismo motivo: STAT y telemetria son paquetes distintos que llegan
// en momentos distintos, y cualquiera puede ser el que llegue.
struct NodeAlarm {
  uint16_t addr;
  int8_t state;
};
static NodeAlarm g_nodeAlarm[8];
static uint8_t g_nodeAlarmCount = 0;

static void nodeAlarmRemember(uint16_t addr, int state) {
  for (uint8_t i = 0; i < g_nodeAlarmCount; i++) {
    if (g_nodeAlarm[i].addr == addr) {
      g_nodeAlarm[i].state = (int8_t)state;
      return;
    }
  }
  if (g_nodeAlarmCount < (uint8_t)(sizeof(g_nodeAlarm) / sizeof(g_nodeAlarm[0]))) {
    g_nodeAlarm[g_nodeAlarmCount].addr = addr;
    g_nodeAlarm[g_nodeAlarmCount].state = (int8_t)state;
    g_nodeAlarmCount++;
  }
}

// Copia en una lectura el ultimo estado de incendio conocido de su nodo.
static void nodeAlarmAttach(SensorSnapshot &snap) {
  for (uint8_t i = 0; i < g_nodeAlarmCount; i++) {
    if (g_nodeAlarm[i].addr == snap.senderAddr) {
      snap.alarmState = g_nodeAlarm[i].state;
      return;
    }
  }
}

// Copia en una lectura el ultimo estado de carga conocido de su nodo.
static void nodeChargeAttach(SensorSnapshot &snap) {
  for (uint8_t i = 0; i < g_nodeChargeCount; i++) {
    if (g_nodeCharge[i].addr == snap.senderAddr) {
      snap.chargePct = g_nodeCharge[i].pct;
      snap.capMv = g_nodeCharge[i].mv;
      return;
    }
  }
}

// =============================================================================
// loop(): servidor web, recepcion LoRa, maquina de estados del modem y OLED
// =============================================================================
void loop() {
  server.handleClient();

  // ---- Cada trama LoRa que haya llegado ----
  LoRaMessage msg;
  while (radio.poll(msg)) {
    Serial.printf("+RCV addr=%u len=%u rssi=%d snr=%d data=\"%.*s\"\n", msg.senderAddr, msg.length,
                  msg.rssi, msg.snr, msg.length, msg.payload);

    // LO PRIMERO, antes de interpretar nada: si Modulo A tiene una orden en cola para
    // este nodo, mandarla ya. Esta subida es el unico momento en que el nodo escucha:
    // abre una ventana de ~2 s justo despues de transmitir y luego queda sordo hasta su
    // siguiente envio. Todo lo que se haga antes de este envio se come esa ventana.
    char nodeName[24];
    nbiotResolveNodeName(msg.senderAddr, nodeName, sizeof(nodeName));
    // SOLO UNA TRAMA DE TELEMETRIA ABRE VENTANA. STAT Y ACK NO.
    //
    // El orden de Modulo C es: transmitir telemetria -> mantener ~2 s la ventana de
    // escucha -> transmitir STAT. La ventana esta ENTRE esas dos tramas. Asi que
    // cuando llega aqui un STAT, el nodo ya la ha cerrado y esta transmitiendo: una
    // orden mandada al recibir el STAT se pierde seguro. Lo mismo con un ACK, que el
    // nodo manda desde dentro de la ventana y despues sale de ella.
    //
    // Disparar con cada trama recibida desperdiciaba un intento cada vez que habia una
    // orden en cola entre la telemetria de un nodo y su STAT. Visto con la peticion 0,
    // 2026-10-06:
    //
    //   +RCV addr=4 data="STAT,4922,8,0,10100,63,3120,86,0"
    //     -> downlink to NodoC-3: CFG,temp_pre_off=47,... (sent)
    //   (sin ACK; recuperado 68 s despues por el reintento)
    //
    // y con la peticion 10 el dia anterior, disparada con el propio ACK del nodo.
    //
    // Una trama de telemetria es cualquier cosa que no sea una de las respuestas o
    // estados del nodo. Se reconoce por el prefijo, que es lo que las distingue en el
    // aire: una lectura empieza por un numero, o por 'B,' si es un lote.
    const bool opensWindow =
        !(msg.length >= 5 && strncmp(msg.payload, "STAT,", 5) == 0) &&
        !(msg.length >= 4 && strncmp(msg.payload, "ACK,", 4) == 0) &&
        !(msg.length >= 5 && strncmp(msg.payload, "NACK,", 5) == 0);

    if (opensWindow) {
      // Preguntar POR DISPOSITIVO, no "hay algo pendiente": con una cola, la primera
      // orden puede ser para otro nodo que no se ha despertado, y eso no debe impedir
      // atender a este en la ventana que acaba de abrir.
      const char *cfg = modem.nodeCommandCfgFor(nodeName);
      if (cfg) {
        bool sent = radio.send(msg.senderAddr, cfg, (uint8_t)strlen(cfg));
        Serial.printf("  -> downlink to %s: %s (%s)\n", nodeName, cfg, sent ? "sent" : "SEND FAILED");
        if (sent) modem.onNodeCommandDelivered(nodeName);
      } else {
        // ---- SINCRONIZACION DE RELOJ ----
        //
        // El nodo no tiene reloj: durmiendo cuenta el tiempo con su oscilador RC,
        // que se desvia un 1-2% y acumula. Aqui si hay hora de red, asi que se la
        // bajamos. No necesita protocolo nuevo - es una clave CFG mas.
        //
        // SOLO CUANDO NO HAY COMANDO PENDIENTE. La ventana de escucha del nodo da
        // para UNA trama, asi que una orden del operador siempre tiene prioridad;
        // la hora puede esperar a la siguiente ventana sin que pase nada.
        //
        // Y no en cada trama: un nodo en alarma transmite cada 3 minutos, y
        // sincronizarlo tan seguido no mejora nada y gasta tiempo de aire, que en
        // EU868 esta limitado por ciclo de trabajo.
        static uint32_t lastTimeSyncMs[sizeof(NBIOT_NODE_NAMES) / sizeof(NBIOT_NODE_NAMES[0])] = {0};
        const unsigned long epochS = modem.netEpochSeconds();
        if (epochS != 0) {
          int slot = -1;
          for (size_t i = 0; i < sizeof(NBIOT_NODE_NAMES) / sizeof(NBIOT_NODE_NAMES[0]); i++) {
            if (NBIOT_NODE_NAMES[i].addr == msg.senderAddr) { slot = (int)i; break; }
          }
          const uint32_t now = millis();
          // El primer envio sale siempre: lastTimeSyncMs arranca en 0 y un nodo
          // recien encendido es justo el que mas necesita la hora.
          if (slot >= 0 && (lastTimeSyncMs[slot] == 0 ||
                            now - lastTimeSyncMs[slot] >= NODE_TIME_SYNC_MIN_GAP_MS)) {
            char timeCfg[32];
            snprintf(timeCfg, sizeof(timeCfg), "CFG,TIME=%lu", epochS);
            bool sent = radio.send(msg.senderAddr, timeCfg, (uint8_t)strlen(timeCfg));
            Serial.printf("  -> reloj a %s: %s (%s)\n", nodeName, timeCfg,
                          sent ? "enviado" : "FALLO");
            if (sent) lastTimeSyncMs[slot] = now;
          }
        }
      }
    }

    // La respuesta de un nodo a una orden, no telemetria: se pasa al modem para que
    // conteste al RPC original, y no se intenta interpretar como lectura (se tiraria
    // por "numero de campos incorrecto").
    if (msg.length >= 4 && strncmp(msg.payload, "ACK,", 4) == 0) {
      modem.onNodeCommandAck(nodeName, msg.payload);
      continue;
    }

    // El nodo entendio la orden pero no aplico nada. Se mira ANTES del interprete de
    // telemetria por el mismo motivo que el ACK: es una respuesta, no una lectura, y
    // si no se tiraria por "numero de campos incorrecto" sin dejar rastro.
    if (msg.length >= 5 && strncmp(msg.payload, "NACK,", 5) == 0) {
      modem.onNodeCommandNack(nodeName, msg.payload + 5);
      continue;
    }

    // STAT - el paquete de salud del propio nodo, no telemetria.
    //
    //   STAT,<despertar>,<motivoReinicio>,<arranques>,<bme><bmv><co2><co><viento>,<antiguedad>,<mV>,<pct>,<alarma>
    //
    // Es un paquete aparte precisamente para no romper el interprete de telemetria de
    // 13 campos, pero el estado de carga que lleva merece estar en ThingsBoard: un nodo
    // a punto de quedarse sin energia deberia decirlo donde alguien lo vea. Asi que los
    // valores se recuerdan por nodo y se adjuntan a la siguiente publicacion de
    // telemetria de ese nodo.
    if (msg.length >= 5 && strncmp(msg.payload, "STAT,", 5) == 0) {
      char buf[96];
      size_t n = msg.length < sizeof(buf) - 1 ? msg.length : sizeof(buf) - 1;
      memcpy(buf, msg.payload, n);
      buf[n] = 0;

      // Los campos 6, 7 y 8 (contando desde 0) son los mV, el porcentaje y la alarma.
      int field = 0;
      int mv = -1, pct = -1, alarm = -1;
      char *tok = strtok(buf, ",");
      while (tok) {
        if (field == 6) mv = atoi(tok);
        if (field == 7) pct = atoi(tok);
        if (field == 8) alarm = atoi(tok);  // 0 normal, 1 pre-alarma, 2 alarma
        field++;
        tok = strtok(nullptr, ",");
      }

      // ESTADO DE INCENDIO - la razon de ser de todo este camino.
      //
      // Se recuerda por nodo y se adjunta a la siguiente publicacion de telemetria de
      // ese nodo, y si esta levantado ademas se salta la espera del lote para que salga
      // YA en vez de hasta un minuto despues. Agrupar en lotes esta bien para la
      // telemetria y mal para un incendio.
      //
      // El campo 8 no existe en firmware antiguo de nodo, lo que deja alarm a -1 y se
      // informa como desconocido, no como NORMAL. Un panel que muestra NORMAL con
      // seguridad para un nodo que nunca lo dijo es peor que uno que no muestra nada.
      if (alarm >= 0) {
        nodeAlarmRemember(msg.senderAddr, alarm);
        // Las lecturas de este nodo ya estan en cola con el estado anterior - el nodo
        // manda la telemetria antes que su STAT. Hay que reetiquetarlas, o la
        // publicacion inmediata de abajo subiria justo el valor que existe para
        // corregir.
        modem.applyAlarmToQueued(msg.senderAddr, (int8_t)alarm);
        if (alarm > 0) {
          char who[32];
          nbiotResolveNodeName(msg.senderAddr, who, sizeof(who));
          Serial.printf("  *** %s EN %s - publicando YA ***\n", who,
                        alarm >= 2 ? "ALARMA" : "PRE-ALARMA");
          Serial.flush();
          modem.requestFlush();
        }
      }

      // TRES casos aqui, no dos. Un nodo que no manda los campos de carga y uno que
      // los manda como "desconocido" son problemas distintos, y juntarlos manda a
      // alguien a buscar una version de firmware que esta bien.
      //
      // pct == -1 puede ser lo NORMAL: significa que el nodo no tiene el divisor
      // calibrado o no lo lleva montado. No es firmware antiguo.
      if (field <= 7) {
        // Paquete corto de verdad: los campos de mV/porcentaje no existen.
        Serial.printf("  STAT has only %d fields - node predates the charge report\n", field);
      } else {
        // Los campos estan. Se recuerdan AUNQUE pct sea -1: los milivoltios en crudo
        // son de donde saldra la calibracion, asi que tirarlos porque el porcentaje
        // es desconocido descartaria justo el numero que permitiria calcularlo.
        nodeChargeRemember(msg.senderAddr, pct, mv);
        if (pct >= 0) {
          Serial.printf("  node health: charge %d%% (%d mV at the sense pin)\n", pct, mv);
        } else {
          Serial.printf("  node health: %d mV at the sense pin, percentage not yet "
                        "calibrated (SUPERCAP_CALIBRATED=0 on the node)\n", mv);
        }
      }
      continue;
    }

    // Un nodo manda o una lectura suelta o un LOTE de ellas. Las dos formas acaban
    // aqui; hay que mantenerlas separadas, porque meter un lote por el interprete de
    // lecturas sueltas es lo que corrompia los datos que llegaban a ThingsBoard: la
    // "B" y el numero se leian como temperatura y humedad, todos los campos se
    // desplazaban dos posiciones y la segunda lectura del lote se tiraba entera.
    SensorSnapshot batch[BATCH_RX_MAX];
    uint16_t ages[BATCH_RX_MAX] = {};
    uint8_t count = 0;
    bool isBatch = (msg.length >= 2 && msg.payload[0] == 'B' && msg.payload[1] == ',');

    if (isBatch) {
      count = TelemetryParser::parseBatch(msg.payload, msg.length, batch, BATCH_RX_MAX, ages);
      if (count == 0) {
        Serial.println("  (batch payload didn't parse - dropped)");
      } else {
        Serial.printf("  batch of %u reading(s)\n", (unsigned)count);
      }
    } else if (TelemetryParser::parse(msg.payload, msg.length, batch[0])) {
      ages[0] = 0;  // una lectura suelta es actual por definicion
      count = 1;
    }

    if (count == 0 && !isBatch) {
      Serial.println("  (payload didn't match the expected schema - dropped, see "
                      "telemetry/TelemetryParser.h)");
    }

    for (uint8_t b = 0; b < count; b++) {
      SensorSnapshot &snap = batch[b];
      snap.hasData = true;
      // Se retrasa la marca de tiempo segun la antiguedad del registro, para que una
      // lectura de un lote quede fechada cuando se TOMO, no cuando llego el paquete.
      // Las lecturas sueltas tienen antiguedad 0 y no cambian.
      // LA EDAD VIENE EN SEGUNDOS REALES, medidos por el propio nodo.
      //
      // Antes se multiplicaba por NODE_TICK_SECONDS = 10, asumiendo que el nodo
      // contaba en ticks de 10 s. El tick real eran 18,8 s - un error del 88% que
      // comprimia todo el historico casi a la mitad - y con el planificador por
      // TIEMPO los despertares pasaron a ser de 5 MINUTOS, con lo que el error se
      // fue a 30x. El rediseno empeoro el bug hasta hacerlo imposible de ignorar.
      //
      // Ahora el nodo manda cuantos SEGUNDOS hace que tomo cada lectura, contados
      // con su propio reloj y corregidos por la deriva que aprende de Modulo B.
      // Aqui no hay nada que adivinar.
      uint32_t ageMs = (uint32_t)ages[b] * 1000UL;
      snap.lastHeardMs = millis() - ageMs;
      snap.senderAddr = msg.senderAddr;
      snap.rssi = msg.rssi;
      snap.snr = msg.snr;
      nodeChargeAttach(snap);  // el ultimo STAT de este nodo, si lo hay
      nodeAlarmAttach(snap);
      g_latest = snap;
      modem.enqueue(snap);  // a la cola de subida a ThingsBoard

      // Y al historico local del panel web.
      uint16_t idx = (uint16_t)((g_historyHead + g_historyCount) % LORA_HISTORY_CAPACITY);
      g_history[idx] = snap;
      if (g_historyCount < LORA_HISTORY_CAPACITY) {
        g_historyCount++;
      } else {
        g_historyHead = (uint16_t)((g_historyHead + 1) % LORA_HISTORY_CAPACITY);
      }
    }
  }

  // ---- El modem: una pequena parte de su trabajo en cada vuelta, sin bloquear ----
  modem.tick();

  // ---- La OLED, cada 500 ms ----
  static uint32_t lastOledMs = 0;
  if (millis() - lastOledMs >= 500) {
    lastOledMs = millis();
    updateOledStatus();
  }
}

#define LORA_NODE_ADDR 1  // direccion AT+ADDRESS del nodo C1

// ---------- Tabla de nombres (direccion LoRa -> nombre del dispositivo en ThingsBoard) ----------
// La usa la publicacion MQTT de pasarela para nombrar cada nodo en el contenido de
// v1/gateway/telemetry. Una entrada por nodo desplegado; anadir aqui los nodos
// nuevos y no quitar ni renumerar nunca una entrada existente (el nombre es lo que
// ThingsBoard usa para identificar el dispositivo en adelante).
struct NodeNameEntry {
  uint16_t addr;
  const char *name;
};

// La direccion 2 es este receptor (LORA_MY_ADDR) y falta a proposito: nunca emite
// telemetria. Todos los nodos salieron con la direccion 1, lo que los hacia
// indistinguibles en el aire; ahora cada uno tiene la suya, fijada en cada
// compilacion con los entornos env:module-c-node*.
static const NodeNameEntry NBIOT_NODE_NAMES[] = {
    {LORA_NODE_ADDR, "NodoC-1"},  // 1
    {3, "NodoC-2"},
    {4, "NodoC-3"},
};

// Traduce addr a su nombre configurado, o a "NodoDesconocido-<addr>" si no esta en
// la tabla de arriba. A proposito nunca se tira una lectura porque su emisor no se
// reconozca: un nodo con un nombre provisional se ve y se arregla en ThingsBoard; un
// nodo cuyos datos se tiran en silencio, no.
inline void nbiotResolveNodeName(uint16_t addr, char *outName, size_t outCap) {
  for (size_t i = 0; i < sizeof(NBIOT_NODE_NAMES) / sizeof(NBIOT_NODE_NAMES[0]); i++) {
    if (NBIOT_NODE_NAMES[i].addr == addr) {
      strncpy(outName, NBIOT_NODE_NAMES[i].name, outCap - 1);
      outName[outCap - 1] = '\0';
      return;
    }
  }
  snprintf(outName, outCap, "NodoDesconocido-%u", addr);
}

// ---------- Lotes / cadencia de publicacion MQTT (ModemNBIoTMqtt) ----------
// Independiente de las constantes NBIOT_BATCH_* de la trama binaria de mas abajo
// (esas son de la via UDP retirada y de su prueba unitaria). La via MQTT acumula las
// lecturas LoRa y las sube en lotes por tiempo:
//
//   MQTT_BATCH_SECONDS       - publicar cada este tiempo.
//   MQTT_BATCH_MAX_READINGS  - disparo por cantidad: publicar antes si el anillo
//                              llega a estas lecturas antes del temporizador (evita
//                              que una subida lenta haga desbordar el anillo).
//   MQTT_PUB_CHUNK_READINGS  - un solo AT+QMTPUB tiene que quedar por debajo del
//                              limite de ~1400 bytes por mensaje del BC660K-GL; un
//                              lote se manda como ceil(N/trozo) QMTPUB seguidos por la
//                              sesion ya abierta. Una lectura con marca de tiempo
//                              son ~230 bytes normalmente y ~300 en el peor caso, asi
//                              que 4 (~1200 B en el peor caso) deja margen bajo 1400.
// Bajado de 600 a 60 (2026-09-14). Diez minutos es mucho tiempo para quedarse una
// lectura de un sensor de INCENDIOS, y ademas hacia imposible comprobar la subida:
// con menos de MQTT_BATCH_MAX_READINGS en cola no se publicaba nada en diez minutos y
// la cadena parecia rota cuando solo estaba esperando. 60UL es MQTT_BATCH_SECONDS_MIN.
// Subirlo otra vez si el volumen de datos NB-IoT pasa a ser el limite: se cambia
// coste de subida por rapidez de deteccion.
#define MQTT_BATCH_SECONDS 60UL
#define MQTT_BATCH_MAX_READINGS 6
#define MQTT_PUB_CHUNK_READINGS 4

// Los dos objetivos de lote se pueden cambiar en marcha desde Modulo A - ver la
// seccion de ordenes de abajo. Se recortan para que una orden mal tecleada no pueda
// parar la telemetria ni agotar los datos de la SIM:
//   suelo de SEGUNDOS 60  - por debajo, la subida cuesta mas en sobrecarga movil de lo
//                           que valen las lecturas.
//   techo de LECTURAS     - nunca por encima de NBIOT_RING_CAPACITY, o el disparo por
//                           cantidad no llegaria nunca antes de que el anillo diera la
//                           vuelta y tirara lecturas.
#define MQTT_BATCH_SECONDS_MIN 60UL
#define MQTT_BATCH_SECONDS_MAX 86400UL
#define MQTT_BATCH_READINGS_MIN 1
#define MQTT_BATCH_READINGS_MAX NBIOT_RING_CAPACITY

// ---------- Ordenes: Modulo A -> Modulo B (-> Modulo C) ----------
// Dos suscripciones, porque ThingsBoard se dirige a la propia pasarela y a sus
// dispositivos hijos por topics distintos:
//   MQTT_TOPIC_DEVICE_RPC_SUB  - RPC a ESTE dispositivo (la pasarela CON-01): ajustes
//                                de Modulo B y la orden configurarModuloC. La
//                                respuesta va a MQTT_TOPIC_DEVICE_RPC_RESP + el id.
//   MQTT_TOPIC_GATEWAY_RPC     - RPC a un dispositivo hijo (NodoC-1, etc.). Modulo B
//                                lo traduce a la gramatica LoRa "CFG,<clave>=<valor>"
//                                que ya entiende Modulo C, lo pone en cola y lo manda
//                                en la siguiente ventana de escucha de ese nodo. La
//                                respuesta se publica en este mismo topic.
#define MQTT_TOPIC_DEVICE_RPC_SUB "v1/devices/me/rpc/request/+"
#define MQTT_TOPIC_DEVICE_RPC_RESP "v1/devices/me/rpc/response/"
#define MQTT_TOPIC_GATEWAY_RPC "v1/gateway/rpc"

// Donde va el commandLog. El protocolo acordado (Protocolo_comandos_ModA_hacia_
// ModC_por_B_v1) informa del RESULTADO de una orden como telemetria y no como
// respuesta RPC, porque el RPC se contesta en el acto con "enviado" mucho antes de
// llegar al nodo. Dos mensajes distintos, dos topics distintos.
#define MQTT_TOPIC_DEVICE_TELEMETRY "v1/devices/me/telemetry"
#define MQTT_DOWNLINK_QOS 1

// Una orden para un nodo solo se puede entregar en la ventana de escucha de ~2 s
// tras cada envio del nodo. Pasado este tiempo se deja de intentar y se informa de
// un timeout, en vez de mantener la orden en cola para siempre contra un nodo muerto.
// (OJO: "3 ventanas perdidas a 5 min" era con el envio cada 5 min; con el calendario
// actual, en estado NORMAL el nodo envia cada 30 min, asi que una orden puede expirar
// antes de que el nodo abra su siguiente ventana.)
#define DOWNLINK_QUEUE_TIMEOUT_MS 900000UL  // 15 min = 3 missed windows at 5 min

// Separacion minima entre dos intentos de entrega de LA MISMA orden.
//
// Un nodo transmite mas de una vez por ventana - telemetria, y el STAT unos 2 s
// despues - y main.cpp ejecuta su gancho de ordenes con cada trama que recibe. Sin un
// minimo aqui el reintento saltaria otra vez con la trama STAT, gastando todos los
// intentos dentro de una misma ventana cuando el nodo ya estaba sordo.
//
// 10 s queda por encima de esa separacion dentro de la ventana y por debajo de
// LORA_TX_PERIOD_MIN_MS (15 s), el periodo de envio mas corto que se le puede poner a
// un nodo, asi que una ventana nunca produce dos intentos y nunca se salta una.
#define DOWNLINK_RETRY_MIN_GAP_MS 10000UL

// Marcas de tiempo por lectura: sin "ts", ThingsBoard pone a todas las lecturas de un
// lote la hora de llegada, aplastando un lote de minutos en un instante. Modulo B no
// tiene reloj, asi que lee la hora de red una vez por registro con AT+CCLK? (necesita
// AT+CTZU=1, puesto en CONFIG) y fecha cada lectura. Si la red no da nunca la hora
// (ano de CCLK < 2023), la subida pasa sola a la forma plana sin "ts".
#define NBIOT_MIN_VALID_EPOCH_MS 1672531200000LL  // 2023-01-01T00:00:00Z - anterior = "sin hora de red"

// KEEPALIVE: corto, porque Modulo B tiene que estar ESCUCHANDO, no solo accesible.
//
// Valia 3600 s - el maximo - con el razonamiento de que hacer ping entre envios
// reales era un desperdicio: "un ritmo de PINGREQ lo bastante corto para importar
// (por ejemplo cada 90 s) supondria ~13 veces el trafico real de telemetria". Esa
// cuenta se hizo cuando los lotes eran cada 10-20 minutos. MQTT_BATCH_SECONDS vale
// ahora 60, y la cifra se equivoca en cuatro ordenes de magnitud:
//
//   PINGREQ + PINGRESP = 4 bytes, cada 120 s       ->    120 B/hora
//   un mensaje de telemetria ~1100 bytes, cada 60 s -> 66000 B/hora
//
//   los ping son el 0,18% de la telemetria - aproximadamente 1 MB al ANO
//
// El valor viejo servia para la SUBIDA, donde una sesion muerta se descubre en el
// siguiente envio y se reconstruye en ese momento. Es incorrecto para la BAJADA de
// ordenes. Con un keepalive de 3600 s ningun extremo cuestiona una conexion TCP caida
// en silencio durante hasta una hora: Modulo B cree que esta conectado, el servidor
// cree que el cliente esta vivo, y todas las ordenes de mientras no van a ninguna
// parte. Nada en los logs parece ir mal.
//
// A 120 s el propio modem manda el PINGREQ (sin trafico AT nuestro - lo gestiona el
// modulo a partir de este ajuste), asi que un enlace muerto se detecta en un par de
// minutos y se reconstruye, y la sesion se mantiene a traves de los tiempos de NAT del
// operador aunque los nodos no tengan nada que decir.
#define MQTT_KEEPALIVE_S 120UL

// El JSON es mucho mas largo que la antigua trama binaria (~15 campos por lectura
// como texto "clave":valor pasan de 100 bytes por lectura, frente a los 30 fijos del
// formato binario): dimensionado con generosidad.
#define MQTT_PAYLOAD_MAX_BYTES 4096

// ---------- Lotes de la via UDP antigua (HISTORICO) ----------
// La subida de la via UDP era una trama binaria compacta (ver nbiot/NbiotProtocol.h),
// no CSV de texto: 10 lecturas completas de 13 campos como texto ocupan ~700-800
// bytes. Solo para la via UDP retirada (ModemNBIoT) y su prueba unitaria; la via MQTT
// usa MQTT_BATCH_* de arriba.
// EXCEPCION: NBIOT_RING_CAPACITY SI se usa hoy - es el tamano del anillo de lecturas
// pendientes de la via MQTT.
#define NBIOT_BATCH_DEFAULT_READINGS 10
#define NBIOT_BATCH_DEFAULT_SECONDS (20UL * 60UL)
#define NBIOT_RING_CAPACITY 24  // > lote maximo, para que una subida lenta no obligue a tirar lecturas en el acto
#define NBIOT_PAYLOAD_MAX_BYTES 500

// Techo que usa el static_assert de la trama binaria en NbiotProtocol.h (HISTORICO,
// solo de la via UDP).
#define NBIOT_CFG_READINGS_MAX 16  // (500 - kBatchHeaderSize) / kBatchRecordSize, ver el static_assert

// ---------- Plazos de cada estado (ms) ----------
#define NBIOT_TIMEOUT_POWERING_MS 3000
// Tiempo que se le da al modem para contestar a un AT tras soltar PWRKEY.
//
// SUBIDO de 8000 a 30000 (2026-09-11). El plazo viejo solo le daba al BC660K-GL
// ~9,7 s desde soltar PWRKEY hasta su primera respuesta antes de que el firmware lo
// diera por fallido y LE CORTARA LA ALIMENTACION: PWRKEY se suelta a t=1300 ms,
// POWERING pasa el testigo a t=3000 ms, y WAIT_AT se rendia 8 s despues. Un
// BC660K-GL necesita a menudo mas que eso para tener el UART listo, asi que la
// recuperacion apagaba un modem que todavia estaba arrancando - para siempre, porque
// cada reintento repetia la misma carrera. Sintoma observado: "-> AT" repitiendose
// sin fin sin un solo "<- OK".
//
// Tiene que seguir siendo holgadamente mayor que el peor arranque del modulo. No
// cuesta nada con el modem sano, porque WAIT_AT sale en cuanto llega el primer OK: es
// un techo, no una espera.
#define NBIOT_TIMEOUT_WAIT_AT_MS 30000
#define NBIOT_TIMEOUT_CONFIG_MS 8000
#define NBIOT_TIMEOUT_ATTACH_MS 60000  // preguntando AT+CEREG? hasta estado 1 o 5
#define NBIOT_TIMEOUT_SOCKET_MS 15000  // cubre el OK y el aviso asincrono posterior
#define NBIOT_TIMEOUT_SEND_MS 10000
#define NBIOT_AT_CMD_TIMEOUT_MS 5000  // techo de un comando simple (AT, ATE0, CPIN?, ...)
#define NBIOT_ATTACH_POLL_INTERVAL_MS 2000  // cada cuanto se repite AT+CEREG? mientras se registra
#define NBIOT_AT_PING_GAP_MS 1000  // cada cuanto se repite el "AT" de comprobacion en WAIT_AT
#define NBIOT_CSQ_POLL_INTERVAL_MS 30000  // cada cuanto se refresca el RSSI para el panel en IDLE

// ---------- Recuperacion escalonada / espera creciente ----------
// reintentar el envio -> reconectar MQTT -> volver a registrarse -> apagar y
// encender. Apagar y encender en frio obliga a un registro completo en la red (lento,
// puede ser de decenas de segundos, y lo mas caro que hace este modem), asi que solo
// puede ser el ultimo recurso, nunca la primera respuesta a un envio fallido.
#define NBIOT_MAX_SEND_RETRIES 2
#define NBIOT_MAX_SOCKET_RETRIES 2
#define NBIOT_MAX_ATTACH_RETRIES 2
#define NBIOT_BACKOFF_BASE_MS 2000
#define NBIOT_BACKOFF_MAX_MS 120000
// Tiempo que EN se mantiene desactivado antes de volver a activarlo en un apagado y
// encendido. SUBIDO (2026-09-11): 500 ms no bastan para que el rail propio del modulo
// caiga de verdad, asi que el "apagado" podia dejarlo medio alimentado en un estado
// indefinido en vez del arranque en frio limpio que se pretendia.
#define NBIOT_POWER_OFF_SETTLE_MS 4000

#pragma once

#include <Arduino.h>

#include "Config.h"
#include "nbiot/NbiotProtocol.h"
#include "telemetry/SensorSnapshot.h"

// =============================================================================
// DRIVER DEL MODEM NB-IoT (Quectel BC660K-GL) CON MQTT HACIA THINGSBOARD
// =============================================================================
//
// Driver NO BLOQUEANTE del cliente MQTT integrado en el BC660K-GL, que publica la
// telemetria de los nodos en ThingsBoard mediante la API de pasarela (Gateway API).
// Es hermano de ModemNBIoT (la via antigua por UDP), no un reemplazo dentro de el:
// el transporte cambia demasiado (QMTOPEN/QMTCONN encadenados sin pausa, una
// publicacion con longitud + aviso '>' + Ctrl-Z, JSON en vez de una trama binaria,
// otros avisos) como para cambiar un solo metodo de ModemNBIoT.
//
// La secuencia de encendido y registro (de POWERING a ATTACHING) esta confirmada en
// el hardware desde el 2026-08-19, y es distinta de la que hacia ModemNBIoT (que
// nunca se valido: pines equivocados, sin canal ni PWRKEY, AT+CFUN=1 despues de
// AT+CPIN? en vez de antes, AT+CGDCONT en vez de AT+QCGDEFCONT). Solo se corrigio
// aqui, porque aquella clase se esta retirando.
//
// Maquina de estados:
//   OFF -> POWERING -> WAIT_AT -> CONFIG -> ATTACHING -> MQTT_CONNECT -> IDLE
//   IDLE <-> PUBLISHING (lote listo; vuelve a IDLE en cualquier caso)
//   IDLE <-> RPC_REPLY (respuestas a ThingsBoard pendientes)
//   cualquier estado -> ERROR ante un plazo agotado sin arreglo; ERROR -> OFF tras
//   una espera creciente (backoff).
//
// Politica de la sesion MQTT: se mantiene abierta entre envios y no se le hace ping
// por iniciativa propia entre lotes (ver MQTT_KEEPALIVE_S en Config.h). Si
// PUBLISHING encuentra la sesion caida (nunca abierta, o +QMTSTAT dijo que se cayo),
// pasa antes por MQTT_CONNECT y despues publica. Ademas IDLE la reconstruye por si
// sola si se cae sin que haya nada que publicar (ver tickIdle()).
//
// Recuperacion escalonada ante fallos: reintentar la publicacion, despues
// reconectar MQTT (QMTOPEN+QMTCONN otra vez), despues volver a registrarse en la
// red, y solo como ultimo recurso apagar y encender el modem.
//
// En sentido contrario (de ThingsBoard a los nodos) recibe RPC por MQTT, los
// guarda en una cola por nodo, y main.cpp los entrega por LoRa en la ventana de
// escucha de cada nodo; el resultado (ACK/NACK/timeout) se devuelve a ThingsBoard.
class ModemNBIoTMqtt {
public:
  enum class State : uint8_t {
    OFF,
    POWERING,
    WAIT_AT,
    CONFIG,
    ATTACHING,
    MQTT_CONNECT,
    IDLE,
    PUBLISHING,
    RPC_REPLY,
    ERROR,
  };

  ModemNBIoTMqtt(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint8_t enPin, uint8_t channelPin,
                 uint8_t pwrkeyPin, uint32_t baud = NBIOT_BAUD);

  // Configura los pines y deja el modem apagado. Se llama una vez desde setup().
  void begin();

  // Hace avanzar la maquina de estados: una pequena parte acotada de trabajo y
  // vuelve, sin bloquear nunca. Se llama en cada vuelta de loop().
  void tick();

  // Mete una lectura en el anillo de lotes. Nunca bloquea. Si el anillo ya esta
  // lleno, se descarta la mas antigua y sube packetsDropped().
  void enqueue(const SensorSnapshot &snap);

  // Publicar en la primera ocasion en vez de esperar al lote.
  //
  // Los lotes existen para que el enlace NB-IoT y los datos de la SIM sean
  // razonables, y para la telemetria es lo correcto. Un incendio no es telemetria:
  // un nodo informando de uno mientras Modulo B se queda el paquete el resto de una
  // ventana de 60 s es un minuto de silencio justo en el momento para el que existe
  // el sistema.
  void requestFlush() { _flushRequested = true; }

  // Publica un registro commandLog: el resultado de una orden que ya se confirmo
  // como "enviado". Usa la misma cola de publicacion que las respuestas RPC.
  void publishCommandLog(long requestId, const char *status);

  // El mismo registro mas un campo "detail": las claves que el nodo aplico de verdad,
  // o el motivo por el que lo rechazo. El documento del protocolo define
  // requestId/status/timestamp; esto anade sin cambiar, asi que un panel que lea solo
  // los campos documentados no se ve afectado.
  // `device` ademas copia el registro en la telemetria de ese nodo, para que se pueda
  // leer tanto si el panel mira la pasarela como si mira el nodo.
  void publishCommandLogApplied(long requestId, const char *status, const char *detail,
                                const char *device = nullptr);

private:
  void handleConfigurarModuloC(const NbiotProtocol::MqttMessage &msg, const char *idStr);

public:

  // Pone el estado de incendio ACTUAL de un nodo en las lecturas suyas que ya estan
  // esperando a salir.
  //
  // Sin esto, la publicacion urgente por alarma subiria un estado viejo, y seria puro
  // teatro. Un nodo transmite PRIMERO sus lecturas y su STAT unos dos segundos
  // despues, asi que cuando Modulo B se entera de que el nodo ha entrado en alarma,
  // la lectura de ese ciclo ya esta en cola con el estado ANTERIOR. requestFlush()
  // sacaria deprisa un paquete diciendo alarmState 0 - justo el paquete cuyo unico
  // proposito era decir lo contrario - y el estado real llegaria en el lote
  // siguiente, hasta un minuto despues.
  //
  // Confirmado en el hardware el 2026-09-28: los tres nodos publicaron alarmState -1
  // en un lote y 0 en el siguiente, siempre un ciclo por detras de sus STAT.
  void applyAlarmToQueued(uint16_t addr, int8_t state) {
    for (uint8_t i = 0; i < _ringCount; i++) {
      if (_ring[i].senderAddr == addr) _ring[i].alarmState = state;
    }
  }

  // ---------- Observabilidad: lo que muestran el panel y la OLED ----------
  State state() const { return _state; }
  // El UART del modem NO se mantiene abierto a traves de los apagados, a proposito:
  // un pin TX conducido alimenta el modulo por la puerta de atras a traves de sus
  // diodos de proteccion e impide que arranque en frio. Ver enterPowering().
  void releaseUart();
  void startUart();

  const char *stateName() const;
  // Nombra un estado sin necesitar una instancia en ese estado: setState() tiene que
  // poder imprimir el estado al que ENTRA antes de asignar _state.
  static const char *stateNameOf(State st);
  const char *lastError() const { return _lastError; }
  int rssiDbm() const { return _rssiDbm; }
  bool attached() const { return _attached; }
  uint32_t attachedUptimeMs() const;
  bool mqttConnected() const { return _mqttConnected; }
  uint32_t packetsSent() const { return _packetsSent; }
  uint32_t packetsFailed() const { return _packetsFailed; }
  uint32_t packetsDropped() const { return _packetsDropped; }
  uint32_t mqttReconnects() const { return _mqttReconnects; }
  uint16_t batchReadingsTarget() const { return _batchReadingsTarget; }
  uint32_t batchSecondsTarget() const { return _batchSecondsTarget; }
  uint8_t ringCount() const { return _ringCount; }
  bool haveNetTime() const { return _haveNetTime; }

  // Epoca UNIX en SEGUNDOS, para bajarsela a los nodos por CFG,TIME. Devuelve 0
  // cuando la red no ha dado hora todavia, y el llamante tiene que tratar ese 0
  // como "no se" y no mandar nada - un cero situaria al nodo en 1970.
  //
  // EN SEGUNDOS Y NO EN MILISEGUNDOS a proposito: cabe en 32 bits hasta 2106, asi
  // que ni este extremo ni el nodo necesitan enteros de 64 bits ni `%lld`, que no
  // es de fiar con la nano-newlib de este build - ya nos costo un fallo aqui
  // mismo. Se pierde precision por debajo del segundo, irrelevante contra
  // periodos de minutos.
  unsigned long netEpochSeconds() const {
    if (!_haveNetTime) return 0;
    const int64_t ms = netNowMs();
    return (ms > 0) ? (unsigned long)(ms / 1000) : 0;
  }

  // ---------- Configuracion en marcha, ajustable desde Modulo A ----------
  // Las dos recortan a los limites MQTT_BATCH_*_MIN/MAX de Config.h y devuelven el
  // valor realmente aplicado, para que un recorte sea visible (y se devuelva en la
  // respuesta RPC) en vez de darse por supuesto.
  uint32_t setBatchSeconds(uint32_t seconds);

  bool _flushRequested = false;  // ver requestFlush()
  uint16_t setBatchReadings(uint16_t readings);

  // ---------- Reenvio de ordenes a un nodo hijo ----------
  // Una orden dirigida a un nodo no se puede entregar al llegar: Modulo C solo
  // escucha ~2 s despues de cada transmision suya. Asi que se aparca aqui y main.cpp
  // la manda por LoRa en cuanto ve la subida de ese nodo.
  //
  // UNA LISTA DE PENDIENTES, como especifica el documento del protocolo ("guardar
  // comando en lista pendiente"), no un unico hueco.
  //
  // Un nodo solo oye en la ventana de ~2 s tras su subida, asi que una orden puede
  // esperar minutos a entregarse. Con un solo hueco, todas las ordenes que llegaban
  // durante esa espera se rechazaban como "ocupado" - y el panel de Modulo A tiene
  // una fila por magnitud, asi que poner la temperatura y luego la humedad hacia que
  // la segunda se rechazara por un motivo que el operador no podia ver ni resolver.
  //
  // Todo va por dispositivo: una orden para C-2 no debe bloquear nunca una para C-3
  // solo porque C-2 aun no se ha despertado.
  const char *nodeCommandCfgFor(const char *device) const {
    const int i = deliverableIndexFor(device);
    return i < 0 ? nullptr : _nodeCmds[i].cfg;
  }
  // main.cpp la llama cuando la cadena CFG de ese dispositivo ya ha salido por LoRa.
  void onNodeCommandDelivered(const char *device);
  // main.cpp la llama cuando vuelve un paquete "ACK,..." de un nodo. Informa del
  // resultado y libera el hueco de ese dispositivo.
  void onNodeCommandAck(const char *device, const char *ackPayload);

  // El nodo entendio la orden y no aplico NADA. Se informa enseguida como el estado
  // "error" del protocolo, en vez de dejar que expire como si el nodo fuera
  // inalcanzable.
  void onNodeCommandNack(const char *device, const char *reason);

private:
  // Tipo de comando AT en curso (los QMT* terminan con un aviso asincrono).
  enum class CmdKind : uint8_t { PLAIN, QMTOPEN, QMTCONN, QMTPUB, QMTSUB };
  enum class CmdOutcome : uint8_t { NONE, OK, ERR, TIMEOUT };
  // Pasos del estado CONFIG, en orden.
  enum class ConfigStep : uint8_t {
    ATE0,
    CMEE,
    CTZU,  // AT+CTZU=1 - actualizacion automatica de la hora de red, para que AT+CCLK? funcione (si se puede)
    QSCLK,
    CFUN_ON_1,
    CPIN,
    CFUN_OFF,
    QCGDEFCONT,
    CFUN_ON_2,
    DONE
  };
  // Pasos del estado MQTT_CONNECT, en orden.
  //
  // SUB_DEVICE/SUB_GATEWAY van despues de CONN para que la bajada de ordenes este
  // activa antes de IDLE. Las dos son "si se puede": una suscripcion fallida pierde
  // el control remoto pero no debe parar la telemetria, que es el trabajo real del
  // modulo.
  //
  // ANNOUNCE va el ultimo, tras las dos suscripciones, y no es opcional.
  //
  // ThingsBoard no encamina un RPC a un dispositivo hijo de una pasarela hasta que la
  // pasarela lo ha anunciado en v1/gateway/connect. La telemetria NO lo necesita -
  // publicar en v1/gateway/telemetry crea el dispositivo solo -, y por eso el olvido
  // era invisible: los nodos aparecian en ThingsBoard con lecturas en vivo mientras
  // todas las ordenes que se les mandaban morian en el servidor, antes de llegar al
  // modem.
  enum class MqttConnectSub : uint8_t {
    CLOSE_FIRST, KEEPALIVE_CFG, SESSION_CFG, OPEN, CONN, SUB_DEVICE, SUB_GATEWAY, ANNOUNCE
  };
  // Hasta donde retroceder al recuperarse de un fallo.
  enum class RecoveryLevel : uint8_t { PUBLISH, MQTT_CONNECT, ATTACH };

  // El comando AT en curso (solo hay uno a la vez).
  struct PendingCmd {
    bool active = false;
    CmdKind kind = CmdKind::PLAIN;
    uint32_t sentAtMs = 0;
    uint32_t timeoutMs = 0;
    bool sawOk = false;           // QMTOPEN/QMTCONN/QMTPUB: ya llego el OK inmediato, se espera el aviso asincrono
    bool awaitingPrompt = false;  // QMTPUB: esperando el '>' que pide los datos
    const uint8_t *sendPayload = nullptr;
    size_t sendPayloadLen = 0;
    char infoLine[96] = {0};      // la linea de respuesta con datos (p. ej. "+CSQ: 20,99")
    bool hasInfoLine = false;
    CmdOutcome outcome = CmdOutcome::NONE;
  };

  // ---------- El unico camino de los comandos ----------
  void issueCommand(const char *cmd, uint32_t timeoutMs, CmdKind kind = CmdKind::PLAIN,
                     const uint8_t *sendPayload = nullptr, size_t sendPayloadLen = 0);
  void pumpSerial();
  void handleLine(const char *line, size_t len);
  void handleUrc(const char *line, size_t len);
  void completeCommand(CmdOutcome outcome);
  void checkCommandTimeout();
  CmdOutcome consumeOutcome();

  // ---------- Maquina de estados ----------
  void setState(State s);
  void enterPowering();
  void enterError(const char *reason);
  void powerCycle();

  // La mitad electrica de un apagado y encendido: TX liberado, VIN y la puerta del
  // canal cortados, PWRKEY en reposo, y el temporizador de espera en apagado armado.
  // La comparten powerCycle() y el camino de reintento ERROR->OFF, que TIENE que
  // quitar la alimentacion en vez de solo volver a pulsar PWRKEY - ver tickError().
  void powerDown();
  void handleFailureAtLevel(RecoveryLevel level, const char *reason);
  void applyBackoffAndSetState(State s);
  uint32_t computeBackoff(uint32_t attempt) const;
  void setLastError(const char *msg);

  // ---------- Ordenes recibidas ----------
  // Reparte un mensaje MQTT entrante segun su topic: el topic de RPC propio de la
  // pasarela significa "reconfigurar Modulo B", y el topic de RPC de pasarela
  // significa "reenviar esto a un nodo hijo".
  void handleInboundMqtt(const NbiotProtocol::MqttMessage &msg);
  void handleOwnRpc(const NbiotProtocol::MqttMessage &msg);
  void handleGatewayRpc(const NbiotProtocol::MqttMessage &msg);
  // Aplica un objeto plano {"CLAVE":valor,...} a la configuracion en marcha de
  // Modulo B, escribiendo en applied lo realmente aplicado (tras recortar).
  bool applyOwnConfig(const char *params, char *applied, size_t cap);
  // Pone en cola una respuesta RPC para que la publique RPC_REPLY.
  void queueRpcReply(const char *topic, const char *payload);
  void tickNodeCommandTimeout();

  // Un tick por estado.
  void tickOff();
  void tickPowering();
  void tickWaitAt();
  void tickConfig();
  void tickAttaching();
  void tickMqttConnect();
  void tickIdle();
  void tickPublishing();
  void tickRpcReply();
  void tickError();

  void onPublishSucceeded();
  void onPublishFailed();

  // ---------- Anillo de lecturas ----------
  const SensorSnapshot &ringPeek(uint8_t i) const;
  void popSentReadings();
  // Construye el contenido de un AT+QMTPUB con hasta maxReadings de las lecturas mas
  // antiguas del anillo (agrupadas por nombre de nodo). Pone en _lastPublishCount
  // cuantas metio de verdad. Con hora de red emite {"ts":<ms>,"values":{...}} por
  // lectura; sin ella, la forma plana sin "ts".
  size_t buildGatewayPayload(uint8_t *out, size_t cap, uint8_t maxReadings);

  // "Ahora" en ms de epoca, a partir de la ultima sincronizacion con AT+CCLK? mas el
  // millis() transcurrido. Solo tiene sentido con _haveNetTime.
  int64_t netNowMs() const;

  // ---------- Configuracion fija ----------
  HardwareSerial &_serial;
  uint8_t _rxPin, _txPin, _enPin, _channelPin, _pwrkeyPin;
  uint32_t _baud;
  bool _uartStarted = false;

  // ---------- Motor de comandos ----------
  PendingCmd _cmd;
  // TIENE QUE CABER UNA LINEA +QMTRECV ENTERA, con su contenido.
  //
  // Valia 96 y rompia en silencio todas las ordenes. Una linea RPC de pasarela es
  //
  //   +QMTRECV: 0,0,"v1/devices/me/rpc/request/123",{...json...}
  //
  // y solo el JSON de la orden de umbral pasa de 150 caracteres, asi que la linea
  // era siempre mas larga que el buffer. Peor que cortarla: el camino de
  // desbordamiento vaciaba el buffer y seguia, asi que se descartaba el PRINCIPIO
  // del mensaje y el FINAL se pasaba al interprete como si fuera una linea completa.
  // parseQmtrecv fallaba con el, y los restos aparecian como
  //
  //   [nbiot-mqtt] urc: n":50,"umbral_alarm_off":55,"umbral_alarm_on":60}}"
  //
  // que parece ruido y es en realidad el final de una orden. Todas las conclusiones
  // de "no llega nada" durante varios dias salian de buscar el mensaje de exito,
  // que solo se imprime cuando la interpretacion funciona.
  //
  // Dimensionado para lo mas grande que puede llegar: topic (72) + contenido (320)
  // mas el envoltorio de +QMTRECV, redondeado hacia arriba.
  static const size_t kLineBufLen = 512;
  char _lineBuf[kLineBufLen];
  bool _lineOverflowed = false;
  uint8_t _lineLen = 0;
  bool _sawBootUrc = false;

  // ---------- Contabilidad de la maquina de estados ----------
  State _state = State::OFF;
  uint32_t _stateEnteredMs = 0;
  uint32_t _backoffUntilMs = 0;
  uint32_t _powerCycleSettleUntilMs = 0;
  uint32_t _consecutiveFailures = 0;
  char _lastError[64] = "none";

  bool _pwrkeyReleased = false;  // tiempos dentro de POWERING: pulsar a bajo y soltar a alto, sin bloquear
  uint32_t _lastAtPingMs = 0;

  ConfigStep _configStep = ConfigStep::ATE0;
  uint32_t _configStepEnteredMs = 0;  // deja a QCGDEFCONT esperar a que se asiente la SIM tras CFUN=0, sin bloquear

  bool _cgpaddrChecked = false;
  bool _csqCheckedAfterAttach = false;  // un AT+CSQ justo tras el registro - ver tickAttaching()
  bool _cclkChecked = false;            // un AT+CCLK? justo tras el de CSQ - ver tickAttaching()
  uint32_t _lastCeregPollMs = 0;
  uint8_t _attachRetries = 0;

  // ---------- Hora de red (marca de tiempo de cada lectura subida) ----------
  // Se lee una vez por registro con AT+CCLK? (necesita AT+CTZU=1). _netEpochMsAtSync
  // es la epoca en ms en el momento en que se tomo _netSyncLocalMs (un valor de
  // millis()); netNowMs() extrapola desde ahi. Se queda en "sin hora" si la red no
  // da nunca un reloj creible, y entonces la subida no lleva "ts".
  bool _haveNetTime = false;
  int64_t _netEpochMsAtSync = 0;
  uint32_t _netSyncLocalMs = 0;

  MqttConnectSub _mqttConnectSub = MqttConnectSub::KEEPALIVE_CFG;
  uint8_t _mqttConnectRetries = 0;
  uint8_t _announceIdx = 0;  // que nodo se esta anunciando en v1/gateway/connect
  bool _mqttConnected = false;
  uint32_t _mqttReconnects = 0;

  uint32_t _lastCsqPollMs = 0;  // refresco del RSSI para el panel mientras esta en IDLE

  bool _publishInFlight = false;
  uint8_t _publishRetries = 0;
  uint8_t _publishPayload[MQTT_PAYLOAD_MAX_BYTES];
  size_t _publishPayloadLen = 0;
  uint8_t _lastPublishCount = 0;

  // ---------- Cola de ordenes para los nodos ----------
  struct NodeCommand {
    bool pending = false;      // hueco ocupado
    bool awaitingAck = false;  // ya enviada por LoRa, esperando el ACK del nodo
    char device[24] = {0};     // nombre del dispositivo hijo en ThingsBoard, p. ej. "NodoC-1"
    char cfg[128] = {0};       // "CFG,INTERVAL=300,BMV080=0" tal como lo espera Modulo C
    char supKey[24] = {0};     // que AJUSTE toca, para que un comando sustituya al anterior
    long rpcId = 0;
    uint32_t queuedMs = 0;
    uint8_t attempts = 0;      // intentos de entrega hasta ahora - se informa, y se reintenta
    uint32_t lastSentMs = 0;   // cuando salio el ultimo intento, para el hueco entre reintentos

    // Que protocolo la pidio, y por tanto como se informa su RESULTADO.
    //
    //   false - el camino de RPC de pasarela: el RPC se deja abierto y se contesta con
    //           el resultado, asi que ThingsBoard espera hasta que el nodo responde o
    //           se agota el plazo.
    //   true  - el protocolo A->C acordado: el RPC ya se contesto "enviado" en cuanto
    //           llego, y el resultado sale despues como un registro de telemetria
    //           commandLog con el mismo requestId.
    //
    // La distincion tiene que ir en la propia orden: cuando llegan el ACK o el plazo
    // agotado, el mensaje original hace mucho que no existe.
    bool viaCommandLog = false;
  };
  // Seis: suficiente para todas las magnitudes de un nodo, o dos por nodo en tres
  // nodos. Una cola llena se INFORMA, nunca se tira en silencio.
  // La cadena CFG se construye aqui antes de elegir un hueco libre, para que una orden
  // que al final no se puede encolar no deje escrito a medias un hueco real.
  char _pendingCfgScratch[128] = {0};

  static const uint8_t kNodeCmdQueue = 6;
  NodeCommand _nodeCmds[kNodeCmdQueue];

  // Hueco de este dispositivo aun no enviado (o -1).
  int undeliveredIndexFor(const char *device) const {
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      if (_nodeCmds[i].pending && !_nodeCmds[i].awaitingAck &&
          strcmp(_nodeCmds[i].device, device) == 0) {
        return (int)i;
      }
    }
    return -1;
  }

  // UN COMANDO NUEVO SUSTITUYE AL ANTERIOR DEL MISMO NODO Y EL MISMO AJUSTE.
  //
  // Sin esto, seis ordenes identicas al mismo nodo ocupan los seis huecos y la
  // cola se bloquea, aunque la primera ya haga el trabajo. Pasa constantemente:
  // el operador repite el comando al no ver respuesta, y ademas ThingsBoard
  // guarda hasta una hora los RPC no entregados y se los descarga a Modulo B de
  // golpe al reconectar, llenando la cola en el arranque.
  //
  // El propio documento v2 ya establece el principio para NxT_enviodatos: "Manda
  // el ultimo recibido". Se aplica a todas las acciones, que es lo coherente -
  // si llegan dos umbrales distintos para la misma variable, el que vale es el
  // ultimo que mando el operador, no el primero que entro en la cola.
  //
  // Solo sustituye comandos que AUN NO SE HAN ENVIADO. Uno ya entregado y
  // esperando ACK se deja en paz: el nodo ya lo tiene, y su resultado hay que
  // reportarlo.
  int supersedableIndexFor(const char *device, const char *supKey) const {
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      const NodeCommand &nc = _nodeCmds[i];
      if (!nc.pending || nc.awaitingAck) continue;
      if (strcmp(nc.device, device) != 0) continue;
      if (strcmp(nc.supKey, supKey) != 0) continue;
      return (int)i;
    }
    return -1;
  }

  // QUE AJUSTE toca este comando, derivado del CFG que se va a mandar al nodo.
  //
  // Agrupar por `action` era un error: TODOS los umbrales comparten la accion
  // `umbrales_de1variable`, asi que configurar temp y despues rh en el mismo nodo
  // cancelaba el primero y lo reportaba como error. La cola esta dimensionada "lo
  // suficiente para cada magnitud de un nodo" precisamente para que eso no pase.
  //
  // Se deriva del CFG y no de la accion porque el CFG es lo que de verdad va a
  // cambiar en el nodo, y eso agrupa bien dos casos que por accion no coinciden:
  // `umbral` y `umbrales_de1variable` son alias, y `NxT_enviodatos` y
  // `NxTglobalenviodatos` producen el mismo CFG,NXT_ENVIO. Dos nombres para un
  // mismo ajuste no deben ocupar dos huecos.
  //
  //   "CFG,temp_pre_off=44,temp_pre_on=50"  ->  "temp"
  //   "CFG,NXT_ENVIO=0.5"                   ->  "NXT_ENVIO"
  //   "CFG,INTERVAL=300"                    ->  "INTERVAL"
  static void supersedeKeyFromCfg(const char *cfg, char *out, size_t cap) {
    out[0] = 0;
    const char *p = strchr(cfg, ',');       // saltar el "CFG"
    if (!p) return;
    p++;
    const char *eq = strchr(p, '=');
    if (!eq) return;

    size_t len = (size_t)(eq - p);
    if (len >= cap) len = cap - 1;
    memcpy(out, p, len);
    out[len] = 0;

    // Un umbral se nombra <magnitud><sufijo>. Recortar el sufijo deja la
    // magnitud, que es el ajuste real: dos ordenes sobre `temp` se sustituyen,
    // una sobre `temp` y otra sobre `rh` no.
    static const char *kSuf[] = {"_pre_off", "_pre_on", "_alarma_off", "_alarma_on",
                                 "_alarm_off", "_alarm_on"};
    for (size_t i = 0; i < sizeof(kSuf) / sizeof(kSuf[0]); i++) {
      const size_t sl = strlen(kSuf[i]);
      const size_t ol = strlen(out);
      if (ol > sl && strcmp(out + ol - sl, kSuf[i]) == 0) {
        out[ol - sl] = 0;
        return;
      }
    }
  }

  // Cuantos huecos quedan. Una accion global necesita uno POR NODO, y hay que
  // saberlo ANTES de encolar nada: encolar la mitad de un comando que va a todos
  // los nodos deja el sistema diciendo que lo aplico cuando no es cierto.
  uint8_t freeNodeCmdSlots() const {
    uint8_t n = 0;
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      if (!_nodeCmds[i].pending) n++;
    }
    return n;
  }

  // UN HUECO QUE A ESTE NODO HAY QUE MANDARLE AHORA - lo que incluye uno ya enviado
  // y aun sin confirmar.
  //
  // Que el nodo haga una subida es en si la prueba de que un intento anterior fallo:
  // si hubiera recibido y aplicado la orden, la habria confirmado dentro de esa
  // ventana y el hueco ya estaria libre. Asi que una orden sin confirmar se vuelve a
  // mandar en la siguiente ventana en vez de esperar al plazo.
  //
  // Es lo que siempre se documento que significaba el plazo de 15 minutos - "3
  // ventanas perdidas a 5 min" - salvo que nada reintentaba en esas ventanas. Una
  // trama perdida costaba los quince minutos enteros y luego informaba de un timeout
  // para una orden que se habia entregado a un nodo sano. Visto con el requestId 7,
  // 2026-10-05.
  //
  // La guarda del hueco entre reintentos es esencial: ver DOWNLINK_RETRY_MIN_GAP_MS.
  int deliverableIndexFor(const char *device) const {
    // UNA ORDEN POR NODO Y POR VENTANA.
    //
    // main.cpp ejecutaba este gancho con cada trama recibida, y un nodo manda varias
    // por ventana - telemetria, su ACK y el STAT. Mirar solo si ESTE hueco se habia
    // mandado hacia poco hacia que la segunda orden en cola saliera con la trama ACK
    // del nodo y la tercera con su STAT, cuando el nodo ya habia cerrado su ventana y
    // estaba transmitiendo. Las dos se perdian y habia que reintentarlas una ventana
    // despues.
    //
    // Visto con cinco ordenes encoladas a la vez el 2026-10-05: se vaciaba una por
    // ventana de todos modos, pero gastando dos tramas cada vez.
    //
    // Asi que si se mando ALGO a este nodo dentro del hueco, el nodo esta
    // transmitiendo y sordo: retener el resto.
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      const NodeCommand &nc = _nodeCmds[i];
      if (!nc.pending || strcmp(nc.device, device) != 0) continue;
      if (nc.awaitingAck && millis() - nc.lastSentMs < DOWNLINK_RETRY_MIN_GAP_MS) return -1;
    }

    // LA MAS ANTIGUA PRIMERO, para que las ordenes lleguen al nodo en el orden en que
    // las mando Modulo A. Los huecos se reparten por el primero libre, asi que el
    // orden de los huecos no es el de llegada - y cuando dos ordenes ponen el mismo
    // umbral a valores distintos, tiene que ganar la ultima que se mando. Se usa la
    // edad y no la marca en bruto, para que el desbordamiento de millis() compare
    // bien.
    int best = -1;
    uint32_t bestAge = 0;
    const uint32_t now = millis();
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      const NodeCommand &nc = _nodeCmds[i];
      if (!nc.pending || strcmp(nc.device, device) != 0) continue;
      const uint32_t age = now - nc.queuedMs;
      if (best < 0 || age > bestAge) {
        best = (int)i;
        bestAge = age;
      }
    }
    return best;
  }

  // Hueco de este dispositivo ya enviado y esperando ACK (o -1).
  int awaitingAckIndexFor(const char *device) const {
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      if (_nodeCmds[i].pending && _nodeCmds[i].awaitingAck &&
          strcmp(_nodeCmds[i].device, device) == 0) {
        return (int)i;
      }
    }
    return -1;
  }

  // Primer hueco libre (o -1).
  int freeNodeCmdSlot() const {
    for (uint8_t i = 0; i < kNodeCmdQueue; i++) {
      if (!_nodeCmds[i].pending) return (int)i;
    }
    return -1;
  }

  // ---------- Cola de respuestas a ThingsBoard ----------
  // UNA COLA, NO UN HUECO.
  //
  // Era un unico hueco que apuntaba "descartando la anterior" y tiraba un mensaje. No
  // habia pasado aun solo por suerte - el modem vaciaba cada respuesta antes de que
  // llegara la siguiente - pero que lleguen dos ordenes en el mismo tick es normal
  // (las peticiones 9 y 10 lo hicieron el 2026-10-05), y la descartada habria sido un
  // "enviado" que Modulo A estaba esperando.
  //
  // Ocho de fondo. Cada resultado produce ahora TRES publicaciones - commandLog en la
  // pasarela (lo que especifica el documento del protocolo), el mismo registro
  // copiado en el nodo, y los atributos del nodo - y pueden llegar dos ordenes en un
  // tick.
  static const uint8_t kRpcReplyQueue = 8;
  struct RpcReply {
    char topic[NbiotProtocol::kMqttTopicLen] = {0};
    char payload[512] = {0};  // la copia en el nodo lleva commandLog + un campo por umbral
  };
  RpcReply _rpcReplies[kRpcReplyQueue];
  uint8_t _rpcReplyHead = 0;   // indice de la siguiente que se manda
  uint8_t _rpcReplyCount = 0;

  bool rpcReplyPending() const { return _rpcReplyCount > 0; }
  void popRpcReply() {
    if (_rpcReplyCount == 0) return;
    _rpcReplyHead = (uint8_t)((_rpcReplyHead + 1) % kRpcReplyQueue);
    _rpcReplyCount--;
  }
  bool _rpcReplyInFlight = false;

  uint16_t _subMsgId = 1;  // id de mensaje de AT+QMTSUB, sube con cada suscripcion
  uint32_t _downlinksReceived = 0;

  // ---------- Anillo de telemetria pendiente de subir ----------
  SensorSnapshot _ring[NBIOT_RING_CAPACITY];
  uint8_t _ringHead = 0;
  uint8_t _ringCount = 0;
  uint32_t _oldestPendingMs = 0;

  // ---------- Lotes ----------
  // PUBLISHING se dispara cuando el anillo llega a _batchReadingsTarget lecturas O
  // pasan _batchSecondsTarget segundos desde la lectura pendiente mas antigua (o por
  // requestFlush()), y entonces vacia el anillo en publicaciones de
  // MQTT_PUB_CHUNK_READINGS lecturas. Los dos objetivos se pueden cambiar desde
  // Modulo A (setBatchSeconds / setBatchReadings).
  uint16_t _batchReadingsTarget = MQTT_BATCH_MAX_READINGS;
  uint32_t _batchSecondsTarget = MQTT_BATCH_SECONDS;

  // ---------- Observabilidad ----------
  bool _attached = false;
  uint32_t _attachedSinceMs = 0;
  int _rssiDbm = 0;
  uint32_t _packetsSent = 0;
  uint32_t _packetsFailed = 0;
  uint32_t _packetsDropped = 0;
};

ModemNBIoTMqtt::ModemNBIoTMqtt(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint8_t enPin,
                                uint8_t channelPin, uint8_t pwrkeyPin, uint32_t baud)
    : _serial(serial),
      _rxPin(rxPin),
      _txPin(txPin),
      _enPin(enPin),
      _channelPin(channelPin),
      _pwrkeyPin(pwrkeyPin),
      _baud(baud) {}

// Pines en su estado de reposo: modem sin alimentacion y canal cerrado.
void ModemNBIoTMqtt::begin() {
  pinMode(_enPin, OUTPUT);
  digitalWrite(_enPin, NBIOT_DISABLE);
  pinMode(_channelPin, OUTPUT);
  digitalWrite(_channelPin, !NBIOT_CHANNEL_ACTIVE);
  pinMode(_pwrkeyPin, OUTPUT);
  digitalWrite(_pwrkeyPin, !NBIOT_PWRKEY_ACTIVE);  // en reposo hasta que POWERING lo pulse
  setState(State::OFF);
}

// Una vuelta de trabajo: leer lo que haya llegado del modem, vigilar el plazo del
// comando en curso, avanzar el estado actual y vigilar los plazos de las ordenes a
// los nodos.
void ModemNBIoTMqtt::tick() {
  pumpSerial();
  checkCommandTimeout();

  switch (_state) {
    case State::OFF: tickOff(); break;
    case State::POWERING: tickPowering(); break;
    case State::WAIT_AT: tickWaitAt(); break;
    case State::CONFIG: tickConfig(); break;
    case State::ATTACHING: tickAttaching(); break;
    case State::MQTT_CONNECT: tickMqttConnect(); break;
    case State::IDLE: tickIdle(); break;
    case State::PUBLISHING: tickPublishing(); break;
    case State::RPC_REPLY: tickRpcReply(); break;
    case State::ERROR: tickError(); break;
  }

  tickNodeCommandTimeout();
}

// ---------- Configuracion en marcha, ajustable desde Modulo A ----------

uint32_t ModemNBIoTMqtt::setBatchSeconds(uint32_t seconds) {
  if (seconds < MQTT_BATCH_SECONDS_MIN) seconds = MQTT_BATCH_SECONDS_MIN;
  if (seconds > MQTT_BATCH_SECONDS_MAX) seconds = MQTT_BATCH_SECONDS_MAX;
  _batchSecondsTarget = seconds;
  Serial.printf("[nbiot-mqtt] uplink period set to %lus\n", (unsigned long)seconds);
  return seconds;
}

uint16_t ModemNBIoTMqtt::setBatchReadings(uint16_t readings) {
  if (readings < MQTT_BATCH_READINGS_MIN) readings = MQTT_BATCH_READINGS_MIN;
  if (readings > MQTT_BATCH_READINGS_MAX) readings = MQTT_BATCH_READINGS_MAX;
  _batchReadingsTarget = readings;
  Serial.printf("[nbiot-mqtt] uplink batch size set to %u readings\n", readings);
  return readings;
}

// ---------- Anillo de lecturas pendientes de subir ----------

// Mete una lectura al final del anillo. Si esta lleno, se descarta la mas antigua.
void ModemNBIoTMqtt::enqueue(const SensorSnapshot &snap) {
  if (_ringCount == 0) _oldestPendingMs = snap.lastHeardMs ? snap.lastHeardMs : millis();

  if (_ringCount >= NBIOT_RING_CAPACITY) {
    _ringHead = (uint8_t)((_ringHead + 1) % NBIOT_RING_CAPACITY);
    _ringCount--;
    _packetsDropped++;
    _oldestPendingMs = ringPeek(0).lastHeardMs;
  }

  uint8_t idx = (uint8_t)((_ringHead + _ringCount) % NBIOT_RING_CAPACITY);
  _ring[idx] = snap;
  _ringCount++;
}

// La lectura numero i contando desde la mas antigua.
const SensorSnapshot &ModemNBIoTMqtt::ringPeek(uint8_t i) const {
  return _ring[(_ringHead + i) % NBIOT_RING_CAPACITY];
}

// Quita del anillo las lecturas que acaba de publicar la ultima publicacion.
void ModemNBIoTMqtt::popSentReadings() {
  uint8_t n = _lastPublishCount;
  if (n > _ringCount) n = _ringCount;
  _ringHead = (uint8_t)((_ringHead + n) % NBIOT_RING_CAPACITY);
  _ringCount = (uint8_t)(_ringCount - n);
  if (_ringCount > 0) _oldestPendingMs = ringPeek(0).lastHeardMs;
}

// ---------- El unico camino de los comandos AT ----------

// Manda un comando AT y lo deja "en curso". No espera: la respuesta la recogen
// pumpSerial()/handleLine() en ticks posteriores, y el estado la consulta con
// consumeOutcome().
void ModemNBIoTMqtt::issueCommand(const char *cmd, uint32_t timeoutMs, CmdKind kind, const uint8_t *sendPayload,
                                   size_t sendPayloadLen) {
  _cmd = PendingCmd{};
  _cmd.active = true;
  _cmd.kind = kind;
  _cmd.sentAtMs = millis();
  _cmd.timeoutMs = timeoutMs;
  _cmd.sendPayload = sendPayload;
  _cmd.sendPayloadLen = sendPayloadLen;
  _cmd.awaitingPrompt = (kind == CmdKind::QMTPUB);
  Serial.print("[nbiot-mqtt] -> ");
  Serial.println(cmd);
  _serial.print(cmd);
  _serial.print("\r\n");
}

// Lee los bytes que haya mandado el modem y los junta en lineas.
void ModemNBIoTMqtt::pumpSerial() {
  while (_serial.available()) {
    char c = (char)_serial.read();

    // AT+QMTPUB contesta con un '>' suelto (sin fin de linea) pidiendo el JSON en
    // crudo; se atrapa aqui, antes de juntar lineas, porque no es una linea. El
    // contenido se termina con Ctrl+Z (0x1A), no con una cabecera de longitud, lo
    // que evita tener que escapar las comillas del JSON en la linea de comando.
    if (_cmd.active && _cmd.kind == CmdKind::QMTPUB && _cmd.awaitingPrompt && c == '>') {
      Serial.print("[nbiot-mqtt] payload (");
      Serial.print((unsigned)_cmd.sendPayloadLen);
      Serial.print(" bytes): ");
      Serial.write(_cmd.sendPayload, _cmd.sendPayloadLen);
      Serial.println();
      _serial.write(_cmd.sendPayload, _cmd.sendPayloadLen);
      _serial.write((uint8_t)0x1A);
      _cmd.awaitingPrompt = false;
      _lineLen = 0;
      continue;
    }

    if (c == '\n') {
      if (_lineOverflowed) {
        // Decirlo. El codigo anterior volvia a empezar el buffer a mitad de linea y
        // trataba el RESTO como una linea completa, lo que convertia un mensaje
        // demasiado largo en un fragmento con apariencia razonable y escondia el
        // fallo real.
        Serial.printf("[nbiot-mqtt] URC LONGER THAN %u BYTES - DISCARDED. A command "
                      "may have been lost; raise kLineBufLen\n",
                      (unsigned)kLineBufLen);
        _lineOverflowed = false;
      } else if (_lineLen > 0) {
        _lineBuf[_lineLen] = '\0';
        handleLine(_lineBuf, _lineLen);
      }
      _lineLen = 0;
    } else if (c != '\r') {
      if (_lineLen < sizeof(_lineBuf) - 1) {
        _lineBuf[_lineLen++] = c;
      } else {
        // Descartar el RESTO de esta linea en vez de empezar una nueva dentro de
        // ella. Una linea a medias nunca es una entrada valida, y fingir lo contrario
        // es lo que pasaba colas de JSON al interprete.
        _lineOverflowed = true;
      }
    }
  }
}

// Decide que es cada linea: un aviso espontaneo, o parte de la respuesta del
// comando en curso (OK, ERROR, el aviso asincrono de un QMT*, o una linea de datos).
void ModemNBIoTMqtt::handleLine(const char *line, size_t len) {
  // UNA ORDEN ENTRANTE ES ESPONTANEA POR DEFINICION, ASI QUE SE ATIENDE ANTES QUE
  // NADA.
  //
  // Antes estaba debajo del reparto por comando, asi que un +QMTRECV que llegaba con
  // un comando en curso solo se salvaba para los cuatro comandos cuyo resultado real
  // llega como aviso asincrono (QMTOPEN/QMTCONN/QMTPUB/QMTSUB). Con cualquier otro
  // comando - AT+CSQ, AT+QMTCFG, un QMTPUB que aun esperaba su OK - caia al final de
  // esta funcion y se guardaba como linea de datos del comando, es decir, se perdia
  // en silencio.
  //
  // Visto en vivo el 2026-10-05: una orden se entrego en el instante de suscribirse
  // (la sesion persistente funcionando como debe - el servidor la tenia en cola) y se
  // la trago el propio intercambio de AT+QMTSUB. Aparecia en el log como
  // "<- +QMTRECV: ..." en vez de "downlink on ...", que es la pista: el prefijo "<- "
  // significa que un comando se la quedo.
  //
  // El modem intercala avisos con respuestas de comandos cuando le parece, asi que no
  // hay ningun estado en el que ignorar uno sea correcto.
  if (strncmp(line, "+QMTRECV", 8) == 0) {
    handleUrc(line, len);
    return;
  }

  if (!_cmd.active) {
    handleUrc(line, len);
    return;
  }
  Serial.print("[nbiot-mqtt] <- ");
  Serial.println(line);

  if (strcmp(line, "OK") == 0) {
    if (_cmd.kind == CmdKind::QMTOPEN || _cmd.kind == CmdKind::QMTCONN || _cmd.kind == CmdKind::QMTPUB ||
        _cmd.kind == CmdKind::QMTSUB) {
      _cmd.sawOk = true;  // confirmacion recibida: el resultado real es el aviso asincrono de abajo
      return;
    }
    completeCommand(CmdOutcome::OK);
    return;
  }
  if (strncmp(line, "ERROR", 5) == 0 || strncmp(line, "+CME ERROR", 10) == 0 ||
      strncmp(line, "+CMS ERROR", 10) == 0) {
    completeCommand(CmdOutcome::ERR);
    return;
  }

  if (_cmd.kind == CmdKind::QMTOPEN && _cmd.sawOk) {
    int idx, result;
    if (NbiotProtocol::parseQmtopen(line, idx, result)) {
      completeCommand(result == 0 ? CmdOutcome::OK : CmdOutcome::ERR);
      return;
    }
    handleUrc(line, len);  // aviso ajeno intercalado: este comando sigue pendiente
    return;
  }
  if (_cmd.kind == CmdKind::QMTCONN && _cmd.sawOk) {
    int idx, result, retCode;
    if (NbiotProtocol::parseQmtconn(line, idx, result, retCode)) {
      completeCommand((result == 0 && retCode == 0) ? CmdOutcome::OK : CmdOutcome::ERR);
      return;
    }
    handleUrc(line, len);
    return;
  }
  if (_cmd.kind == CmdKind::QMTPUB && _cmd.sawOk) {
    int idx, msgId, result;
    if (NbiotProtocol::parseQmtpub(line, idx, msgId, result)) {
      completeCommand(result == 0 ? CmdOutcome::OK : CmdOutcome::ERR);
      return;
    }
    handleUrc(line, len);
    return;
  }
  if (_cmd.kind == CmdKind::QMTSUB && _cmd.sawOk) {
    int idx, msgId, result;
    if (NbiotProtocol::parseQmtsub(line, idx, msgId, result)) {
      completeCommand(result == 0 ? CmdOutcome::OK : CmdOutcome::ERR);
      return;
    }
    handleUrc(line, len);
    return;
  }

  // Cualquier otra linea con un comando simple en curso es su linea de datos (p. ej.
  // "+CPIN: READY" o "+CEREG: 1,5" antes del OK).
  strncpy(_cmd.infoLine, line, sizeof(_cmd.infoLine) - 1);
  _cmd.hasInfoLine = true;
}

// Avisos espontaneos del modem: sesion MQTT caida, orden entrante, arranque.
void ModemNBIoTMqtt::handleUrc(const char *line, size_t len) {
  (void)len;

  int idx, errCode;
  if (NbiotProtocol::parseQmtstat(line, idx, errCode)) {
    if (idx == MQTT_CLIENT_IDX) {
      _mqttConnected = false;
      Serial.printf("[nbiot-mqtt] +QMTSTAT fired (err=%d) - session marked dead, "
                    "will reconnect from IDLE\n",
                    errCode);
    }
    return;
  }

  // Mensaje MQTT entrante - el camino de las ordenes desde Modulo A. Llega como aviso
  // espontaneo en cualquier momento, incluso en mitad de la respuesta de otro
  // comando, y por eso se atiende aqui y no en una rama de comando.
  NbiotProtocol::MqttMessage inbound;
  if (NbiotProtocol::parseQmtrecv(line, inbound)) {
    _downlinksReceived++;
    Serial.printf("[nbiot-mqtt] downlink on \"%s\": %s\n", inbound.topic, inbound.payload);
    handleInboundMqtt(inbound);
    return;
  }

  if (strstr(line, "RDY") || strstr(line, "QNBIOTEVENT")) {
    _sawBootUrc = true;
    return;
  }

  Serial.printf("[nbiot-mqtt] urc: %s\n", line);
}

// ---------- Ordenes entrantes: Modulo A -> Modulo B (-> Modulo C) ----------

// Reparte una orden entrante segun su topic.
void ModemNBIoTMqtt::handleInboundMqtt(const NbiotProtocol::MqttMessage &msg) {
  // El topic de RPC propio de la pasarela lleva el id de la peticion en el propio
  // topic:  v1/devices/me/rpc/request/<id>
  if (strncmp(msg.topic, "v1/devices/me/rpc/request/", 26) == 0) {
    handleOwnRpc(msg);
    return;
  }
  if (strcmp(msg.topic, MQTT_TOPIC_GATEWAY_RPC) == 0) {
    handleGatewayRpc(msg);
    return;
  }
  Serial.printf("[nbiot-mqtt] downlink on an unhandled topic, ignored: %s\n", msg.topic);
}

// RPC dirigido al propio dispositivo pasarela = reconfigurar Modulo B, o la orden
// configurarModuloC del protocolo A->C.
//   {"method":"cfg","params":{"UPLINK":300,"BATCH":6}}
void ModemNBIoTMqtt::handleOwnRpc(const NbiotProtocol::MqttMessage &msg) {
  const char *idStr = msg.topic + 26;  // lo que va detras de "v1/devices/me/rpc/request/"

  char method[24] = {0};
  NbiotProtocol::jsonString(msg.payload, "method", method, sizeof(method));

  char topic[NbiotProtocol::kMqttTopicLen];
  snprintf(topic, sizeof(topic), "%s%s", MQTT_TOPIC_DEVICE_RPC_RESP, idStr);

  char params[160] = {0};
  char reply[224];

  // ---- configurarModuloC: el protocolo A -> C acordado ----
  //
  // Modulo A se dirige a la PASARELA y nombra el nodo de destino dentro de params, en
  // vez de usar el topic de RPC de pasarela. La respuesta se manda EN EL ACTO -
  // "enviado" - y el resultado real llega despues como un registro commandLog.
  //
  // Esa respuesta inmediata es la idea del diseno, no un atajo. Un nodo duerme entre
  // ventanas de subida y puede estar a minutos de oir nada; un RPC abierto tanto
  // tiempo expira en ThingsBoard e informa de un fallo para una orden que en realidad
  // se entrego y se aplico.
  if (strcmp(method, "configurarModuloC") == 0) {
    handleConfigurarModuloC(msg, idStr);
    return;
  }

  if (strcmp(method, "cfg") != 0) {
    snprintf(reply, sizeof(reply),
             "{\"error\":\"unknown method '%s', expected 'configurarModuloC' or 'cfg'\"}", method);
    queueRpcReply(topic, reply);
    return;
  }
  if (!NbiotProtocol::jsonObject(msg.payload, "params", params, sizeof(params))) {
    snprintf(reply, sizeof(reply), "{\"error\":\"missing params\"}");
    queueRpcReply(topic, reply);
    return;
  }

  char applied[128] = {0};
  if (!applyOwnConfig(params, applied, sizeof(applied))) {
    snprintf(reply, sizeof(reply), "{\"error\":\"no recognized keys (want UPLINK and/or BATCH)\"}");
    queueRpcReply(topic, reply);
    return;
  }

  // Devolver los valores tras el recorte, para que un recorte sea visible.
  snprintf(reply, sizeof(reply), "{\"applied\":{%s}}", applied);
  queueRpcReply(topic, reply);
}

// Aplica {"UPLINK":300,"BATCH":6} a la configuracion en marcha de este modulo:
// UPLINK = segundos entre subidas, BATCH = lecturas por lote.
bool ModemNBIoTMqtt::applyOwnConfig(const char *params, char *applied, size_t cap) {
  const char *cur = params;
  char key[24], value[24];
  size_t pos = 0;
  bool any = false;

  auto appendKv = [&](const char *k, unsigned long v) {
    int n = snprintf(applied + pos, cap - pos, "%s\"%s\":%lu", pos ? "," : "", k, v);
    if (n > 0 && (size_t)n < cap - pos) pos += (size_t)n;
  };

  while (NbiotProtocol::jsonNextPair(cur, key, sizeof(key), value, sizeof(value))) {
    long v = strtol(value, nullptr, 10);
    if (strcmp(key, "UPLINK") == 0) {
      appendKv("UPLINK", setBatchSeconds((uint32_t)(v < 0 ? 0 : v)));
      any = true;
    } else if (strcmp(key, "BATCH") == 0) {
      appendKv("BATCH", setBatchReadings((uint16_t)(v < 0 ? 0 : v)));
      any = true;
    } else {
      Serial.printf("[nbiot-mqtt] rpc: unrecognized own-config key \"%s\", ignored\n", key);
    }
  }
  applied[pos] = '\0';
  return any;
}

// RPC dirigido a un dispositivo hijo = reenviarlo a ese nodo por LoRa.
//   {"device":"NodoC-1","data":{"id":42,"method":"cfg","params":{"INTERVAL":300}}}
// Se guarda en la cola; main.cpp lo entrega en la siguiente ventana del nodo.
void ModemNBIoTMqtt::handleGatewayRpc(const NbiotProtocol::MqttMessage &msg) {
  char device[24] = {0};
  char data[224] = {0};
  if (!NbiotProtocol::jsonString(msg.payload, "device", device, sizeof(device)) ||
      !NbiotProtocol::jsonObject(msg.payload, "data", data, sizeof(data))) {
    Serial.println("[nbiot-mqtt] gateway rpc missing device/data, ignored");
    return;
  }

  long rpcId = 0;
  NbiotProtocol::jsonInt(data, "id", rpcId);

  char reply[224];
  auto replyErr = [&](const char *err) {
    snprintf(reply, sizeof(reply), "{\"device\":\"%s\",\"id\":%ld,\"data\":{\"error\":\"%s\"}}", device, rpcId,
             err);
    queueRpcReply(MQTT_TOPIC_GATEWAY_RPC, reply);
  };

  char method[24] = {0};
  NbiotProtocol::jsonString(data, "method", method, sizeof(method));

  // Se aceptan LOS DOS nombres de metodo en LOS DOS topics.
  //
  // Por que topic llega una orden depende solo de a que dispositivo de ThingsBoard
  // se dirija Modulo A: la pasarela CON-1 llega a v1/devices/me/rpc/request/<id>, un
  // dispositivo hijo llega aqui por v1/gateway/rpc. El documento del protocolo
  // especifica configurarModuloC y describe el camino de la pasarela, pero un panel
  // construido por nodo se dirigira naturalmente al nodo - y rechazarlo por el nombre
  // seria negarse por un tecnicismo cuando la intencion no tiene dudas.
  //
  // Los dos llevan lo mismo: un nodo de destino y unas claves que aplicar.
  if (strcmp(method, "cfg") != 0 && strcmp(method, "configurarModuloC") != 0) {
    replyErr("unknown method, expected 'configurarModuloC' or 'cfg'");
    return;
  }

  char params[160] = {0};
  if (!NbiotProtocol::jsonObject(data, "params", params, sizeof(params))) {
    replyErr("missing params");
    return;
  }

  const int slot = freeNodeCmdSlot();
  if (slot < 0) {
    replyErr("command queue full");
    return;
  }

  // Traducir el objeto params a la cadena "CFG,K=V,..." que ya entiende
  // applyConfigCommand() de Modulo C. Las claves no se validan aqui: Modulo C ignora
  // lo que no reconoce y solo confirma lo que aplico, asi que el nodo sigue siendo
  // la autoridad sobre su propio vocabulario de configuracion.
  const char *cur = params;
  char key[24], value[24];
  size_t pos = 0;
  pos += (size_t)snprintf(_pendingCfgScratch, sizeof(_pendingCfgScratch), "CFG");
  bool any = false;
  while (NbiotProtocol::jsonNextPair(cur, key, sizeof(key), value, sizeof(value))) {
    int n = snprintf(_pendingCfgScratch + pos, sizeof(_pendingCfgScratch) - pos, ",%s=%s", key, value);
    if (n > 0 && (size_t)n < sizeof(_pendingCfgScratch) - pos) {
      pos += (size_t)n;
      any = true;
    }
  }
  if (!any) {
    _pendingCfgScratch[0] = '\0';
    replyErr("params object was empty");
    return;
  }

  NodeCommand &nc = _nodeCmds[slot];
  strncpy(nc.cfg, _pendingCfgScratch, sizeof(nc.cfg) - 1);
  nc.cfg[sizeof(nc.cfg) - 1] = '\0';
  strncpy(nc.device, device, sizeof(nc.device) - 1);
  nc.device[sizeof(nc.device) - 1] = '\0';
  nc.rpcId = rpcId;
  nc.queuedMs = millis();
  nc.awaitingAck = false;
  nc.viaCommandLog = false;
  nc.pending = true;

  Serial.printf("[nbiot-mqtt] queued for %s: \"%s\" (waiting for its next uplink window)\n", nc.device,
                nc.cfg);
}

// main.cpp acaba de mandar la orden de este nodo por LoRa: queda esperando su ACK.
void ModemNBIoTMqtt::onNodeCommandDelivered(const char *device) {
  const int i = deliverableIndexFor(device);
  if (i < 0) return;
  NodeCommand &nc = _nodeCmds[i];
  nc.awaitingAck = true;
  nc.lastSentMs = millis();
  if (nc.attempts < 255) nc.attempts++;
  if (nc.attempts == 1) {
    Serial.printf("[nbiot-mqtt] sent to %s over LoRa, awaiting ACK\n", nc.device);
  } else {
    Serial.printf("[nbiot-mqtt] RESENT to %s over LoRa (attempt %u - no ACK to the "
                  "previous one), awaiting ACK\n", nc.device, (unsigned)nc.attempts);
  }
}

// El nodo contesto NACK: entendio la orden pero no aplico nada. Se informa como
// "error" y se libera el hueco.
void ModemNBIoTMqtt::onNodeCommandNack(const char *device, const char *reason) {
  int i = awaitingAckIndexFor(device);
  if (i < 0) i = undeliveredIndexFor(device);
  if (i < 0) return;
  NodeCommand &nc = _nodeCmds[i];

  if (nc.viaCommandLog) {
    publishCommandLogApplied(nc.rpcId, "error", reason, nc.device);
  } else {
    char reply[224];
    snprintf(reply, sizeof(reply), "{\"device\":\"%s\",\"id\":%ld,\"data\":{\"error\":\"%s\"}}",
             nc.device, nc.rpcId, reason);
    queueRpcReply(MQTT_TOPIC_GATEWAY_RPC, reply);
  }
  Serial.printf("[nbiot-mqtt] %s REJECTED the command: %s\n", nc.device, reason);
  nc = NodeCommand{};
}

// El nodo contesto ACK: se informa de lo que aplico y se libera el hueco.
void ModemNBIoTMqtt::onNodeCommandAck(const char *device, const char *ackPayload) {
  // Emparejar el ACK con la orden que lo espera DE ESE NODO. Con varias ordenes en
  // curso a nodos distintos, liberar la equivocada daria por confirmado el requestId
  // equivocado: una orden pareceria haber funcionado mientras otra expiraba en
  // silencio.
  int i = awaitingAckIndexFor(device);
  if (i < 0) i = undeliveredIndexFor(device);  // el ACK llego antes de apuntar la entrega
  if (i < 0) return;
  NodeCommand &_nodeCmd = _nodeCmds[i];

  // "ACK,INTERVAL=300,BMV080=0" -> devolver los valores aplicados (tras el recorte)
  // como resultado del RPC, para que un recorte en el nodo se vea en ThingsBoard.
  const char *body = strncmp(ackPayload, "ACK,", 4) == 0 ? ackPayload + 4 : ackPayload;
  if (_nodeCmd.viaCommandLog) {
    // Lleva QUE se aplico, no solo que se aplico algo. El nodo solo confirma las
    // claves que de verdad tomo, asi que es la diferencia entre que Modulo A crea que
    // un umbral cambio y poder mostrar el valor que el nodo usa realmente.
    publishCommandLogApplied(_nodeCmd.rpcId, "confirmado", body, _nodeCmd.device);
  } else {
    char reply[224];
    snprintf(reply, sizeof(reply), "{\"device\":\"%s\",\"id\":%ld,\"data\":{\"applied\":\"%s\"}}",
             _nodeCmd.device, _nodeCmd.rpcId, body);
    queueRpcReply(MQTT_TOPIC_GATEWAY_RPC, reply);
  }

  Serial.printf("[nbiot-mqtt] %s acked: %s\n", _nodeCmd.device, body);
  _nodeCmd = NodeCommand{};
}

// Ordenes que llevan demasiado tiempo en cola (DOWNLINK_QUEUE_TIMEOUT_MS) sin
// confirmacion: se informan como timeout y se liberan.
void ModemNBIoTMqtt::tickNodeCommandTimeout() {
  // Cada hueco se cronometra por separado: un nodo inalcanzable no debe mantener viva
  // la orden de otro nodo mas alla de su propio plazo.
  for (uint8_t qi = 0; qi < kNodeCmdQueue; qi++) {
    NodeCommand &_nodeCmd = _nodeCmds[qi];
    if (!_nodeCmd.pending) continue;
    if (millis() - _nodeCmd.queuedMs < DOWNLINK_QUEUE_TIMEOUT_MS) continue;

  // Se construye una vez, antes de la bifurcacion, porque la linea de log del final
  // tambien lo usa - llegue la orden por el protocolo que llegue.
  char why[72];
  if (_nodeCmd.awaitingAck) {
    snprintf(why, sizeof(why), "delivered %u time(s), no ACK", (unsigned)_nodeCmd.attempts);
  } else {
    snprintf(why, sizeof(why), "node never uplinked - not delivered");
  }

  if (_nodeCmd.viaCommandLog) {
    // DECIR QUE MITAD FALLO. Antes se publicaba un "timeout" a secas, tirando el unico
    // dato que necesita el operador: si la orden llego a salir al aire. Modulo B ya lo
    // sabe - awaitingAck significa que el nodo hizo su subida, la orden salio en su
    // ventana de escucha y luego no dijo nada; sin el, el nodo no transmitio nunca.
    //
    // Piden respuestas opuestas. "Entregada, sin ACK" es un nodo vivo que rechaza o
    // se cuelga - mirar el nodo. "Nunca entregada" es un nodo que no transmite -
    // mirar la radio, la alimentacion o la ubicacion. Una sola palabra para las dos
    // te manda a la mitad equivocada, y esto es una torre en un bosque, no una placa
    // en una mesa.
    //
    // Va en `detail`, que el documento del protocolo deja libre, asi que los cuatro
    // valores de status documentados no cambian.
    publishCommandLogApplied(_nodeCmd.rpcId, "timeout", why, _nodeCmd.device);
  } else {
    char reply[224];
    snprintf(reply, sizeof(reply), "{\"device\":\"%s\",\"id\":%ld,\"data\":{\"error\":\"%s\"}}", _nodeCmd.device,
             _nodeCmd.rpcId,
             _nodeCmd.awaitingAck ? "delivered but the node never acked" : "node never uplinked - not delivered");
    queueRpcReply(MQTT_TOPIC_GATEWAY_RPC, reply);
  }
    Serial.printf("[nbiot-mqtt] command for %s timed out (%s)\n", _nodeCmd.device, why);
    _nodeCmd = NodeCommand{};
  }
}

static size_t appendI64(char *out, int64_t v);  // definida mas abajo, junto al constructor del JSON

// ---- Resultado de una orden (commandLog / seguimientoCmd) ----
//
// La mitad del resultado del protocolo acordado. El RPC en si se contesto "enviado"
// al llegar; esto es lo que dice de verdad si el nodo hizo lo que se le pidio.
//
// Se publica como TELEMETRIA, no como respuesta RPC, porque el RPC se cerro hace
// minutos - un nodo puede estar a toda una ventana de subida de oir una orden, y
// mantener un RPC abierto tanto tiempo es lo que hacia parecer fallos a ordenes que
// solo eran lentas.
//
// appendKvAsJson: convierte el cuerpo del ACK del nodo - "temp_pre_off=48,temp_pre_on=50"
// - en campos JSON:  ,"temp_pre_off":48,"temp_pre_on":50
//
// POR QUE COMO CLAVES SUELTAS Y NO SOLO EL TEXTO: Modulo A necesita saber el umbral
// que el nodo usa ahora, y `detail` solo lo lleva como texto que habria que
// interpretar. Como claves de telemetria cada uno es un valor que un panel puede
// enlazar, graficar y comparar con lo que pidio. Los nombres son el vocabulario del
// propio nodo - magnitud mas sufijo, los nombres de la hoja de umbrales - asi que aqui
// no se inventa nada.
//
// Los numeros salen sin comillas para que se guarden como numeros; lo que no se pueda
// leer limpiamente como numero va entre comillas en vez de emitir JSON roto. Estas
// parejas llegan por radio, asi que no se dan por bien formadas.
static size_t appendKvAsJson(char *out, size_t cap, const char *kv) {
  size_t pos = 0;
  const char *p = kv;
  while (*p && pos + 8 < cap) {
    const char *comma = strchr(p, ',');
    const char *end = comma ? comma : p + strlen(p);
    const char *eq = nullptr;
    for (const char *q = p; q < end; q++) {
      if (*q == '=') { eq = q; break; }
    }
    if (eq && eq > p && (end - eq) > 1) {
      const size_t klen = (size_t)(eq - p);
      const size_t vlen = (size_t)(end - eq - 1);

      // Es numerico? Se mira caracter a caracter en vez de fiarse del resto de strtod.
      bool numeric = vlen > 0;
      bool dot = false;
      for (size_t i = 0; i < vlen && numeric; i++) {
        const char c = eq[1 + i];
        if (c == '-' && i == 0) continue;
        if (c == '.' && !dot) { dot = true; continue; }
        if (c < '0' || c > '9') numeric = false;
      }

      const size_t need = klen + vlen + (numeric ? 4 : 6);
      if (pos + need >= cap) break;
      out[pos++] = ',';
      out[pos++] = '"';
      memcpy(out + pos, p, klen); pos += klen;
      out[pos++] = '"';
      out[pos++] = ':';
      if (!numeric) out[pos++] = '"';
      memcpy(out + pos, eq + 1, vlen); pos += vlen;
      if (!numeric) out[pos++] = '"';
      out[pos] = 0;
    }
    if (!comma) break;
    p = comma + 1;
  }
  return pos;
}

void ModemNBIoTMqtt::publishCommandLog(long requestId, const char *status) {
  publishCommandLogApplied(requestId, status, nullptr, nullptr);
}

// Publica el resultado de una orden en tres sitios: la telemetria de la pasarela, la
// telemetria del nodo y los atributos del nodo.
void ModemNBIoTMqtt::publishCommandLogApplied(long requestId, const char *status,
                                              const char *detail, const char *device) {
  // EL OBJETO SE CONSTRUYE UNA VEZ Y LOS TRES ENVIOS SE COMPONEN DE EL.
  //
  // Antes el espejo y los atributos reutilizaban `payload` entero recortandole
  // llaves. Mientras `payload` fue plano colo; en cuanto la v2 del protocolo le
  // metio su propio {"ts":...,"values":{...}} salieron las dos cosas rotas a la
  // vez, y asi se vieron en el log del 2026-10-07:
  //
  //   espejo:    {"NodoC-3":[{"ts":N,"values":{"ts":N,"values":{...}}}]}   <- duplicado
  //   atributos: {"NodoC-3":{"seguimientoCmd":{...}},"temp_pre_off":20,...}}  <- descuadrado
  //
  // Contar llaves a mano sobre una cadena que otro sitio construye es fragil por
  // naturaleza: el dia que cambia el formato, rompe en silencio y lejos de aqui.
  // Con una sola fuente no hay nada que recortar.
  //
  // appendI64() y no %lld: nano newlib no lo trae de forma fiable.

  // ---- el objeto, sin llaves exteriores: "seguimientoCmd":{...} ----
  char inner[224];
  size_t ip = (size_t)snprintf(inner, sizeof(inner),
                               "\"seguimientoCmd\":{\"requestId\":%ld", requestId);

  // moduloC_id es OBLIGATORIO en la v2: el requestId lo genera MODULO-A por cada
  // conexion, empieza en 0 y se reinicia al reconectar, asi que por si solo no
  // identifica un comando.
  if (device && *device && ip + 32 < sizeof(inner)) {
    ip += (size_t)snprintf(inner + ip, sizeof(inner) - ip, ",\"moduloC_id\":\"%s\"", device);
  }
  ip += (size_t)snprintf(inner + ip, sizeof(inner) - ip, ",\"status\":\"%s\"", status);

  // `detail` no esta en el documento: lleva las claves que el nodo aplico de
  // verdad, o el motivo del rechazo. El documento acota los valores de status, no
  // prohibe campos extra. Las comillas y barras se descartan porque un paquete
  // malformado no puede poder emitir telemetria invalida.
  if (detail && *detail && ip + 24 < sizeof(inner)) {
    ip += (size_t)snprintf(inner + ip, sizeof(inner) - ip, ",\"detail\":\"");
    for (const char *p = detail; *p && ip < sizeof(inner) - 8; p++) {
      if (*p == '"' || *p == '\\' || (unsigned char)*p < 0x20) continue;
      inner[ip++] = *p;
    }
    if (ip < sizeof(inner) - 4) inner[ip++] = '"';
  }
  if (ip < sizeof(inner) - 2) inner[ip++] = '}';
  inner[ip] = 0;

  // ---- los umbrales aplicados, como campos sueltos ----
  // Solo una confirmacion lleva valores aplicados: el detail de un timeout es
  // prosa y el de un error un motivo de rechazo, no umbrales.
  char extra[200];
  extra[0] = 0;
  if (detail && *detail && strcmp(status, "confirmado") == 0 && strchr(detail, '=')) {
    appendKvAsJson(extra, sizeof(extra), detail);
  }

  const bool stamped = _haveNetTime;
  const int64_t nowMs = stamped ? netNowMs() : 0;

  // ---- 1. telemetria del gateway: la forma exacta que exige la v2 ----
  // Sin reloj de red se manda plano y ThingsBoard sella al recibir, que es mejor
  // que emitir un ts de 1970: ordenaria el registro como el mas antiguo de todos.
  char payload[288];
  size_t pos = 0;
  if (stamped) {
    pos += (size_t)snprintf(payload, sizeof(payload), "{\"ts\":");
    pos += appendI64(payload + pos, nowMs);
    pos += (size_t)snprintf(payload + pos, sizeof(payload) - pos, ",\"values\":{%s}}", inner);
  } else {
    snprintf(payload, sizeof(payload), "{%s}", inner);
  }
  queueRpcReply(MQTT_TOPIC_DEVICE_TELEMETRY, payload);

  if (device && *device) {
    // ---- 2. el mismo registro en la telemetria del propio nodo ----
    // v1/devices/me/telemetry lleva el token del gateway, asi que seguimientoCmd
    // aterriza en el dispositivo GATEWAY y nunca en el nodo. Esto lo pone tambien
    // bajo el nodo, con los umbrales como claves propias para que se puedan
    // graficar.
    char mirror[560];
    if (stamped) {
      size_t mp = (size_t)snprintf(mirror, sizeof(mirror), "{\"%s\":[{\"ts\":", device);
      mp += appendI64(mirror + mp, nowMs);
      snprintf(mirror + mp, sizeof(mirror) - mp, ",\"values\":{%s%s}}]}", inner, extra);
    } else {
      snprintf(mirror, sizeof(mirror), "{\"%s\":[{%s%s}]}", device, inner, extra);
    }
    queueRpcReply("v1/gateway/telemetry", mirror);

    // ---- 3. atributos del nodo ----
    // La seccion 5 del documento dice que el dashboard lee ATRIBUTOS y que
    // MODULO-A los copia desde la telemetria. Escribirlos aqui quita esa
    // dependencia. Son atributos de CLIENTE: por MQTT un dispositivo no puede
    // escribir los de servidor.
    char attrs[560];
    snprintf(attrs, sizeof(attrs), "{\"%s\":{%s%s}}", device, inner, extra);
    queueRpcReply("v1/gateway/attributes", attrs);
  }

  Serial.printf("[nbiot-mqtt] seguimientoCmd %ld -> %s%s\n", requestId, status,
                (device && *device) ? " (gateway + nodo + atributos)" : "");
}

// ---- configurarModuloC: el protocolo A -> C ----
//   {"method":"configurarModuloC",
//    "params":{"moduloC_id":"NodoC-1","action":"frecuencia","value":"freq_alarma"}}
//
// Contesta el RPC en el acto con "enviado" y deja la orden en cola para la siguiente
// ventana de subida del nodo. El resultado llega despues como commandLog.
//
// Acciones: umbrales_de1variable (alias umbral), NxT_lectura, NxT_enviodatos,
// NxTglobalenviodatos (a todos los nodos), y frecuencia (antigua, aun aceptada).
void ModemNBIoTMqtt::handleConfigurarModuloC(const NbiotProtocol::MqttMessage &msg,
                                             const char *idStr) {
  const long requestId = strtol(idStr, nullptr, 10);

  char respTopic[NbiotProtocol::kMqttTopicLen];
  snprintf(respTopic, sizeof(respTopic), "%s%s", MQTT_TOPIC_DEVICE_RPC_RESP, idStr);

  char params[224] = {0};
  char device[24] = {0};
  char action[24] = {0};
  char value[32] = {0};
  char reply[224];

  auto refuse = [&](const char *err) {
    // Un rechazo se contesta en el propio RPC, no como commandLog: no se llego a
    // encolar nada, asi que no habra resultado que informar despues.
    snprintf(reply, sizeof(reply), "{\"status\":\"error\",\"error\":\"%s\"}", err);
    queueRpcReply(respTopic, reply);
    Serial.printf("[nbiot-mqtt] configurarModuloC %ld rechazado: %s\n", requestId, err);
  };

  if (!NbiotProtocol::jsonObject(msg.payload, "params", params, sizeof(params))) {
    refuse("missing params");
    return;
  }
  // moduloC_id NO es obligatorio para todas las acciones: NxTglobalenviodatos va
  // dirigido a todos los MODULO-C del MODULO-B y por eso no lo lleva. Se exige
  // mas abajo, una vez se sabe que accion es.
  NbiotProtocol::jsonString(params, "moduloC_id", device, sizeof(device));
  NbiotProtocol::jsonString(params, "action", action, sizeof(action));

  // jsonScalar y no jsonString: la v2 manda la N como NUMERO JSON (1.5), sin
  // comillas, y jsonString() exige comillas. Acepta las dos formas.
  NbiotProtocol::jsonScalar(params, "value", value, sizeof(value));

  const bool isGlobal = (strcmp(action, "NxTglobalenviodatos") == 0);
  if (!isGlobal && !device[0]) {
    refuse("missing moduloC_id");
    return;
  }

  // Una accion global necesita un hueco POR NODO, no uno.
  const uint8_t nodeCount = (uint8_t)(sizeof(NBIOT_NODE_NAMES) / sizeof(NBIOT_NODE_NAMES[0]));
  const uint8_t slotsNeeded = isGlobal ? nodeCount : 1;

  // La comprobacion de huecos libres NO puede ir aqui: necesita saber QUE ajuste
  // toca el comando, y eso sale del CFG, que todavia no se ha construido. Va justo
  // antes de encolar.

  // ---- accion -> el vocabulario CFG que ya entiende Modulo C ----
  char cfg[128] = {0};
  if (strcmp(action, "frecuencia") == 0) {
    // Los tres ritmos con nombre. ESTOS SEGUNDOS SON UNA ELECCION, NO DE LA
    // ESPECIFICACION, que nombra los ajustes sin dar valores: son la traduccion
    // razonable a CFG,INTERVAL, y lo evidente es revisarlos cuando se acuerden los
    // numeros reales. freq_alarma es el suelo LORA_TX_PERIOD_MIN_MS.
    // (OJO: CFG,INTERVAL ajusta un periodo heredado; en el nodo actual el envio lo
    // marca el calendario, ver LORA_TX_PERIOD_MS en el firmware de Modulo C.)
    unsigned secs = 0;
    if (strcmp(value, "freq_normal") == 0) secs = 300;
    else if (strcmp(value, "freq_prealarma") == 0) secs = 60;
    else if (strcmp(value, "freq_alarma") == 0) secs = 15;
    if (secs == 0) {
      refuse("value must be freq_normal, freq_prealarma or freq_alarma");
      return;
    }
    snprintf(cfg, sizeof(cfg), "CFG,INTERVAL=%u", secs);

  } else if (strcmp(action, "umbral") == 0 || strcmp(action, "umbrales_de1variable") == 0) {
    // LA MAGNITUD Y EL UMBRAL LLEGAN EN CAMPOS SEPARADOS, y hay que juntarlos antes
    // de que el nodo reconozca nada. Visto en vivo el 2026-10-05, y no es lo que
    // muestra el documento del protocolo:
    //
    //   {"action":"umbrales_de1variable","variable":"temp",
    //    "umbral_pre_off":48,"umbral_pre_on":50,
    //    "umbral_alarm_off":55,"umbral_alarm_on":60}
    //
    // El nodo nombra un umbral como <magnitud><sufijo> - temp_pre_on - asi que una
    // clave "umbral_pre_on" no coincide con ninguna regla y se ignora en silencio.
    // Todos los umbrales de la orden se perderian y el nodo los rechazaria con NACK,
    // lo que parece un fallo de firmware y es en realidad un desacuerdo de nombres.
    //
    // Asi que "umbral_" se reescribe como "<variable>_". Una clave que ya lleva su
    // magnitud (temp_pre_on, como en la propia tabla del panel) pasa sin tocar - las
    // dos formas funcionan.
    char variable[24] = {0};
    NbiotProtocol::jsonString(params, "variable", variable, sizeof(variable));

    const char *cur = params;
    char k[40], v[32];
    size_t pos = (size_t)snprintf(cfg, sizeof(cfg), "CFG");
    bool any = false;
    while (NbiotProtocol::jsonNextPair(cur, k, sizeof(k), v, sizeof(v))) {
      if (strcmp(k, "moduloC_id") == 0 || strcmp(k, "action") == 0 ||
          strcmp(k, "variable") == 0 || strcmp(k, "value") == 0) {
        continue;  // son de enrutado, no umbrales
      }

      int n;
      if (strncmp(k, "umbral_", 7) == 0 && variable[0]) {
        n = snprintf(cfg + pos, sizeof(cfg) - pos, ",%s_%s=%s", variable, k + 7, v);
      } else {
        n = snprintf(cfg + pos, sizeof(cfg) - pos, ",%s=%s", k, v);
      }
      if (n > 0 && (size_t)n < sizeof(cfg) - pos) {
        pos += (size_t)n;
        any = true;
      }
    }
    if (!any) {
      refuse("umbral with no threshold keys");
      return;
    }
    // (strncmp con longitud 0 siempre da 0: la condicion equivale a !variable[0].)
    if (strncmp(params, "", 0) == 0 && !variable[0]) {
      // No es fatal - las claves pueden llevar ya su magnitud - pero conviene
      // decirlo, porque si no la llevan el nodo las rechazara todas.
      Serial.println("[nbiot-mqtt] umbral without a \"variable\" field - keys must "
                     "already name their magnitude");
    }

  } else if (strcmp(action, "NxT_lectura") == 0) {
    // "Cada sensor se lee cada N x su periodo minimo de lectura. N >= 1."
    // La N viaja en crudo: multiplica el NODO, que es quien conoce sus periodos
    // base y quien puede validar y confirmar lo que de verdad aplico.
    if (!value[0]) {
      refuse("NxT_lectura sin value");
      return;
    }
    snprintf(cfg, sizeof(cfg), "CFG,NXT_LECTURA=%s", value);

  } else if (strcmp(action, "NxT_enviodatos") == 0 || isGlobal) {
    // Mismo comando para el nodo; lo unico que cambia es a cuantos va.
    // NxTglobalenviodatos "sustituye al antiguo action: frecuencia, que deja de
    // usarse" - de ahi que `frecuencia` siga aceptandose pero ya no se documente.
    if (!value[0]) {
      refuse(isGlobal ? "NxTglobalenviodatos sin value" : "NxT_enviodatos sin value");
      return;
    }
    snprintf(cfg, sizeof(cfg), "CFG,NXT_ENVIO=%s", value);

  } else {
    refuse("unknown action, expected 'umbrales_de1variable', 'NxT_lectura', "
           "'NxT_enviodatos' or 'NxTglobalenviodatos'");
    return;
  }

  // Ya se sabe que CFG se va a mandar, asi que ya se puede saber que ajuste toca
  // y cuantos huecos hacen falta de verdad.
  char supKey[24];
  supersedeKeyFromCfg(cfg, supKey, sizeof(supKey));

  // Los huecos que este comando puede REUTILIZAR por sustituir a otro del mismo
  // ajuste no cuentan como ocupados: si no, ordenes repetidas seguirian llenando
  // la cola aunque solo una vaya a sobrevivir.
  uint8_t reusables = 0;
  for (uint8_t t = 0; t < slotsNeeded; t++) {
    const char *target = isGlobal ? NBIOT_NODE_NAMES[t].name : device;
    if (supersedableIndexFor(target, supKey) >= 0) reusables++;
  }
  if (freeNodeCmdSlots() + reusables < slotsNeeded) {
    refuse("command queue full");
    return;
  }

  // UNA ENTRADA POR NODO DESTINO, TODAS CON EL MISMO requestId.
  //
  // El documento lo pide asi para la accion global: "Una sola respuesta
  // inmediata. Un seguimientoCmd por cada MODULO-C, cada uno en su propia
  // publicacion y todos con el mismo requestId." Como cada hueco reporta su
  // propio resultado, eso sale solo: basta con encolar uno por nodo.
  //
  // Y sirve para algo mas que la simetria - cada nodo puede responder distinto.
  // El ejemplo del propio documento lo ensena: NodoC-1 confirmado y NodoC-2
  // timeout, bajo el mismo numero de peticion.
  for (uint8_t t = 0; t < slotsNeeded; t++) {
    const char *target = isGlobal ? NBIOT_NODE_NAMES[t].name : device;

    // Sustituir antes que encolar. El comando anterior se cierra como `error`
    // diciendo por que: el documento acota status a confirmado/error/timeout, y
    // dejarlo sin respuesta seria peor - Modulo A lo mostraria "en transito"
    // para siempre, que es exactamente el sintoma que estamos arreglando.
    int slot = supersedableIndexFor(target, supKey);
    if (slot >= 0) {
      const long viejo = _nodeCmds[slot].rpcId;
      if (viejo != requestId) {
        char motivo[72];
        snprintf(motivo, sizeof(motivo), "sustituido por el comando %ld", requestId);
        // SIN `device`, a proposito: eso publica solo el registro del gateway en
        // vez de tres mensajes. Un comando sustituido no llego a aplicar nada, asi
        // que el espejo en el nodo y los atributos no tienen valores que llevar.
        //
        // Y evita desbordar el anillo de respuestas: una sustitucion global son 3
        // nodos, que a tres mensajes cada uno serian 9 mas el `enviado` = 10 en un
        // anillo de 8, y el mas viejo que se tira puede ser el `enviado` de otro
        // comando - justo el fallo del que avisa el comentario del anillo.
        publishCommandLogApplied(viejo, "error", motivo, nullptr);
        Serial.printf("[nbiot-mqtt] %ld sustituye al %ld en %s (misma accion)\n",
                      requestId, viejo, target);
      }
    } else {
      slot = freeNodeCmdSlot();
    }
    if (slot < 0) break;  // comprobado arriba; defensivo

    NodeCommand &nc = _nodeCmds[slot];
    strncpy(nc.device, target, sizeof(nc.device) - 1);
    nc.device[sizeof(nc.device) - 1] = 0;
    strncpy(nc.cfg, cfg, sizeof(nc.cfg) - 1);
    nc.cfg[sizeof(nc.cfg) - 1] = 0;
    strncpy(nc.supKey, supKey, sizeof(nc.supKey) - 1);
    nc.supKey[sizeof(nc.supKey) - 1] = 0;
    nc.rpcId = requestId;
    nc.queuedMs = millis();
    nc.awaitingAck = false;
    nc.attempts = 0;
    nc.lastSentMs = 0;
    nc.viaCommandLog = true;   // el resultado sale como seguimientoCmd, no por este RPC
    nc.pending = true;
  }

  // CONTESTAR YA. El nodo puede estar a minutos de su siguiente ventana de subida, y
  // un RPC abierto tanto tiempo expira en ThingsBoard e informa de un fallo para una
  // orden que se entrego y se aplico.
  snprintf(reply, sizeof(reply), "{\"status\":\"enviado\",\"requestId\":%ld}", requestId);
  queueRpcReply(respTopic, reply);

  if (isGlobal) {
    Serial.printf("[nbiot-mqtt] configurarModuloC %ld -> LOS %u NODOS: \"%s\" "
                  "(enviado; esperando sus ventanas)\n",
                  requestId, (unsigned)slotsNeeded, cfg);
  } else {
    Serial.printf("[nbiot-mqtt] configurarModuloC %ld -> %s: \"%s\" (enviado; esperando ventana)\n",
                  requestId, device, cfg);
  }
}

// Pone un mensaje en la cola de publicaciones (respuestas RPC y commandLog).
void ModemNBIoTMqtt::queueRpcReply(const char *topic, const char *payload) {
  if (_rpcReplyCount >= kRpcReplyQueue) {
    // Llena. Se va la MAS ANTIGUA, no la mas nueva: un registro nuevo sustituye a uno
    // viejo de la misma orden, y el mas nuevo es el que aun merece la pena mandar.
    Serial.println("[nbiot-mqtt] rpc reply queue full - dropping the oldest");
    popRpcReply();
  }
  const uint8_t slot = (uint8_t)((_rpcReplyHead + _rpcReplyCount) % kRpcReplyQueue);
  RpcReply &r = _rpcReplies[slot];
  strncpy(r.topic, topic, sizeof(r.topic) - 1);
  r.topic[sizeof(r.topic) - 1] = '\0';
  strncpy(r.payload, payload, sizeof(r.payload) - 1);
  r.payload[sizeof(r.payload) - 1] = '\0';
  _rpcReplyCount++;
}

// ---- Resultado del comando AT en curso ----

void ModemNBIoTMqtt::completeCommand(CmdOutcome outcome) {
  _cmd.active = false;
  _cmd.outcome = outcome;
}

void ModemNBIoTMqtt::checkCommandTimeout() {
  if (_cmd.active && millis() - _cmd.sentAtMs > _cmd.timeoutMs) completeCommand(CmdOutcome::TIMEOUT);
}

// Devuelve el resultado del ultimo comando una sola vez (y lo borra).
ModemNBIoTMqtt::CmdOutcome ModemNBIoTMqtt::consumeOutcome() {
  CmdOutcome o = _cmd.outcome;
  _cmd.outcome = CmdOutcome::NONE;
  return o;
}

// ---------- Maquina de estados ----------

void ModemNBIoTMqtt::setState(State s) {
  // Apuntar cada transicion. La OLED solo muestra el nombre del estado ACTUAL, asi que
  // un bucle rapido POWERING -> WAIT_AT -> ERROR -> OFF parece en pantalla un unico
  // estado atascado - que es exactamente como "NBIoT: WAIT_AT" se leyo como
  // "conectado a la banda NB-IoT". Por el puerto serie el ciclo no tiene duda.
  if (_state != s) {
    Serial.printf("[nbiot-mqtt] state %s -> %s\n", stateName(), stateNameOf(s));
    Serial.flush();
  }
  _state = s;
  _stateEnteredMs = millis();
  switch (s) {
    case State::ATTACHING:
      _cgpaddrChecked = false;
      _csqCheckedAfterAttach = false;
      _cclkChecked = false;
      _lastCeregPollMs = 0;
      break;
    case State::IDLE:
      _lastCsqPollMs = 0;  // una consulta poco despues de entrar en IDLE
      break;
    case State::PUBLISHING:
      _publishInFlight = false;
      break;
    case State::RPC_REPLY:
      _rpcReplyInFlight = false;
      break;
    default:
      break;
  }
}

// Dejar de conducir el UART del modem por completo. Cerrar el periferico no basta:
// el pin tambien tiene que dejar de ser salida, o conserva su ultimo nivel y sigue
// alimentando al modulo sin corriente.
void ModemNBIoTMqtt::releaseUart() {
  if (_uartStarted) {
    _serial.end();
    _uartStarted = false;
  }
  pinMode(_txPin, INPUT);
  pinMode(_rxPin, INPUT);
}

void ModemNBIoTMqtt::startUart() {
  if (_uartStarted) return;
  _serial.begin(_baud, SERIAL_8N1, _rxPin, _txPin);
  _uartStarted = true;
  while (_serial.available()) _serial.read();  // tirar el ruido del encendido
}

// Dar alimentacion al modem.
void ModemNBIoTMqtt::enterPowering() {
  // PRIMERO EL UART ABAJO, con la linea TX liberada, antes de tocar la alimentacion:
  // un pin TX conducido alimenta el modulo mientras VIN esta apagado e impide un
  // arranque en frio de verdad. El UART se vuelve a abrir en tickPowering() cuando el
  // rail ha subido y se ha asentado.
  releaseUart();

  digitalWrite(_channelPin, NBIOT_CHANNEL_ACTIVE);
  digitalWrite(_enPin, NBIOT_EN_ACTIVE);
  _sawBootUrc = false;
  _pwrkeyReleased = false;
  setState(State::POWERING);
}

void ModemNBIoTMqtt::enterError(const char *reason) {
  setLastError(reason);
  _consecutiveFailures++;
  _backoffUntilMs = millis() + computeBackoff(_consecutiveFailures);
  setState(State::ERROR);
}

void ModemNBIoTMqtt::powerDown() {
  // Soltar TX antes de cortar VIN: si no, el modulo se sigue alimentando por el y esto
  // no seria un apagado en absoluto.
  releaseUart();
  digitalWrite(_enPin, NBIOT_DISABLE);
  digitalWrite(_channelPin, !NBIOT_CHANNEL_ACTIVE);
  digitalWrite(_pwrkeyPin, !NBIOT_PWRKEY_ACTIVE);
  _attached = false;
  _mqttConnected = false;
  _powerCycleSettleUntilMs = millis() + NBIOT_POWER_OFF_SETTLE_MS;
}

// Ultimo recurso: apagar el modem y volver a empezar desde OFF.
void ModemNBIoTMqtt::powerCycle() {
  setLastError("power cycling modem (last resort)");
  powerDown();
  setState(State::OFF);
}

// Recuperacion escalonada: reintentar en el mismo nivel unas pocas veces y, si no,
// bajar un nivel (publicar -> reconectar MQTT -> registrarse -> apagar y encender).
void ModemNBIoTMqtt::handleFailureAtLevel(RecoveryLevel level, const char *reason) {
  setLastError(reason);
  _consecutiveFailures++;

  switch (level) {
    case RecoveryLevel::PUBLISH:
      _publishRetries++;
      if (_publishRetries <= NBIOT_MAX_SEND_RETRIES) {
        applyBackoffAndSetState(State::PUBLISHING);
        return;
      }
      _publishRetries = 0;
      _mqttConnected = false;  // forzar un QMTOPEN/QMTCONN nuevo, no solo otro intento de publicar
      _mqttConnectSub = MqttConnectSub::CLOSE_FIRST;
      applyBackoffAndSetState(State::MQTT_CONNECT);
      return;

    case RecoveryLevel::MQTT_CONNECT:
      _mqttConnectRetries++;
      if (_mqttConnectRetries <= NBIOT_MAX_SOCKET_RETRIES) {
        _mqttConnectSub = MqttConnectSub::CLOSE_FIRST;
        applyBackoffAndSetState(State::MQTT_CONNECT);
        return;
      }
      _mqttConnectRetries = 0;
      applyBackoffAndSetState(State::ATTACHING);
      return;

    case RecoveryLevel::ATTACH:
      _attachRetries++;
      if (_attachRetries <= NBIOT_MAX_ATTACH_RETRIES) {
        applyBackoffAndSetState(State::ATTACHING);
        return;
      }
      _attachRetries = 0;
      powerCycle();
      return;
  }
}

void ModemNBIoTMqtt::applyBackoffAndSetState(State s) {
  _backoffUntilMs = millis() + computeBackoff(_consecutiveFailures);
  setState(s);
}

// Espera creciente: base x 2^intento, con un techo.
uint32_t ModemNBIoTMqtt::computeBackoff(uint32_t attempt) const {
  if (attempt > 10) attempt = 10;
  uint32_t v = (uint32_t)NBIOT_BACKOFF_BASE_MS << attempt;
  if (v > NBIOT_BACKOFF_MAX_MS || v < NBIOT_BACKOFF_BASE_MS) v = NBIOT_BACKOFF_MAX_MS;
  return v;
}

void ModemNBIoTMqtt::setLastError(const char *msg) {
  strncpy(_lastError, msg, sizeof(_lastError) - 1);
  _lastError[sizeof(_lastError) - 1] = '\0';
  // Tambien IMPRIMIRLO. Antes solo se guardaba el texto, y por eso un modem que no
  // contestaba nunca producia un log con nada mas que "-> AT": el abandono, el
  // apagado y el motivo eran invisibles, y el unico sitio donde aparecia el error era
  // una funcion de consulta que nadie llamaba. Un bucle de recuperacion silencioso no
  // se distingue de uno colgado.
  Serial.printf("[nbiot-mqtt] ERROR: %s\n", msg);
  Serial.flush();
}

const char *ModemNBIoTMqtt::stateName() const { return stateNameOf(_state); }

const char *ModemNBIoTMqtt::stateNameOf(State st) {
  switch (st) {
    case State::OFF: return "OFF";
    case State::POWERING: return "POWERING";
    case State::WAIT_AT: return "WAIT_AT";
    case State::CONFIG: return "CONFIG";
    case State::ATTACHING: return "ATTACHING";
    case State::MQTT_CONNECT: return "MQTT_CONNECT";
    case State::IDLE: return "IDLE";
    case State::PUBLISHING: return "PUBLISHING";
    case State::RPC_REPLY: return "RPC_REPLY";
    case State::ERROR: return "ERROR";
  }
  return "?";
}

uint32_t ModemNBIoTMqtt::attachedUptimeMs() const {
  return _attached ? (millis() - _attachedSinceMs) : 0;
}

// ---- OFF: esperar a que se descargue tras un apagado y volver a encender ----
void ModemNBIoTMqtt::tickOff() {
  if (millis() < _powerCycleSettleUntilMs) return;
  enterPowering();
}

// ---- POWERING: pulso de PWRKEY y espera al arranque ----
void ModemNBIoTMqtt::tickPowering() {
  // Pulso de PWRKEY sin bloquear: esperar a que VIN y el canal se asienten, poner
  // PWRKEY a bajo NBIOT_PWRKEY_PULSE_MS, soltarlo, y esperar un aviso de arranque o un
  // plazo fijo antes de seguir. Los reintentos de AT de WAIT_AT son el verdadero
  // punto de sincronizacion; esto es solo una ventaja de salida.
  // VIN y la puerta del canal tienen que estar estables ANTES de pulsar PWRKEY: el
  // modulo puede ignorar del todo un pulso sobre un rail que aun esta subiendo.
  // Subido de 300 a 2000 (2026-09-11) junto con el techo de WAIT_AT; los dos formaban
  // parte de la misma carrera de encendido demasiado agresiva.
  // (Nota: segun Config.h, PWRKEY no esta cableado en esta placa; el pulso es
  // inofensivo.)
  static const uint32_t kPowerSettleMs = 2000;
  uint32_t elapsed = millis() - _stateEnteredMs;

  if (elapsed < kPowerSettleMs) return;

  if (elapsed < kPowerSettleMs + NBIOT_PWRKEY_PULSE_MS) {
    digitalWrite(_pwrkeyPin, NBIOT_PWRKEY_ACTIVE);
    return;
  }

  if (!_pwrkeyReleased) {
    digitalWrite(_pwrkeyPin, !NBIOT_PWRKEY_ACTIVE);
    _pwrkeyReleased = true;
  }

  if (_sawBootUrc || elapsed > NBIOT_TIMEOUT_POWERING_MS) {
    // Solo ahora, con el modulo alimentado y asentado, es seguro conducir TX.
    startUart();
    _lastAtPingMs = 0;
    setState(State::WAIT_AT);
  }
}

// ---- WAIT_AT: preguntar "AT" hasta que conteste ----
void ModemNBIoTMqtt::tickWaitAt() {
  if (millis() - _stateEnteredMs > NBIOT_TIMEOUT_WAIT_AT_MS) {
    enterError("modem never responded to AT");
    return;
  }
  if (_cmd.active) return;

  if (consumeOutcome() == CmdOutcome::OK) {
    _configStep = ConfigStep::ATE0;
    _configStepEnteredMs = millis();
    setState(State::CONFIG);
    return;
  }

  if (millis() - _lastAtPingMs >= NBIOT_AT_PING_GAP_MS) {
    issueCommand("AT", NBIOT_AT_CMD_TIMEOUT_MS);
    _lastAtPingMs = millis();
  }
}

// ---- CONFIG: la secuencia de configuracion, un comando por paso ----
//   ATE0 -> CMEE=2 -> CTZU=1 -> QSCLK=0 -> CFUN=1 -> CPIN? -> CFUN=0 ->
//   QCGDEFCONT (APN) -> CFUN=1 -> ATTACHING
void ModemNBIoTMqtt::tickConfig() {
  if (millis() - _stateEnteredMs > NBIOT_TIMEOUT_CONFIG_MS) {
    enterError("CONFIG timed out");
    return;
  }
  if (_cmd.active) return;

  if (_cmd.outcome != CmdOutcome::NONE) {
    CmdOutcome outcome = consumeOutcome();
    // CTZU y QSCLK son "si se puede" (ver sus casos abajo); todos los demas pasos
    // son obligatorios, y un ERROR o un plazo agotado en ellos es fatal.
    if (outcome != CmdOutcome::OK && _configStep != ConfigStep::QSCLK &&
        _configStep != ConfigStep::CTZU) {
      enterError("a CONFIG step returned ERROR/timeout");
      return;
    }
    _configStep = (ConfigStep)((uint8_t)_configStep + 1);
    _configStepEnteredMs = millis();
  }

  switch (_configStep) {
    case ConfigStep::ATE0:
      issueCommand("ATE0", NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    case ConfigStep::CMEE:
      issueCommand("AT+CMEE=2", NBIOT_AT_CMD_TIMEOUT_MS);  // errores con texto ("ue not power on"...), no un ERROR a secas
      break;
    case ConfigStep::CTZU:
      // Actualizacion automatica de zona y hora desde la red, para que AT+CCLK? de
      // despues devuelva la hora real con la que fechar cada lectura. Antes de CFUN=1
      // para que se sincronice durante el registro. Si se puede: si el modulo lo
      // rechaza, simplemente no se pone "ts" por lectura (ver buildGatewayPayload).
      issueCommand("AT+CTZU=1", NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    case ConfigStep::QSCLK:
      // AJUSTE DE BANCO, NO UNA DECISION DE DISENO: la pasarela funciona con energia
      // solar y un modem que no duerme nunca es un consumo real y continuo. Esto
      // desactiva el sueno para siempre solo porque obtener una puesta en marcha
      // limpia del camino de QMTPUB importaba mas que el consumo. El diseno real es:
      // dejar el sueno activo, despertar el modem explicitamente antes de publicar,
      // confirmar que esta despierto y publicar - eso es otra iteracion, no esta hecha.
      //
      // Sin esto el modem puede entrar solo en sueno profundo en mitad de un comando
      // (confirmado el 2026-08-19: un AT+QMTPUB mandado mientras dormia no recibio
      // nunca el '>' - solo un "+QNBIOTEVENT: EXIT DEEPSLEEP" cuando desperto por su
      // cuenta, con la publicacion perdida y sin error ni timeout). Si no se admite,
      // se sigue igualmente en vez de bloquear la puesta en marcha.
      issueCommand("AT+QSCLK=0", NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    case ConfigStep::CFUN_ON_1:
      // TIENE que ir antes de AT+CPIN?: este modulo no activa solo su radio al
      // arrancar (confirmado el 2026-08-19: AT+CPIN? falla con
      // "+CME ERROR: ue not power on" sin esto antes).
      issueCommand("AT+CFUN=1", NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    case ConfigStep::CPIN:
      issueCommand("AT+CPIN?", NBIOT_AT_CMD_TIMEOUT_MS);  // SIM lista?
      break;
    case ConfigStep::CFUN_OFF:
      issueCommand("AT+CFUN=0", NBIOT_AT_CMD_TIMEOUT_MS);  // el APN solo se acepta con la radio apagada
      break;
    case ConfigStep::QCGDEFCONT: {
      // Dejar que se asiente la transicion de la SIM tras CFUN=0 (dice NOT READY un
      // momento): confirmado el 2026-08-19 que necesita ~3 s, o AT+QCGDEFCONT llega
      // antes de tiempo. Sin bloquear: se vuelve a mirar hasta que toca.
      if (millis() - _configStepEnteredMs < 3000) return;
      char cmd[64];
      snprintf(cmd, sizeof(cmd), "AT+QCGDEFCONT=\"IP\",\"%s\"", NBIOT_APN);
      issueCommand(cmd, NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    }
    case ConfigStep::CFUN_ON_2:
      issueCommand("AT+CFUN=1", NBIOT_AT_CMD_TIMEOUT_MS);
      break;
    case ConfigStep::DONE:
      setState(State::ATTACHING);
      break;
  }
}

// ---- ATTACHING: esperar al registro en la red, y leer cobertura y hora ----
//   AT+CEREG? (hasta registrado) -> AT+CGPADDR -> AT+CSQ -> AT+CCLK? -> MQTT_CONNECT
void ModemNBIoTMqtt::tickAttaching() {
  if (millis() < _backoffUntilMs) return;

  if (!_cgpaddrChecked) {
    if (millis() - _stateEnteredMs > NBIOT_TIMEOUT_ATTACH_MS) {
      handleFailureAtLevel(RecoveryLevel::ATTACH, "attach timed out");
      return;
    }
    if (_cmd.active) return;

    if (_cmd.outcome == CmdOutcome::NONE) {
      if (millis() - _lastCeregPollMs >= NBIOT_ATTACH_POLL_INTERVAL_MS) {
        issueCommand("AT+CEREG?", NBIOT_AT_CMD_TIMEOUT_MS);
        _lastCeregPollMs = millis();
      }
      return;
    }

    if (consumeOutcome() != CmdOutcome::OK) {
      handleFailureAtLevel(RecoveryLevel::ATTACH, "AT+CEREG? failed");
      return;
    }

    NbiotProtocol::CeregState st;
    if (!NbiotProtocol::parseCereg(_cmd.infoLine, st) || !NbiotProtocol::cieregIsAttached(st)) {
      _attached = false;
      return;  // aun no registrado: seguir preguntando; el plazo de arriba es la red de seguridad
    }

    _attached = true;
    _attachedSinceMs = millis();
    _attachRetries = 0;
    _consecutiveFailures = 0;

    char cmd[24];
    snprintf(cmd, sizeof(cmd), "AT+CGPADDR=%d", NBIOT_CONTEXT_ID);
    issueCommand(cmd, NBIOT_AT_CMD_TIMEOUT_MS);
    _cgpaddrChecked = true;
    return;
  }

  if (_cmd.active) return;
  if (_cmd.outcome == CmdOutcome::NONE) return;

  if (!_csqCheckedAfterAttach) {
    consumeOutcome();  // diga lo que diga CGPADDR, CEREG ya confirmo el registro: no merece fallar por ello
    // Un AT+CSQ justo aqui, sin esperar a que MQTT_CONNECT funcione: si no, la
    // lectura de cobertura (y lo que la muestra, como las barras de la OLED) no se
    // actualizaria nunca mientras Modulo A o el servidor fueran inalcanzables, aunque
    // el modem este registrado con senal real.
    issueCommand("AT+CSQ", NBIOT_AT_CMD_TIMEOUT_MS);
    _csqCheckedAfterAttach = true;
    return;
  }

  if (!_cclkChecked) {
    CmdOutcome csqOutcome = consumeOutcome();
    if (csqOutcome == CmdOutcome::OK && _cmd.hasInfoLine) {
      int dbm;
      if (NbiotProtocol::parseCsq(_cmd.infoLine, dbm)) _rssiDbm = dbm;
    }
    // Un AT+CCLK? por registro, para fijar la relacion epoca<->millis() con la que se
    // fecha cada lectura en la subida MQTT. Si se puede: un fallo (o un reloj
    // increible) deja _haveNetTime a false y la subida va sin "ts".
    issueCommand("AT+CCLK?", NBIOT_AT_CMD_TIMEOUT_MS);
    _cclkChecked = true;
    return;
  }

  CmdOutcome cclkOutcome = consumeOutcome();
  if (cclkOutcome == CmdOutcome::OK && _cmd.hasInfoLine) {
    int64_t epochMs;
    if (NbiotProtocol::parseCclk(_cmd.infoLine, epochMs) && epochMs >= NBIOT_MIN_VALID_EPOCH_MS) {
      _netEpochMsAtSync = epochMs;
      _netSyncLocalMs = millis();
      _haveNetTime = true;
      Serial.printf("[nbiot-mqtt] network time acquired: epoch %lld ms - uplink readings will carry per-reading ts\n",
                    (long long)epochMs);
    } else {
      Serial.println("[nbiot-mqtt] no usable network time (AT+CCLK?) - uplink will use ThingsBoard receipt time");
    }
  }

  _mqttConnectSub = MqttConnectSub::KEEPALIVE_CFG;
  setState(State::MQTT_CONNECT);
}

// "Ahora" en ms de epoca: la hora de red de la ultima sincronizacion mas el millis()
// transcurrido desde entonces.
int64_t ModemNBIoTMqtt::netNowMs() const {
  return _netEpochMsAtSync + (int64_t)(int32_t)(millis() - _netSyncLocalMs);
}

// ---- MQTT_CONNECT: abrir la sesion con ThingsBoard, paso a paso ----
//   QMTCLOSE -> keepalive -> sesion persistente -> QMTOPEN -> QMTCONN ->
//   suscripciones (RPC propio y RPC de pasarela) -> anunciar cada nodo -> IDLE
void ModemNBIoTMqtt::tickMqttConnect() {
  if (millis() < _backoffUntilMs) return;
  if (millis() - _stateEnteredMs > NBIOT_TIMEOUT_SOCKET_MS) {
    handleFailureAtLevel(RecoveryLevel::MQTT_CONNECT, "MQTT connect timed out");
    return;
  }
  if (_cmd.active) return;

  switch (_mqttConnectSub) {
    case MqttConnectSub::CLOSE_FIRST: {
      // Cierre defensivo antes de cada intento de (re)conexion: el estado del cliente
      // MQTT vive en el modem, no en el ESP32, y sobrevive a un reflasheo o reinicio
      // del ESP32. Una sesion sobrante de un intento anterior dejaba bloqueado el
      // client_idx 0, y AT+QMTOPEN fallaba sin mas con
      // "+CME ERROR: operation not allowed" hasta que se anadio esto (2026-08-19). Si
      // se puede: un ERROR aqui solo significa que no habia nada que cerrar.
      if (_cmd.outcome != CmdOutcome::NONE) {
        consumeOutcome();
        _mqttConnectSub = MqttConnectSub::KEEPALIVE_CFG;
        return;
      }
      char cmd[24];
      snprintf(cmd, sizeof(cmd), "AT+QMTCLOSE=%d", MQTT_CLIENT_IDX);
      issueCommand(cmd, NBIOT_AT_CMD_TIMEOUT_MS);
      return;
    }

    case MqttConnectSub::KEEPALIVE_CFG: {
      if (_cmd.outcome != CmdOutcome::NONE) {
        consumeOutcome();  // si se puede: seguir aunque se rechace el QMTCFG
        _mqttConnectSub = MqttConnectSub::SESSION_CFG;
        return;
      }
      char cmd[48];
      snprintf(cmd, sizeof(cmd), "AT+QMTCFG=\"keepalive\",%d,%lu", MQTT_CLIENT_IDX, (unsigned long)MQTT_KEEPALIVE_S);
      issueCommand(cmd, NBIOT_AT_CMD_TIMEOUT_MS);
      return;
    }

        case MqttConnectSub::SESSION_CFG: {
      // SESION PERSISTENTE - setCleanSession(false) en el documento del protocolo.
      //
      // clean_session = 0 le dice al servidor que CONSERVE la sesion de este cliente
      // a traves de una desconexion: sus suscripciones sobreviven, y los mensajes QoS 1
      // publicados mientras esta fuera un momento se ponen EN COLA en vez de tirarse.
      //
      // Aqui importa mas que en un enlace con cable. Este modulo no manda PINGREQ - ver
      // MQTT_KEEPALIVE_S - asi que la conexion solo la mantiene viva su propia
      // telemetria, y un enlace NB-IoT que se cae entre publicaciones no se nota hasta
      // el siguiente envio. Con sesion limpia, el servidor tiraria todas las ordenes
      // mandadas en ese hueco sin avisar a nadie. Con sesion persistente, estan
      // esperando cuando vuelve el enlace.
      //
      // Si se puede: un modem que rechace el ajuste sigue teniendo una subida que
      // funciona, y negarse a conectar por ello cambiaria una bajada fragil por no
      // tener enlace en absoluto.
      if (_cmd.outcome != CmdOutcome::NONE) {
        if (consumeOutcome() != CmdOutcome::OK) {
          Serial.println("[nbiot-mqtt] persistent session REJECTED - commands sent while "
                         "briefly offline will be lost");
        }
        _mqttConnectSub = MqttConnectSub::OPEN;
        return;
      }
      char cmd[48];
      snprintf(cmd, sizeof(cmd), "AT+QMTCFG=\"session\",%d,0", MQTT_CLIENT_IDX);
      issueCommand(cmd, NBIOT_AT_CMD_TIMEOUT_MS);
      return;
    }

    case MqttConnectSub::OPEN: {
      if (_cmd.outcome != CmdOutcome::NONE) {
        if (consumeOutcome() != CmdOutcome::OK) {
          handleFailureAtLevel(RecoveryLevel::MQTT_CONNECT, "QMTOPEN failed");
          return;
        }
        // CRITICO: encadenar directamente con QMTCONN, sin pausa ni espera - el
        // servidor cierra el socket ~5-6 s despues de QMTOPEN si no llega el CONNECT
        // (+QMTSTAT: <idx>,7). La unica demora es la del siguiente tick(), que en la
        // practica es inmediata.
        _mqttConnectSub = MqttConnectSub::CONN;
        return;
      }
      char cmd[96];
      snprintf(cmd, sizeof(cmd), "AT+QMTOPEN=%d,\"%s\",%d", MQTT_CLIENT_IDX, MQTT_BROKER_HOST, MQTT_BROKER_PORT);
      issueCommand(cmd, NBIOT_TIMEOUT_SOCKET_MS, CmdKind::QMTOPEN);
      return;
    }

    case MqttConnectSub::CONN: {
      if (_cmd.outcome != CmdOutcome::NONE) {
        if (consumeOutcome() != CmdOutcome::OK) {
          handleFailureAtLevel(RecoveryLevel::MQTT_CONNECT, "QMTCONN rejected/failed");
          return;
        }
        _mqttConnected = true;
        _mqttConnectRetries = 0;
        _consecutiveFailures = 0;
        _mqttConnectSub = MqttConnectSub::SUB_DEVICE;  // activar la bajada de ordenes antes de IDLE
        return;
      }
      // Autenticacion de ThingsBoard por token de acceso: el token va como usuario,
      // sin contrasena. (2026-08-19: se probo a poner una contrasena "" explicita tras
      // un CONNACK 5 "no autorizado" con un token correcto, y el modem lo rechazo con
      // "+CME ERROR: Incorrect parameters", asi que esta forma de 3 argumentos es la
      // correcta. La causa real del CONNACK 5 fue muy probablemente una sesion vieja
      // abierta en el modem desde un flasheo anterior - ver CLOSE_FIRST arriba.)
      // OJO: issueCommand() imprime el comando por el puerto serie, asi que el token
      // aparece en el log de depuracion.
      char cmd[128];
      snprintf(cmd, sizeof(cmd), "AT+QMTCONN=%d,\"%s\",\"%s\"", MQTT_CLIENT_IDX, MQTT_CLIENT_ID, MQTT_ACCESS_TOKEN);
      issueCommand(cmd, NBIOT_TIMEOUT_SOCKET_MS, CmdKind::QMTCONN);
      return;
    }

    // Las dos suscripciones son "si se puede": perder la bajada cuesta el control
    // remoto, pero la telemetria es el trabajo real de este modulo y no debe
    // bloquearla un SUBSCRIBE rechazado. Los fallos se apuntan y se sigue hacia IDLE.
    case MqttConnectSub::SUB_DEVICE: {
      if (_cmd.outcome != CmdOutcome::NONE) {
        if (consumeOutcome() != CmdOutcome::OK)
          Serial.println("[nbiot-mqtt] subscribe to this device's RPC topic FAILED - no remote config of Module B");
        _mqttConnectSub = MqttConnectSub::SUB_GATEWAY;
        return;
      }
      char cmd[96];
      snprintf(cmd, sizeof(cmd), "AT+QMTSUB=%d,%u,\"%s\",%d", MQTT_CLIENT_IDX, _subMsgId++,
               MQTT_TOPIC_DEVICE_RPC_SUB, MQTT_DOWNLINK_QOS);
      issueCommand(cmd, NBIOT_TIMEOUT_SOCKET_MS, CmdKind::QMTSUB);
      return;
    }

    case MqttConnectSub::SUB_GATEWAY: {
      if (_cmd.outcome != CmdOutcome::NONE) {
        if (consumeOutcome() != CmdOutcome::OK)
          Serial.println("[nbiot-mqtt] subscribe to the gateway RPC topic FAILED - no remote config of the nodes");
        else
          Serial.println("[nbiot-mqtt] downlink live - announcing the nodes next");
        _announceIdx = 0;
        _mqttConnectSub = MqttConnectSub::ANNOUNCE;
        return;
      }
      char cmd[96];
      snprintf(cmd, sizeof(cmd), "AT+QMTSUB=%d,%u,\"%s\",%d", MQTT_CLIENT_IDX, _subMsgId++,
               MQTT_TOPIC_GATEWAY_RPC, MQTT_DOWNLINK_QOS);
      issueCommand(cmd, NBIOT_TIMEOUT_SOCKET_MS, CmdKind::QMTSUB);
      return;
    }

    case MqttConnectSub::ANNOUNCE: {
      // Un v1/gateway/connect por nodo, uno por cada pasada por este estado.
      //
      // SIN ESTO, UN RPC A UN NODO NO LLEGA NUNCA. ThingsBoard solo encamina una orden
      // a un dispositivo hijo de una pasarela cuando la pasarela ha dicho que ese
      // dispositivo esta conectado a traves de ella. La telemetria no necesita ese
      // anuncio, asi que los nodos aparecian con datos y sanos mientras todas las
      // ordenes a ellos se tiraban en el servidor - y durante dos sesiones parecio un
      // fallo de firmware en este extremo.
      //
      // Se repite en cada conexion MQTT, porque el anuncio va ligado a la sesion: si
      // no, una reconexion des-registra en silencio todos los nodos.
      if (_cmd.outcome != CmdOutcome::NONE) {
        consumeOutcome();  // un anuncio fallido no es fatal: la telemetria sigue fluyendo
        _announceIdx++;
      }

      const size_t nodeCount = sizeof(NBIOT_NODE_NAMES) / sizeof(NBIOT_NODE_NAMES[0]);
      if (_announceIdx >= nodeCount) {
        Serial.printf("[nbiot-mqtt] %u node(s) announced - downlink live\n", (unsigned)nodeCount);
        setState(State::IDLE);
        return;
      }

      const char *name = NBIOT_NODE_NAMES[_announceIdx].name;
      _publishPayloadLen = (size_t)snprintf((char *)_publishPayload, sizeof(_publishPayload),
                                            "{\"device\":\"%s\"}", name);
      char header[96];
      snprintf(header, sizeof(header), "AT+QMTPUB=%d,0,0,0,\"v1/gateway/connect\"", MQTT_CLIENT_IDX);
      Serial.printf("[nbiot-mqtt] announcing %s to the gateway\n", name);
      issueCommand(header, NBIOT_TIMEOUT_SEND_MS, CmdKind::QMTPUB, _publishPayload, _publishPayloadLen);
      return;
    }
  }
}

// ---- IDLE: esperar; reconectar si la sesion cayo, contestar, o publicar si toca ----
void ModemNBIoTMqtt::tickIdle() {
  if (_cmd.active) return;

  CmdOutcome outcome = consumeOutcome();
  if (outcome == CmdOutcome::OK && _cmd.hasInfoLine) {
    int dbm;
    if (NbiotProtocol::parseCsq(_cmd.infoLine, dbm)) _rssiDbm = dbm;
  }

  // UNA SESION MUERTA SE RECONSTRUYE AQUI, no se deja para la siguiente publicacion.
  //
  // handleUrc() pone _mqttConnected a false cuando llega +QMTSTAT y dice que se
  // reconectara "en la siguiente publicacion". Pero solo se publica cuando llegan
  // datos por LoRa, e IDLE es justo el estado en el que no llegan. Un nodo que pasa
  // horas sin lecturas - o que simplemente esta apagado - significa que no se publica,
  // y por tanto que no se reconecta NUNCA: la maquina de estados se queda aqui
  // preguntando CSQ con una sesion muerta y la OLED diciendo "MQTT: caido" para siempre.
  //
  // Visto el 2026-09-14: Modulo B en IDLE durante dias, la red movil bien (CSQ 31/31,
  // registrado), MQTT caido en pantalla y sin reintentarse nunca, porque nada
  // transmitia para provocar una publicacion.
  //
  // Va condicionado a la espera creciente compartida, para que un servidor caido de
  // verdad se reintente con calma en vez de machacarse: los fallos de MQTT_CONNECT
  // alimentan esa misma espera a traves de handleFailureAtLevel().
  if (!_mqttConnected && millis() >= _backoffUntilMs) {
    Serial.println("[nbiot-mqtt] idle with a dead MQTT session - reconnecting now "
                   "(not waiting for a publish)");
    Serial.flush();
    _mqttConnectSub = MqttConnectSub::CLOSE_FIRST;
    setState(State::MQTT_CONNECT);
    return;
  }

  // Una respuesta RPC sale antes que el lote de telemetria: es lo que alguien esta
  // esperando en la pantalla de ThingsBoard, y el lote no pierde nada por esperar una
  // vuelta mas.
  if (rpcReplyPending() && _mqttConnected) {
    setState(State::RPC_REPLY);
    return;
  }

  // _flushRequested se salta la espera del lote - ver requestFlush(). Se borra cuando
  // la publicacion sale bien, no aqui, para que una publicacion urgente que falla se
  // reintente en vez de olvidarse sin decir nada.
  bool batchDue = _ringCount > 0 && (_flushRequested ||
                                      _ringCount >= _batchReadingsTarget ||
                                      (millis() - _oldestPendingMs) >= _batchSecondsTarget * 1000UL);
  if (batchDue) {
    setState(State::PUBLISHING);
    return;
  }

  // Mientras tanto, refrescar la cobertura para el panel cada cierto tiempo.
  if (millis() - _lastCsqPollMs >= NBIOT_CSQ_POLL_INTERVAL_MS) {
    issueCommand("AT+CSQ", NBIOT_AT_CMD_TIMEOUT_MS);
    _lastCsqPollMs = millis();
  }
}

// ---- PUBLISHING: subir el anillo a v1/gateway/telemetry, por trozos ----
void ModemNBIoTMqtt::tickPublishing() {
  if (millis() < _backoffUntilMs) return;

  if (!_mqttConnected) {
    // Reconexion perezosa: no intentar nunca publicar sobre una sesion que ya se sabe
    // muerta. La comprobacion batchDue de IDLE vuelve aqui en cuanto MQTT_CONNECT
    // funciona.
    _mqttReconnects++;
    _mqttConnectSub = MqttConnectSub::KEEPALIVE_CFG;
    setState(State::MQTT_CONNECT);
    return;
  }

  if (_cmd.active) return;

  if (!_publishInFlight) {
    _publishPayloadLen = buildGatewayPayload(_publishPayload, sizeof(_publishPayload), MQTT_PUB_CHUNK_READINGS);
    if (_publishPayloadLen == 0) {
      setState(State::IDLE);  // nada que publicar - no deberia pasar, batchDue implica anillo no vacio
      return;
    }
    char header[48];
    snprintf(header, sizeof(header), "AT+QMTPUB=%d,0,0,0,\"v1/gateway/telemetry\"", MQTT_CLIENT_IDX);
    issueCommand(header, NBIOT_TIMEOUT_SEND_MS, CmdKind::QMTPUB, _publishPayload, _publishPayloadLen);
    _publishInFlight = true;
    return;
  }

  CmdOutcome outcome = consumeOutcome();
  if (outcome == CmdOutcome::NONE) return;

  _publishInFlight = false;
  if (outcome == CmdOutcome::OK) {
    onPublishSucceeded();
  } else {
    onPublishFailed();
  }
}

// ---- RPC_REPLY: publicar las respuestas en cola ----
// Publica una respuesta RPC en cola y vuelve directamente a IDLE. A proposito mas
// simple que PUBLISHING: sin reintentos, sin trozos, sin anillo que vaciar. Si falla,
// la respuesta se tira y ThingsBoard deja expirar el RPC por su cuenta: reintentar el
// resultado viejo de una orden es peor que no contestar.
void ModemNBIoTMqtt::tickRpcReply() {
  if (!rpcReplyPending()) {
    setState(State::IDLE);
    return;
  }
  if (!_mqttConnected) {
    popRpcReply();  // no se puede contestar; que ThingsBoard lo deje expirar
    setState(State::IDLE);
    return;
  }
  if (_cmd.active) return;

  RpcReply &r = _rpcReplies[_rpcReplyHead];

  if (!_rpcReplyInFlight) {
    _publishPayloadLen = strlen(r.payload);
    if (_publishPayloadLen >= sizeof(_publishPayload)) _publishPayloadLen = sizeof(_publishPayload) - 1;
    memcpy(_publishPayload, r.payload, _publishPayloadLen);

    char header[128];
    snprintf(header, sizeof(header), "AT+QMTPUB=%d,0,0,0,\"%s\"", MQTT_CLIENT_IDX, r.topic);
    issueCommand(header, NBIOT_TIMEOUT_SEND_MS, CmdKind::QMTPUB, _publishPayload, _publishPayloadLen);
    _rpcReplyInFlight = true;
    return;
  }

  CmdOutcome outcome = consumeOutcome();
  if (outcome == CmdOutcome::NONE) return;

  if (outcome != CmdOutcome::OK) Serial.println("[nbiot-mqtt] rpc reply publish failed - dropped");
  _rpcReplyInFlight = false;
  popRpcReply();
  // Seguir directamente si hay mas esperando - IDLE volveria a entrar aqui de todos
  // modos, pero asi la cola se vacia de una pasada en vez de una por tick.
  setState(rpcReplyPending() ? State::RPC_REPLY : State::IDLE);
}

// ---- ERROR: esperar, apagar el modem de verdad y volver a empezar ----
void ModemNBIoTMqtt::tickError() {
  if (millis() < _backoffUntilMs) return;

  // UN REINTENTO TIENE QUE QUITAR LA ALIMENTACION DE VERDAD, y esto no lo hacia.
  //
  // El bucle era ERROR -> OFF -> enterPowering() -> POWERING -> WAIT_AT -> ERROR.
  // enterPowering() solo ACTIVA las puertas EN y del canal - que ya estaban activas,
  // porque nada en este camino las bajaba - y luego pulsa PWRKEY. powerCycle(), la
  // unica funcion que de verdad corta VIN, no estaba en este camino, asi que
  // _powerCycleSettleUntilMs seguia a 0 y tickOff() pasaba de largo.
  //
  // EN UN MODULO QUECTEL, PWRKEY CONMUTA. Un pulso a un modulo apagado lo enciende; el
  // mismo pulso a un modulo FUNCIONANDO lo APAGA. Asi que cada "reintento" pulsaba
  // PWRKEY sobre un modulo cuyo estado no se conocia y que, tras el primer intento,
  // probablemente estaba encendido - de modo que el reintento tenia tantas
  // posibilidades de apagar el modem como de arrancarlo. Eso explica tambien por que
  // el fallo a veces se arreglaba "solo" tras varios ciclos, y por que afectaba igual
  // a las dos placas: es paridad, no hardware.
  //
  // Cortar VIN primero hace que el pulso de PWRKEY no sea ambiguo: se sabe que el
  // modulo esta apagado, asi que un pulso solo puede significar "enciende".
  // NBIOT_POWER_OFF_SETTLE_MS (4 s) da tiempo a que el rail baje de verdad antes de
  // volver a subir; tickOff() espera al temporizador que arma powerDown().
  powerDown();
  setState(State::OFF);
}

// Publicacion correcta: quitar del anillo lo publicado y seguir si queda algo.
void ModemNBIoTMqtt::onPublishSucceeded() {
  _packetsSent++;
  _flushRequested = false;  // solo una vez que ha salido de verdad
  _consecutiveFailures = 0;
  _publishRetries = 0;
  popSentReadings();
  // Un lote se vacia en varias QMTPUB de MQTT_PUB_CHUNK_READINGS lecturas: se queda
  // en PUBLISHING hasta que el anillo esta vacio, y solo entonces vuelve a IDLE. La
  // sesion ya esta abierta, asi que el siguiente trozo sale en el tick siguiente sin
  // reconectar. Llegan lecturas mucho mas despacio de lo que se vacian (un trozo por
  // tick), asi que esto siempre termina.
  if (_ringCount > 0) {
    _publishInFlight = false;
    return;
  }
  setState(State::IDLE);
}

void ModemNBIoTMqtt::onPublishFailed() {
  _packetsFailed++;
  handleFailureAtLevel(RecoveryLevel::PUBLISH, "publish failed");
}

// Escribe en out las cifras decimales de un int64 no negativo (sin '\0') y devuelve
// cuantas son. Evita depender de que %lld este compilado en el snprintf de este
// compilador (la nano newlib a menudo no lo trae).
static size_t appendI64(char *out, int64_t v) {
  if (v < 0) v = 0;  // aqui la epoca en ms siempre es positiva - solo por precaucion
  char tmp[24];
  size_t n = 0;
  uint64_t u = (uint64_t)v;
  do {
    tmp[n++] = (char)('0' + (int)(u % 10));
    u /= 10;
  } while (u);
  for (size_t i = 0; i < n; i++) out[i] = tmp[n - 1 - i];
  return n;
}

// ---------- El JSON de telemetria para la API de pasarela de ThingsBoard ----------
//
// Construye un mensaje de telemetria de pasarela con hasta maxReadings de las
// lecturas pendientes mas antiguas del anillo:
//   {"<nodo>":[<registro>,<registro>,...], "<otroNodo>":[...]}
// Las lecturas se agrupan por nombre de nodo (ver nbiotResolveNodeName en Config.h)
// en vez de suponer un solo nodo.
//
// Dos formas de registro, y la eleccion no es libre - el interprete de pasarela de
// ThingsBoard solo acepta el envoltorio "values" si va con un "ts":
//   _haveNetTime : {"ts":<epoca_ms>,"values":{<claves>}}  - hora real de cada lectura
//   si no        : {<claves>}                             - el servidor pone la hora al recibir
// El ts de cada lectura es el momento en que se TOMO: netNowMs() trasladado a traves
// del lastHeardMs de la propia lectura (un valor de millis() ya corregido por su
// antiguedad), asi que un lote de lecturas espaciadas llega a ThingsBoard como una
// serie real, no como varios puntos en el mismo instante. Sin reloj de red se guarda
// igual, solo que con la resolucion de la llegada del lote.
size_t ModemNBIoTMqtt::buildGatewayPayload(uint8_t *out, size_t cap, uint8_t maxReadings) {
  char *buf = reinterpret_cast<char *>(out);
  size_t pos = 0;
  auto append = [&](const char *s) {
    size_t n = strlen(s);
    if (pos + n < cap) {
      memcpy(buf + pos, s, n);
      pos += n;
    }
  };

  uint8_t count = _ringCount;
  if (count > maxReadings) count = maxReadings;

  bool grouped[NBIOT_RING_CAPACITY] = {false};
  char groupName[24];

  append("{");
  bool firstNode = true;
  for (uint8_t i = 0; i < count; i++) {
    if (grouped[i]) continue;
    nbiotResolveNodeName(ringPeek(i).senderAddr, groupName, sizeof(groupName));

    if (!firstNode) append(",");
    firstNode = false;
    append("\"");
    append(groupName);
    append("\":[");

    bool firstReading = true;
    for (uint8_t j = i; j < count; j++) {
      if (grouped[j]) continue;
      const SensorSnapshot &snap = ringPeek(j);
      char thisName[24];
      nbiotResolveNodeName(snap.senderAddr, thisName, sizeof(thisName));
      if (strcmp(thisName, groupName) != 0) continue;

      grouped[j] = true;
      if (!firstReading) append(",");
      firstReading = false;

      // UN CAMPO QUE EL NODO NO MIDIO NO VIAJA.
      //
      // El nodo manda -999 en los campos que esa lectura concreta no midio. El
      // CM1106 va cada 30 min y el VOC del BME690 cada 20, asi que la mayoria de
      // filas no llevan CO2 ni gas. Antes el nodo repetia el ultimo valor
      // conocido, lo que hacia INDISTINGUIBLE "el CO2 no se movio" de "nadie miro
      // el CO2" - el mismo fallo que el viento congelado y el co2=663 fantasma de
      // C1, que costo dias de no entender nada.
      //
      // Y reenviar el -999 seria aun peor que repetir: la grafica lo pintaria como
      // una medida de -999 ppm. Se OMITE la clave, que es lo que de verdad
      // ocurrio, y ThingsBoard lo trata bien - no registra punto para esa serie en
      // ese instante.
      char keys[500];  // +3 claves para chargePct/capMv/alarmState
      int kn = 0;
      keys[kn++] = '{';
      keys[kn] = '\0';
#define SIN_MEDIR_B(v) ((v) < -900.0f)
#define CLAVE_F(nombre, valor)                                                        \
  do {                                                                                \
    if (!SIN_MEDIR_B(valor)) {                                                        \
      kn += snprintf(keys + kn, sizeof(keys) - (size_t)kn, "%s\"" nombre "\":%.2f",    \
                     (kn > 1) ? "," : "", (double)(valor));                            \
    }                                                                                 \
  } while (0)

      CLAVE_F("temp", snap.temp);
      CLAVE_F("rh", snap.hum);
      CLAVE_F("pres", snap.pres);
      CLAVE_F("gas", snap.gas);
      CLAVE_F("pm1", snap.pm1);
      CLAVE_F("pm25", snap.pm25);
      CLAVE_F("pm10", snap.pm10);
      CLAVE_F("co2", snap.co2);
      CLAVE_F("co", snap.co);
      CLAVE_F("coTemp", snap.coTemp);
      CLAVE_F("windAngle", snap.windAngle);
      CLAVE_F("windSpeed", snap.windSpeed);
#undef CLAVE_F
#undef SIN_MEDIR_B

      // Estos van SIEMPRE: no son medidas de sensor sino el estado del enlace y de
      // la placa, y ahi la ausencia si seria una perdida de informacion.
      kn += snprintf(keys + kn, sizeof(keys) - (size_t)kn,
                     "%s\"windValid\":%s,\"rssi\":%d,\"snr\":%d,"
                     "\"chargePct\":%d,\"capMv\":%d,\"alarmState\":%d}",
                     (kn > 1) ? "," : "", snap.windValid ? "true" : "false",
                     (int)snap.rssi, (int)snap.snr, (int)snap.chargePct, (int)snap.capMv,
                     (int)snap.alarmState);

      if (_haveNetTime) {
        int64_t ts = _netEpochMsAtSync + (int64_t)(int32_t)(snap.lastHeardMs - _netSyncLocalMs);
        char tsField[40];
        size_t tn = 0;
        memcpy(tsField, "{\"ts\":", 6);
        tn = 6;
        tn += appendI64(tsField + tn, ts);
        memcpy(tsField + tn, ",\"values\":", 10);
        tn += 10;
        tsField[tn] = '\0';
        append(tsField);
        append(keys);
        append("}");
      } else {
        append(keys);
      }
    }
    append("]");
  }
  append("}");

  _lastPublishCount = count;
  return pos;
}

#pragma once

#include <stdint.h>
#include <stddef.h>

#include "Config.h"
#include "telemetry/SensorSnapshot.h"

// =============================================================================
// PROTOCOLO DEL MODEM NB-IoT: INTERPRETES DE RESPUESTAS AT Y DE JSON
// =============================================================================
//
// Todo este archivo esta, a proposito, libre de Arduino (sin Arduino.h ni
// HardwareSerial) para poder probarlo con una prueba unitaria nativa en el PC (ver
// test/test_nbiot_protocol) sin el compilador del ESP32. ModemNBIoTMqtt es lo unico
// que habla con el UART; a este modulo le pasa lineas de texto y recibe resultados
// ya interpretados.
//
// QUE SE USA HOY: los interpretes de CEREG, CSQ, CCLK, los de MQTT (QMT*) y las
// funciones de JSON. Las de socket UDP (QIOPEN, QIURC, QIRD), parseDownlink() y la
// trama binaria (encodeBatch*) son de la via antigua por UDP, ya retirada: solo las
// usa la prueba unitaria.

namespace NbiotProtocol {

// ---------- Interpretes de lineas de respuesta AT ----------
// Cada uno recibe UNA linea de respuesta ya separada y terminada en '\0' (sin
// CR/LF, tal como la entrega el lector byte a byte de ModemNBIoTMqtt) y devuelve
// false si la linea no encaja en absoluto, para que quien llama pruebe el
// siguiente interprete o la trate como un aviso no reconocido en vez de adivinar.

// Estado de registro en la red movil (respuesta a AT+CEREG?).
enum class CeregState : uint8_t {
  NOT_REGISTERED = 0,
  REGISTERED_HOME = 1,      // registrado en la red propia
  SEARCHING = 2,
  DENIED = 3,
  UNKNOWN = 4,
  REGISTERED_ROAMING = 5,   // registrado en itinerancia
};

// Registrado = red propia o itinerancia.
inline bool cieregIsAttached(CeregState s) {
  return s == CeregState::REGISTERED_HOME || s == CeregState::REGISTERED_ROAMING;
}

// "+CEREG: <n>,<stat>[,...]" (aviso espontaneo) o "+CEREG: <stat>" (respuesta a la
// consulta con <n>=0). Solo hace falta <stat>, asi que se aceptan las dos formas.
bool parseCereg(const char *line, CeregState &out);

// "+CSQ: <rssi>,<ber>". <rssi>=99 significa "desconocido/no detectable".
// Devuelve false si la linea no se entiende; rssiDbm solo vale si devuelve true.
bool parseCsq(const char *line, int &rssiDbm);

// "+CCLK: \"yy/MM/dd,hh:mm:ss[+/-zz]\"" (las comillas y la zona horaria final son
// opcionales). <zz> es un numero con signo de pasos de 15 minutos respecto a UTC.
// Convierte a milisegundos de epoca Unix (UTC). Devuelve false si la linea no se
// entiende o los campos estan fuera de rango; quien llama tiene que comprobar
// ademas el valor (un modem sin hora de red contesta con una fecha de relleno de
// hacia el ano 2000, que se interpreta bien pero no es una hora real).
// OJO: este modem (BC660K-GL) da la hora YA EN UTC; ver el .cpp.
bool parseCclk(const char *line, int64_t &epochMs);

// HISTORICO (via UDP): "+QIOPEN: <connectID>,<err>", el aviso que sigue al OK de
// AT+QIOPEN. err==0 significa que el socket se abrio.
bool parseQiopen(const char *line, int &connectId, int &err);

// HISTORICO (via UDP): "+QIURC: \"recv\",<connectID>,<dataLen>" - avisa de que hay
// <dataLen> bytes esperando para leerlos con AT+QIRD=<connectID>,<dataLen>.
bool parseQiurcRecv(const char *line, int &connectId, int &dataLen);

// HISTORICO (via UDP): "+QIRD: <len>" - la cabecera de la respuesta de AT+QIRD,
// seguida de <len> bytes en crudo en la(s) linea(s) siguiente(s). Aqui solo se
// interpreta la cabecera.
bool parseQirdHeader(const char *line, int &len);

// ---------- Avisos MQTT (juego de comandos MQTT del BC660K-GL) ----------
// La sintaxis sigue la forma QMT estandar de Quectel, comun a las familias
// BC66/BC660K/EC. Si alguno vuelve sin entenderse, comprobarlo contra la "BC660K-GL
// MQTT Application Note" antes de suponer que la disposicion de campos esta mal.

// "+QMTOPEN: <client_idx>,<result>" - aviso tras el OK de AT+QMTOPEN.
// result==0 significa que se abrio la conexion TCP con el servidor.
bool parseQmtopen(const char *line, int &clientIdx, int &result);

// "+QMTCONN: <client_idx>,<result>,<retCode>" - aviso tras el OK de AT+QMTCONN.
// result==0 significa que el paquete CONNECT salio y hubo respuesta; retCode==0
// significa que el servidor acepto la conexion.
bool parseQmtconn(const char *line, int &clientIdx, int &result, int &retCode);

// "+QMTPUB: <client_idx>,<msgId>,<result>[,<value>]" - aviso tras AT+QMTPUB.
// result==0 significa publicacion correcta (value solo importa con QoS>0, que aqui
// no se usa: la telemetria se publica con QoS 0).
bool parseQmtpub(const char *line, int &clientIdx, int &msgId, int &result);

// "+QMTSTAT: <client_idx>,<errCode>" - espontaneo: el cliente MQTT se ha caido (el
// servidor o la red lo cerraron, se agoto el keepalive, etc.). No va ligado a ningun
// comando en curso: siempre entra por el camino de avisos.
bool parseQmtstat(const char *line, int &clientIdx, int &errCode);

// "+QMTSUB: <client_idx>,<msgID>,<result>[,<value>]" - aviso tras el OK de
// AT+QMTSUB. result==0 significa que la suscripcion se acepto.
bool parseQmtsub(const char *line, int &clientIdx, int &msgId, int &result);

// ---------- Mensaje MQTT entrante (ordenes desde Modulo A / ThingsBoard) ----------

constexpr size_t kMqttTopicLen = 72;
constexpr size_t kMqttPayloadLen = 320;

struct MqttMessage {
  int clientIdx = 0;
  int msgId = 0;
  char topic[kMqttTopicLen] = {0};
  char payload[kMqttPayloadLen] = {0};
};

// "+QMTRECV: <client_idx>,<msgID>,"<topic>"[,<payload_len>],"<payload>""
// Se aceptan las dos formas (con y sin la longitud explicita): cual emite el modulo
// depende de AT+QMTCFG="recv/mode". El contenido se toma del ultimo tramo entre
// comillas de la linea, para que un contenido con comas (todo JSON las tiene) no
// descoloque los campos de detras.
bool parseQmtrecv(const char *line, MqttMessage &out);

// ---------- Extraccion minima de campos JSON ----------
// Lo justo para los pequenos mensajes RPC de ThingsBoard: sin memoria dinamica, sin
// libreria, sin analizar el JSON entero. Cada funcion busca "<clave>" y lee el
// valor que la sigue, lo que no es ambiguo en estos mensajes porque sus claves son
// distintas y anidan como mucho dos niveles. NO reutilizarlo para JSON arbitrario:
// no entiende de ambitos, asi que una clave repetida a otra profundidad
// coincidiria con la primera aparicion.
bool jsonString(const char *json, const char *key, char *out, size_t cap);

// Lee el valor de `key` sea cadena entrecomillada O NUMERO JSON DESNUDO, y lo
// devuelve como texto tal cual venia.
//
// jsonString() exige comillas, y la v2 del protocolo A->C dice explicitamente:
// "Numeros con decimal: van como numero JSON (1.5), no como texto". Los tres
// comandos NxT mandan su N asi, de modo que sin esto no se pueden leer.
//
// Se devuelve como TEXTO y no como float a proposito: el valor viaja despues
// dentro de una cadena CFG y es el nodo quien lo interpreta, asi que convertirlo
// aqui solo anadiria una conversion de ida y vuelta donde se pierde precision.
bool jsonScalar(const char *json, const char *key, char *out, size_t cap);
bool jsonInt(const char *json, const char *key, long &out);
// Extrae el objeto con llaves equilibradas que sigue a "<clave>": - se usa para
// sacar los subobjetos "data" y "params" de un RPC de pasarela.
bool jsonObject(const char *json, const char *key, char *out, size_t cap);

// Recorre las parejas clave/valor del cuerpo de un objeto JSON plano. Se llama con
// el cursor apuntando al objeto (o justo despues de su '{'); devuelve false cuando
// no quedan parejas. Los valores se devuelven como texto sin las comillas, asi que
// tanto {"A":1} como {"A":"1"} dan "1".
bool jsonNextPair(const char *&cursor, char *key, size_t keyCap, char *value, size_t valueCap);

// ---------- HISTORICO: gramatica de respuesta de la via UDP ----------
// El servidor contestaba a un datagrama de subida con:
//   ACK,<seq>
//   ACK,<seq>,CFG,<version>,<clave>=<valor>,<clave>=<valor>,...
// Texto plano. Aqui solo se extrae la estructura; que claves son validas y en que
// rango lo decidia ModemNBIoT con las constantes NBIOT_CFG_*_MIN/MAX de Config.h.
// Ya no lo usa la via MQTT.
constexpr uint8_t kMaxDownlinkKv = 4;
constexpr uint8_t kDownlinkKeyLen = 8;
constexpr uint8_t kDownlinkValueLen = 16;

struct DownlinkKv {
  char key[kDownlinkKeyLen] = {0};
  char value[kDownlinkValueLen] = {0};
};

struct DownlinkAck {
  bool valid = false;
  uint16_t seq = 0;
  bool hasCfg = false;
  uint8_t cfgVersion = 0;
  DownlinkKv kv[kMaxDownlinkKv];
  uint8_t kvCount = 0;
};

// text es el contenido UDP en crudo (ya terminado en '\0' por quien llama).
// Devuelve false solo si ni siquiera empieza por "ACK,<numero>"; lo que no encaje en
// la gramatica CFG se deja sin interpretar (out.hasCfg sigue a false) en vez de
// rechazar el ACK entero.
bool parseDownlink(const char *text, DownlinkAck &out);

// ---------- HISTORICO: trama binaria de lote de la via UDP ----------
// Un CSV de texto con una lectura completa de 13 campos mas hora y calidad de enlace
// ocupa ~75-80 bytes por lectura, y un lote de 10 superaba el presupuesto de 500
// bytes. Con campos binarios de coma fija una lectura baja a kBatchRecordSize bytes.
// (La via MQTT actual publica JSON y no usa esto.)
//
// Todos los campos de varios bytes van en big-endian ("orden de red"), escritos a
// mano byte a byte en vez de copiar una estructura con memcpy: sin depender del
// relleno del compilador ni de que el ESP32 y el servidor coincidan en orden por
// casualidad.
//
// Disposicion:
//   Cabecera (kBatchHeaderSize = 5 bytes):
//     [0]   magic        0xF0
//     [1:2] seq          uint16 BE, sube una vez por lote (da la vuelta)
//     [3]   count        uint8, numero de registros que siguen
//     [4]   cfgVersion   uint8, version del ultimo CFG aplicado (0 = ninguno)
//   Registro (kBatchRecordSize = 30 bytes), repetido `count` veces:
//     [0:1]   dtSec        uint16 BE, segundos desde la lectura MAS ANTIGUA del lote
//     [2:3]   rssi         int16 BE, RSSI LoRa (dBm) al recibir
//     [4]     snr          int8,  SNR LoRa
//     [5:6]   temp_x10     int16 BE, grados C * 10
//     [7:8]   hum_x10      uint16 BE, %HR * 10
//     [9:10]  pres_x10     uint16 BE, hPa * 10
//     [11:12] gas_x100ohm  uint16 BE, resistencia en unidades de 100 ohmios
//     [13:14] pm1_x10      uint16 BE, ug/m3 * 10
//     [15:16] pm25_x10     uint16 BE, ug/m3 * 10
//     [17:18] pm10_x10     uint16 BE, ug/m3 * 10
//     [19:20] co2_ppm      uint16 BE
//     [21:22] co_x100      uint16 BE, ppm * 100
//     [23:24] coTemp_x10   int16 BE, grados C * 10
//     [25:26] windAngle_x10 uint16 BE, grados * 10 (0-3599)
//     [27:28] windSpeed_x10 uint16 BE
//     [29]    flags        uint8, bit0 = windValid, bits 1-7 reservados (0)
constexpr uint8_t kBatchMagic = 0xF0;
constexpr size_t kBatchHeaderSize = 5;
constexpr size_t kBatchRecordSize = 30;

static_assert(kBatchHeaderSize + (size_t)NBIOT_CFG_READINGS_MAX * kBatchRecordSize <=
                  NBIOT_PAYLOAD_MAX_BYTES,
              "NBIOT_CFG_READINGS_MAX * kBatchRecordSize + header must fit under "
              "NBIOT_PAYLOAD_MAX_BYTES - lower NBIOT_CFG_READINGS_MAX or shrink the record");

// Escribe kBatchHeaderSize bytes en out (buffer de quien llama, con sitio suficiente).
void encodeBatchHeader(uint16_t seq, uint8_t count, uint8_t cfgVersion, uint8_t *out);

// Escribe kBatchRecordSize bytes en out (buffer de quien llama, con sitio suficiente).
void encodeBatchRecord(const SensorSnapshot &snap, uint16_t dtSec, uint8_t *out);

}  // namespace NbiotProtocol

// Devuelve un puntero justo despues de la coma numero `count` contando desde p, o
// nullptr si no hay tantas. No modifica el texto: strtol() se para sola en el
// siguiente caracter no numerico, asi que nunca hace falta una subcadena terminada
// en '\0' para leer un campo numerico.
const char *skipCommas(const char *p, int count) {
  for (int i = 0; i < count; i++) {
    p = strchr(p, ',');
    if (!p) return nullptr;
    p++;
  }
  return p;
}

void putU16BE(uint8_t *out, uint16_t v) {
  out[0] = (uint8_t)(v >> 8);
  out[1] = (uint8_t)(v & 0xFF);
}

int16_t clampToInt16(long v) {
  if (v > 32767) return 32767;
  if (v < -32768) return -32768;
  return (int16_t)v;
}

uint16_t clampToUint16(long v) {
  if (v < 0) return 0;
  if (v > 65535) return 65535;
  return (uint16_t)v;
}

bool parseCereg(const char *line, CeregState &out) {
  if (strncmp(line, "+CEREG:", 7) != 0) return false;
  const char *p = line + 7;
  while (*p == ' ') p++;
  // Solo se trata la forma de respuesta a la consulta "AT+CEREG?":
  // "+CEREG: <n>,<stat>[,tac,ci,act]". El modem siempre pregunta en vez de activar
  // los avisos espontaneos +CEREG, asi que es la unica forma que necesita leer. (La
  // forma espontanea es ambigua sin ese contexto: con +CEREG=2 su primer campo ES
  // <stat>, no <n> - una ambiguedad que las consultas no tienen.)
  const char *statField = skipCommas(p, 1);
  if (!statField) return false;
  long stat = strtol(statField, nullptr, 10);
  if (stat < 0 || stat > 5) return false;
  out = static_cast<CeregState>(stat);
  return true;
}

bool parseCclk(const char *line, int64_t &epochMs) {
  if (strncmp(line, "+CCLK:", 6) != 0) return false;
  const char *p = line + 6;
  while (*p == ' ' || *p == '"') p++;

  // yy/MM/dd,hh:mm:ss[+/-zz]
  char *end = nullptr;
  long yy = strtol(p, &end, 10);
  if (end == p || *end != '/') return false;
  p = end + 1;
  long mon = strtol(p, &end, 10);
  if (end == p || *end != '/') return false;
  p = end + 1;
  long day = strtol(p, &end, 10);
  if (end == p || *end != ',') return false;
  p = end + 1;
  long hh = strtol(p, &end, 10);
  if (end == p || *end != ':') return false;
  p = end + 1;
  long mm = strtol(p, &end, 10);
  if (end == p || *end != ':') return false;
  p = end + 1;
  long ss = strtol(p, &end, 10);
  if (end == p) return false;
  p = end;

  long tzQuarters = 0;
  if (*p == '+' || *p == '-') {
    int sign = (*p == '-') ? -1 : 1;
    p++;
    const char *q = p;
    long v = strtol(q, &end, 10);
    if (end != q) tzQuarters = sign * v;
  }

  if (mon < 1 || mon > 12 || day < 1 || day > 31 || hh < 0 || hh > 23 || mm < 0 || mm > 59 || ss < 0 ||
      ss > 60)
    return false;

  // days_from_civil (Howard Hinnant): dias desde el 1970-01-01, calendario
  // gregoriano, sin depender de ninguna libreria de fechas ni de zonas horarias.
  long y = 2000 + yy;
  unsigned m = (unsigned)mon;
  unsigned d = (unsigned)day;
  y -= (m <= 2);
  long era = (y >= 0 ? y : y - 399) / 400;
  unsigned yoe = (unsigned)(y - era * 400);
  unsigned doy = (153u * (m + (m > 2 ? -3 : 9)) + 2u) / 5u + d - 1u;
  unsigned doe = yoe * 365u + yoe / 4u - yoe / 100u + doy;
  long long days = (long long)era * 146097 + (long long)doe - 719468;

  long long secs = days * 86400LL + hh * 3600LL + mm * 60LL + ss;

  // LA HORA QUE DA ESTE MODULO YA ESTA EN UTC: no restar la zona horaria.
  //
  // La norma 3GPP dice que +CCLK? devuelve la hora LOCAL con la zona al lado, asi que
  // lo obvio seria "restar la zona para tener UTC". Este BC660K-GL no se comporta
  // asi: da UTC en el campo de la hora aunque siga anunciando la zona local. Restarla
  // corrige dos veces y pone todas las marcas de tiempo una zona entera en el pasado.
  //
  // Medido en el hardware el 2026-09-14:
  //   modem          +CCLK: 26/09/14,08:46:08+08   (+08 cuartos de hora = +2 h, horario de verano)
  //   UTC del PC     2026-09-14 08:45:54
  //   antes salia -> 2026-09-14 06:46:08    = 2,00 horas ANTES
  //
  // El campo de la hora coincide con UTC con los 14 s de ida y vuelta, asi que es UTC.
  //
  // tzQuarters se sigue leyendo - hay que consumirlo para validar la linea y sirve
  // para diagnosticar - pero a proposito NO se aplica. Si alguna vez se usa otro
  // modulo, comprobarlo contra un reloj conocido antes de volver a restar; el fallo
  // es silencioso y parece un problema de visualizacion del servidor, no de este
  // codigo.
  (void)tzQuarters;

  epochMs = (int64_t)secs * 1000LL;
  return true;
}

bool parseQmtrecv(const char *line, MqttMessage &out) {
  if (strncmp(line, "+QMTRECV:", 9) != 0) return false;
  const char *p = line + 9;
  while (*p == ' ') p++;

  char *end = nullptr;
  long idx = strtol(p, &end, 10);
  if (end == p || *end != ',') return false;
  p = end + 1;
  long msg = strtol(p, &end, 10);
  if (end == p || *end != ',') return false;
  p = end + 1;

  // topic: el primer tramo entre comillas
  if (*p != '"') return false;
  const char *topicStart = p + 1;
  const char *topicEnd = strchr(topicStart, '"');
  if (!topicEnd) return false;
  size_t topicLen = (size_t)(topicEnd - topicStart);
  if (topicLen >= kMqttTopicLen) topicLen = kMqttTopicLen - 1;

  // contenido: desde la comilla que se abre tras el topic hasta la ULTIMA comilla de
  // la linea. Saltarse lo que haya en medio (un campo de longitud opcional) hace que
  // funcionen las dos formas del aviso, y tomar la ultima comilla impide que las
  // comas del JSON lo corten.
  const char *afterTopic = topicEnd + 1;
  const char *payStart = strchr(afterTopic, '"');
  if (!payStart) return false;
  payStart++;
  const char *payEnd = strrchr(payStart, '"');
  if (!payEnd || payEnd < payStart) return false;
  size_t payLen = (size_t)(payEnd - payStart);
  if (payLen >= kMqttPayloadLen) payLen = kMqttPayloadLen - 1;

  out.clientIdx = (int)idx;
  out.msgId = (int)msg;
  memcpy(out.topic, topicStart, topicLen);
  out.topic[topicLen] = '\0';
  memcpy(out.payload, payStart, payLen);
  out.payload[payLen] = '\0';
  return true;
}

// Devuelve un puntero justo despues del ':' que sigue a "<clave>", o nullptr.
const char *jsonSeekValue(const char *json, const char *key) {
  char quoted[40];
  size_t n = strlen(key);
  if (n + 3 > sizeof(quoted)) return nullptr;
  quoted[0] = '"';
  memcpy(quoted + 1, key, n);
  quoted[n + 1] = '"';
  quoted[n + 2] = '\0';

  const char *at = strstr(json, quoted);
  if (!at) return nullptr;
  const char *p = at + n + 2;
  while (*p == ' ') p++;
  if (*p != ':') return nullptr;
  p++;
  while (*p == ' ') p++;
  return p;
}

}  // namespace

bool jsonScalar(const char *json, const char *key, char *out, size_t cap) {
  const char *p = jsonSeekValue(json, key);
  if (!p) return false;
  if (*p == '"') return jsonString(json, key, out, cap);

  // Token desnudo: numero, true, false o null. Termina en el primer separador.
  const char *e = p;
  while (*e && *e != ',' && *e != '}' && *e != ']' &&
         *e != ' ' && *e != '\t' && *e != '\r' && *e != '\n') {
    e++;
  }
  const size_t len = (size_t)(e - p);
  if (len == 0 || len >= cap) return false;
  memcpy(out, p, len);
  out[len] = '\0';
  return true;
}

bool jsonString(const char *json, const char *key, char *out, size_t cap) {
  const char *p = jsonSeekValue(json, key);
  if (!p || *p != '"') return false;
  p++;
  const char *e = strchr(p, '"');
  if (!e) return false;
  size_t len = (size_t)(e - p);
  if (len >= cap) len = cap - 1;
  memcpy(out, p, len);
  out[len] = '\0';
  return true;
}

bool jsonInt(const char *json, const char *key, long &out) {
  const char *p = jsonSeekValue(json, key);
  if (!p) return false;
  if (*p == '"') p++;  // aceptar tambien un numero entre comillas
  char *end = nullptr;
  long v = strtol(p, &end, 10);
  if (end == p) return false;
  out = v;
  return true;
}

bool jsonObject(const char *json, const char *key, char *out, size_t cap) {
  const char *p = jsonSeekValue(json, key);
  if (!p || *p != '{') return false;
  int depth = 0;
  const char *start = p;
  bool inStr = false;
  for (; *p; p++) {
    if (inStr) {
      if (*p == '\\' && p[1]) p++;
      else if (*p == '"') inStr = false;
      continue;
    }
    if (*p == '"') inStr = true;
    else if (*p == '{') depth++;
    else if (*p == '}') {
      depth--;
      if (depth == 0) {
        size_t len = (size_t)(p - start) + 1;
        if (len >= cap) len = cap - 1;
        memcpy(out, start, len);
        out[len] = '\0';
        return true;
      }
    }
  }
  return false;
}

bool jsonNextPair(const char *&cursor, char *key, size_t keyCap, char *value, size_t valueCap) {
  const char *p = cursor;
  // avanzar hasta la siguiente clave entre comillas
  while (*p && *p != '"') {
    if (*p == '}') return false;  // fin de este objeto
    p++;
  }
  if (*p != '"') return false;
  p++;
  const char *ke = strchr(p, '"');
  if (!ke) return false;
  size_t klen = (size_t)(ke - p);
  if (klen >= keyCap) klen = keyCap - 1;
  memcpy(key, p, klen);
  key[klen] = '\0';

  p = ke + 1;
  while (*p == ' ') p++;
  if (*p != ':') return false;
  p++;
  while (*p == ' ') p++;

  if (*p == '"') {
    p++;
    const char *ve = strchr(p, '"');
    if (!ve) return false;
    size_t vlen = (size_t)(ve - p);
    if (vlen >= valueCap) vlen = valueCap - 1;
    memcpy(value, p, vlen);
    value[vlen] = '\0';
    cursor = ve + 1;
  } else {
    const char *ve = p;
    while (*ve && *ve != ',' && *ve != '}' && *ve != ' ') ve++;
    size_t vlen = (size_t)(ve - p);
    if (vlen >= valueCap) vlen = valueCap - 1;
    memcpy(value, p, vlen);
    value[vlen] = '\0';
    cursor = ve;
  }
  return true;
}

bool parseDownlink(const char *text, DownlinkAck &out) {
  out = DownlinkAck{};
  if (strncmp(text, "ACK,", 4) != 0) return false;

  const char *p = text + 4;
  char *end = nullptr;
  long seq = strtol(p, &end, 10);
  if (end == p || seq < 0) return false;
  out.seq = (uint16_t)seq;
  out.valid = true;

  p = end;
  if (*p != ',') return true;  // "ACK,<seq>" a secas: valido, sin CFG
  p++;
  if (strncmp(p, "CFG,", 4) != 0) return true;  // sufijo desconocido: el ACK en si sigue valiendo
  p += 4;

  long version = strtol(p, &end, 10);
  if (end == p || version < 0) return true;  // cabecera CFG mal formada: se ignora solo la parte CFG
  out.cfgVersion = (uint8_t)version;
  out.hasCfg = true;
  p = end;

  while (*p == ',' && out.kvCount < kMaxDownlinkKv) {
    p++;
    const char *eq = strchr(p, '=');
    if (!eq) break;
    const char *nextComma = strchr(eq, ',');
    size_t keyLen = (size_t)(eq - p);
    size_t valLen = nextComma ? (size_t)(nextComma - eq - 1) : strlen(eq + 1);

    if (keyLen > 0 && keyLen < kDownlinkKeyLen && valLen > 0 && valLen < kDownlinkValueLen) {
      DownlinkKv &slot = out.kv[out.kvCount++];
      memcpy(slot.key, p, keyLen);
      slot.key[keyLen] = '\0';
      memcpy(slot.value, eq + 1, valLen);
      slot.value[valLen] = '\0';
    }
    // Un elemento mal formado o demasiado largo simplemente no se guarda; el resto de
    // la linea merece la pena leerlo, asi que se avanza p en cualquier caso.
    p = nextComma ? nextComma : eq + 1 + valLen;
  }
  return true;
}

void encodeBatchHeader(uint16_t seq, uint8_t count, uint8_t cfgVersion, uint8_t *out) {
  out[0] = kBatchMagic;
  putU16BE(out + 1, seq);
  out[3] = count;
  out[4] = cfgVersion;
}

void encodeBatchRecord(const SensorSnapshot &snap, uint16_t dtSec, uint8_t *out) {
  putU16BE(out + 0, dtSec);
  putU16BE(out + 2, (uint16_t)clampToInt16((long)snap.rssi));
  out[4] = (uint8_t)(int8_t)snap.snr;
  putU16BE(out + 5, (uint16_t)clampToInt16(lroundf(snap.temp * 10.0f)));
  putU16BE(out + 7, clampToUint16(lroundf(snap.hum * 10.0f)));
  putU16BE(out + 9, clampToUint16(lroundf(snap.pres * 10.0f)));
  putU16BE(out + 11, clampToUint16(lroundf(snap.gas / 100.0f)));
  putU16BE(out + 13, clampToUint16(lroundf(snap.pm1 * 10.0f)));
  putU16BE(out + 15, clampToUint16(lroundf(snap.pm25 * 10.0f)));
  putU16BE(out + 17, clampToUint16(lroundf(snap.pm10 * 10.0f)));
  putU16BE(out + 19, clampToUint16(lroundf(snap.co2)));
  putU16BE(out + 21, clampToUint16(lroundf(snap.co * 100.0f)));
  putU16BE(out + 23, (uint16_t)clampToInt16(lroundf(snap.coTemp * 10.0f)));
  putU16BE(out + 25, clampToUint16(lroundf(snap.windAngle * 10.0f)));
  putU16BE(out + 27, clampToUint16(lroundf(snap.windSpeed * 10.0f)));
  out[29] = snap.windValid ? 0x01 : 0x00;
}

// =============================================================================
// UNA LECTURA DE UN NODO, TAL COMO LA GUARDA MODULO B
// =============================================================================
//
// Las ultimas lecturas conocidas mas la calidad del enlace y su frescura. Esta
// estructura es lo unico que conoce la parte web: no sabe que existen LoRa ni el
// interprete, solo lee la lectura que se le da.
struct SensorSnapshot {
  bool hasData = false;     // false hasta que llega el primer paquete valido
  uint32_t lastHeardMs = 0;  // millis() del momento en que se TOMO la lectura (ya corregido por su antiguedad)
  uint16_t senderAddr = 0;  // direccion LoRa AT+ADDRESS del nodo que la envio
  int16_t rssi = 0;         // intensidad de la senal recibida (dBm)
  int8_t snr = 0;           // relacion senal/ruido (dB)

  float temp = 0;    // BME690, grados C
  float hum = 0;      // BME690, %
  float pres = 0;      // BME690, hPa
  float gas = 0;        // BME690, ohmios

  float pm1 = 0;    // BMV080, ug/m3
  float pm25 = 0;
  float pm10 = 0;

  float co2 = 0;   // CM1106, ppm

  float co = 0;      // SEN0466, ppm
  float coTemp = 0;   // SEN0466, temperatura interna, grados C

  float windAngle = 0;  // Calypso, grados
  float windSpeed = 0;   // Calypso
  bool windValid = false;

  // Salud del nodo. Llega en el paquete STAT, aparte de la telemetria. Se rellena
  // con el ultimo STAT recibido de ese nodo, para que el estado de carga viaje a
  // ThingsBoard junto con los datos de los sensores en vez de quedarse en Modulo B.
  //
  // chargePct vale -1 hasta que se recibe un STAT de ese nodo: un 0% real y "aun no
  // se sabe" son cosas distintas, y un nodo que dijera 0% simplemente por no haber
  // hablado todavia alarmaria sin motivo.
  int16_t chargePct = -1;
  int16_t capMv = -1;

  // Estado de incendio segun el STAT del nodo: 0 normal, 1 pre-alarma, 2 alarma.
  // -1 significa que el nodo nunca lo dijo (un nodo antiguo cuyo STAT solo tiene ocho
  // campos). "Desconocido" y "normal" tienen que seguir siendo distintos: un panel que
  // muestra NORMAL con seguridad para un nodo que no informa es peor que uno que no
  // muestra nada.
  int8_t alarmState = -1;
};

#include "SensorSnapshot.h"

// =============================================================================
// INTERPRETE DE LA TELEMETRIA DE LOS NODOS (Modulo C)
// =============================================================================
//
// Convierte el texto que manda el nodo en una SensorSnapshot. No sabe nada de LoRa
// ni del servidor web: entra texto y sale una estructura, asi que se puede probar o
// reutilizar sin ninguno de los dos.
//
// FORMATO: CSV de orden fijo, sin cabecera, exactamente 13 campos:
//
//   temp,hum,pres,gas,pm1,pm25,pm10,co2,co,coTemp,windAngle,windSpeed,windValid
//
// Todos son numeros decimales salvo windValid (literalmente "0" o "1"). Un campo que
// el nodo no midio llega como -999. Si cambia la codificacion en el nodo, hay que
// cambiar kFieldCount y la asignacion de campos del .cpp: los dos extremos tienen
// que coincidir en el orden exacto (ver transmitStore() en el firmware de Modulo C).
class TelemetryParser {
public:
  // Una lectura suelta: exactamente 13 campos separados por comas, ni uno mas. Si
  // no, devuelve false y deja `out` sin tocar: se rechaza entera en vez de rellenar a
  // medias una lectura a partir de un texto mal formado o desincronizado.
  // RECHAZA un lote ("B,<n>,...") - ver parseBatch.
  static bool parse(const char *payload, uint8_t len, SensorSnapshot &out);

  // Un lote: "B,<cuantas>,<registro>;<registro>;..." donde cada registro son los
  // mismos 13 campos mas, al final, su antiguedad en el momento del envio (hoy en
  // SEGUNDOS reales; antes eran ticks de 10 s, de ahi el nombre del parametro).
  //
  // Devuelve cuantos registros se leyeron de verdad, escribiendo hasta maxOut
  // lecturas y (opcionalmente) sus antiguedades. El numero de la cabecera no se cree:
  // un paquete cortado devuelve los registros que realmente estan.
  static uint8_t parseBatch(const char *payload, uint8_t len, SensorSnapshot *out,
                            uint8_t maxOut, uint16_t *ageTicksOut = nullptr);

private:
  static constexpr uint8_t kFieldCount = 13;
  static constexpr uint8_t kBatchFieldCount = 14;  // 13 + antiguedad
};

#include "TelemetryParser.h"
#include <stdlib.h>
#include <string.h>

// Parte un registro separado por comas en EXACTAMENTE kFieldCount campos.
//
// Devuelve false si hay de menos O DE MAS - y el caso "de mas" es el que importa.
// La version anterior dejaba de mirar al tener 13 campos y luego comprobaba
// `fieldIdx != kFieldCount`, que por construccion no podia fallar nunca. Cualquier
// cosa con MAS de 13 campos se aceptaba en silencio usando solo los 13 primeros, asi
// que un lote "B,2,<registro>;<registro>;" se interpretaba como una sola lectura con
// la cabecera metida en los datos:
//
//   "B"   -> temp = 0.0      (atof de una letra)
//   "2"   -> rh   = 2.0      (el numero de registros del lote)
//   42.4  -> pres            (en realidad la temperatura)
//   11.3  -> gas             (en realidad la humedad)
//
// que es exactamente la corrupcion que se vio en ThingsBoard: todos los campos
// desplazados dos posiciones, con 0.0 y 2.0 apareciendo de la nada. La segunda
// lectura de cada lote se perdia por completo.
static bool splitRecord(char *buf, uint8_t len, char **fields, uint8_t want) {
  uint8_t fieldIdx = 0;
  fields[fieldIdx++] = buf;

  for (uint8_t i = 0; i < len; i++) {
    if (buf[i] == ',') {
      if (fieldIdx >= want) return false;  // MAS campos de los esperados: no es esta forma
      buf[i] = '\0';
      fields[fieldIdx++] = buf + i + 1;
    }
  }
  return fieldIdx == want;
}

// Copia los 13 campos de texto, en su orden fijo, a la estructura de lectura.
static void fillSnapshot(char **fields, SensorSnapshot &out) {
  out.temp = atof(fields[0]);
  out.hum = atof(fields[1]);
  out.pres = atof(fields[2]);
  out.gas = atof(fields[3]);
  out.pm1 = atof(fields[4]);
  out.pm25 = atof(fields[5]);
  out.pm10 = atof(fields[6]);
  out.co2 = atof(fields[7]);
  out.co = atof(fields[8]);
  out.coTemp = atof(fields[9]);
  out.windAngle = atof(fields[10]);
  out.windSpeed = atof(fields[11]);
  out.windValid = (fields[12][0] == '1');
}

// Una lectura suelta: exactamente 13 campos.
bool TelemetryParser::parse(const char *payload, uint8_t len, SensorSnapshot &out) {
  if (len == 0 || len > 200) return false;

  // Un lote tiene otra forma y no se debe meter a la fuerza en esta - ver
  // parseBatch(). Rechazarlo aqui es lo que evita la corrupcion por desplazamiento.
  if (len >= 2 && payload[0] == 'B' && payload[1] == ',') return false;

  char buf[241];
  memcpy(buf, payload, len);
  buf[len] = '\0';

  char *fields[kFieldCount] = {};
  if (!splitRecord(buf, len, fields, kFieldCount)) return false;

  fillSnapshot(fields, out);
  return true;
}

// Un lote: "B,<cuantas>,<registro>;<registro>;...". Devuelve cuantos registros se
// pudieron leer, y la antiguedad de cada uno en ageTicksOut.
uint8_t TelemetryParser::parseBatch(const char *payload, uint8_t len, SensorSnapshot *out,
                                    uint8_t maxOut, uint16_t *ageTicksOut) {
  if (len < 4 || len > 240) return 0;
  if (!(payload[0] == 'B' && payload[1] == ',')) return 0;

  char buf[241];
  memcpy(buf, payload, len);
  buf[len] = '\0';

  // Saltar la cabecera "B,<cuantas>,". El numero NO se toma como cuantos registros
  // hay, a proposito: los registros se cuentan mientras se leen, asi que un paquete
  // cortado o mal formado devuelve los registros que de verdad estan, no un numero
  // prometido que no se cumple.
  char *p = strchr(buf, ',');
  if (!p) return 0;
  p = strchr(p + 1, ',');
  if (!p) return 0;
  p++;

  uint8_t n = 0;
  while (*p && n < maxOut) {
    char *end = strchr(p, ';');
    if (end) *end = '\0';

    uint8_t recLen = (uint8_t)strlen(p);
    if (recLen > 0) {
      // Un registro de lote lleva UN campo mas al final que una lectura suelta: su
      // antiguedad en el momento del envio. Hoy son SEGUNDOS reales medidos por el
      // nodo (antes eran ticks; el nombre ageTicksOut viene de entonces).
      char *fields[kBatchFieldCount] = {};
      if (splitRecord(p, recLen, fields, kBatchFieldCount)) {
        fillSnapshot(fields, out[n]);
        if (ageTicksOut) ageTicksOut[n] = (uint16_t)atoi(fields[13]);
        n++;
      }
    }

    if (!end) break;
    p = end + 1;
  }

  return n;
}

#pragma once

#include <Arduino.h>

// =============================================================================
// DRIVER DE LA RADIO LoRa REYAX RYLR998
// =============================================================================
//
// Lo comparten Modulo C (transmite) y Modulo B (recibe). Antes habia una copia en
// cada modulo que solo se diferenciaba en los comentarios; se unifico aqui porque
// el driver era identico. Ver radio/README.md para los parametros de enlace de
// partida y como cambian alcance por tiempo en el aire.
//
// La radio se maneja con comandos de texto AT por un puerto serie (UART): se le
// manda "AT+SEND=..." y contesta "+OK" o "+ERR=..."; cuando llega un paquete por
// el aire, la radio escribe sola una linea "+RCV=...".

// Un paquete recibido, interpretado a partir de una linea
// "+RCV=<direccion>,<longitud>,<datos>,<rssi>,<snr>".
// El limite de 240 bytes de carga es el que documenta habitualmente AT+SEND del
// RYLR998. No se ha sacado de una hoja de datos incluida en este repositorio (no
// hay ninguna), asi que es una suposicion de partida a verificar contra el
// firmware real del modulo.
struct LoRaMessage {
  uint16_t senderAddr = 0;    // direccion AT+ADDRESS de quien lo envio
  char payload[241] = {0};    // los datos, terminados en '\0'
  uint8_t length = 0;         // longitud de los datos
  int16_t rssi = 0;           // intensidad de la senal recibida (dBm)
  int8_t snr = 0;             // relacion senal/ruido (dB)
};

// AT+PARAMETER=<factorDeEnsanchamiento>,<anchoDeBanda>,<tasaDeCodificacion>,<preambulo>.
// El ancho de banda es el INDICE/codigo del modulo, no un valor en kHz - ver
// radio/README.md para los valores de partida y su efecto en alcance y tiempo.
struct RYLR998Params {
  uint8_t spreadingFactor;  // SF: mas alto = mas alcance y mas tiempo en el aire
  uint8_t bandwidth;        // codigo de ancho de banda (7 = 125 kHz)
  uint8_t codingRate;       // tasa de codificacion (1 = 4/5)
  uint16_t preamble;        // longitud del preambulo
};

// Driver minimo por comandos AT.
//
// begin() y send() son BLOQUEANTES: esperan la respuesta rapida +OK/+ERR del
// modulo. poll() NO lo es: solo vacia los bytes que ya estan en el buffer de
// recepcion del UART, asi que se puede llamar en cada vuelta de loop() sin que
// la radio detenga nunca el programa (en Modulo B en particular, nunca debe
// detener el servidor web de handleClient()).
class RYLR998 {
public:
  RYLR998(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint32_t baud = 115200)
      : _serial(serial), _rxPin(rxPin), _txPin(txPin), _baud(baud) {}

  // Bloqueante: ejecuta AT, AT+ADDRESS, AT+NETWORKID, AT+BAND y AT+PARAMETER por
  // orden, y abandona en el primero que no conteste +OK. Si devuelve false, revisa
  // primero el cableado y la velocidad (rylr998_bridge.cpp) antes de suponer que
  // un parametro es incorrecto.
  // Pasando un flujo `debug` (por ejemplo &Serial) imprime cada comando y su
  // respuesta en el momento, para confirmar que el modulo acepto cada ajuste en
  // vez de fiarse de un unico verdadero/falso al final.
  bool begin(uint16_t addr, uint16_t networkId, uint32_t bandHz, const RYLR998Params &params,
             Print *debug = nullptr);

  // Bloqueante: escribe AT+SEND=<direccion>,<longitud>,<datos> y espera +OK, con
  // un limite de kSendTimeoutMs. El tiempo real en el aire depende de la longitud
  // y del SF/BW/CR configurados, pero esta llamada nunca espera indefinidamente.
  bool send(uint16_t destAddr, const char *data, uint8_t len);

  // Bloqueante: AT+CRFOP=<dbm>, de 0 a 22. El modulo arranca a 22 dBm, y su rafaga
  // de transmision pide ~120 mA - suficiente para tumbar una alimentacion que no
  // aguanta un escalon de carga, que es exactamente lo que le pasa a la placa de
  // Modulo C (se reiniciaba en cada transmision). Menos potencia cambia alcance
  // por una rafaga mas pequena. NO se incluye en begin() a proposito: este archivo
  // lo comparte Modulo B, que tiene alimentacion de sobra y ningun motivo para
  // renunciar a alcance.
  bool setTxPower(uint8_t dbm, Print *debug = nullptr);

  // No bloqueante: vacia lo que haya en el buffer y devuelve true (con outMsg
  // relleno) en cuanto encuentra una linea +RCV= completa y bien formada. Todo lo
  // demas que vea por el camino (ecos +OK sueltos, basura) se descarta.
  bool poll(LoRaMessage &outMsg);

private:
  static constexpr uint16_t kLineMax = 260;  // "+RCV=" + dir + long + 240 B de datos + rssi + snr
  static constexpr uint32_t kSendTimeoutMs = 2000;

  HardwareSerial &_serial;
  uint8_t _rxPin, _txPin;
  uint32_t _baud;

  // Linea en construccion para poll(): los bytes llegan sueltos y se juntan aqui
  // hasta ver el fin de linea.
  char _line[kLineMax + 1] = {0};
  uint16_t _lineLen = 0;

  // Un paquete que llego mientras se esperaba el +OK de un comando. Se guarda
  // aqui para que el siguiente poll() lo entregue en vez de perderlo.
  LoRaMessage _pending;
  bool _hasPending = false;

  bool sendATCommand(const char *cmd, Print *debug, uint32_t timeoutMs = 1000);
  bool waitForOK(uint32_t timeoutMs, Print *debug);
  static bool tryParseLine(char *line, uint16_t len, LoRaMessage &outMsg);
};

#include "RYLR998.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Ajusta la potencia de transmision (0..22 dBm). Ver la explicacion en RYLR998.h.
bool RYLR998::setTxPower(uint8_t dbm, Print *debug) {
  if (dbm > 22) dbm = 22;
  char cmd[24];
  snprintf(cmd, sizeof(cmd), "AT+CRFOP=%u", dbm);
  return sendATCommand(cmd, debug);
}

// Abre el UART y configura la radio paso a paso. Cada comando tiene que contestar
// +OK; al primer fallo se abandona y se devuelve false.
bool RYLR998::begin(uint16_t addr, uint16_t networkId, uint32_t bandHz, const RYLR998Params &params,
                     Print *debug) {
  _serial.begin(_baud, SERIAL_8N1, _rxPin, _txPin);

  char cmd[48];

  // Contesta el modulo? Si esto falla, es cableado, velocidad o alimentacion.
  if (!sendATCommand("AT", debug)) return false;

  // Direccion propia de esta radio. Tiene que ser distinta en cada nodo.
  snprintf(cmd, sizeof(cmd), "AT+ADDRESS=%u", addr);
  if (!sendATCommand(cmd, debug)) return false;

  // Red: solo se oyen radios con el mismo NETWORKID.
  snprintf(cmd, sizeof(cmd), "AT+NETWORKID=%u", networkId);
  if (!sendATCommand(cmd, debug)) return false;

  // Frecuencia en Hz (868 MHz en Europa).
  snprintf(cmd, sizeof(cmd), "AT+BAND=%lu", (unsigned long)bandHz);
  if (!sendATCommand(cmd, debug)) return false;

  // Parametros de modulacion. Emisor y receptor tienen que coincidir.
  snprintf(cmd, sizeof(cmd), "AT+PARAMETER=%u,%u,%u,%u", params.spreadingFactor, params.bandwidth,
           params.codingRate, params.preamble);
  if (!sendATCommand(cmd, debug)) return false;

  return true;
}

// Envia un paquete: cabecera "AT+SEND=<destino>,<longitud>," + los datos tal cual
// + fin de linea, y espera el +OK del modulo.
bool RYLR998::send(uint16_t destAddr, const char *data, uint8_t len) {
  char header[24];
  snprintf(header, sizeof(header), "AT+SEND=%u,%u,", destAddr, len);
  _serial.print(header);
  _serial.write(reinterpret_cast<const uint8_t *>(data), len);
  _serial.print("\r\n");
  return waitForOK(kSendTimeoutMs, nullptr);
}

// Manda un comando AT (opcionalmente mostrandolo por `debug`) y espera +OK.
bool RYLR998::sendATCommand(const char *cmd, Print *debug, uint32_t timeoutMs) {
  if (debug) {
    debug->print("  -> ");
    debug->println(cmd);
  }
  _serial.print(cmd);
  _serial.print("\r\n");
  bool ok = waitForOK(timeoutMs, debug);
  if (debug && !ok) debug->println("     (no +OK - check wiring/baud or that value is accepted)");
  return ok;
}

// Lee lineas del modulo hasta ver +OK (true), +ERR (false) o agotar el tiempo
// (false).
bool RYLR998::waitForOK(uint32_t timeoutMs, Print *debug) {
  uint32_t start = millis();
  char line[kLineMax + 1];
  uint16_t lineLen = 0;

  while (millis() - start < timeoutMs) {
    while (_serial.available()) {
      char c = _serial.read();

      if (c == '\n') {
        if (lineLen > 0) {
          line[lineLen] = '\0';
          if (debug) {
            debug->print("  <- ");
            debug->println(line);
          }
          if (strncmp(line, "+OK", 3) == 0) return true;
          if (strncmp(line, "+ERR", 4) == 0) return false;
          // Puede ser un +RCV= de un paquete que llego justo mientras se esperaba
          // la respuesta a este comando. Se guarda para poll() en vez de tragarse
          // en silencio un paquete real.
          if (!_hasPending && tryParseLine(line, lineLen, _pending)) _hasPending = true;
        }
        lineLen = 0;
      } else if (c != '\r') {
        if (lineLen < kLineMax) {
          line[lineLen++] = c;
        } else {
          lineLen = 0;  // linea demasiado larga: es basura, se descarta
        }
      }
    }
  }
  return false;
}

// No bloqueante. Primero entrega el paquete guardado por waitForOK(), si lo hay;
// despues procesa solo los bytes que ya esten en el buffer del UART.
bool RYLR998::poll(LoRaMessage &outMsg) {
  if (_hasPending) {
    outMsg = _pending;
    _hasPending = false;
    return true;
  }

  while (_serial.available()) {
    char c = _serial.read();

    if (c == '\n') {
      bool matched = false;
      if (_lineLen > 0) {
        _line[_lineLen] = '\0';
        matched = tryParseLine(_line, _lineLen, outMsg);
      }
      _lineLen = 0;
      if (matched) return true;
    } else if (c != '\r') {
      if (_lineLen < kLineMax) {
        _line[_lineLen++] = c;
      } else {
        _lineLen = 0;
      }
    }
  }
  return false;
}

// Interpreta "+RCV=<direccion>,<longitud>,<datos>,<rssi>,<snr>".
//
// Se guia por la LONGITUD declarada en vez de partir ingenuamente por comas, para
// que una coma dentro de los datos (las tramas de telemetria estan llenas de ellas)
// no descoloque los campos que vienen detras.
//
// Lo que NO resiste: un salto de linea dentro de los datos, porque poll() y
// waitForOK() cortan las lineas por '\n' antes de llegar aqui. Ni la telemetria de
// Modulo C ni los comandos CFG/ACK llevan ninguno, pero no esta protegido.
bool RYLR998::tryParseLine(char *line, uint16_t len, LoRaMessage &outMsg) {
  if (len < 7 || memcmp(line, "+RCV=", 5) != 0) return false;

  // Direccion del emisor.
  char *p = line + 5;
  char *comma1 = strchr(p, ',');
  if (!comma1) return false;
  *comma1 = '\0';
  int addr = atoi(p);

  // Longitud de los datos.
  char *lenStart = comma1 + 1;
  char *comma2 = strchr(lenStart, ',');
  if (!comma2) return false;
  *comma2 = '\0';
  int dataLen = atoi(lenStart);
  if (dataLen < 0 || dataLen > (int)(sizeof(outMsg.payload) - 1)) return false;

  // Los datos: exactamente dataLen bytes, seguidos de una coma.
  char *dataStart = comma2 + 1;
  if ((dataStart - line) + dataLen + 1 > len) return false;
  if (dataStart[dataLen] != ',') return false;

  // RSSI y SNR, despues de los datos.
  char *rssiStart = dataStart + dataLen + 1;
  char *comma3 = strchr(rssiStart, ',');
  if (!comma3) return false;
  *comma3 = '\0';
  int rssi = atoi(rssiStart);

  char *snrStart = comma3 + 1;
  int snr = atoi(snrStart);

  memcpy(outMsg.payload, dataStart, dataLen);
  outMsg.payload[dataLen] = '\0';
  outMsg.length = (uint8_t)dataLen;
  outMsg.senderAddr = (uint16_t)addr;
  outMsg.rssi = (int16_t)rssi;
  outMsg.snr = (int8_t)snr;
  return true;
}
