/*
 * =============================================================================
 * CHIP FOREST - MODULO C (NODO SENSOR) - FIRMWARE DE PRODUCCION
 * =============================================================================
 *
 * Que hace este programa:
 *   Un nodo de deteccion de incendios forestales. Mide cinco sensores, decide si
 *   hay riesgo de incendio y manda los datos por radio LoRa a Modulo B, que los
 *   sube a ThingsBoard. Funciona con bateria, asi que pasa casi todo el tiempo en
 *   sueno profundo.
 *
 *   Sensores:  BME690 (temperatura, humedad, presion, gas)   BMV080 (particulas)
 *              CM1106 (CO2)   SEN0466 (CO)   Calypso (viento)
 *   Radio:     RYLR998, LoRa 868 MHz, hacia Modulo B (direccion 2)
 *
 * Como esta organizado el archivo (de arriba abajo):
 *   1. Configuracion: pines, parametros de radio, tiempos, umbrales.
 *   2. LED de estado.
 *   3. Memoria que sobrevive al sueno (RTC) y el RELOJ del nodo.
 *   4. El CALENDARIO: cada sensor tiene su periodo, por estado de alarma.
 *   5. Comandos que llegan por radio (CFG,...).
 *   6. Arranque: guardia de arranques (modo seguro), rails, I2C, setup().
 *   7. Sueno profundo.
 *   8. Lectura de sensores, de uno en uno ("slots").
 *   9. Almacen de lecturas y ALARMA (NORMAL / PRE-ALARMA / ALARMA).
 *  10. Transmision a Modulo B y paquete de estado (STAT).
 *  11. Bateria e hibernacion.
 *  12. loop().
 *
 * Ciclo normal, en una frase: cada despertar es un arranque nuevo del ESP32;
 * setup() mira en el calendario que trabajos vencen, enciende solo lo necesario,
 * lee esos sensores de uno en uno, evalua la alarma, transmite si toca y vuelve a
 * dormir hasta el siguiente vencimiento. En PRE-ALARMA y ALARMA el nodo no duerme.
 *
 * ---- Formato de la telemetria (debe coincidir con el TelemetryParser de B) ----
 *   temp,hum,pres,gas,pm1,pm25,pm10,co2,co,coTemp,windAngle,windSpeed,windValid
 * Una sola lectura va con ese formato tal cual. Varias van en LOTE:
 *   B,<cuantas>,<lectura>;<lectura>;...   (cada lectura + su edad en segundos)
 * Ver src/module_b/communications/telemetry/TelemetryParser.h. Un campo que esa
 * lectura no midio va como -999 y Modulo B no lo publica.
 *
 * ---- Parametros LoRa (SF7/BW7/CR1/preambulo 12) ----
 * SF7 (no el "7,7,1,4" generico, que este modulo rechazo con +ERR=18: el preambulo
 * tenia que ser 12). A SF7/BW125/CR4-5/preambulo 12, un paquete de ~75 bytes son
 * ~137 ms en el aire. El ciclo de trabajo del 1% de EU868 permite ~36 s por hora; en
 * estado normal se transmite una vez cada 30 minutos, muy por debajo del limite.
 * SF7 cuesta ~5-6 dB de enlace frente a SF9 (aproximadamente la mitad de alcance,
 * peor a traves de vegetacion); ninguna prueba de alcance en campo ha confirmado aun
 * que el despliegue funcione a SF7. TIENE que coincidir con LORA_PARAM_SF del
 * Config.h de Modulo B.
 *
 * ---- Comandos que puede recibir el nodo (Modulo B -> este nodo) ----
 * Solo se oyen en la ventana de LORA_POST_TX_LISTEN_MS que sigue a cada envio: es
 * el UNICO momento en que el nodo escucha. Formato:
 *   CFG,<clave>=<valor>[,<clave>=<valor>...]
 * Claves: TIME (hora de red), INTERVAL (periodo de envio), NXT_LECTURA y NXT_ENVIO
 * (multiplicadores de los periodos), ALARM (bajar la alarma a mano), BME690 /
 * SEN0466 / BMV080 / CM1106 / CALYPSO (activar o silenciar un sensor, 0/1), y los
 * umbrales de alarma por nombre. Las claves desconocidas se ignoran.
 * Solo se aceptan comandos de LORA_RX_ADDR (Modulo B): es un filtro por direccion
 * del emisor, NO una autenticacion - el sistema no tiene autenticacion en este
 * enlace. Si se aplica algo, el nodo contesta ACK,<clave>=<valor>,... con los
 * valores realmente aplicados (tras recortarlos a sus limites), para que el
 * recorte sea visible. Un comando que no se entiende o no aplica nada no recibe
 * ACK: no se gasta tiempo de aire contestando ruido.
 */

#define LORA_TX_PIN 5  // RXD del modulo <- TX del ESP

// ---------- Parametros del enlace LoRa - TIENEN que coincidir con el Config.h de Modulo B ----------
#define LORA_BAND_HZ 868000000UL  // EU868, despliegue en Espana (confirmado)
#define LORA_NETWORK_ID 5
// SF7 (antes SF9): reduce a la mitad el tiempo en el aire. Cuesta ~5-6 dB de enlace
// frente a SF9. TIENE que coincidir exactamente con LORA_PARAM_SF del Config.h de
// Modulo B.
#define LORA_PARAM_SF 7
#define LORA_PARAM_BW 7
#define LORA_PARAM_CR 1
#define LORA_PARAM_PREAMBLE 12
// Potencia de transmision, dBm (AT+CRFOP, 0..22). Los 22 dBm por defecto del modulo
// piden ~120 mA en la rafaga de transmision, y la alimentacion de esta placa no
// aguanta ese escalon: la placa se reiniciaba en todas las transmisiones, 3 de 3,
// justo despues de "[tx] sending". 14 dBm reduce mucho la rafaga y sigue siendo un
// enlace LoRa normal. Es una mitigacion, no una cura: el arreglo de verdad es mas
// capacidad en el rail (ver la nota en rylr998_bridge.cpp). Subirla cuando la
// alimentacion lo aguante.
#define LORA_TX_POWER_DBM 14

// Direccion AT+ADDRESS de este nodo: el numero que Modulo B ve como
// "+RCV addr=<n>", y lo UNICO que distingue a un nodo de otro en el aire. Todos los
// nodos salieron de fabrica con 1, asi que Modulo B no podia distinguirlos y las
// lecturas de dos nodos se mezclaban bajo una misma identidad.
//
// Se puede cambiar en cada compilacion para que un mismo firmware sirva a todos los
// nodos: ver los entornos env:module-c-node* de platformio.ini. La direccion 2 es
// Modulo B y no la puede usar nunca un nodo. Mantenerlas al dia con
// NBIOT_NODE_NAMES del Config.h de Modulo B, y no renumerar nunca un nodo que
// ThingsBoard ya conoce.
//
//   1 = C1 (NodoC-1)   2 = Modulo B (receptor)   3 = C2   4 = C3
#ifndef LORA_MY_ADDR
#define LORA_MY_ADDR 1
#endif
#define LORA_RX_ADDR 2  // direccion AT+ADDRESS de Modulo B

#define LORA_TX_PERIOD_MIN_MS 15000UL   // suelo de seguridad: CFG,INTERVAL no puede bajar de aqui
#define LORA_TX_PERIOD_MAX_MS 86400000UL  // techo de 24 h: protege de un CFG mal tecleado
#define FIRST_TX_DELAY_MS 30000UL       // HISTORICO: ya no se usa en ningun sitio

#define LORA_TX_MIN_GAP_MS 15000UL      // HISTORICO: ya no se usa

//
// CICLO DE TRABAJO, DICHO CLARAMENTE: a SF7 un paquete de ~116 bytes son ~0,2 s en
// el aire. Cada 5 s serian un ~4%, POR ENCIMA del limite del 1% de EU868. Era
// deliberado y acotado (solo mientras un nodo informa de un incendio).
// ESTADO ACTUAL: esta constante ya no se usa; en alarma el envio lo marca el
// calendario (JOB_TX: 15 min en pre-alarma, 3 min en alarma), muy por debajo del
// limite.
#define ALARM_TX_MIN_GAP_MS 5000UL

// ---------- Encendido por turnos del BMV080 ----------
// El laser consume ~68 mA, con diferencia la mayor carga del sistema. Un firmware
// anterior lo tenia encendido sin parar (unas 90 veces su presupuesto de energia) y
// la placa se reiniciaba en cada arranque en que estaba midiendo. Ahora solo se

#define BMV080_MEASURE_TIMEOUT_MS 30000UL  // dejar de esperar un dato pasado este tiempo
#define BMV080_SETTLE_MS 2000UL            // precalentamiento del laser antes de que los datos valgan

// ---------- Umbrales de deteccion de incendio ----------

// Las cifras reales de la hoja de alarma/prealarma del proyecto se transcribieron

// Lecturas seguidas con todas las reglas por debajo antes de bajar de nivel. Mas de
// una, para que un solo descenso no termine una alarma en un incendio real.
#define ALARM_CLEAR_CONSECUTIVE 5

// ---- VENTANA DE CONFIRMACION DE LA PRE-ALARMA ----
//
// La pre-alarma es una pregunta, no un veredicto. El nodo la mantiene este tiempo y
// entonces decide: lo que siga por encima de su umbral cuando se acabe la ventana
// sube a ALARMA, porque una condicion que dura una hora no es ruido del sensor. Lo
// que vuelva por debajo antes baja a NORMAL por el camino habitual de lecturas
// limpias.
//
// Una lectura de nivel ALARMA sigue subiendo AL MOMENTO y no espera a esto. Esperar
// una hora con 60 C seria absurdo: la ventana existe para confirmar una senal
// dudosa, no para retrasar una evidente.
//
// 60 MINUTOS, subido desde 10 el 2026-10-14 (decision del usuario). El coste es
// energia: en pre-alarma el nodo no duerme, asi que ahora puede pasar seis veces mas
// tiempo despierto con una senal que nunca llega al umbral de alarma.
//
// Se puede cambiar desde platformio.ini (-DPREALARM_CONFIRM_MS=...) para los builds
// de medicion, que necesitan otra duracion para poder medir intervalos.
#ifndef PREALARM_CONFIRM_MS
#define PREALARM_CONFIRM_MS 3600000UL  // 60 minutes
#endif

// ---------------------------------------------------------------------------
// SOLO PARA BUILDS DE MEDICION - QUITAR ANTES DEL DESPLIEGUE
//
// Fija el estado de alarma para poder medir con el PPK2 cada uno de los tres
// perfiles de consumo el tiempo que haga falta, sin esperar a que haya humo real:
//
//   0  sin forzar - firmware normal, la maquina de estados funciona como esta disenada
//   1  PRE-ALARMA fija
//   2  ALARMA fija
//
// Forzado, updateAlarmState() vuelve en el acto: el estado no puede ni bajar ni
// subir, asi que una PRE-ALARMA fija sigue en pre-alarma en vez de subir a alarma al
// acabar la ventana de confirmacion.
//
// POR QUE ES PELIGROSO DEJARLO. Un nodo compilado con 1 o 2 esta SORDO A SUS PROPIOS
// SENSORES: informa de un incendio que no existe y, peor, nunca podria informar de
// uno real, porque el estado esta congelado. Ademas nunca duerme, asi que con
// bateria se agota en horas. Cada arranque imprime un aviso y loop() lo repite,
// precisamente para que un build de prueba no se confunda con uno real. Los
// entornos que lo activan son los medir-* de platformio.ini.
#ifndef FORCE_ALARM_STATE
#define FORCE_ALARM_STATE 0
#endif

// El LED de estado falsea las medidas de consumo en los estados altos.

// ledOff() solo se ejecuta al entrar en sueno profundo, y los estados altos nunca

//
// Conviene saberlo mas alla de la prueba: es el comportamiento REAL. Un nodo de
// verdad en alarma deja el LED encendido indefinidamente, que es corriente
// desperdiciada justo cuando mas importa la bateria.
#ifndef FORCE_LED_OFF
#define FORCE_LED_OFF 0
#endif

// =============================================================================
// 2. LED DE ESTADO
// =============================================================================
//
// LED direccionable WS2812 en PIN_STATUS_LED (GPIO21); ver pins.h para saber por
// que es ese pin y no el PIN_NEOPIXEL 48 de la variante.
//
// Colores, para leer el comportamiento del nodo desde el otro lado de la sala:
//
//   VERDE (onda)  en cada despertar del sueno profundo. Prueba de que el
//                 temporizador salto y el nodo volvio: si falta, el nodo no
//                 despierta.
//   AZUL          mantenido durante una lectura de sensores.
//   BLANCO        mantenido durante una transmision.
//   ROJO          algo va mal y el nodo no lo puede arreglar solo (un sensor que se
//                 esperaba y no contesto, o modo seguro).
//   APAGADO       dormido.
//
// EL BRILLO ES UNA DECISION DE CONSUMO, no estetica.
//
// Estos valores eran 120 y 120/80, descritos en un comentario como "un pequeno
// porcentaje del maximo". Era falso: 120/255 es casi la mitad, en dos canales a la
// vez. Paso desapercibido tanto tiempo porque el LED NO SE ENCENDIA: el orden de
// colores era incorrecto (ver ledWrite), asi que el pixel recibia una trama que
// nunca aceptaba y apenas consumia nada.
//
// En cuanto se arreglo el orden de colores y el LED se encendio de verdad, la placa
// empezo a sufrir reinicios POWERON: un colapso real de la alimentacion, no un fallo
// de software. Un WS2812 en blanco al maximo son ~60 mA, comparable al laser del
// BMV080, y el color de trabajo se mantenia durante toda la lectura incluida la
// rafaga de ~120 mA de la radio. Es justo la carga simultanea que los turnos de este
// firmware existen para evitar.
//
// Asi que: lo bastante bajo para ser barato y lo bastante alto para verse desde
// lejos. Un WS2812 es muy brillante: 30/255 se ve de sobra en interior y cuesta unos
// pocos mA.
#define LED_GREEN_R 0
#define LED_GREEN_G 30
#define LED_GREEN_B 0
#define LED_RED_R 30
#define LED_RED_G 0
#define LED_RED_B 0
#define LED_BLUE_R 0
#define LED_BLUE_G 0
#define LED_BLUE_B 30
// Blanco = transmitiendo. Los tres canales, asi que consume unas tres veces lo que
// un color de un solo canal con el mismo valor; por eso se deja en 30. Es seguro
// porque el LED se apaga durante la propia rafaga de la radio (ver transmitStore),
// asi que nunca esta encendido en el momento mas critico para la alimentacion.
#define LED_WHITE_R 30
#define LED_WHITE_G 30
#define LED_WHITE_B 30
#define LED_FLASH_MS 120  // HISTORICO: ya no se usa

// Forma del indicador de despertar: una rampa suave de subida y bajada en vez de un
// parpadeo seco. El brillo va en los valores RGB que se le mandan al WS2812.
//
// COSTE, porque se dispara en CADA despertar: un despertar sin trabajo era casi
// gratis, unos cientos de milisegundos despierto a ~40 mA, y una onda de 500 ms lo
// duplica aproximadamente. Merece la pena en el banco, donde lo que se quiere es VER
// trabajar al nodo; para el despliegue, bajar LED_WAVE_MS.
#define LED_WAVE_MS 500
#define LED_WAVE_STEPS 24

// Este LED es NEO_RGB, NO el NEO_GRB estandar del WS2812B que neopixelWrite() lleva
// fijo. Se establecio en el nodo C2 probando los seis ordenes de color y pidiendo
// verde puro a cada uno: solo NEO_RGB mostro verde.
//
// Esa diferencia es la razon de que todos los colores salieran mal, y de que no se
// pudiera arreglar intercambiando canales: neopixelWrite() pone el byte del verde
// primero en el cable y esta pieza lee el primer byte como rojo, asi que el error
// esta en la TRAMA, no en los argumentos. Un driver con orden de color explicito lo
// resuelve; no "simplificarlo" volviendo a neopixelWrite().
//
// La version antigua hacia ademas un digitalWrite() despues de escribir el pixel,
// por si el LED era uno normal y no direccionable. Ya no: el tipo es conocido, y
// mantener la linea de datos alta entre tramas estropea la temporizacion de la
// siguiente.
static Adafruit_NeoPixel g_statusLed(1, PIN_STATUS_LED, NEO_RGB + NEO_KHZ800);
static bool g_ledBegun = false;

// Pone el LED de un color. El argumento `level` es de la version antigua y ya no
// hace nada.
static void ledWrite(uint8_t r, uint8_t g, uint8_t b, bool level) {
  (void)level;  // argumento heredado del LED normal, ya sin significado
#if FORCE_LED_OFF
  // Forzado a oscuras para una medida de consumo - ver FORCE_LED_OFF. Se hace AQUI
  // y no en cada llamada, para que ningun color se escape por un camino olvidado.
  r = g = b = 0;
#endif
  if (!g_ledBegun) {
    g_statusLed.begin();
    g_ledBegun = true;
  }
  g_statusLed.setPixelColor(0, g_statusLed.Color(r, g, b));
  g_statusLed.show();
}

static void ledOff() {
  ledWrite(0, 0, 0, false);
}

// Una onda: subir hasta el color completo y volver a bajar. Bloqueante, como todo
// lo demas en el camino del despertar.
//
// El triangulo se calcula como un nivel 0..mitad..0 que se aplica a cada canal, asi
// que un color con un canal a cero conserva su tono durante toda la onda.
static void ledWave(uint8_t r, uint8_t g, uint8_t b) {
  const int half = LED_WAVE_STEPS / 2;
  const int stepMs = LED_WAVE_MS / LED_WAVE_STEPS;
  for (int i = 0; i <= LED_WAVE_STEPS; i++) {
    int level = (i <= half) ? i : (LED_WAVE_STEPS - i);  // 0 -> mitad -> 0
    ledWrite((uint8_t)((int)r * level / half), (uint8_t)((int)g * level / half),
             (uint8_t)((int)b * level / half), level > 0);
    delay(stepMs);
  }
  ledOff();
}

// Onda verde: "me he despertado". Se dispara en cada despertar, asi que si falta es
// que el nodo no despierta.
static void ledFlashWake() {
  ledWave(LED_GREEN_R, LED_GREEN_G, LED_GREEN_B);
}

// Blanco, mantenido durante toda la transmision (no un parpadeo), para que su
// duracion signifique algo.
static void ledWorking() {
  ledWrite(LED_WHITE_R, LED_WHITE_G, LED_WHITE_B, true);
}

// Rojo: algo va mal que el nodo no puede arreglar solo - un sensor que se esperaba
// y no contesto, o modo seguro. A proposito con el MISMO brillo que los demas: es un
// cambio de color legible de lejos, no un destello para llamar la atencion.
//
// Importa sobre todo en pruebas sin cable: sin puerto serie, el LED y la radio son
// lo unico que puede decir que el nodo tiene un problema.
static void ledTrouble() {
  ledWrite(LED_RED_R, LED_RED_G, LED_RED_B, true);
}

// Azul, mantenido durante una lectura de sensores. Distinguir leer de transmitir de
// un vistazo es util: una lectura completa dura ~19 s y una transmision apenas un
// segundo.
static void ledReading() {
  ledWrite(LED_BLUE_R, LED_BLUE_G, LED_BLUE_B, true);
}

// ---------- Sueno profundo ----------
// El ESP mantiene su propio rail de 3V3 y se duerme a si mismo con un temporizador
// (~10-15 uA), conservando la memoria RTC. Cada despertar es un ciclo completo:
// encender lo necesario -> leer -> transmitir si toca -> apagar -> dormir.
//
// Con 0 el nodo no duerme: se queda despierto, lee sin parar y transmite con un
// temporizador simple. Util en el banco cuando se necesita que el puerto serie no
// desaparezca y poder mirar el LED. Se puede cambiar desde platformio.ini
// (-DSLEEP_ENABLED=0) sin tocar el codigo - ver el entorno medir-tiempos.
#ifndef SLEEP_ENABLED
#define SLEEP_ENABLED 1
#endif

// Cadencia de envio cuando SLEEP_ENABLED vale 0 (sin sueno no hay calendario que la
// marque).
#define NOSLEEP_TX_GAP_MS 15000UL
// SUELO DEL SUENO, y nada mas. Ya no es "un tick": el trabajo se programa por
// TIEMPO - cada sensor con su periodo en minutos - y la duracion de cada sueno la
// calcula schedSleepSeconds() segun lo que falte hasta el proximo vencimiento.
//
// Sobrevive solo como valor por defecto del argumento de enterDeepSleep(), para
// las llamadas defensivas que no tienen un vencimiento que consultar.
#define SLEEP_CYCLE_SECONDS 10UL
// Se encendieron los rails de sensores en este despertar?
//
// Se lleva con una variable en vez de leer los pines: son salidas, y en un despertar
// sin trabajo o de hibernacion nada las ha conducido desde que el sueno las enclavo
// apagadas, asi que leer el pin describiria el enclavamiento, no este arranque.
static bool g_railsUp = false;
static bool railsArePowered() { return g_railsUp; }


// Por debajo de esto la lectura no es una bateria vacia: es que falta el divisor.
//
// La placa no puede funcionar con una tension tan baja (el rail cae hacia 1,9 V),
// asi que un nodo que esta transmitiendo este paquete mientras dice 0,5 V esta
// hablando de su cableado, no de su energia. Una placa antigua sin divisor flota en
// ~100 mV, que a traves del factor son ~0,12 V, muy por debajo de esto.
#define SUPERCAP_V_SANITY_FLOOR 0.8f

// ---- HIBERNACION POR BATERIA BAJA ----
//
// Por debajo de HIB_ENTER_V el nodo deja de ser un nodo sensor y pasa a ser un
// voltimetro que duerme: sin lecturas, los dos rails apagados, la radio apagada y
// en silencio. Solo despierta para mirar la bateria, y vuelve a funcionar cuando la
// carga la ha subido hasta HIB_EXIT_V.
//
// LO QUE SE PROTEGE SON LAS CELDAS, NO EL TIEMPO EN MARCHA. Un pack de
// supercondensadores de 3,8 V como este se dana si se descarga a fondo, asi que la
// hibernacion es una parada de proteccion: mejor un nodo que se para y vuelve que
// uno que agota su pack y obliga a cambiar el hardware.
// OJO: estos valores son de la epoca de los supercondensadores. Desde el
// 2026-09-24 la fuente es una bateria de litio, y no se han recalculado.
//
// La separacion entre los dos umbrales es histeresis y no es opcional. Sin ella, un
// pack justo en el limite volveria a funcionar, consumiria, bajaria del limite,
// hibernaria, se recuperaria... oscilando justo cuando menos energia tiene que
// gastar. 0,3 V es suficiente para que solo una carga de verdad lo cruce.
#define HIB_ENTER_V 2.2f
#define HIB_EXIT_V 2.5f

// Cada cuanto despierta la hibernacion para mirar la bateria.
//
// El ESP32-S3 no tiene despertar analogico por umbral - nada puede interrumpir al
// chip cuando una tension cruza un nivel - asi que "esperar a que se cargue" tiene
// que ser preguntar de vez en cuando. Una comprobacion es un arranque, una lectura
// del ADC y vuelta a dormir sin ningun rail encendido: unos 400 ms. Cada 10 minutos
// son unos 0,8 mAh al dia.
//
// Mas tiempo ahorra poco y cuesta respuesta: el nodo seguiria hibernando durante la
// primera parte de una manana soleada con el pack ya lleno.
#define HIB_CHECK_SECONDS 600UL

// Un STAT al entrar en hibernacion, antes de quedarse callado.
//
// La hibernacion en si no transmite NADA: esa es la idea. Pero un nodo que
// simplemente se calla, sin ultima palabra, no se distingue en Modulo A de un nodo
// que ha muerto, y distinguirlos cuesta que alguien se meta en el bosque. Un paquete
// al apagarse convierte un misterio en un estado.
//
// Con 0, silencio total desde el momento en que se cruza el umbral.
#define HIB_ANNOUNCE_ENTRY 1

// Cuanto esperar antes de dormir para que el PC termine de leer el buffer USB.
//
// Serial.flush() NO basta en el USB-Serial/JTAG del ESP32-S3: solo espera a que el
// buffer pase a la pila USB, y si el PC no esta leyendo en ese momento los bytes se
// quedan en la FIFO del endpoint y mueren con el periferico. Con 50 ms se perdia
// toda la cola de cada lectura - [read], [store] y [sleep] desaparecian mientras
// las lineas anteriores sobrevivian -, que es como se ve una perdida de cola frente
// a un cuelgue.
//
// En el campo no hay ningun PC leyendo el buffer, asi que es tiempo despierto puro en
// cada despertar: se bajo a 60 ms para funcionar sin cable. Subirlo para depurar
// con cable.
#define SLEEP_USB_DRAIN_MS 60

// ---------- Planificacion por ticks (HISTORICO) ----------
// El firmware anterior funcionaba por "ticks": cada despertar era un tick, y unos
// contadores decidian cada cuantos ticks se leia y se transmitia (SENSOR_READ,
// LORA_TRANS, BURST). Lo SUSTITUYO el calendario por tiempo (seccion 4, kJobs[]),
// porque un tick no medio nunca tiempo real: duraba 17,8-18,8 s en vez de 10.
//
// Las constantes de abajo quedan de esa epoca. BURST_EVERY_DEFAULT,
// SENSOR_READ_EVERY_DEFAULT, LORA_TRANS_EVERY_DEFAULT, COUNTER_EVERY_MIN/MAX,
// READINGS_PER_PACKET_MAX, BATCH_READING_WORST_BYTES y readingsPerPacket() ya no
// se usan en ningun sitio.
#ifndef BURST_EVERY_DEFAULT
#define BURST_EVERY_DEFAULT 90   // ticks; 90 x 10s nominal = 15 min, 26.7 min real
#endif

#define SENSOR_READ_EVERY_DEFAULT 3
#define LORA_TRANS_EVERY_DEFAULT 4
#define COUNTER_EVERY_MIN 1
#define COUNTER_EVERY_MAX 360           // 1 hour at a 10s tick

// El RYLR998 admite como mucho 240 bytes por paquete. Una lectura en formato de lote
// ocupa ~56 bytes normalmente y hasta ~70 en el peor caso.
#define READINGS_PER_PACKET_MAX 3
#define BATCH_READING_WORST_BYTES 70

// CUANTAS LECTURAS CABEN ESPERANDO AL PROXIMO ENVIO.
//
// Valia READINGS_PER_PACKET_MAX + 1 = 4, dimensionado para el planificador por
// ticks, donde la relacion entre leer y transmitir se validaba para que un lote
// cupiera siempre en UN paquete.
//
// Con los periodos en tiempo real esa cuenta ya no sale, y el 2026-10-09 se vio
// perdiendo datos en el nodo: el log decia "[store] reading 4/4 kept" y aun
// faltaban dos lecturas para el envio.
//
//   estado      lectura mas rapida   envio    lecturas acumuladas
//   NORMAL          5 min            30 min          6     <- no caben en 4
//   PREALARMA       3 min            15 min          5     <- tampoco
//   ALARMA          1 min             3 min          3
//
// Ocho cubre el peor caso con margen. NO hay que confundirlo con el limite del
// paquete: transmitStore() ya parte el lote en varios paquetes cuando no cabe en
// los 240 bytes del RYLR998, asi que almacenar mas no obliga a enviar mas en uno.
//
// El margen importa porque los periodos son modificables en caliente desde Modulo
// A: NXT_LECTURA y NXT_ENVIO pueden empeorar la relacion sin que nadie recompile.
// g_storeDropped cuenta lo que se pierda si aun asi se desborda, y el envio lo
// reporta - esa es la red de seguridad, no este numero.
#define READING_STORE_MAX 8

// HISTORICO (planificador por ticks): lecturas que acumularia un par de contadores
// entre envios. Ya no se llama desde ningun sitio.
static uint16_t readingsPerPacket(uint16_t sensorReadEvery, uint16_t loraTransEvery) {
  if (sensorReadEvery == 0) return 0xFFFF;
  return (uint16_t)((loraTransEvery + sensorReadEvery - 1) / sensorReadEvery);
}

// =============================================================================
// 3. MEMORIA QUE SOBREVIVE AL SUENO (RTC_NOINIT) Y EL RELOJ DEL NODO
// =============================================================================
//
// RTC_NOINIT_ATTR, a proposito y NO RTC_DATA_ATTR: esta ultima vive en .rtc.data,
// que se recarga desde la imagen de la flash en cada reinicio, asi que un contador
// ahi valdria 1 siempre y no probaria nada. NOINIT sobrevive al sueno y a cualquier
// reinicio que mantenga alimentado el dominio RTC, y solo se pierde con un corte de
// alimentacion de verdad. Al encender en frio contiene basura, y por eso existe la
// palabra magica de abajo.
//
// SUBIR ESTE NUMERO CADA VEZ QUE SE ANADE ESTADO EN RTC_NOINIT. No es opcional.
//
// La memoria RTC_NOINIT sobrevive al RE-FLASHEO, no solo al sueno. Asi que un
// firmware nuevo con variables nuevas arranca encontrando el magic ANTERIOR
// intacto, se salta el bloque de siembra entero, y esas variables se quedan con
// la basura que hubiera en esas direcciones.
//
// Visto en vivo el 2026-10-08 con este mismo cambio: g_clockDrift, los
// multiplicadores y el calendario quedaron sin sembrar. El sueno se divide por la
// deriva, asi que una deriva basura da un sueno arbitrario, y el planificador
// nunca llego a re-fasear - en el log no aparecia ni una linea [sched].
// (Antes ya habia costado tres sesiones de depuracion: g_hibernating con basura
// podia hacer hibernar al nodo con la bateria llena.)
//
// 0xC0FFEE0A: StoredReading.tickAge pasa a takenS. MISMO TAMANO, OTRO SIGNIFICADO:
//             antes era una cuenta de despertares y ahora son segundos del reloj
//             propio del nodo. Como el tamano no cambia, la RTC sobrevivia al
//             reflasheo y el almacen arrastraba lecturas viejas que se leian con
//             el significado nuevo - visto en vivo: edades de 26906 s para
//             lecturas de hace un minuto. Hay que subirlo tambien cuando cambia lo
//             que un campo SIGNIFICA, no solo cuando cambia la disposicion.
// 0xC0FFEE09: READING_STORE_MAX de 4 a 8, y fuera los contadores de tick. Las dos
//             cosas mueven las direcciones de todo lo que viene despues.
// 0xC0FFEE08: planificador por tiempo (g_nextDueMs, multiplicadores,
//             g_schedPhasedForState) y reloj (g_nodeBaseMs, g_clockDrift, ...)
// 0xC0FFEE07: g_thr[][] sembrado desde kAlarmRules
#define BOOTCOUNT_MAGIC 0xC0FFEE0AUL
RTC_NOINIT_ATTR uint32_t g_bootMagic;
// Arranques desde la ultima perdida de alimentacion. Un valor > 1 dejando la placa
// sola prueba que se reinicia por si misma, y no porque el PC toque DTR/RTS por USB:
// distingue un fallo de alimentacion de un efecto de las herramientas.
RTC_NOINIT_ATTR uint32_t g_bootCount;

// Cuenta de despertares. Sobrevive al sueno: permite que cada despertar diga en que
// punto de la serie esta.
RTC_NOINIT_ATTR uint32_t g_wakeCount;

// Se pone justo antes de esp_deep_sleep_start() y se borra en el arranque
// siguiente. ES ESTO lo que identifica un despertar, NO
// esp_sleep_get_wakeup_cause(): un reinicio del chip borra la causa, y en el banco
// un reinicio es lo normal (al despertar cae el USB, el PC lo vuelve a enumerar y
// abrir el puerto reinicia el ESP32-S3). Fiarse de la causa hacia que cada
// despertar pareciera un arranque en frio, asi que el nodo se quedaba 30 s despierto
// cada vez y nunca dormia de verdad. Visto en el nodo C2, 2026-09-08. Esta marca
// sobrevive a ese reinicio porque RTC_NOINIT lo hace.
RTC_NOINIT_ATTR uint32_t g_sleepFlag;
#define SLEEP_FLAG_MAGIC 0x5EEDBEEFUL

// True mientras el nodo esta en hibernacion por bateria baja. RTC_NOINIT para que
// sobreviva al sueno entre comprobaciones: la maquina de estados entera es este bool
// mas el ADC.
RTC_NOINIT_ATTR bool g_hibernating;
// El periodo de envio ajustado por CFG,INTERVAL. Sin guardarlo en RTC volveria al
// valor compilado en cada despertar, y pareceria que el comando se ignora.
RTC_NOINIT_ATTR uint32_t g_txPeriodMsPersist;

// ---------------------------------------------------------------------------
// RELOJ DEL NODO
// ---------------------------------------------------------------------------
//
// EL PROBLEMA. El nodo no tiene reloj. Durmiendo, el tiempo lo cuenta el
// oscilador RC interno del ESP32, que tiene entre un 1% y un 2% de error y ademas
// deriva con la temperatura - que en un bosque cambia 20 grados entre el dia y la
// noche. Sobre un periodo de 30 minutos eso son 18-36 segundos, y lo peor es que
// SE ACUMULA: si cada sueno se pasa 18 s, al decimo envio llevas 3 minutos de
// retraso y sigue creciendo. Una estrategia que dice "mide cada 5 minutos" no
// significa nada si nadie sabe cuanto dura un minuto.
//
// LA SOLUCION. Modulo B si tiene la hora de red (AT+CCLK? por cada enganche,
// `netNowMs()`), y el nodo ya abre una ventana de escucha de 2 s despues de cada
// transmision por donde recibe comandos CFG. Asi que la sincronizacion no
// necesita protocolo nuevo: es una clave CFG mas, `CFG,TIME=<epoca en SEGUNDOS>`.
//
// DOS RELOJES, Y HAY QUE NO CONFUNDIRLOS:
//
//   * nodeNowMs()  - monotono, propio del nodo, empieza en 0 al energizar y
//                    sobrevive al sueno. SIEMPRE existe. Es con lo que se
//                    planifica, porque no depende de haber hablado con nadie.
//   * wallNowMs()  - hora real, epoca en ms. Solo existe tras la primera
//                    sincronizacion. Es con lo que se FECHAN las lecturas.
//
// `millis()` se reinicia en cada despertar, asi que el tiempo del nodo se
// acumula explicitamente en RTC: antes de dormir se suma lo que se estuvo
// despierto mas lo que se va a dormir. La diferencia entre lo que se pidio
// dormir y lo que de verdad se durmio ES el error del RC, y es justo lo que
// corrige el factor de deriva.
RTC_NOINIT_ATTR uint64_t g_nodeBaseMs;     // tiempo del nodo acumulado ANTES de este arranque
RTC_NOINIT_ATTR uint64_t g_epochMsAtSync;  // epoca real recibida en la ultima sincronizacion
RTC_NOINIT_ATTR uint64_t g_nodeMsAtSync;   // tiempo del nodo en ese mismo instante
RTC_NOINIT_ATTR uint8_t g_clockSynced;     // 0 = nunca sincronizado; planifica relativo
RTC_NOINIT_ATTR uint32_t g_clockSyncCount; // cuantas veces, para el diagnostico

// Cuantos milisegundos REALES vale un milisegundo contado por el nodo. 1.0 = el
// oscilador es perfecto; 1.013 = va un 1,3% lento y hay que dormir menos.
//
// Es el refinamiento que de verdad mata la deriva, y sale gratis: en cada
// sincronizacion el nodo compara cuanto tiempo cree el que ha pasado contra
// cuanto dice la red, y se corrige. Tras un par de sincronizaciones incluso los
// intervalos ENTRE sincronizaciones son precisos, sin tocar hardware.
RTC_NOINIT_ATTR float g_clockDrift;

// Fuera de estos limites el factor no se cree: un RC puede desviarse un 2%, no un
// 20%. Un valor absurdo viene de una hora de red mala o de un desbordamiento, y
// aplicarlo dejaria al nodo midiendo a un ritmo arbitrario sin forma de volver.
// UN OSCILADOR RC SE DESVIA UN 1-2%, NO UN 10%.
//
// La guarda valia [0.90, 1.10] y era demasiado ancha para lo que pretende filtrar.
// El 2026-10-12 el factor salto de 803 ppm a 99.882 - un 9,99%, justo por debajo
// del tope - y paso el filtro. Como el sueno se divide por el, TODOS los periodos
// se acortaron un 9%: 273 s donde tocaban 300, 1090 donde 1200, 1630 donde 1800.
// El error identico en los seis trabajos fue lo que delato que no era el
// planificador sino la base de tiempo.
//
// Y ese 10% NO ERA DERIVA. El tiempo del nodo se acumula en enterDeepSleep(), asi
// que CUALQUIER arranque que no termine durmiendo - un reflasheo, un reinicio, la
// ventana de arranque en frio - pierde su tiempo despierto de la cuenta. El nodo
// se queda atras respecto al reloj real y el aprendizaje lo atribuye al oscilador,
// que no tiene la culpa. Con una guarda ancha eso se convierte en una correccion
// que empeora las cosas en vez de arreglarlas.
//
// [0.97, 1.03] cubre con holgura lo que un RC puede hacer de verdad, incluida la
// variacion con la temperatura. Fuera de ahi la causa es otra y corregir es peor
// que no corregir.
#define CLOCK_DRIFT_MIN 0.97f
#define CLOCK_DRIFT_MAX 1.03f

// Por debajo de esto no se aprende deriva: medir un error del 1% sobre un
// intervalo corto es medir ruido. Una sincronizacion por envio en estado normal
// son 31 minutos, de sobra.
#define CLOCK_DRIFT_MIN_SPAN_MS 300000ULL  // 5 min

// Tiempo propio del nodo, monotono y a prueba de suenos. Es lo unico con lo que
// se puede planificar antes de haber hablado con Modulo B.
static uint64_t nodeNowMs() { return g_nodeBaseMs + (uint64_t)millis(); }

// Hora real en epoca-ms, o 0 si el nodo todavia no se ha sincronizado nunca.
// CERO ES "NO SE", no "1 de enero de 1970": quien la use tiene que comprobarlo,
// porque fechar una lectura con un cero la situa 56 anos en el pasado y eso en
// ThingsBoard no se distingue de un dato legitimo antiguo.
static uint64_t wallNowMs() {
  if (!g_clockSynced) return 0;
  const uint64_t elapsed = nodeNowMs() - g_nodeMsAtSync;
  return g_epochMsAtSync + (uint64_t)((double)elapsed * (double)g_clockDrift);
}

  uint8_t windValid;
  // CUANDO se tomo, en tiempo propio del nodo (segundos). Antes era la cuenta de
  // despertares, que Modulo B multiplicaba por NODE_TICK_SECONDS=10 para fechar
  // la lectura: con los ticks el error era 1,9x y con despertares de 5 minutos
  // pasa a 30x. En segundos reales no hay nada que adivinar.
  uint32_t takenS;
};
RTC_NOINIT_ATTR StoredReading g_store[READING_STORE_MAX];
RTC_NOINIT_ATTR uint16_t g_storeCount;
RTC_NOINIT_ATTR uint32_t g_storeDropped;  // lecturas perdidas por almacen lleno

// Estado de alarma. En RTC para que una alarma sobreviva a los suenos (que en estado
// alto casi no hay). Tres niveles, como la hoja de umbrales.
//
// Ordenados para que "al menos pre-alarma" sea una comparacion >=: varios sitios
// tienen que tratar igual los dos estados altos (quedarse despierto, no dormir), y
// >= no puede olvidarse de uno como lo haria una cadena de ==.
enum AlarmState : uint8_t { ALARM_NORMAL = 0, ALARM_PREALARM = 1, ALARM_FIRE = 2 };
RTC_NOINIT_ATTR uint8_t g_alarmState;
RTC_NOINIT_ATTR uint16_t g_alarmClearRun;   // lecturas limpias seguidas mientras el estado esta alto

// =============================================================================
// 4. EL CALENDARIO (PLANIFICADOR POR TIEMPO)
// =============================================================================
//
// Resumen: cada sensor es un "trabajo" con su periodo en minutos (distinto en
// NORMAL, PRE-ALARMA y ALARMA) y su tiempo de preparacion. El nodo duerme hasta el
// siguiente vencimiento menos la preparacion, atiende lo que toca y vuelve a dormir.
// Funciones: schedRephase() rehace el calendario, schedEvaluate() decide que toca
// en este despertar, schedMarkDone() pasa un trabajo hecho a su siguiente
// vencimiento, schedSleepSeconds() calcula cuanto dormir.
//
// Sustituye al planificador por ticks. El viejo contaba "cada N ticks", y un tick
// no media tiempo: el sueno eran 10 s fijos ADEMAS del trabajo, asi que el tick
// real salia 18,8 s y variaba con lo que tardara cada lectura. Toda cadencia
// quedaba ~1,9x mas lenta que la configurada y ninguna constante podia acertar.
//
// Aqui cada trabajo tiene su periodo EN MINUTOS y por estado de alarma. La tabla
// es la especificacion del usuario del 2026-10-08; ver
// notes/estrategia_lectura_por_tiempo.md.
//
// ALINEADO AL RELOJ DE PARED, no al arranque. Una lectura de periodo P cae en los
// multiplos de P de la hora real, asi que los tres nodos miden en EL MISMO
// INSTANTE y sus datos son comparables entre si. Mientras el nodo no haya
// recibido su primer CFG,TIME usa su tiempo propio con la misma formula, y se
// re-fasea al sincronizar.
enum SensorJobId : uint8_t {
  JOB_BME_THP = 0,  // temperatura, humedad, presion - readFast(), calefactor OFF
  JOB_BME_VOC,      // resistencia de gas - read(), calefactor a 320 C
  JOB_CM1106,       // CO2
  JOB_SEN0466,      // CO
  JOB_BMV080,       // PM1 / PM2.5 / PM10
  JOB_CALYPSO,      // viento: angulo Y velocidad
  JOB_TX,           // transmision a Modulo B
  JOB_COUNT
};

struct SensorJob {
  const char *name;
  uint16_t periodMin[3];  // indexado por AlarmState: NORMAL, PREALARMA, FUEGO
  uint32_t prepMs;        // cuanto ANTES hay que despertar para que el dato exista al vencer
  uint32_t offsetMs;      // desfase sobre el multiplo; solo el envio lo usa
};

// EL DESFASE DEL ENVIO ES DELIBERADO. En el minuto 30 vencen cinco lecturas a la
// vez y los sensores no pueden leerse en paralelo - en paralelo reinician la placa
// - asi que van en fila y tardan ~22 s calculados, 19,3 s medidos. Transmitir en
// el minuto exacto mandaria el paquete con las particulas y el CO2 a medias. Un
// minuto de desfase da margen de sobra.
//
// NO es un periodo de 31 minutos: eso derivaria un minuto por vuelta hasta perder
// toda relacion con el barrido. Es periodo 30 + desfase 1.
#define TX_OFFSET_NORMAL_MS 60000UL

static const SensorJob kJobs[JOB_COUNT] = {
    //  nombre        NORM  PREA  FUEGO   preparacion   desfase
    {"BME690_T/H/P",  {   5,    3,    1},       1200UL,                   0},
    {"BME690_VOC",    {  20,   10,    5},      11000UL,                   0},
    {"CM1106_CO2",    {  30,   30,   30},       4000UL,                   0},
    {"SEN0466_CO",    {  30,   10,    5},        500UL,                   0},
    {"BMV080_PM",     {  30,   10,    5},       8000UL,                   0},
    {"Calypso_viento",{   5,    5,    5}, CALYPSO_BOOT_MS + CALYPSO_READ_WINDOW_MS, 0},
    {"ENVIO",         {  30,   15,    3},        500UL, TX_OFFSET_NORMAL_MS},
};

// Cuando vence cada trabajo, en la misma base de tiempo que devuelve schedNowMs().
// En RTC porque el calendario tiene que sobrevivir al sueno: si se reiniciara en
// cada despertar, el nodo volveria a medirlo todo cada vez.
RTC_NOINIT_ATTR uint64_t g_nextDueMs[JOB_COUNT];

// Para que estado se calculo el calendario que hay guardado. En RTC porque en
// operacion normal cada despertar es un arranque nuevo: una variable normal
// valdria siempre su valor inicial y re-fasearia en cada despertar.
RTC_NOINIT_ATTR uint8_t g_schedPhasedForState;

// Multiplicadores de Modulo A. NXT_LECTURA escala TODOS los periodos de sensor,
// NXT_ENVIO solo el de transmision. Sustituyen a SENSOR_READ / BURST / LORA_TRANS,
// que contaban ticks y ya no existen.
RTC_NOINIT_ATTR float g_readMultiplier;
RTC_NOINIT_ATTR float g_txMultiplier;

// Ventana para agrupar trabajos que vencen casi a la vez en un solo barrido.
//
// Dos segundos: lo justo para que dos trabajos del MISMO periodo - que vencen en
// el mismo instante salvo redondeo - caigan juntos, sin arrastrar a uno que
// venza de verdad mas tarde y adelantar su medida fuera de su minuto.
#define SCHED_COALESCE_MS 2000ULL

#define SCHED_MULT_MIN 0.1f
#define SCHED_MULT_MAX 10.0f

// LA BASE DE TIEMPO DEL PLANIFICADOR, y por que no es siempre la misma.
//
// Con reloj sincronizado se planifica sobre la hora real, que es lo que alinea a
// los tres nodos. Sin el - antes del primer CFG,TIME, o tras una perdida de
// corriente - se planifica sobre el tiempo propio del nodo, que siempre existe.
// La formula de alineacion es la misma; lo unico que cambia es el origen.
static uint64_t schedNowMs() { return g_clockSynced ? wallNowMs() : nodeNowMs(); }

// Siguiente instante ESTRICTAMENTE posterior a `now` que sea multiplo de
// `periodMs` mas `offsetMs`. Es lo que alinea las medidas a minutos redondos del
// reloj: con periodMs = 5 min cae en :00, :05, :10...
static uint64_t alignedNextDue(uint64_t now, uint32_t periodMs, uint32_t offsetMs) {
  if (periodMs == 0) return now + 60000ULL;  // defensivo: nunca un bucle cerrado
  const uint64_t shifted = (now >= offsetMs) ? (now - offsetMs) : 0;
  return (shifted / periodMs + 1ULL) * (uint64_t)periodMs + (uint64_t)offsetMs;
}

// Periodo efectivo de un trabajo: su minuto de tabla para el estado actual, por el
// multiplicador que corresponda.
static uint32_t jobPeriodMs(uint8_t job, uint8_t state) {
  if (job >= JOB_COUNT) return 60000UL;
  if (state > ALARM_FIRE) state = ALARM_NORMAL;
  const float mult = (job == JOB_TX) ? g_txMultiplier : g_readMultiplier;
  const double ms = (double)kJobs[job].periodMin[state] * 60000.0 * (double)mult;
  // Un periodo de cero pararia el trabajo para siempre; el suelo son 10 s, que es
  // mas rapido de lo que cualquier sensor de esta placa puede leer.
  if (ms < 10000.0) return 10000UL;
  if (ms > 86400000.0) return 86400000UL;  // 24 h
  return (uint32_t)ms;
}

// El desfase tambien se escala, o el envio dejaria de caer despues del barrido en
// cuanto Modulo A cambiara el multiplicador.
static uint32_t jobOffsetMs(uint8_t job, uint8_t state) {
  if (job != JOB_TX) return 0;
  // Solo en normal: en prealarma y alarma el nodo no duerme y transmite seguido,
  // asi que no hay barrido que esperar.
  if (state != ALARM_NORMAL) return 0;
  return kJobs[job].offsetMs;
}

// Recalcula el calendario entero desde `now`. Se llama al arrancar, al cambiar de
// estado de alarma y al sincronizar el reloj - los tres momentos en los que el
// calendario viejo deja de significar lo que decia.
static void schedRephase(uint64_t now, uint8_t state, const char *why) {
  for (uint8_t j = 0; j < JOB_COUNT; j++) {
    g_nextDueMs[j] = alignedNextDue(now, jobPeriodMs(j, state), jobOffsetMs(j, state));
  }
  Serial.printf("[sched] re-fasado (%s) en t=%lus, estado %u\n", why,
                (unsigned long)(now / 1000ULL), (unsigned)state);
}

// Instante en el que hay que DESPERTAR, que no es cuando vence el primer trabajo
// sino antes: cada sensor necesita su tiempo de preparacion para que el dato
// exista EN el instante pedido en vez de empezar a calentar entonces. El VOC del
// BME690 son 10,8 s de calefactor; el BMV080, 5 s de arranque mas 2 de laser.
static uint64_t schedNextWakeMs(uint8_t state) {
  uint64_t best = UINT64_MAX;
  for (uint8_t j = 0; j < JOB_COUNT; j++) {
    const uint64_t due = g_nextDueMs[j];
    const uint64_t prep = (uint64_t)kJobs[j].prepMs;
    const uint64_t wake = (due > prep) ? (due - prep) : 0;
    if (wake < best) best = wake;
  }
  (void)state;
  return best;
}

// Que debe ESTE despertar. No va en RTC: se recalcula en cada arranque a partir
// del calendario, que es lo que si persiste.
static bool g_jobDue[JOB_COUNT];

static void schedEvaluate() {
  const uint64_t now = schedNowMs();

  for (uint8_t j = 0; j < JOB_COUNT; j++) {
    // El margen de preparacion cuenta como vencido: si al BMV080 le faltan 3 s
    // para vencer y necesita 8 de arranque, este es el despertar que le toca -
    // volver a dormir 3 s y despertar otra vez no le daria tiempo.
    g_jobDue[j] = (now + (uint64_t)kJobs[j].prepMs >= g_nextDueMs[j]);
  }

  // AGRUPAR LOS TRABAJOS QUE VENCEN EN EL MISMO INSTANTE, aunque sus
  // preparaciones sean distintas.
  //
  // Sin esto, dos trabajos del MISMO periodo se separan en despertares distintos
  // solo porque uno necesita mas antelacion. El T/H/P y el Calypso van los dos a
  // 5 minutos y vencen a la vez, pero el Calypso pide 21,5 s de preparacion y el
  // T/H/P 1,2: el nodo despierta para el Calypso, lo atiende, y el T/H/P todavia
  // no vence - asi que duerme un segundo, despierta otra vez y lo atiende aparte.
  //
  // Medido el 2026-10-11 en una captura de 2 h: 49 despertares donde deberian ser
  // la mitad, y lo que importa de verdad - cada despertar guarda SU PROPIA FILA,
  // asi que en 30 minutos se almacenaban 12 lecturas en vez de 6. El almacen son
  // 8, y el log lo confirmaba: los tres envios llevaban 8, 8 y 8. Se estaban
  // TIRANDO lecturas por duplicar filas, no por medir demasiado.
  //
  // Se agrupa por VENCIMIENTO, no por tiempo de despertar: si un trabajo vence
  // dentro de la ventana del barrido que ya se va a hacer, entra en el. Mas alla
  // de esa ventana no, o se adelantarian medidas que tienen que caer en su
  // minuto.
  uint64_t primerVenc = UINT64_MAX;
  for (uint8_t j = 0; j < JOB_COUNT; j++) {
    if (g_jobDue[j] && g_nextDueMs[j] < primerVenc) primerVenc = g_nextDueMs[j];
  }
  if (primerVenc != UINT64_MAX) {
    for (uint8_t j = 0; j < JOB_COUNT; j++) {
      if (!g_jobDue[j] && g_nextDueMs[j] <= primerVenc + SCHED_COALESCE_MS) {
        g_jobDue[j] = true;
      }
    }
  }
}

static bool schedAnySensorDue() {
  for (uint8_t j = 0; j < JOB_TX; j++) {  // JOB_TX va el ultimo y no es un sensor
    if (g_jobDue[j]) return true;
  }
  return false;
}

// Un trabajo ya hecho vuelve a la rejilla: su proximo vencimiento es el siguiente
// multiplo del reloj, NO "ahora mas el periodo". La diferencia importa - lo
// segundo acumularia el tiempo que costo la lectura y el calendario se iria
// desplazando hasta perder la alineacion con los otros nodos.
static void schedMarkDone(uint8_t job, uint8_t state) {
  // AVANZAR DESDE EL VENCIMIENTO, NO DESDE "AHORA", y la diferencia no es sutil.
  //
  // Un trabajo se marca vencido cuando `ahora + preparacion >= vencimiento`, asi
  // que se EJECUTA ANTES de su instante: el Calypso, con 21,5 s de preparacion,
  // corre 21,5 s antes. Si al terminar se calcula el siguiente vencimiento desde
  // "ahora", el siguiente multiplo que sale es EL MISMO VENCIMIENTO que acaba de
  // atenderse - y el trabajo se vuelve a disparar en el acto.
  //
  // Medido el 2026-10-10 en pre-alarma, donde el nodo no duerme y loop() gira
  // cada 2 s, asi que el bucle se ve sin filtro:
  //
  //     Calypso     intervalos 4,1  4,6  10,7 s   (deberian ser 300)
  //     BME690 VOC  intervalo  10,8 s             (deberian ser 600)
  //     BME690 T/H/P  179,6  180,6 s              correcto: su preparacion es 1,2 s
  //
  // El T/H/P se salvaba porque su preparacion es tan corta que casi nunca adelanta
  // la ejecucion al otro lado de un multiplo. En estado NORMAL el fallo salia como
  // suenos de 1 segundo, que se habian dado por anecdoticos.
  //
  // Tomando como base el vencimiento que se acaba de atender, el siguiente cae un
  // periodo mas alla, que es lo que significa "cada N minutos".
  const uint64_t ahora = schedNowMs();
  const uint64_t base = (g_nextDueMs[job] > ahora) ? g_nextDueMs[job] : ahora;
  g_nextDueMs[job] = alignedNextDue(base, jobPeriodMs(job, state), jobOffsetMs(job, state));
}

// Cuanto dormir, en segundos, hasta el proximo despertar.
static uint32_t schedSleepSeconds(uint8_t state) {
  const uint64_t now = schedNowMs();
  const uint64_t wake = schedNextWakeMs(state);
  const uint64_t deltaReal = (wake > now) ? (wake - now) : 1000ULL;

  // DE TIEMPO REAL A TIEMPO DEL NODO. El nodo duerme contando con su oscilador RC,
  // y g_clockDrift dice cuantos milisegundos reales vale cada milisegundo que el
  // cuenta. Para dormir X reales hay que pedirle X/deriva de los suyos. Esta
  // division es la que convierte la correccion aprendida en precision de verdad:
  // sin ella el factor se mediria y no se usaria para nada.
  double nodeMs = (double)deltaReal / (double)g_clockDrift;

  if (nodeMs < 1000.0) nodeMs = 1000.0;          // nunca un sueno de cero
  if (nodeMs > 3600000.0) nodeMs = 3600000.0;    // tope de 1 h, defensivo
  return (uint32_t)((nodeMs + 500.0) / 1000.0);  // al segundo mas cercano
}
// Milisegundos acumulados en pre-alarma, y por que no es una simple marca de millis().
//
// g_alarmState es RTC_NOINIT y sobrevive a un reinicio, asi que un nodo que se cae
// por tension a mitad de una pre-alarma vuelve todavia en pre-alarma - pero millis()
// empieza de cero. Una hora de inicio guardada le daria una ventana nueva en cada
// reinicio, y un nodo que se reinicie bajo la carga continua de ~35 mA de la
// pre-alarma podria aplazar una alarma de incendio real indefinidamente sin que nada
// pareciera ir mal.
//
// Acumulando, la ventana sobrevive haga lo que haga el nodo.
RTC_NOINIT_ATTR uint32_t g_prealarmMs;
RTC_NOINIT_ATTR float g_gasBaseline;        // resistencia de gas del BME690 en aire limpio

// Configuracion modificable en marcha mediante comandos CFG - ver la cabecera del
// archivo. Solo en RAM: vuelve a estos valores compilados en cada reinicio, a
// proposito (salvo lo que se guarda expresamente en RTC).
uint32_t g_txPeriodMs = LORA_TX_PERIOD_MS;  // CFG,INTERVAL=<segundos>

// Interpreta y aplica un comando "CFG,<clave>=<valor>[,<clave>=<valor>...]" sobre
// las variables de configuracion de arriba. Las claves desconocidas se apuntan en el
// log y se saltan, sin fallar. Devuelve true (con ackPayload relleno: los
// "<clave>=<valor>" realmente aplicados, tras recortarlos a sus limites, separados
// por comas) solo si se reconocio al menos una clave - ver en la cabecera por que un
// comando que no aplica nada no recibe ACK.
static bool applyConfigCommand(const char *payload, uint8_t len, char *ackPayload, size_t ackCap) {
  static const char kPrefix[] = "CFG,";
  static const size_t kPrefixLen = sizeof(kPrefix) - 1;
  if (len <= kPrefixLen || memcmp(payload, kPrefix, kPrefixLen) != 0) return false;

  // Copia del cuerpo (lo que va detras de "CFG,") para poder trocearla.
  char buf[241];
  uint8_t bodyLen = len - (uint8_t)kPrefixLen;
  if (bodyLen >= sizeof(buf)) bodyLen = sizeof(buf) - 1;
  memcpy(buf, payload + kPrefixLen, bodyLen);
  buf[bodyLen] = '\0';

  ackPayload[0] = '\0';
  size_t ackLen = 0;
  bool appliedAny = false;

  // Recorrer cada "<clave>=<valor>" separado por comas.
  char *saveptr = nullptr;
  char *tok = strtok_r(buf, ",", &saveptr);
  while (tok != nullptr) {
    char *eq = strchr(tok, '=');
    if (eq) {
      *eq = '\0';
      const char *key = tok;
      const char *valueStr = eq + 1;
      char appliedKv[24] = {0};
      bool matched = true;

      // ---- TIME: la hora de red que Modulo B baja en la ventana de escucha ----
      //
      // EN SEGUNDOS, NO EN MILISEGUNDOS, a proposito. Una epoca en segundos cabe
      // en 32 bits hasta 2106, asi que ni el parseo ni el ACK necesitan enteros de
      // 64 bits - y `%lld` no es de fiar con la nano-newlib que lleva este build,
      // que ya nos costo un fallo en Modulo B. Se pierde precision por debajo del
      // segundo, que contra periodos de minutos es irrelevante.
      if (strcmp(key, "TIME") == 0) {
        const unsigned long epochS = strtoul(valueStr, nullptr, 10);
        // Antes de 2023 no es una hora, es un modem que todavia no ha cogido la
        // hora de red y esta devolviendo su valor de fabrica. Modulo B ya filtra
        // por esto, pero aceptarla aqui situaria todas las lecturas en 1970.
        if (epochS < 1672531200UL) {  // 2023-01-01
          Serial.printf("[clock] TIME=%lu ignorado: anterior a 2023\n", epochS);
          matched = false;
        } else {
          const uint64_t epochMs = (uint64_t)epochS * 1000ULL;
          const uint64_t nodeNow = nodeNowMs();

          // APRENDER LA DERIVA. Solo se puede a partir de la SEGUNDA
          // sincronizacion, que es la primera con dos puntos que comparar.
          if (g_clockSynced && epochMs > g_epochMsAtSync && nodeNow > g_nodeMsAtSync) {
            const uint64_t dReal = epochMs - g_epochMsAtSync;
            const uint64_t dNode = nodeNow - g_nodeMsAtSync;
            if (dNode >= CLOCK_DRIFT_MIN_SPAN_MS) {
              const float f = (float)((double)dReal / (double)dNode);
              if (f >= CLOCK_DRIFT_MIN && f <= CLOCK_DRIFT_MAX) {
                Serial.printf("[clock] deriva %ld ppm -> %ld ppm\n",
                              (long)((g_clockDrift - 1.0f) * 1000000.0f),
                              (long)((f - 1.0f) * 1000000.0f));
                g_clockDrift = f;
              } else {
                // Fuera de rango no es deriva: es una hora mala o un salto. Un
                // factor absurdo dejaria al nodo midiendo a un ritmo arbitrario
                // sin forma de volver, asi que se conserva el anterior.
                Serial.printf("[clock] factor %ld ppm fuera de rango, descartado\n",
                              (long)((f - 1.0f) * 1000000.0f));
              }
            }
          }

          const bool primera = !g_clockSynced;
          g_epochMsAtSync = epochMs;
          g_nodeMsAtSync = nodeNow;
          g_clockSynced = 1;
          g_clockSyncCount++;

          // LA PRIMERA SINCRONIZACION CAMBIA LA BASE DE TIEMPO bajo los pies del
          // planificador: hasta ahora contaba sobre el tiempo propio del nodo, que
          // empieza en cero al energizar, y a partir de aqui cuenta sobre la epoca
          // real, que son 1,76e12 ms. Los vencimientos guardados quedan
          // astronomicamente en el pasado, asi que todos los trabajos vencerian a
          // la vez y para siempre. Hay que re-fasear.
          //
          // Y es justo lo que queremos ademas: ese re-faseo es el que ALINEA al
          // nodo con el reloj de pared, y por tanto con los otros nodos.
          if (primera) schedRephase(schedNowMs(), g_alarmState, "primera hora de red");
          Serial.printf("[clock] sincronizado #%lu: epoca %lu, nodo %lus, deriva %ld ppm\n",
                        (unsigned long)g_clockSyncCount, epochS,
                        (unsigned long)(nodeNow / 1000ULL),
                        (long)((g_clockDrift - 1.0f) * 1000000.0f));
          snprintf(appliedKv, sizeof(appliedKv), "TIME=%lu", epochS);
        }
      } else if (strcmp(key, "INTERVAL") == 0) {
        // Periodo de envio heredado (ver LORA_TX_PERIOD_MS). Se recorta a sus
        // limites, y se guarda en RTC al dormir para que sobreviva al sueno.
        long seconds = atol(valueStr);
        const long minSeconds = LORA_TX_PERIOD_MIN_MS / 1000;
        const long maxSeconds = LORA_TX_PERIOD_MAX_MS / 1000;
        if (seconds < minSeconds) {
          Serial.printf("[cfg] INTERVAL=%ld below floor, clamped to %ld\n", seconds, minSeconds);
          seconds = minSeconds;
        } else if (seconds > maxSeconds) {
          Serial.printf("[cfg] INTERVAL=%ld above ceiling, clamped to %ld\n", seconds, maxSeconds);
          seconds = maxSeconds;
        }
        g_txPeriodMs = (uint32_t)seconds * 1000UL;
        snprintf(appliedKv, sizeof(appliedKv), "INTERVAL=%ld", seconds);
      // ---- NxT: multiplicadores de la v2 del protocolo A->C ----
      //
      // LA MULTIPLICACION LA HACE EL NODO, NO MODULO B. Los periodos base son
      // constantes de ESTE fichero (la tabla kJobs[]). Si Modulo B los calculara
      // tendria que llevarlos duplicados, y se desincronizarian en cuanto alguien
      // tocara uno aqui. Modulo B manda la N en crudo; el nodo multiplica contra sus
      // propios valores, valida y confirma lo que de verdad aplico.
      // (El texto original de este comentario hablaba de ticks; desde el
      // calendario por tiempo, N multiplica minutos.)
      } else if (strcmp(key, "NXT_LECTURA") == 0) {
        // "Cada sensor se lee cada N x su periodo minimo de lectura. N >= 1."
        //
        // Con el planificador por tiempo esto es literal: N multiplica los MINUTOS
        // de la tabla de cada sensor, en los tres estados a la vez. Antes escalaba
        // dos contadores de ticks, que es lo mismo que queria decir el documento
        // pero expresado en una unidad que no media tiempo.
        const float n = (float)atof(valueStr);
        if (n < 1.0f || n > SCHED_MULT_MAX) {
          Serial.printf("[downlink] NXT_LECTURA=%.2f fuera de [1..%.0f] - ignorado\n",
                        (double)n, (double)SCHED_MULT_MAX);
          matched = false;
        } else {
          g_readMultiplier = n;
          // El calendario viejo se calculo con el multiplicador anterior, asi que
          // hay que rehacerlo o los periodos nuevos no empezarian hasta que cada
          // trabajo venciera una vez mas con el viejo.
          schedRephase(schedNowMs(), g_alarmState, "NXT_LECTURA");
          Serial.printf("[cfg] NXT_LECTURA=%.2f -> T/H/P cada %lus en estado %u\n",
                        (double)n, (unsigned long)(jobPeriodMs(JOB_BME_THP, g_alarmState) / 1000UL),
                        (unsigned)g_alarmState);
          snprintf(appliedKv, sizeof(appliedKv), "NXT_LECTURA=%.2f", (double)n);
        }
      } else if (strcmp(key, "NXT_ENVIO") == 0) {
        // "Los datos se mandan cada N x el periodo base de envio." Multiplica los
        // minutos de la fila ENVIO de la tabla: 30 en normal, 15 en prealarma, 3 en
        // alarma. N < 1 es legitimo aqui - sirve para pedir envios MAS seguidos.
        const float n = (float)atof(valueStr);
        if (n < SCHED_MULT_MIN || n > SCHED_MULT_MAX) {
          Serial.printf("[downlink] NXT_ENVIO=%.2f fuera de [%.1f..%.0f] - ignorado\n",
                        (double)n, (double)SCHED_MULT_MIN, (double)SCHED_MULT_MAX);
          matched = false;
        } else {
          g_txMultiplier = n;
          schedRephase(schedNowMs(), g_alarmState, "NXT_ENVIO");
          Serial.printf("[cfg] NXT_ENVIO=%.2f -> envio cada %lus en estado %u\n",
                        (double)n, (unsigned long)(jobPeriodMs(JOB_TX, g_alarmState) / 1000UL),
                        (unsigned)g_alarmState);
          snprintf(appliedKv, sizeof(appliedKv), "NXT_ENVIO=%.2f", (double)n);
        }
      } else if (strcmp(key, "ALARM") == 0) {
        // Bajar la alarma a mano desde Modulo A. ACTIVARLA a distancia NO se admite,
        // a proposito: una alarma la tienen que levantar los sensores, no una orden.
        if (atoi(valueStr) == 0) {
          g_alarmState = ALARM_NORMAL;
          g_alarmClearRun = 0;
          Serial.println("[downlink] pre-alarm force-cleared by Module A");
          snprintf(appliedKv, sizeof(appliedKv), "ALARM=0");
        } else {
          Serial.println("[downlink] ALARM can only be cleared (ALARM=0), not set - ignored");
          matched = false;
        }
      // ---- Activar (1) o silenciar (0) un sensor sin reflashear ----
      } else if (strcmp(key, "BME690") == 0) {
        g_bmeEnabled = atoi(valueStr) != 0;
        snprintf(appliedKv, sizeof(appliedKv), "BME690=%d", g_bmeEnabled ? 1 : 0);
      } else if (strcmp(key, "SEN0466") == 0) {
        g_sen0466Enabled = atoi(valueStr) != 0;
        snprintf(appliedKv, sizeof(appliedKv), "SEN0466=%d", g_sen0466Enabled ? 1 : 0);
      } else if (strcmp(key, "BMV080") == 0) {
        g_bmvEnabled = atoi(valueStr) != 0;
        snprintf(appliedKv, sizeof(appliedKv), "BMV080=%d", g_bmvEnabled ? 1 : 0);
      } else if (strcmp(key, "CM1106") == 0) {
        g_cm1106Enabled = atoi(valueStr) != 0;
        snprintf(appliedKv, sizeof(appliedKv), "CM1106=%d", g_cm1106Enabled ? 1 : 0);
      } else if (strcmp(key, "CALYPSO") == 0) {
        g_calypsoEnabled = atoi(valueStr) != 0;
        snprintf(appliedKv, sizeof(appliedKv), "CALYPSO=%d", g_calypsoEnabled ? 1 : 0);

      // ---- Umbrales de incendio, p. ej. CFG,temp_pre_on=50 ----
      //
      // Se prueban LOS ULTIMOS, despues de todas las claves fijas, para que un nombre
      // de umbral nunca pueda tapar una clave de configuracion. Los nombres son los
      // de las reglas de kAlarmRules con el sufijo _pre_on / _pre_off / _alarma_on /
      // _alarma_off, que es exactamente lo que manda el panel de Modulo A.
      //
      // EL ACK DEVUELVE LO QUE SE GUARDO, NO LO QUE LLEGO.
      //
      // Antes devolvia valueStr - los caracteres que mando Modulo A -, lo que solo
      // prueba que la clave coincidio con una regla. Modulo A usa esto para mostrar
      // el umbral que el nodo usa de verdad, asi que devolver la peticion haria que
      // el panel estuviera de acuerdo consigo mismo por construccion y no revelara
      // nunca un desacuerdo.
      //
      // Se formatea con %.2f y se recortan los ceros, en vez de usar %g: el snprintf
      // de la nano-newlib de este compilador no es de fiar con las conversiones
      // menos comunes - el mismo motivo por el que existe appendI64() en vez de %lld
      // - y un formato que no emitiera nada pondria un ACK mal formado en el aire.
      } else if (float *slot = thresholdSlot(key)) {
        *slot = (float)atof(valueStr);
        char num[20];
        snprintf(num, sizeof(num), "%.2f", (double)*slot);
        char *dot = strchr(num, '.');
        if (dot) {
          char *end = num + strlen(num) - 1;
          while (end > dot && *end == '0') *end-- = '\0';
          if (end == dot) *end = '\0';
        }
        snprintf(appliedKv, sizeof(appliedKv), "%s=%s", key, num);

      } else {
        matched = false;
      }

      // Anadir lo aplicado a la respuesta ACK, separado por comas.
      if (matched) {
        appliedAny = true;
        size_t kvLen = strlen(appliedKv);
        size_t sep = (ackLen > 0) ? 1 : 0;
        if (ackLen + sep + kvLen < ackCap) {
          if (sep) ackPayload[ackLen++] = ',';
          memcpy(ackPayload + ackLen, appliedKv, kvLen);
          ackLen += kvLen;
          ackPayload[ackLen] = '\0';
        }
      } else {
        Serial.printf("[cfg] unrecognized key \"%s\", ignored\n", key);
      }
    }
    tok = strtok_r(nullptr, ",", &saveptr);
  }

  return appliedAny;
}

// Nombre corto de cada estado de lectura, para el log.
static const char *statusName(ReadingStatus s) {
  switch (s) {
    case ReadingStatus::NotInitialized: return "NotInit";
    case ReadingStatus::NotReady:       return "NotReady";
    case ReadingStatus::Timeout:        return "Timeout";
    case ReadingStatus::NoAck:          return "NoAck";
    case ReadingStatus::InvalidFrame:   return "BadFrame";
    case ReadingStatus::Ok:             return "OK";
  }
  return "?";
}

// Atiende un mensaje LoRa recibido en la ventana de escucha tras un envio: un CFG
// reconocido (se aplica y se contesta ACK) o cualquier otra cosa (solo se apunta en
// el log).
static void handleInboundMessage(const LoRaMessage &msg) {
  Serial.printf("[downlink] from addr %u (%d bytes, rssi %d, snr %d): %.*s\n", msg.senderAddr, msg.length,
                msg.rssi, msg.snr, msg.length, msg.payload);

  // Solo se aceptan ordenes de Modulo B (filtro por direccion, no autenticacion).
  if (msg.senderAddr != LORA_RX_ADDR) {
    Serial.println("[downlink] ignored - not from Module B (LORA_RX_ADDR)");
    return;
  }

  // CALLARSE ES LA RESPUESTA EQUIVOCADA A UNA ORDEN ENTENDIDA PERO NO APLICABLE.
  //
  // Antes simplemente volvia. Modulo B no puede distinguir el silencio por rechazo
  // del silencio por falta de cobertura, asi que agotaba el plazo entero y decia
  // "timeout" - el estado equivocado de los cuatro del protocolo, un cuarto de hora
  // despues. Al operador se le decia que el nodo era inalcanzable cuando estaba ahi
  // habiendo rechazado la orden.
  //
  // Un CFG reconocido del que no se aplica nada recibe ahora un NACK explicito, que
  // Modulo B convierte de inmediato en el estado "error".
  const bool looksLikeCfg = (msg.length >= 4 && strncmp(msg.payload, "CFG,", 4) == 0);

  char kv[200];
  if (!applyConfigCommand(msg.payload, msg.length, kv, sizeof(kv))) {
    if (looksLikeCfg && radioReady) {
      const char *nack = "NACK,no recognized keys";
      bool sent = radio.send(LORA_RX_ADDR, nack, (uint8_t)strlen(nack));
      Serial.printf("[cfg] nothing applied - NACK sent (%s)\n", sent ? "ok" : "FAILED");
    } else if (looksLikeCfg) {
      Serial.println("[cfg] nothing applied, and the radio is down - cannot NACK");
    }
    return;
  }

  if (!radioReady) {
    Serial.println("[cfg] applied, but ACK skipped - radio not initialized");
    return;
  }

  char ack[220];
  int ackLen = snprintf(ack, sizeof(ack), "ACK,%s", kv);
  bool sent = radio.send(LORA_RX_ADDR, ack, (uint8_t)ackLen);
  Serial.printf("[cfg] applied, ACK sent: %s (%s)\n", ack, sent ? "ok" : "FAILED");
}

// Instrumentacion de la puesta en marcha. Cada paso se anuncia ANTES de ejecutarlo
// y se vacia el buffer: el USB CDC guarda los datos, y lo que siga en cola se pierde
// si el chip se cae, que es justo el caso que se buscaba (reinicios justo despues de
// "BMV080: OK", 2026-09-04).
static void step(const char *what) {
  Serial.printf("[step] %s ...\n", what);
  Serial.flush();
  delay(20);
}

// ---------------- Guardia de arranques (NVS) ----------------
// El contador de arranques en RTC dice "encendido real" en CADA reinicio cuando el
// dominio RTC se pierde - es decir, cuando la alimentacion se cae de verdad, no
// cuando una tarea falla. La RAM RTC no puede medir eso, porque el propio suceso la
// destruye.
//
// La NVS vive en la flash y sobrevive a un corte, asi que si puede. Si la cuenta en
// NVS sube 1, 2, 3, 4 mientras el contador RTC dice 1 en cada arranque, la teoria de
// la alimentacion queda confirmada: dos contadores, los mismos reinicios, y solo uno
// sobrevive.
//
// Tambien gobierna el MODO SEGURO: con 12 o mas arranques seguidos sin una vuelta lo
// bastante larga para darla por buena (ver g_safeMode en setup()), los rails se
// quedan APAGADOS para que el puerto siga sirviendo para reflashear en vez de que la
// placa entre en un bucle sin fin.
//
// OJO: la cuenta vive en la particion NVS (0x9000), que un flasheo normal NO borra.
// El mensaje de arranque que dice "clear it by flashing again" es incorrecto; para
// borrarla hay que borrar esa particion (esptool erase_region 0x9000 0x5000).
static Preferences bootStore;
static uint32_t nvsBoots = 0;

// CUENTA FALLOS, NO DESPERTARES. Esa distincion es lo importante.
//
// Antes sumaba en CADA arranque, y bootGuardMarkGood() solo la borraba tras 15 s de
// funcionamiento. Eso es incompatible con el sueno profundo: un despertar sin trabajo
// se duerme en mucho menos de 15 s, asi que la cuenta nunca se borraba, subia en cada
// despertar y activaba el modo seguro a los 12. TODOS los nodos sanos que dormian lo
// hacian, a los dos minutos de encenderlos.
//
// Lo demostro el propio paquete de socorro del nodo: "SAFE,12,8,17" - doce arranques,
// motivo de reinicio 8 = DEEPSLEEP, diecisiete despertares. Doce despertares
// PERFECTAMENTE NORMALES contados como doce fallos.
//
// Un despertar del sueno es el firmware funcionando tal como esta disenado y nunca
// debe contar en su contra. Solo un arranque que NO viene de nuestro propio sueno -
// un fallo, una caida de tension, un perro guardian, un corte - es indicio de
// problemas, y eso es lo que cuenta ahora la guardia.
static void bootGuardBegin(bool wokeFromTimer) {
  bootStore.begin("boot", false);
  nvsBoots = bootStore.getUInt("n", 0);

  if (wokeFromTimer) {
    // Llegar a un despertar programado significa que el ciclo anterior termino y se
    // durmio a proposito. Es una vuelta sana por corta que fuera, asi que la cuenta
    // se borra aqui en vez de esperar al temporizador de 15 s de loop(), al que un
    // despertar corto no llega nunca.
    if (nvsBoots != 0) {
      bootStore.putUInt("n", 0);
      Serial.printf("Boot guard: timer wake - counter cleared (was %lu)\n",
                    (unsigned long)nvsBoots);
      nvsBoots = 0;
    }
  } else {
    nvsBoots++;
    bootStore.putUInt("n", nvsBoots);
    Serial.printf("NVS boot count (non-sleep boots since last good run): %lu\n",
                  (unsigned long)nvsBoots);
  }
  Serial.flush();
}

// Se llama una vez que una vuelta ha durado lo bastante para darla por sana.
static void bootGuardMarkGood() {
  bootStore.putUInt("n", 0);
  Serial.println("Boot guard: run marked good, NVS counter cleared");
  Serial.flush();
}

// DIAGNOSTICO: lectura cruda del ID del BME690, utilizable en cualquier punto de
// setup() e independiente del driver. Antes este registro solo se miraba DESPUES de
// que bme690.begin() hubiera fallado, y eso no distingue "el aparato nunca contesta"
// de "algo entre el escaneo del bus y begin() lo estropea". Llamarla en dos momentos
// distintos si los distingue.
static void probeBmeChipId(const char *when) {
  const uint8_t addrs[2] = {0x76, 0x77};
  for (int i = 0; i < 2; i++) {
    uint8_t addr = addrs[i];

    // Intentar la LECTURA DEL REGISTRO aunque falle la prueba de direccion.
    //
    // Antes se saltaba la direccion si fallaba una escritura de longitud cero, lo que
    // supone que un aparato que no da ACK a una transaccion vacia no puede contestar
    // a una de verdad. Eso no es cierto en todas las piezas I2C, y significaba
    // declarar ausente un sensor sin haberle preguntado nada. Se informan los dos
    // resultados.
    Wire.beginTransmission(addr);
    uint8_t probe = Wire.endTransmission();

    Wire.beginTransmission(addr);
    Wire.write(0xD0);  // BME69X_REG_CHIP_ID
    uint8_t wr = Wire.endTransmission(false);
    if (wr != 0) {
      Serial.printf("[bme probe %s] 0x%02X addr-probe=%u, register write NACKed "
                    "(endTransmission=%u) - no answer either way\n",
                    when, addr, (unsigned)probe, (unsigned)wr);
      continue;
    }
    if (Wire.requestFrom((int)addr, 1) != 1) {
      // NO DIGAS QUE "ACEPTO LA ESCRITURA". Esto costo una hora de diagnostico el
      // 2026-10-08: el mensaje anterior afirmaba que el sensor habia reconocido su
      // direccion, lo que llevo a descartar que estuviera sin alimentar y a buscar
      // un bus atascado que no existia.
      //
      // `endTransmission(false)` NO EJECUTA la transaccion: pide un repeated start,
      // asi que ESP32 la deja encolada y devuelve 0 porque todavia no ha fallado
      // nada - no porque nadie haya contestado. El unico dato fiable de que hay
      // alguien ahi es `probe`, la escritura de longitud cero que SI se ejecuta.
      Serial.printf("[bme probe %s] 0x%02X sin respuesta a la lectura "
                    "(addr-probe=%u: %s)\n",
                    when, addr, (unsigned)probe,
                    probe == 0 ? "el chip SI acuso su direccion"
                               : "NADIE acuso esa direccion - sensor ausente o sin alimentar");
      continue;
    }
    uint8_t id = (uint8_t)Wire.read();
    Serial.printf("[bme probe %s] 0x%02X (addr-probe=%u) chip ID = 0x%02X %s\n", when, addr,
                  (unsigned)probe, id,
                  id == 0x61 ? "(0x61 = BME69x, CORRECT)" : "(expected 0x61)");
  }
  Serial.flush();
}

// DIAGNOSTICO: escaneo del bus I2C. El BME690 y el BMV080 comparten bus, asi que un
// escaneo separa "el sensor ha desaparecido del bus" de "el sensor esta pero no
// contesta" - y prueba el propio bus con lo que conteste en los mismos cables en el
// mismo instante. El nodo C1 perdio su BME690 a mitad de una sesion y esto fue lo
// que lo identifico como fisico en una linea en vez de en una tarde.
static void scanI2C(const char *when) {
  Serial.printf("--- I2C scan (%s) ---\n", when);
  int found = 0;
  for (uint8_t addr = 0x08; addr <= 0x77; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) {
      Serial.printf("  ACK at 0x%02X\n", addr);
      found++;
    }
  }
  if (found == 0) Serial.println("  (nothing acked)");
  Serial.flush();
}

// RECUPERACION DEL BUS I2C ANTES DE USARLO.
//
// EL SINTOMA QUE LA MOTIVA, visto en C3 el 2026-10-08: los TRES sensores de I2C
// dan NOT FOUND mientras los dos de UART responden perfectamente, y el log dice
// que el chip "acepta la escritura de registro pero no devuelve datos", con error
// 263 (timeout) en la lectura.
//
// Esa contradiccion solo tiene una explicacion: SDA PEGADO A NIVEL BAJO. El ACK de
// I2C se senaliza poniendo SDA bajo, asi que si la linea ya esta baja el maestro
// lee un ACK que nadie ha dado, y a partir de ahi toda lectura expira. Y afecta a
// TODOS los dispositivos del bus a la vez, que es exactamente el patron observado:
// no son tres averias independientes, es el bus.
//
// POR QUE SE QUEDA ASI. Un esclavo interrumpido a mitad de transferencia sigue
// sujetando SDA mientras espera los pulsos de reloj que nunca llegaron. Aqui pasa
// con facilidad: cada despertar es un reset del maestro, y si cae mientras un
// sensor estaba transmitiendo, el sensor se queda colgado y el maestro arranca sin
// saberlo. Un reset no lo arregla - solo un corte de alimentacion del esclavo, o
// esto.
//
// LA CURA es la estandar: darle los pulsos que le faltan. Nueve ciclos de SCL
// bastan para que termine el byte que tenia a medias y suelte la linea, y despues
// una condicion de STOP deja el bus en reposo.
//
// Devuelve si el bus quedo utilizable. Imprime siempre que haya tenido que actuar,
// porque un bus que se atasca a menudo es un sintoma por si mismo.
//
// ---- Diagnostico de las lineas (i2cLineState / i2cReportPullups) ----
//
// SI EL RAIL DE 3V3 LLEGA DE VERDAD AL BUS, medido por el propio ESP32 y sin
// polimetro. Las resistencias de pull-up del I2C se alimentan del rail de 3V3:
//
//   * leidas SIN pull-up interno, si las lineas siguen ALTAS es que las externas
//     tienen tension -> el rail llega al bus.
//   * si caen a BAJO, las externas estan sin alimentar -> el rail NO llega, por
//     mucho que la puerta del transistor este conduciendo.
//
// Esa distincion es justo la que no se puede hacer mirando el GPIO de la puerta:
// "el transistor esta prendido" y "el rail da 3,3 V en el sensor" son cosas
// distintas, y entre una y otra hay una pista, un conector y el propio transistor.
//
// CUIDADO AL LEERLO: un ALTO aqui no prueba que los sensores funcionen, solo que
// hay tension en las pull-up. Y con los rails APAGADOS es normal ver las lineas
// SUJETAS: los sensores sin alimentar tiran de ellas hacia masa (visto en C3 el
// 2026-10-14: sujetas antes de encender los rails, LIBRES despues).
//
// Estado de una linea del bus, distinguiendo las TRES causas que se confunden
// entre si cuando una linea lee BAJO:
//
//   LIBRE     sube sola y se queda arriba        -> hay pull-up con tension
//   FLOTANTE  se puede subir, pero no se sostiene -> no hay pull-up (o sin tension)
//   SUJETA    no sube ni forzandola               -> corto a masa, o un esclavo
//                                                   colgado sujetandola
//
// La diferencia importa: "flotante" apunta al rail que alimenta las pull-up, y
// "sujeta" apunta a un corto o a un chip colgado. Mirar solo el nivel no las
// separa, y es lo que hacia la version anterior de esta funcion.
static const char *i2cLineState(uint8_t pin) {
  pinMode(pin, INPUT);  // sin pull-up interno: solo queda lo que haya fuera
  delayMicroseconds(200);
  if (digitalRead(pin) == HIGH) return "LIBRE";

  // Forzarla alta un instante y soltarla. Si rebota a bajo de inmediato hay algo
  // sujetandola; si aguanta aunque sea un momento, es capacidad flotando.
  pinMode(pin, OUTPUT);
  digitalWrite(pin, HIGH);
  delayMicroseconds(50);
  pinMode(pin, INPUT);
  delayMicroseconds(5);
  const int justoAlSoltar = digitalRead(pin);
  delayMicroseconds(200);
  const int despues = digitalRead(pin);

  if (justoAlSoltar == LOW) return "SUJETA a masa";
  if (despues == LOW) return "FLOTANTE (sin pull-up)";
  return "LIBRE (lenta)";
}

// Imprime el estado de SDA y SCL en un momento dado del arranque.
static void i2cReportPullups(const char *cuando) {
  const char *sda = i2cLineState(PIN_I2C_SDA);
  const char *scl = i2cLineState(PIN_I2C_SCL);
  Serial.printf("[i2c] bus (%s): SDA=%s  SCL=%s\n", cuando, sda, scl);
  Serial.flush();
}

// Recuperacion del bus: si SDA esta bajo, hasta 9 pulsos de SCL y un STOP.
static bool i2cRecoverBus() {
  i2cReportPullups("antes de Wire.begin");
  pinMode(PIN_I2C_SCL, OUTPUT_OPEN_DRAIN);
  pinMode(PIN_I2C_SDA, INPUT_PULLUP);
  digitalWrite(PIN_I2C_SCL, HIGH);
  delayMicroseconds(10);

  if (digitalRead(PIN_I2C_SDA) == HIGH) return true;  // bus en reposo, nada que hacer

  Serial.println("[i2c] SDA PEGADO A NIVEL BAJO - recuperando el bus");
  uint8_t pulsos = 0;
  for (; pulsos < 9 && digitalRead(PIN_I2C_SDA) == LOW; pulsos++) {
    digitalWrite(PIN_I2C_SCL, LOW);
    delayMicroseconds(10);
    digitalWrite(PIN_I2C_SCL, HIGH);
    delayMicroseconds(10);
  }

  const bool libre = (digitalRead(PIN_I2C_SDA) == HIGH);
  if (libre) {
    // STOP: SDA sube mientras SCL esta alto. Sin esto algun esclavo puede quedarse
    // creyendo que la transferencia sigue abierta.
    pinMode(PIN_I2C_SDA, OUTPUT_OPEN_DRAIN);
    digitalWrite(PIN_I2C_SDA, LOW);
    delayMicroseconds(10);
    digitalWrite(PIN_I2C_SCL, HIGH);
    delayMicroseconds(10);
    digitalWrite(PIN_I2C_SDA, HIGH);
    delayMicroseconds(10);
  }
  Serial.printf("[i2c] %u pulsos -> SDA %s\n", (unsigned)pulsos,
                libre ? "LIBRE" : "SIGUE BAJO (fallo fisico, no un esclavo colgado)");
  Serial.flush();
  return libre;
}

// ENCIENDE LOS RAILS, de uno en uno: 3V3, 5V, LoRa, y la habilitacion del BMV080.
//
// Primero los apaga los dos (cada puerta con su nivel inactivo, que son opuestos).
// En modo seguro se quedan apagados y vuelve.
//
// La rampa de arranque suave que habia aqui se quito. Se escribio para la teoria de
// la corriente de arranque, que las pruebas desmintieron: hacia que la placa muriera
// ANTES, a media rampa, y la causa real resulto ser varios sensores consumiendo a la
// vez. Ahora son escrituras simples, identicas a las del programa minimo de rails
// que mantiene estos mismos rails mas de 80 s sin un solo reinicio.
//
// Cada paso se imprime ANTES y DESPUES: si sobrevive "about to" pero no "asserted",
// muere en la propia escritura; si sobreviven los dos pero no "settled", muere en la
// espera que sigue. Son averias distintas. (Asi se vio el 2026-10-14 el corto del
// rail LoRa en C3: el log se cortaba siempre en "LoRa: about to assert".)
static void enableRails(bool safeMode) {
  pinMode(PIN_PCB_EN_A, OUTPUT);
  pinMode(PIN_PCB_EN_B, OUTPUT);
  // Los dos APAGADOS primero - cada puerta con su propio nivel inactivo, porque son opuestos.
  digitalWrite(PIN_PCB_EN_A, (PIN_PCB_EN_A_ACTIVE == LOW) ? HIGH : LOW);
  digitalWrite(PIN_PCB_EN_B, (PIN_PCB_EN_B_ACTIVE == LOW) ? HIGH : LOW);

  if (safeMode) {
    Serial.println("SAFE MODE: rails stay off, port is yours");
    Serial.println("  (3 boots without a good run - clear it by flashing again)");
    Serial.flush();
    return;
  }

  delay(2000);  // dejar que el USB enumere antes de cargar nada la alimentacion
  Serial.println("3V3: about to assert");
  Serial.flush();
  g_railsUp = true;  // para que la hibernacion sepa si aun se puede dormir a los sensores
  digitalWrite(PIN_PCB_EN_A, PIN_PCB_EN_A_ACTIVE);
  Serial.println("3V3: asserted");
  Serial.flush();
  delay(500);
  Serial.println("3V3: settled");
  Serial.flush();

  // Escritura simple, exactamente como en el programa minimo de rails.
  Serial.println("5V: about to assert (plain write)");
  Serial.flush();
  digitalWrite(PIN_PCB_EN_B, PIN_PCB_EN_B_ACTIVE);
  Serial.println("5V: asserted");
  Serial.flush();
  delay(500);
  Serial.println("5V: settled");
  Serial.flush();

  Serial.println("LoRa: about to assert");
  Serial.flush();
  pinMode(PIN_LORA_EN, OUTPUT);
  digitalWrite(PIN_LORA_EN, PIN_LORA_EN_ACTIVE);
  Serial.println("LoRa: asserted");

  // Habilitacion del BMV080 - ver PIN_BMV080_EN en pins.h. Sin ella el sensor abre
  // pero no enciende el laser.
  pinMode(PIN_BMV080_EN, OUTPUT);
  digitalWrite(PIN_BMV080_EN, PIN_BMV080_EN_ACTIVE);
  Serial.printf("BMV080 EN: GPIO%d -> %s\n", PIN_BMV080_EN,
                PIN_BMV080_EN_ACTIVE == LOW ? "LOW" : "HIGH");

  // La medida de bateria es una ENTRADA. Se deja como entrada expresamente, porque
  // este pin se condujo como salida por error durante la busqueda del BMV080.
  pinMode(PIN_SUPERCAP_SENSE, INPUT);
  Serial.flush();
  delay(500);

  Serial.println("rails up");
  Serial.flush();
}

// Cuanto aguanta un arranque en modo seguro antes de reiniciarse en modo normal. Lo
// bastante para pillarlo por el puerto serie y reflashear, y lo bastante corto para
// que el nodo no este mucho tiempo fuera de servicio. El modo seguro es una pausa,
// no un estado final.
#define SAFE_MODE_RECOVER_MS 60000UL

// UN PAQUETE AL ENTRAR EN HIBERNACION, y despues silencio.
//
// Mismo motivo que sendSafeModeDistress(): un nodo que simplemente se para parece
// exactamente un nodo que ha muerto. A proposito NO usa el formato de telemetria,
// para que nunca se confunda con una lectura: Modulo B apunta tal cual una trama que
// no reconoce, asi que aparece como kind=other en vez de perderse.
//
// Enciende SOLO el rail de LoRa. Los sensores siguen apagados: la bateria ya esta
// baja y todo este estado existe para dejar de gastarla.
//   Trama: HIB,<mV de la bateria>,<mV para salir>,<despertares>
static void sendHibernationNotice(float volts) {
  Serial.println("[HIBERNATE] sending one notice, then going silent");
  Serial.flush();

  pinMode(PIN_LORA_EN, OUTPUT);
  digitalWrite(PIN_LORA_EN, PIN_LORA_EN_ACTIVE);
  delay(PIN_LORA_EN_SETTLE_MS);

  if (!radio.begin(LORA_MY_ADDR, LORA_NETWORK_ID, LORA_BAND_HZ,
                   {LORA_PARAM_SF, LORA_PARAM_BW, LORA_PARAM_CR, LORA_PARAM_PREAMBLE},
                   &Serial)) {
    Serial.println("[HIBERNATE] radio init failed - hibernating unannounced");
    Serial.flush();
    return;
  }

  char msg[64];
  int len = snprintf(msg, sizeof(msg), "HIB,%d,%d,%lu", (int)(volts * 1000.0f),
                     (int)(HIB_EXIT_V * 1000.0f), (unsigned long)g_wakeCount);
  if (len < 0) len = 0;
  if (len >= (int)sizeof(msg)) len = (int)sizeof(msg) - 1;

  bool ok = radio.send(LORA_RX_ADDR, msg, (uint8_t)len);
  Serial.printf("[HIBERNATE] %s (%s)\n", msg, ok ? "sent" : "FAILED");
  Serial.flush();

  digitalWrite(PIN_LORA_EN, (PIN_LORA_EN_ACTIVE == LOW) ? HIGH : LOW);
}

// PAQUETE DE SOCORRO DEL MODO SEGURO.
//
// Enciende SOLO el rail de LoRa y la radio - ningun sensor, porque la carga de los
// sensores es el motivo habitual de estar en modo seguro - y manda un paquete
// diciendolo. Sin esto un nodo en modo seguro no se distingue de uno muerto, que es
// exactamente lo que costo dias de investigacion con el nodo C1.
//   Trama: SAFE,<arranques fallidos>,<motivo del reinicio>,<despertares>
//
// OJO: si lo que falla es el propio rail de LoRa (como en C3 el 2026-10-14, con un
// corto), este paquete tumba la placa en cada arranque y el modo seguro no puede
// salir solo: hay que reparar el hardware y borrar la cuenta en NVS.
static void sendSafeModeDistress() {
  Serial.println("[safe mode] sending distress packet (radio only, sensors stay off)");
  Serial.flush();

  pinMode(PIN_LORA_EN, OUTPUT);
  digitalWrite(PIN_LORA_EN, PIN_LORA_EN_ACTIVE);
  delay(PIN_LORA_EN_SETTLE_MS);

  if (!radio.begin(LORA_MY_ADDR, LORA_NETWORK_ID, LORA_BAND_HZ,
                   {LORA_PARAM_SF, LORA_PARAM_BW, LORA_PARAM_CR, LORA_PARAM_PREAMBLE},
                   &Serial)) {
    Serial.println("[safe mode] radio init failed - cannot report");
    Serial.flush();
    return;
  }

  // A proposito NO es el formato de telemetria: no debe confundirse con una lectura.
  // Modulo B apunta tal cual una trama que no reconoce en vez de tirarla, asi que
  // aparece en porthole_data.csv como kind=other.
  char msg[96];
  int len = snprintf(msg, sizeof(msg), "SAFE,%lu,%d,%lu", (unsigned long)nvsBoots,
                     (int)esp_reset_reason(), (unsigned long)g_wakeCount);
  if (len < 0) len = 0;
  if (len >= (int)sizeof(msg)) len = (int)sizeof(msg) - 1;

  bool ok = radio.send(LORA_RX_ADDR, msg, (uint8_t)len);
  Serial.printf("[safe mode] distress \"%s\" (%s)\n", msg, ok ? "sent" : "FAILED");
  Serial.flush();
}

// Un arranque en frio (encendido o reset, no un despertar del sueno) siempre
// inicializa todo y se queda despierto: es la ventana en la que se puede reflashear
// la placa.
static bool isColdBoot();

// Que debe este despertar: leer sensores y/o transmitir. Se decide ANTES de encender
// nada, para que un despertar sin trabajo pueda volver a dormir sin encender ni un
// solo rail.
static bool g_readDue = false;
static bool g_txDue = false;

// =============================================================================
// setup(): SE EJECUTA EN CADA DESPERTAR (cada despertar es un arranque nuevo)
// =============================================================================
//
// Orden:
//   1. Diagnostico del bus I2C antes de tocar nada; esperar al USB solo en frio.
//   2. Motivo del reinicio, guardia de arranques (modo seguro), y siembra de la
//      memoria RTC si es un encendido en frio (BOOTCOUNT_MAGIC).
//   3. Medir la bateria (antes de encender nada) y decidir si hiberna.
//   4. Consultar el calendario: que vence en este despertar. Si nada, a dormir sin
//      encender ningun rail.
//   5. Soltar los pines enclavados, encender el SEN0466 y los rails, recuperar el
//      bus I2C, arrancar SOLO los sensores que tocan, y la radio si hay envio.
// El trabajo en si (leer, transmitir, dormir) lo hace loop().
void setup() {
  Serial.begin(115200);

  // LO PRIMERO DE TODO, antes de que este firmware haya tocado un solo pin.
  //
  // Separa software de hardware sin discusion posible: si SCL ya esta sujeto a
  // masa AQUI, ningun codigo nuestro lo ha hecho - no ha corrido nada todavia.
  // Si en cambio aparece libre aqui y sujeto mas adelante, lo sujeta algo que
  // corre entre los dos puntos, y entonces se puede bisecar.
  delay(300);  // que el puerto USB llegue a enumerar y se vea esta linea
  i2cReportPullups("PRIMERA LINEA de setup");
  {
    // CORTO DURO o SOLO FALTA LA PULL-UP EXTERNA. El pull-up interno del ESP32
    // son unos 45 kOhm: no basta para un bus I2C de verdad, pero si para
    // levantar una linea que simplemente no tiene nada tirando de ella. Si
    // consigue subirla no hay cortocircuito; si no, la linea esta a masa.
    pinMode(PIN_I2C_SDA, INPUT_PULLUP);
    pinMode(PIN_I2C_SCL, INPUT_PULLUP);
    delay(2);
    const int sda = digitalRead(PIN_I2C_SDA);
    const int scl = digitalRead(PIN_I2C_SCL);
    Serial.printf("[i2c] con pull-up INTERNO: SDA=%s SCL=%s -> %s\n",
                  sda ? "ALTO" : "BAJO", scl ? "ALTO" : "BAJO",
                  scl ? "SCL sin corto: solo le falta la pull-up externa"
                      : "SCL A MASA de verdad: corto o pista a GND");
    Serial.flush();
  }

  // ESPERAR A UN PC POR USB SOLO EN UN ARRANQUE EN FRIO - nunca al despertar.
  //
  // Antes esperaba siempre:
  //
  //     while (!Serial && millis() - waitStart < 3000) delay(10);
  //     delay(500);
  //
  // En el campo NO hay PC, asi que `!Serial` sigue siendo verdad y cada despertar -
  // incluidos los que no tienen trabajo y deberian ser casi gratis - gastaba los 3 s
  // enteros, mas otros 500 ms. Con un despertar cada 10 s eran 3,5 s de cada 10: un
  // 35% del tiempo a ~40 mA, unos 14 mA de media, esperando un puerto serie que no
  // iba a aparecer nunca.
  //
  // Ademas se escondia durante el desarrollo: en el banco SI hay USB, `Serial` se
  // pone a verdad en milisegundos y la espera desaparece - asi que todas las medidas
  // hechas con cable subestimaban el consumo real en el campo.
  //
  // g_sleepFlag es RTC_NOINIT y sobrevive al sueno, asi que se puede leer aqui antes
  // de que corra nada mas. Un despertar del temporizador significa que nadie esta
  // mirando: directo al trabajo. Un arranque en frio significa que alguien puede
  // haberlo enchufado y quiere ver el mensaje de arranque, asi que ahi se espera.
  if (g_sleepFlag != SLEEP_FLAG_MAGIC) {
    uint32_t waitStart = millis();
    while (!Serial && millis() - waitStart < 3000) delay(10);
    delay(500);
  }

  Serial.println();
  Serial.println("=== CHIP FOREST + LoRa TX: sensors -> RYLR998 -> ground station ===");
#if FORCE_ALARM_STATE
  // Build de medicion: aviso bien visible de que el estado de alarma esta forzado.
  g_alarmState = (FORCE_ALARM_STATE >= 2) ? ALARM_FIRE : ALARM_PREALARM;
  g_alarmClearRun = 0;
  g_prealarmMs = 0;
  Serial.println("********************************************************");
  Serial.printf("*** TEST BUILD - ALARM STATE FORCED TO %-14s ***\n",
                (FORCE_ALARM_STATE >= 2) ? "ALARM" : "PRE-ALARM");
  Serial.println("*** The node is DEAF to its own sensors and NEVER     ***");
  Serial.println("*** sleeps. NOT FOR DEPLOYMENT. Rebuild without       ***");
  Serial.println("*** FORCE_ALARM_STATE before this board goes out.     ***");
  Serial.println("********************************************************");
  Serial.flush();
#endif
  Serial.flush();

  // Motivo del ultimo reinicio, en texto, para el log.
  esp_reset_reason_t rr = esp_reset_reason();
  const char *rrName = rr == ESP_RST_POWERON    ? "POWERON"
                       : rr == ESP_RST_EXT      ? "EXT"
                       : rr == ESP_RST_SW       ? "SW"
                       : rr == ESP_RST_PANIC    ? "PANIC (crash)"
                       : rr == ESP_RST_INT_WDT  ? "INT_WDT"
                       : rr == ESP_RST_TASK_WDT ? "TASK_WDT"
                       : rr == ESP_RST_WDT      ? "WDT"
                       : rr == ESP_RST_BROWNOUT ? "BROWNOUT (rail collapsed)"
                       : rr == ESP_RST_DEEPSLEEP ? "DEEPSLEEP"
                                                 : "OTHER";
  esp_sleep_wakeup_cause_t wakeCause = esp_sleep_get_wakeup_cause();

  // La marca manda; la causa del despertar solo se imprime como diagnostico.
  g_wokeFromTimer = (g_sleepFlag == SLEEP_FLAG_MAGIC);
  g_sleepFlag = 0;

  // La guardia de arranques necesita saber si este arranque es nuestro propio
  // despertar, asi que va AQUI y no antes: un despertar del sueno no debe contarse
  // como fallo. (Sigue yendo antes de tocar ningun rail, que era la idea original.)
  bootGuardBegin(g_wokeFromTimer);

  // ---- Siembra de la memoria RTC tras un encendido real ----
  // La palabra magica no coincide = la RTC trae basura (o es un firmware con otra
  // disposicion). Se ponen todas las variables RTC_NOINIT a un valor conocido.
  if (g_bootMagic != BOOTCOUNT_MAGIC) {
    g_bootMagic = BOOTCOUNT_MAGIC;
    g_bootCount = 0;
    g_wakeCount = 0;
    g_sleepFlag = 0;
    g_txPeriodMsPersist = LORA_TX_PERIOD_MS;
    // El reloj arranca sin hora y sin deriva conocida. Es lo correcto tras una
    // perdida de corriente: la hora que hubiera guardada seria de antes del corte
    // y no hay forma de saber cuanto duro. El nodo planifica con su tiempo propio
    // hasta que Modulo B le mande CFG,TIME en la primera ventana de escucha.
    //
    // La deriva arranca en 1.0 - "supongo que mi oscilador es perfecto" - y se
    // corrige sola en la SEGUNDA sincronizacion, que es la primera que tiene dos
    // puntos que comparar.
    g_nodeBaseMs = 0;
    g_epochMsAtSync = 0;
    g_nodeMsAtSync = 0;
    g_clockSynced = 0;
    g_clockSyncCount = 0;
    g_clockDrift = 1.0f;

    // El calendario arranca vacio y se re-fasea mas abajo, ya con el estado de
    // alarma sembrado. Los multiplicadores a 1.0 = los minutos de la tabla tal
    // cual, que es lo que el usuario especifico.
    g_readMultiplier = 1.0f;
    g_txMultiplier = 1.0f;
    for (uint8_t j = 0; j < JOB_COUNT; j++) g_nextDueMs[j] = 0;
    g_schedPhasedForState = ALARM_NORMAL;
    // La cache de lecturas tambien vive en RTC_NOINIT, asi que en un encendido real
    // contiene basura y hay que sembrarla aqui con todo lo demas.
    g_temp = g_hum = g_pres = g_gas = 0.0f;
    g_pm1 = g_pm25 = g_pm10 = 0.0f;
    g_co2 = 0.0f;
    g_co = g_coTemp = 0.0f;
    g_windAngle = g_windSpeed = 0.0f;
    g_windValid = false;
    // Un nodo recien sembrado arranca despierto. Si la bateria esta baja de verdad,
    // la comprobacion de setup() lo devuelve a hibernacion en este mismo arranque.
    g_hibernating = false;

    // EL ESTADO DE ALARMA TAMBIEN HAY QUE SEMBRARLO.
    //
    // Las tres viven en RTC_NOINIT, que sobrevive al deep sleep - que es lo que
    // se quiere - pero NO a un corte de alimentacion: ahi vuelven con basura. Y
    // este bloque, que existe justo para limpiar esa basura, se las saltaba.
    //
    // Lo vimos en vivo el 2026-10-07: C3 arranco en frio y, al entrar en
    // pre-alarma, escalo a ALARMA en 26 segundos en lugar de los 600 de la
    // ventana de confirmacion, porque g_prealarmMs vino con un valor cualquiera
    // mayor que el umbral.
    //
    // Lo grave no es ese. Es que g_alarmState tampoco se sembraba: un nodo que
    // pierda alimentacion puede arrancar directamente en ALARMA DE INCENDIO sin
    // que ningun sensor haya cruzado nada. En un sistema de deteccion de
    // incendios eso es una falsa alarma que se dispara justo cuando una placa se
    // reinicia en campo, que es cuando menos se puede ir a mirar.
    // (Las tres asignaciones de alarma estan repetidas dos veces; es inofensivo.)
    g_alarmState = ALARM_NORMAL;
    g_prealarmMs = 0;
    g_alarmClearRun = 0;
    g_storeCount = 0;
    g_storeDropped = 0;
    g_alarmState = ALARM_NORMAL;
    g_alarmClearRun = 0;
    g_prealarmMs = 0;
    seedThresholds();
    g_gasBaseline = 0.0f;
    g_lastReadHealth = 0;
    g_lastReadWake = 0;
    Serial.println("Boot counter initialised (true power-on, or RTC domain lost)");
  }

#if FORCE_ALARM_STATE
  // RE-APLICAR LA BANDERA DESPUES DE LA SIEMBRA, y este es el unico sitio valido.
  //
  // El bloque de arriba pone g_alarmState = ALARM_NORMAL, asi que forzar el estado
  // antes - donde esta el cartel - no sirve de nada en cuanto el magic no coincide:
  // la siembra lo pisa. Y el magic no coincide precisamente cuando mas se usa esta
  // bandera, que es al cambiar de build de medicion a otro.
  //
  // Visto el 2026-10-10: el arranque imprimia "ALARM STATE FORCED TO PRE-ALARM" y
  // acto seguido "alarma 0". El cartel decia una cosa y el nodo hacia otra, que es
  // la peor forma posible de fallar.
  g_alarmState = (FORCE_ALARM_STATE >= 2) ? ALARM_FIRE : ALARM_PREALARM;
  g_alarmClearRun = 0;
  g_prealarmMs = 0;
  g_schedPhasedForState = g_alarmState;  // o el envoltorio re-fasearia en la 1a vuelta
#endif
  if (g_wokeFromTimer) {
    g_wakeCount++;
    Serial.printf("Deep-sleep wake #%lu (cause=%d)\n", (unsigned long)g_wakeCount,
                  (int)wakeCause);
  } else {
    Serial.printf("COLD boot (cause=%d) - staying awake at least %lums so the port\n"
                  "  stays open long enough to reflash\n",
                  (int)wakeCause, (unsigned long)SLEEP_COLD_BOOT_AWAKE_MS);
  }
  // Recuperar el periodo de envio ajustado por comando, que sobrevive al sueno.
  g_txPeriodMs = g_txPeriodMsPersist;

  // El motivo del reinicio LO PRIMERO, antes de que cualquier salida temprana se lo
  // salte. Cuesta una linea por despertar y es la diferencia entre diagnosticar un
  // reinicio y adivinarlo.
  Serial.printf("Last reset reason: %d = %s\n", (int)rr, rrName);
  Serial.printf("Node address: %d (Module B sees this as +RCV addr=%d)\n", LORA_MY_ADDR,
                LORA_MY_ADDR);
  // Estado del RELOJ, que es de lo que depende el calendario entero.
  //
  // Sustituye a la linea "Tick: sensor_read N/M, lora_trans N/M" del planificador
  // viejo. Esa seguia imprimiendose con contadores que ya no decidian nada, y
  // hacia creer que el nodo continuaba funcionando por ticks - justo el tipo de
  // resto que manda un diagnostico por el camino equivocado.
  if (g_clockSynced) {
    Serial.printf("Reloj: %lu sincronizaciones, deriva %ld ppm | alarma %u\n",
                  (unsigned long)g_clockSyncCount,
                  (long)((g_clockDrift - 1.0f) * 1000000.0f), (unsigned)g_alarmState);
  } else {
    Serial.printf("Reloj: SIN sincronizar, planificando con tiempo propio | alarma %u\n",
                  (unsigned)g_alarmState);
  }

  // Verde: me he despertado. Antes de la posible vuelta a dormir de abajo, para que
  // todos los despertares lo muestren.
  ledFlashWake();

  // ---- HIBERNACION POR BATERIA BAJA: se decide antes de encender nada ----
  //
  // Se mide aqui, al principio del despertar, por un motivo concreto: la tension
  // CAE con la carga. Una lectura tras una rafaga de LoRa o un turno de sensor
  // mediria la caida, no la bateria, y dejaria parado horas a un nodo sano. En este
  // punto no se ha encendido nada desde que el sueno lo enclavo todo apagado, asi
  // que es la bateria en reposo - el numero mas justo posible.
  //
  // El divisor esta directamente sobre la bateria y no necesita rail propio, asi
  // que cuesta una lectura del ADC.
  float packV = supercapVolts();

  if (g_hibernating) {
    // En hibernacion el UNICO trabajo es decidir si la carga ya la ha recuperado.
    // Sin sensores, sin radio, sin rails: mirar y volver a dormir.
    if (packV >= HIB_EXIT_V) {
      Serial.printf("[HIBERNATE] pack recovered to %.2fV (>= %.2fV) - resuming normal operation\n",
                    packV, (double)HIB_EXIT_V);
      Serial.flush();
      g_hibernating = false;
      // Sigue desde aqui como un despertar normal.
    } else {
      Serial.printf("[HIBERNATE] pack %.2fV, waiting for %.2fV - back to sleep for %lus\n",
                    packV, (double)HIB_EXIT_V, HIB_CHECK_SECONDS);
      Serial.flush();
      enterDeepSleep("still hibernating", HIB_CHECK_SECONDS);
      return;  // no se llega nunca
    }
  } else if (packV <= HIB_ENTER_V && packV >= SUPERCAP_V_SANITY_FLOOR && !isColdBoot()) {
    // GUARDA DEL SUELO DE CORDURA: una placa antigua no tiene divisor y lee ~0,12 V,
    // lo que de otro modo haria hibernar para siempre a cada placa asi en su primer
    // despertar. Una lectura tan baja significa "no hay divisor", no "bateria
    // vacia" - ver la misma guarda en transmitStatus().
    //
    // ARRANQUE EN FRIO EXCLUIDO para que un nodo con la bateria vacia se pueda
    // reflashear. El sueno profundo apaga el USB; hibernar nada mas encender no
    // dejaria ventana para reprogramarlo, y un nodo que no se puede reprogramar es
    // peor que uno que gasta un poco mas. Primero corre la ventana de 30 s del
    // arranque en frio, y loop() hiberna al terminarla.
    enterHibernation(packV, true);
    return;  // no se llega nunca
  }

  // Los umbrales, en CADA arranque y no solo cuando cambia el magic: viven en
  // RTC_NOINIT y pueden corromperse sin que el magic se entere.
  validateThresholds();


  // ---- Que debe este despertar, segun el PLANIFICADOR POR TIEMPO ----
  //
  // En arranque en frio el calendario guardado no significa nada: o es basura de
  // RTC_NOINIT sin inicializar, o es de antes de un corte de corriente y no hay
  // forma de saber cuanto duro. Se re-fasea desde ahora.
  //
  // Y SE FUERZA UNA TRANSMISION. Con la tabla nueva el primer envio caeria hasta
  // 31 minutos despues de encender el nodo: media hora de silencio desde que lo
  // enchufas hasta que da senales de vida. Peor aun, la hora de red solo baja en
  // la ventana de escucha POSTERIOR a una transmision, asi que sin esto el nodo
  // estaria media hora sin reloj - y por tanto sin poder alinearse.
  if (isColdBoot()) {
    schedRephase(schedNowMs(), g_alarmState, "arranque en frio");
    // TODO vence ya, no solo el envio.
    //
    // Re-fasear deja todos los vencimientos en el FUTURO, asi que sin esto un nodo
    // recien encendido transmitia vacio y se dormia sin haber medido nada - visto
    // en el arranque del 2026-10-08, donde no aparecio ni una linea [t] y el envio
    // fue "nothing stored this cycle".
    //
    // Al encender un nodo se quiere una lectura completa inmediata: confirma que
    // cada sensor responde, le da al estado de alarma algo real sobre lo que
    // decidir en vez de los ceros de la siembra, y hace que la primera trama lleve
    // datos. El envio va incluido porque es el ultimo del bucle.
    for (uint8_t j = 0; j < JOB_COUNT; j++) g_nextDueMs[j] = 0;
  }
  schedEvaluate();

  g_readDue = schedAnySensorDue();
  g_txDue = g_jobDue[JOB_TX];

  // ---- Despertar sin trabajo: no encender nada, volver a dormir ----
  // Esta decision vivia antes en loop(), lo que significaba que cada despertar
  // encendia primero los tres rails, pagaba los 5 s de precalentamiento del BMV080 e
  // iniciaba la radio - para descubrir despues que no tenia nada que hacer.
  //
  // No es solo energia. Encender tres rails y un laser es un escalon de carga grande
  // y simultaneo, y hacerlo mas veces de las necesarias son mas ocasiones de que se
  // caiga una alimentacion con poco margen.
  //
  // Los rails siguen apagados aqui (el sueno los enclavo apagados y nada los ha
  // conducido desde entonces), asi que no hay nada que deshacer: solo dormir otra vez.
#if SLEEP_ENABLED
  // TODOS los trabajos tienen que contar aqui. Olvidar uno no se salta un despertar:
  // ese trabajo no se hace NUNCA, porque esta salida duerme antes de encender ningun
  // rail. (En el planificador por ticks se olvido asi una de las banderas al anadir
  // el grupo caro de sensores.)
  if (!g_readDue && !g_txDue && !isColdBoot()) {
    // Con el planificador por tiempo este caso casi no se da: el nodo despierta
    // porque ALGO vence, no porque haya pasado un tick. Se conserva como red de
    // seguridad - un despertar espurio no debe encender ningun rail - y ahora
    // vuelve a dormir hasta el proximo vencimiento de verdad, no 10 s fijos.
    enterDeepSleep("nada vence - sin encender rails", schedSleepSeconds(g_alarmState));
    return;  // no se llega nunca: enterDeepSleep no vuelve
  }
#else
  // Modo continuo: lo inicializa todo cada vez; no hay despertares que saltarse, y
  // salir aqui dejaria setup() a medias.
  g_readDue = true;
  g_txDue = true;
#endif

  // El color dice QUE trabajo hace este despertar, y se mantiene hasta dormir:
  //   AZUL    lectura - rails y sensores
  //   BLANCO  solo transmision - radio
  // Un despertar que hace las dos empieza azul y pasa a blanco al transmitir.
  if (g_readDue) {
    ledReading();
  } else {
    ledWorking();
  }
  g_bootCount++;
  Serial.printf("Boot #%lu since last power loss\n", (unsigned long)g_bootCount);
  Serial.flush();

  // El sueno profundo dejo estos pines enclavados (ver kHeldGates). No se pueden
  // conducir hasta soltar el enclavamiento, asi que esto tiene que ir ANTES de los
  // pinMode de abajo - si no, las escrituras se ignoran en silencio y los rails se
  // quedan apagados. Es IMPRESCINDIBLE para que el sueno funcione: sin soltarlos,
  // los rails no vuelven a encenderse nunca despues del primer despertar.
  for (size_t i = 0; i < sizeof(kHeldGates) / sizeof(kHeldGates[0]); i++) {
    gpio_hold_dis(kHeldGates[i]);
  }
  gpio_deep_sleep_hold_dis();

  // Alimentacion del sensor de CO, lo primero de todo: es la pieza de la placa que
  // mas tarda en ser utilizable, asi que cada segundo que se encienda antes es un
  // segundo de calentamiento. Tambien en modo seguro: no es la carga que el modo
  // seguro existe para quitar.
  pinMode(PIN_SEN0466_EN, OUTPUT);
  digitalWrite(PIN_SEN0466_EN, SEN0466_GATE_ALWAYS_ON ? PIN_SEN0466_EN_ACTIVE : !PIN_SEN0466_EN_ACTIVE);
  Serial.printf("SEN0466 EN: GPIO%d -> %s\n", PIN_SEN0466_EN,
                digitalRead(PIN_SEN0466_EN) == LOW ? "LOW" : "HIGH");

  // MODO SEGURO: con 12 o mas arranques seguidos sin una vuelta buena, los rails se
  // quedan apagados, para que una placa que no aguanta su alimentacion siga dando un
  // puerto serie utilizable. (El umbral se subio desde 3: mientras se reproducia el
  // fallo a proposito, un disparo a los 3 arranques secuestraba cada prueba antes de
  // que saliera lo interesante.)
  g_safeMode = (nvsBoots >= 12);
  // Medido ANTES y DESPUES de encender los rails. Si el bus esta igual en los dos
  // momentos, el rail no esta llegando a las pull-up y encenderlo no cambia nada -
  // que es exactamente la pregunta abierta.
  i2cReportPullups("ANTES de encender rails");


  // NOTA: aqui hubo un corte de 1500 ms del rail de 3V3 para descolgar un posible
  // esclavo de I2C. Se quito porque la medida lo desmintio: con el rail cortado,
  // SCL seguia sujeto a masa. La causa real es un esclavo que NO esta en ese rail
  // - el SEN0466 iba cableado directo a corriente saltandose el transistor - asi
  // que no habia nada que el firmware pudiera apagar para reiniciarlo. Desde el
  // 2026-10-14 tiene su propia puerta en GPIO16 (PIN_SEN0466_EN): cortarla es
  // ahora la forma de descolgarlo sin desenchufar la placa.
  enableRails(g_safeMode);
  i2cReportPullups("DESPUES de encender rails");

  if (g_safeMode) {
    Serial.println("Rails off - skipping sensor and radio init this boot.");
    Serial.flush();
    return;  // sin alimentacion no puede funcionar nada de lo de abajo
  }

  power.enable3V3Sensors();  // stub: solo imprime, ver PowerManager.h
  // Antes de Wire.begin: si un esclavo quedo colgado sujetando SDA de un reset
  // anterior, el bus esta inservible y Wire no lo detecta - devuelve ACK falsos.
  i2cRecoverBus();
  Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL);

  // Diagnostico del bus en cada arranque.
  scanI2C("after rails, before sensor init");
  probeBmeChipId("right after scan");

  // ---- ARRANQUE SELECTIVO: solo los sensores que vencen en este despertar ----
  //
  // SOLO SI EL BMV080 VA A LEERSE. Son cinco segundos, y pagarlos en un
  // despertar que solo viene a por la temperatura retrasa esa temperatura cinco
  // segundos sobre el instante en el que tenia que existir.
  if (g_jobDue[JOB_BMV080]) {
    Serial.printf("Waiting %dms for BMV080 startup...\n", BMV080_STARTUP_DELAY_MS);
    delay(BMV080_STARTUP_DELAY_MS);
  }

  probeBmeChipId("after BMV080 wait, before begin");
  // El BME690 sirve a DOS trabajos sobre el mismo chip - T/H/P con el calefactor
  // apagado y VOC con el encendido - asi que arranca si vence cualquiera.
  if (g_jobDue[JOB_BME_THP] || g_jobDue[JOB_BME_VOC]) {
    step("bme690.begin");
    bmeReady = bme690.begin();
  } else {
    bmeReady = false;
  }
  // begin() prueba la direccion configurada y luego la otra, asi que se imprime en
  // cual se quedo: un "OK" a secas no distingue 0x76 de 0x77.
  if (bmeReady) {
    Serial.printf("BME690: OK (bound to 0x%02X)\n", bme690.address());
  } else if (bme690.chipIdValid()) {
    // El driver lo rechazo pero el registro se sigue leyendo. 0x61 significa que la
    // pieza esta y esta sana y el fallo es del DRIVER; cualquier otro valor
    // significa que el bus corrompe una lectura de varios bytes aunque siga dando ACK
    // a una prueba de direccion.
    Serial.printf("BME690: NOT FOUND at 0x%02X - chip ID 0x%02X via mode %d (expect 0x61) -> %s\n",
                  bme690.address(), bme690.lastChipId(), bme690.chipIdMode(),
                  bme690.lastChipId() == 0x61 ? "part is FINE, driver init failed"
                                              : "BUS CORRUPTION or wrong part");
  } else {
    Serial.printf("BME690: NOT FOUND at 0x%02X - chip ID register does not respond\n",
                  bme690.address());
  }

#if SLEEP_ENABLED && SLEEP_SKIP_SEN0466
  // Fuera del ciclo: sus 210 s de estabilizacion dominarian cada despertar. Se
  // desactiva en vez de simplemente no leerlo. (Hoy SLEEP_SKIP_SEN0466 vale 0, asi
  // que esta rama no se compila.)
  g_sen0466Enabled = false;
  sen0466Ready = false;
  Serial.println("SEN0466: SKIPPED (210s settle - excluded from the sleeping cycle)");
#else
  if (g_jobDue[JOB_SEN0466]) {
    step("sen0466.begin");
    sen0466Ready = sen0466.begin();
  }
  Serial.println(sen0466Ready ? "SEN0466: OK" : "SEN0466: NOT FOUND");
#endif

  // ---- ORDEN DE ARRANQUE INTERCAMBIADO (diagnostico del 2026-09-04) ----
  // El orden original era bmv080 y despues cm1106, y la placa se reiniciaba en
  // cm1106.begin() en todos los arranques en que el BMV080 arrancaba - nunca cuando
  // no lo hacia. Intercambiarlos decia que tipo de fallo era (carga acumulada o un
  // problema de tiempos). La respuesta fue la carga simultanea, y el orden se quedo
  // asi; los textos "SWAPPED" del log son de esa prueba.
  // El CM1106 hace un ciclo de alimentacion y espera su calentamiento en begin(),
  // asi que tambien cuesta segundos que no hay que pagar si no vence.
  if (g_jobDue[JOB_CM1106]) {
    step("cm1106.begin (SWAPPED: now before bmv080)");
    cm1106.begin();
    step("cm1106.begin returned");
  }

  if (g_jobDue[JOB_BMV080]) {
    step("bmv080.begin (SWAPPED: now after cm1106)");
    bmvReady = bmv080.begin();
    // En que pasada abrio, para saber si los reintentos estan salvando arranques
    // o si el sensor abria igual a la primera.
    if (bmvReady) {
      Serial.printf("BMV080: abrio en la pasada %u de %u\n",
                    (unsigned)bmv080.openPasses(), 3u);
    }
  } else {
    bmvReady = false;
  }
  if (bmvReady) {
    Serial.println("BMV080: OK");
  } else if (g_jobDue[JOB_BMV080]) {
    // SOLO SI DE VERDAD SE INTENTO ABRIRLO.
    //
    // Con el arranque selectivo, un despertar en el que al BMV080 no le toca medir
    // no llama a begin() - y hasta el 2026-10-12 este mensaje salia igual, con el
    // estado del SDK a 0, que es E_BMV080_OK: el valor inicial de una variable que
    // nadie habia tocado.
    //
    // El efecto era un log que decia "NOT FOUND" 26 veces en dos horas mientras el
    // sensor abria las 5 veces que se le pidio. Se paso un buen rato diagnosticando
    // una fiabilidad del 16% que en realidad era del 100%.
    //
    // Y se sigue diciendo QUE etapa fallo: "NOT FOUND" a secas no distingue un
    // sensor ausente de uno presente que no abre, y esas dos cosas apuntan al bus y
    // a la alimentacion respectivamente.
    Serial.printf("BMV080: NOT FOUND - open failed at every strap address "
                  "(0x57/0x56/0x55/0x54), last SDK status %d\n",
                  bmv080.lastOpenStatus());
  }
  // Laser apagado hasta que de verdad se quiera una medida. (begin() ya no lo deja
  // midiendo, ver Bmv080Sensor.cpp; esto es una garantia extra.)
  if (bmvReady) {
    bmv080.stopMeasurement();
    g_bmvPhase = BmvPhase::Idle;
  }

  step("calypso.begin next");
  calypso.begin();
  step("calypso.begin returned - radio next");

  // La radio se configura en cada despertar con trabajo, toque enviar o no.
  Serial.println("Configuring RYLR998 (link parameters must match pp1-lora-receiver):");
  radioReady = radio.begin(LORA_MY_ADDR, LORA_NETWORK_ID, LORA_BAND_HZ,
                            {LORA_PARAM_SF, LORA_PARAM_BW, LORA_PARAM_CR, LORA_PARAM_PREAMBLE}, &Serial);
  if (radioReady) {
    bool powerSet = radio.setTxPower(LORA_TX_POWER_DBM, &Serial);
    Serial.printf("LoRa TX power -> %d dBm: %s\n", LORA_TX_POWER_DBM,
                  powerSet ? "OK" : "REJECTED (still at module default, likely 22)");
  }
  Serial.println(radioReady ? "Radio init OK"
                             : "Radio init FAILED - not transmitting until this is fixed "
                               "(see the AT exchange above for which step was rejected)");

  Serial.println("Setup complete.\n");
  g_setupDoneMs = millis();
}

// =============================================================================
// 7. SUENO PROFUNDO
// =============================================================================

// Lo apaga todo y duerme. Los rails se APAGAN y sus puertas se enclavan, asi que
// nada de lo que cuelga de ellos consume durante el sueno: es ese corte, y no los
// ~10 uA del propio microcontrolador, de donde sale el ahorro de verdad.
static void enterDeepSleep(const char *why, uint32_t seconds) {
#if !SLEEP_ENABLED
  (void)why;
  return;
#else
  // Apagar el LED antes de dormir. Un WS2812 mantiene su ultimo valor, asi que sin
  // esto seguiria encendido todo el sueno - gastando y dejando el indicador sin
  // sentido.
  ledOff();

  Serial.printf("\n[sleep] %s - sleeping %lus\n", why, (unsigned long)seconds);

  // Poner cada aparato en reposo antes de que desaparezca su rail, en vez de
  // quitarle la corriente con el laser o un calefactor en marcha.
  if (bmvReady) bmv080.stopMeasurement();
  cm1106.sleep();

  // Guardar lo que necesite el siguiente despertar, mientras la RAM aun existe.
  g_txPeriodMsPersist = g_txPeriodMs;

  // Rails apagados. Las dos puertas tienen polaridad OPUESTA, asi que cada una
  // necesita el inverso de su propio nivel activo - un unico "nivel de apagado"
  // compartido apagaria un rail y ENCENDERIA el otro durante todo el sueno.
  // (GPIO16, la puerta del SEN0466, no se toca aqui: conserva el nivel que le dio
  // setup() y se enclava con las demas.)
  digitalWrite(PIN_PCB_EN_A, (PIN_PCB_EN_A_ACTIVE == LOW) ? HIGH : LOW);
  digitalWrite(PIN_PCB_EN_B, (PIN_PCB_EN_B_ACTIVE == LOW) ? HIGH : LOW);
  digitalWrite(PIN_LORA_EN, (PIN_LORA_EN_ACTIVE == LOW) ? HIGH : LOW);

  // Enclavar esos niveles durante el sueno - ver kHeldGates.
  for (size_t i = 0; i < sizeof(kHeldGates) / sizeof(kHeldGates[0]); i++) {
    gpio_hold_en(kHeldGates[i]);
  }
  gpio_deep_sleep_hold_en();

  Serial.flush();
  delay(SLEEP_USB_DRAIN_MS);

  g_sleepFlag = SLEEP_FLAG_MAGIC;  // para que el siguiente arranque sepa que viene del sueno

  // EL TIEMPO DEL NODO SE ACUMULA AQUI, Y ESTE ES EL UNICO SITIO.
  //
  // `millis()` se reinicia en cada despertar, asi que sin esto el nodo no tendria
  // ninguna nocion de cuanto lleva vivo mas alla del arranque actual. Se suma lo
  // que se estuvo despierto MAS lo que se va a dormir, porque despues de
  // esp_deep_sleep_start() no se ejecuta ninguna linea mas: no hay un "despues"
  // donde apuntarlo.
  //
  // Lo que se apunta es lo que se PIDIO dormir, no lo que de verdad se durmio -
  // que nadie puede medir sin un reloj mejor que el que estamos usando. Esa
  // diferencia es exactamente el error del oscilador RC, y es lo que corrige
  // g_clockDrift cuando llega la siguiente sincronizacion con Modulo B.
  g_nodeBaseMs += (uint64_t)millis() + (uint64_t)seconds * 1000ULL;

  esp_sleep_enable_timer_wakeup((uint64_t)seconds * 1000000ULL);
  esp_deep_sleep_start();
  // no vuelve nunca
#endif
}

// Un nodo arrancado en frio tiene que quedarse despierto lo suficiente para
// reflashearlo. El sueno profundo apaga el USB-Serial/JTAG, y el puerto solo
// existiria uno o dos segundos cada vez - casi imposible de pillar. Asi que tras un
// encendido o un reset (no un despertar del temporizador) el nodo se niega a dormir
// hasta que pasa esta ventana. Un despertar del temporizador se la salta y duerme en
// cuanto acaba su trabajo.
// Arranque en frio = cualquier arranque que no venga de nuestro propio sueno. Se
// decide con la marca en RTC, no con la causa del despertar, porque un reinicio del
// chip borra la causa (ver la declaracion de la marca).
static bool isColdBoot() { return !g_wokeFromTimer; }

static bool coldBootWindowOpen() {
  if (g_wokeFromTimer) return false;
  uint32_t awake = millis() - g_setupDoneMs;
  if (awake >= SLEEP_COLD_BOOT_AWAKE_MS) return false;
  static uint32_t lastNoticeMs = 0;
  if (millis() - lastNoticeMs >= 5000) {
    lastNoticeMs = millis();
    Serial.printf("[boot] holding awake %lus more for reflashing (cold boot)\n",
                  (unsigned long)((SLEEP_COLD_BOOT_AWAKE_MS - awake) / 1000));
  }
  return true;
}

// =============================================================================
// 8. LECTURA DE SENSORES, DE UNO EN UNO ("SLOTS")
// =============================================================================

// Que sensores se esperaban y aun no tienen lectura fresca, por nombre. Lo usa
// readFastSensorsOnce(), que solo se ejecuta en el modo sin sueno (SLEEP_ENABLED=0).
//
// Un sensor se "espera" solo si esta activado Y presente. Uno que no arranco
// (bmvReady == false) o que se silencio por comando no debe esperarse nunca. El
// Calypso no tiene indicador propio de "listo", asi que solo cuenta como esperado
// cuando ha dado ALGUNA lectura alguna vez: un anemometro desenchufado no debe
// retener los datos de los demas sensores.
static bool calypsoEverSeen = false;

static uint8_t pendingSensors(char *out, size_t outCap) {
  uint8_t pending = 0;
  if (out && outCap) out[0] = 0;
  struct { bool expected; bool fresh; const char *name; } checks[] = {
      {bmeReady && g_bmeEnabled, g_bmeFresh, "BME690"},
      {bmvReady && g_bmvEnabled, g_bmvFresh, "BMV080"},
      {sen0466Ready && g_sen0466Enabled, g_coFresh, "SEN0466"},
      {g_cm1106Enabled, g_co2Fresh, "CM1106"},
      {g_calypsoEnabled && calypsoEverSeen, g_calFresh, "Calypso"},
  };
  for (size_t i = 0; i < sizeof(checks) / sizeof(checks[0]); i++) {
    if (checks[i].expected && !checks[i].fresh) {
      pending++;
      if (out && outCap) {
        if (out[0]) strncat(out, ",", outCap - strlen(out) - 1);
        strncat(out, checks[i].name, outCap - strlen(out) - 1);
      }
    }
  }
  return pending;
}

// Borra todas las marcas de "fresco". Se llama al empezar cada barrido.
static void clearFreshFlags() {
  g_bmeFresh = g_gasFresh = g_bmvFresh = g_co2Fresh = g_coFresh = g_calFresh = false;
}

// ---------------- Turnos de sensor ("slots") ----------------
// UN SOLO sensor activo cada vez, sin solaparse nunca. Este es el arreglo de los
// reinicios que bloquearon el proyecto durante dias, y es la propiedad mas
// importante de este archivo: no "optimizarlo" volviendo a una sola pasada.
//
// Las medidas que lo zanjaron, en este hardware:
//
//   todos los sensores a la vez      31 reinicios / 140 s
//   BMV080 apagado, el resto a la vez  1 reinicio  / 140 s
//   turnos, todo encendido            2 reinicios / 150 s   (los dos fueron el reflasheo)
//
// El laser del BMV080 solo ya son ~68 mA - un orden de magnitud mas que cualquier
// otra cosa - y el codigo antiguo lo dejaba midiendo mientras se leian el BME690,
// el CM1106 y el SEN0466. Ese solape es lo que tumbaba la placa. Escalonar los rails
// y arrancar la puerta suavemente fallaron porque atacaban la CORRIENTE DE ARRANQUE,
// y el problema era el solape en regimen permanente.
//
// notes/power_budget.md lo especificaba desde el principio: "rails conmutados para
// que el BMV080 de 68 mA y el SEN0466 de 5 mA/210 s solo consuman en sus ventanas".
enum Slot {
  SLOT_BME690 = 0,
  SLOT_BMV080,
  SLOT_CM1106,
  SLOT_SEN0466,
  SLOT_CALYPSO,
  SLOT_COUNT
};

// Nombre del turno para el log (lo usa readFastSensorsOnce()).
static const char *slotName(int slot) {
  switch (slot) {
    case SLOT_BME690: return "BME690";
    case SLOT_BMV080: return "BMV080";
    case SLOT_CM1106: return "CM1106";
    case SLOT_SEN0466: return "SEN0466";
    case SLOT_CALYPSO: return "Calypso";
  }
  return "?";
}

// Lo maximo que puede estar encendido el laser del BMV080 esperando un dato. Los
// datos llegan normalmente 2-4 s despues del precalentamiento. Esto acota un turno;
// BMV080_MEASURE_TIMEOUT_MS es demasiado largo para gastarlo dentro de una pasada
// secuencial (y de hecho ya no se usa).
#define BMV080_SLOT_MS 8000UL

// Hueco entre turnos normales: lo justo para que una carga pequena deje de consumir
// antes de que empiece la siguiente.
#define SLOT_GAP_MS 1500UL

// Cuanto se mantienen APAGADOS los rails de sensores despues de que funcione el
// calefactor de gas del BME690. Un simple hueco no bastaba, ni a 250 ms ni a 1500
// ms: solo un ciclo real del rail dejaba arrancar despues al BMV080. Este es el
// tiempo apagado de ese ciclo.
#define HEATER_RECOVERY_OFF_MS 800UL

// Espera tras cortar los rails de sensores y antes de que la radio transmita. Lo
// bastante para que los rails descarguen de verdad sus cargas y la capacidad de la
// alimentacion se recupere antes de la rafaga.
#define TX_RAIL_SETTLE_MS 400UL

// RAILS DE SENSORES APAGADOS - no solo los sensores en reposo.
//
// quiesceAll() para el BMV080 y baja el EN del CM1106, pero deja los dos rails de
// sensores alimentados, asi que cada pieza sigue consumiendo su corriente de reposo
// durante la transmision. Consecuencia medida: la placa sufre un reinicio POWERON -
// un colapso real de la alimentacion - justo despues de "Setup complete." en un
// despertar de envio, es decir, exactamente en radio.send().
//
// La radio no necesita ninguno de los dos rails. Asi que la transmision tiene su
// turno exclusivo: 3V3 y 5V apagados, rail de LoRa encendido, espera, y envio. Es el
// mismo principio que arreglo los reinicios de los sensores, aplicado a la unica
// carga que aun compartia.
//
// Los rails NO se vuelven a encender despues: transmitir es lo ultimo que hace un
// despertar antes de dormir, y enterDeepSleep() los apaga igualmente.
//
// ---- TX_KEEP_5V_RAIL: mantener el rail de 5 V durante la transmision ----
//
// Cortar los dos rails para transmitir fue el arreglo de un reinicio POWERON en
// radio.send() - un colapso real de la alimentacion -, y eso no se deshace a la
// ligera. Pero se diagnostico cuando la placa iba con SUPERCONDENSADORES, cuya ESR no
// podia con el pico de 151 mA de la radio encima de los sensores. Desde entonces la
// fuente es una bateria de litio de 10 Ah, mucho mas firme.
//
// Importa porque el SEN0466 necesitaba 210 s de 5 V CONTINUOS y el nodo transmite a
// menudo en alarma, asi que con los dos rails cortados el sensor de CO no podria
// calentarse nunca.
//
// ACTIVADO POR DEFECTO el 2026-10-07, y la razon ya no es solo el CO.
//
// El Calypso tambien vive en el rail de 5V, y mostro el mismo sintoma: la primera
// lectura tras el arranque recibio 27 bytes - cuando el rail llevaba todo el
// setup() alimentado - y las doce siguientes cero. Dos sensores distintos
// fallando igual por la misma causa dejan poca duda.
//
// Medido en C3: 290 s en alarma, 11 transmisiones con el rail de 5V levantado,
// SIN UN SOLO REINICIO (nvsBoots=1).
//
// AVISO QUE SIGUE EN PIE: esa prueba se hizo con C3 alimentado desde una fuente
// de banco, NO desde el pack de litio montado. Demuestra que el firmware
// funciona, no que la alimentacion real lo aguante. El corte original se puso
// contra un reset POWERON de verdad en radio.send(), aunque diagnosticado con
// SUPERCONDENSADORES, cuya ESR no podia con los 151 mA de pico de la radio. Hay
// que repetirlo en un nodo con bateria antes de darlo por bueno en campo; si
// aparece un POWERON tras "[tx] 3V3 off", volver a 0 y la respuesta es hardware.
#ifndef TX_KEEP_5V_RAIL
#define TX_KEEP_5V_RAIL 1
#endif

// Apaga los rails de sensores antes de transmitir (el de 5 V solo si
// TX_KEEP_5V_RAIL vale 0) y espera a que se estabilice la alimentacion.
static void powerDownSensorRails() {
  g_railsUp = false;
  digitalWrite(PIN_PCB_EN_A, (PIN_PCB_EN_A_ACTIVE == LOW) ? HIGH : LOW);
#if !TX_KEEP_5V_RAIL
  digitalWrite(PIN_PCB_EN_B, (PIN_PCB_EN_B_ACTIVE == LOW) ? HIGH : LOW);
  Serial.println("[tx] sensor rails off for the transmit");
#else
  Serial.println("[tx] 3V3 off for the transmit, 5V HELD UP (TX_KEEP_5V_RAIL)");
#endif
  Serial.flush();
  delay(TX_RAIL_SETTLE_MS);
}

// Todo en reposo. Se llama entre turnos y antes de cada envio, para que la rafaga
// de ~120 mA de la radio nunca caiga encima de un laser o de un calefactor.
static void quiesceAll() {
  if (bmvReady && g_bmvPhase != BmvPhase::Idle) {
    bmv080.stopMeasurement();
    g_bmvPhase = BmvPhase::Idle;
  }
  cm1106.sleep();  // EN a nivel bajo
}

// Ejecuta el turno de UN sensor. Bloqueante A PROPOSITO: la idea es que no consuma
// nada mas mientras este lo hace. Cada turno acota su propio tiempo.
static void runSlot(int slot) {
  switch (slot) {
    case SLOT_BME690: {
      if (!bmeReady || !g_bmeEnabled) break;
      Reading env = bme690.read();
      g_bmeSt = env.status;
      if (env.ok()) {
        g_bmeFresh = true;
        // ESTE es el unico sitio donde el gas se mide de verdad: la lectura
        // completa, con el calefactor a 320 C. readBmeFastOnly() deja values[3] a
        // cero y no toca g_gas ni esta bandera.
        g_gasFresh = true;
        g_temp = env.values[0];
        g_hum = env.values[1];
        g_pres = env.values[2];
        g_gas = env.values[3];
      }
      break;
    }

    case SLOT_BMV080: {
      if (!bmvReady || !g_bmvEnabled) break;
      // Laser encendido solo durante este turno y apagado antes de que acabe, por
      // cualquier camino.
      if (!bmv080.startMeasurement()) {
        Serial.printf("[bmv080] start FAILED - SDK status %d (see bmv080_status_code_t)\n",
                      bmv080.lastStartStatus());
        Serial.flush();

        // NOTA: aqui habia un barrido de GPIO12/GPIO14, de antes de que se
        // describiera la placa. Se QUITO y no debe volver asi: GPIO12 va al CM1106 y
        // GPIO14 es la medida analogica de la bateria, asi que ese bucle conducia la
        // linea de un sensor que funciona y peleaba con un divisor analogico. La
        // habilitacion resulto ser GPIO15 (PIN_BMV080_EN), que se activa una vez con
        // los rails.
        g_bmvSt = ReadingStatus::NoAck;
        break;
      }
      g_bmvPhase = BmvPhase::Measuring;
      uint32_t t0 = millis();
      delay(BMV080_SETTLE_MS);  // precalentamiento: los datos de antes no valen
      while (millis() - t0 < BMV080_SLOT_MS) {
        Reading pm = bmv080.read();
        g_bmvSt = pm.status;
        if (pm.ok()) {
          g_bmvFresh = true;
          g_pm1 = pm.values[0];
          g_pm25 = pm.values[1];
          g_pm10 = pm.values[2];
          break;
        }
        delay(50);
      }
      bmv080.stopMeasurement();
      g_bmvPhase = BmvPhase::Idle;
      Serial.printf("[bmv080] laser off after %lums (pm2.5=%.1f, %s)\n",
                    (unsigned long)(millis() - t0), g_pm25,
                    g_bmvFresh ? "frame" : "no frame");
      break;
    }

    case SLOT_CM1106: {
      if (!g_cm1106Enabled) break;
      // Pieza de un solo disparo: mide al encenderse y repite ese valor el resto del
      // tiempo, asi que un numero NUEVO cuesta un ciclo de alimentacion con EN cada
      // vez. No es un apano de una averia: es como se comporta la pieza.
      cm1106.powerCycle();
      Reading co2 = cm1106.read();
      g_co2St = co2.status;
      if (co2.ok()) {
        g_co2Fresh = true;
        g_co2 = co2.values[0];
      }

      // Los bytes del cable, no los ppm interpretados. Un ppm que no se mueve es
      // ambiguo; tramas identicas byte a byte no. Ojo: resp[6] es un contador que
      // sube en cada lectura, asi que frozenValueRun compara los bytes del VALOR,
      // nunca tramas enteras.
      Serial.print("[cm1106 raw] ");
      if (cm1106.lastRawLen() == 0) {
        Serial.print("(nothing received)");
      } else {
        for (uint8_t i = 0; i < cm1106.lastRawLen(); i++)
          Serial.printf("%02X ", cm1106.lastRaw()[i]);
      }
      Serial.printf("| status=%s frozenValueRun=%u\n", statusName(co2.status),
                    (unsigned)cm1106.frozenValueRun());

      cm1106.sleep();  // EN apagado otra vez antes de que empiece el siguiente turno
      break;
    }

    case SLOT_SEN0466: {
      if (!sen0466Ready || !g_sen0466Enabled) break;
      Reading co = sen0466.read();
      g_coSt = co.status;
      if (co.ok()) {
        g_coFresh = true;
        g_co = co.values[0];
        g_coTemp = co.values[1];
      }
      break;
    }

    case SLOT_CALYPSO: {
      if (!g_calypsoEnabled) break;

      // HAY QUE ESPERAR A QUE EL SENSOR ARRANQUE, no solo a que llegue una trama.
      //
      // El Calypso es un anemometro ULTRASONICO y cuelga del rail de 5V
      // CONMUTADO, asi que pierde la alimentacion en cada sueno y vuelve a
      // arrancar de cero en cada despertar. Antes de emitir nada ejecuta un
      // autotest de varios segundos, y lo primero que emite puede ser una trama
      // valida con CEROS - medida sin medir todavia.
      //
      // Eso es exactamente lo que estabamos publicando. Medido el 2026-10-09 en
      // una captura limpia de 75 min: cinco de siete lecturas agotaron la ventana
      // sin recibir nada, y las dos que si recibieron dieron 0,00 m/s con validez
      // 1 - mientras habia viento real. El sensor no estaba roto: le preguntabamos
      // 1,5 s despues de encenderlo.
      //
      // El comentario de calypso_probe.cpp ya lo habia anticipado palabra por
      // palabra: "la espera previa de 1,5 s mas 4 s de escucha habria cerrado la
      // ventana antes de que el sensor dijera su primera palabra - produciendo
      // exactamente el resultado de cero bytes que vimos, sin que nada este mal".
      //
      // LA SOLUCION NO ES ESPERAR AQUI, sino despertar antes: el planificador le
      // da ahora CALYPSO_BOOT_MS de preparacion, asi que para cuando se ejecuta
      // este slot el sensor lleva ya ese tiempo alimentado. Esperar aqui moveria
      // la medida fuera del minuto pedido, que es justo lo que no se quiere.
      // ESPERAR A QUE EL SENSOR ESTE ARRANCADO ANTES DE ESCUCHAR.
      //
      // Despertar pronto no basta: el planificador despierta 21,5 s antes del
      // minuto, pero luego ejecuta este slot de inmediato - a los ~6 s de
      // arrancar - y el sensor todavia no emite. Medido el 2026-10-09 en 30
      // lecturas:
      //
      //     ~6,0 s encendido   ->  sin trama  (28 veces)
      //     ~12,4 s encendido  ->  sin trama  (2 veces)
      //      22,7 s encendido  ->  TRAMA
      //      29,2 s encendido  ->  TRAMA
      //
      // La frontera esta entre 12,4 y 22,7 s, y por eso CALYPSO_BOOT_MS son 20.
      //
      // millis() sirve como reloj del rail porque este sensor cuelga del 5V
      // conmutado y ese rail sube en setup(), un par de segundos despues del
      // arranque: usar millis() sobreestima un poco el tiempo encendido, que es
      // el lado seguro. (OJO: en PRE-ALARMA/ALARMA el nodo no duerme y millis() no
      // vuelve a cero, asi que esta espera solo actua tras un arranque.)
      //
      // Y esta espera es la que hace que la medida caiga EN SU MINUTO: el nodo
      // desperto 21,5 s antes a proposito, asi que esperar hasta
      // CALYPSO_BOOT_MS consume justo esa antelacion.
      if (millis() < CALYPSO_BOOT_MS) {
        const uint32_t falta = CALYPSO_BOOT_MS - millis();
        Serial.printf("[calypso] esperando %lums a que arranque el sensor\n",
                      (unsigned long)falta);
        Serial.flush();
        delay(falta);
      }

      calypso.flushInput();
      Reading wind;
      uint32_t t0 = millis();
      do {
        wind = calypso.read();
        if (wind.ok()) break;
        delay(10);
      } while (millis() - t0 < CALYPSO_READ_WINDOW_MS);
      g_calSt = wind.status;

      // CUANTO LLEVABA ENCENDIDO. Sirve para ajustar CALYPSO_BOOT_MS con un dato
      // en vez de con el 20 s a ojo que usa la sonda: si las tramas validas
      // aparecen siempre pasados N segundos, ese N es el tiempo de arranque real.
      Serial.printf("[calypso] rail lleva %lums encendido, escuche %lums -> %s",
                    (unsigned long)millis(), (unsigned long)(millis() - t0),
                    wind.ok() ? "TRAMA VALIDA" : "sin trama");
      if (wind.ok()) {
        Serial.printf(" ang=%.1f vel=%.2f valid=%d", (double)wind.values[0],
                      (double)wind.values[1], (int)(wind.values[2] > 0.5f));
        // LA TRAMA CRUDA. Es lo unico que distingue "el sensor dice cero" de
        // "este driver lee el campo equivocado".
        Serial.printf("\n[calypso] trama: %s", calypso.lastFrame());
      }
      Serial.println();
      Serial.flush();

      if (wind.ok()) {
        g_calFresh = true;
        calypsoEverSeen = true;  // solo ahora pasa a contar como sensor esperado
        g_windAngle = wind.values[0];
        g_windSpeed = wind.values[1];
        g_windValid = wind.values[2] > 0.5f;
      }
      break;
    }

    default:
      break;
  }
}

// Guarda en el almacen RTC una fila con lo medido en este barrido. Si el almacen
// esta lleno se descarta la mas antigua y se cuenta la perdida: un lote que cubre
// menos tiempo del que dice sin avisar seria peor que uno que admite el hueco.
static void storeCurrentReading() {
  if (g_storeCount >= READING_STORE_MAX) {
    for (uint16_t i = 1; i < READING_STORE_MAX; i++) g_store[i - 1] = g_store[i];
    g_storeCount = READING_STORE_MAX - 1;
    g_storeDropped++;
  }
  StoredReading &r = g_store[g_storeCount++];

  // CADA CAMPO LLEVA SU MEDIDA O EL CENTINELA, nunca el valor cacheado de hace
  // media hora. Las banderas *Fresh las limpia runDueSensorJobs() al empezar cada
  // barrido y solo las pone el sensor que de verdad leyo, asi que dicen
  // exactamente "esta lectura midio esto".
  //
  // El gas va aparte del resto del BME690 a proposito: son DOS trabajos sobre el
  // mismo chip, y la lectura rapida de T/H/P deja el gas sin tocar. Sin
  // g_gasFresh, una fila de solo temperatura publicaria el ultimo VOC como si
  // acabara de medirlo.
  r.temp = g_bmeFresh ? g_temp : SIN_MEDIR;
  r.hum = g_bmeFresh ? g_hum : SIN_MEDIR;
  r.pres = g_bmeFresh ? g_pres : SIN_MEDIR;
  r.gas = g_gasFresh ? g_gas : SIN_MEDIR;
  r.pm1 = g_bmvFresh ? g_pm1 : SIN_MEDIR;
  r.pm25 = g_bmvFresh ? g_pm25 : SIN_MEDIR;
  r.pm10 = g_bmvFresh ? g_pm10 : SIN_MEDIR;
  r.co2 = g_co2Fresh ? g_co2 : SIN_MEDIR;

  // UN SENSOR QUE NO SE LEE NO DEBE PUBLICAR UN NUMERO.
  //
  // g_co se queda en 0.00 mientras el SEN0466 esta desactivado, y 0,00 ppm de CO es
  // una lectura perfectamente plausible de aire limpio - asi que un nodo que NO
  // ESTA MIDIENDO CO y uno que mide un cero sano se veian identicos en el panel. Un
  // operador que leyera "CO 0.00" concluiria que no hay monoxido, cuando significa
  // que nadie esta mirando. Es el mismo fallo que el co2=663 congelado de C1, que
  // estuvo dias sin explicacion, solo que publicado a proposito.
  //
  // El CO ya usaba un centinela, pero era -1 y ademas solo miraba si el sensor
  // estaba habilitado, no si habia leido en ESTE barrido. Ahora las dos cosas.
  const bool coMeasured = sen0466Ready && g_sen0466Enabled && g_coFresh;
  r.co = coMeasured ? g_co : SIN_MEDIR;
  r.coTemp = coMeasured ? g_coTemp : SIN_MEDIR;

  const bool vientoMedido = g_calypsoEnabled && g_calFresh && g_windValid;
  r.windAngle = vientoMedido ? g_windAngle : SIN_MEDIR;
  r.windSpeed = vientoMedido ? g_windSpeed : SIN_MEDIR;

  // UN DATO VIEJO NO ES UN DATO VALIDO.
  //
  // g_windValid se pone a true cuando una lectura del Calypso sale bien y NUNCA
  // se vuelve a poner a false, porque vive en RTC. g_calFresh, en cambio, se
  // limpia al empezar cada ciclo de lectura y solo se pone si ESTE ciclo recibio
  // datos. Publicando solo g_windValid, un nodo que leyo el viento una vez al
  // arrancar seguia diciendo "valid" indefinidamente con el mismo valor.
  //
  // Medido en C3 el 2026-10-07: la primera lectura tras el arranque recibio 27
  // bytes y las doce siguientes cero, y las trece publicaron valid=1 con
  // windSpeed=0.00. Es el mismo patron que el co2=663 congelado de C1, que costo
  // dias de no entender nada.
  //
  // La regla de alarma ya exigia las dos (ruleHasData comprueba g_calFresh &&
  // g_windValid), asi que el fuego nunca estuvo en juego - lo que mentia era la
  // telemetria que ve el operador.
  r.windValid = (g_windValid && g_calFresh) ? 1 : 0;
  r.takenS = (uint32_t)(nodeNowMs() / 1000ULL);  // segundo del nodo; al enviar se convierte en antiguedad
  Serial.printf("[store] reading %u/%u kept (tick %lu)\n", (unsigned)g_storeCount,
                (unsigned)READING_STORE_MAX, (unsigned long)g_wakeCount);
}

// ---- UMBRALES DE INCENDIO, de la hoja de alarma/prealarma del proyecto ----
//
// Transcritos el 2026-09-28. Cuatro numeros por magnitud, como las cuatro columnas
// de la hoja: un umbral ON para ENTRAR en cada nivel y uno OFF para SALIR. La
// separacion entre ellos es histeresis: sin ella una lectura justo en el limite
// haria entrar y salir al nodo de la pre-alarma solo por el ruido del sensor, y en
// pre-alarma el nodo no duerme.
//
// LA DIRECCION ES POR MAGNITUD y no es un detalle. Temperatura, particulas, CO y
// viento disparan al SUBIR. LA HUMEDAD DISPARA AL BAJAR - el aire seco es el indicio
// de incendio - asi que sus umbrales descienden (25 -> 20 -> 15 -> 10) y la
// comparacion se invierte con ellos. Equivocarse aqui dejaria mudo al nodo justo en
// las condiciones que existe para detectar.
//
// UNA sola magnitud basta para cambiar de nivel (O, no Y).
//
// Una tabla en vez de #defines sueltos para poder leerla de arriba abajo contra la
// hoja y comprobarla: es la unica forma de que siga siendo fiel cuando cambien los
// numeros.
struct AlarmRule {
  const char *name;
  const float *value;
  uint8_t source;       // que sensor tiene que haber leido en este barrido - ver ruleHasData()
  bool higherIsWorse;   // true: dispara al subir. false: dispara al bajar (humedad)
  float preOff, preOn;  // salir / entrar en PRE-ALARMA
  float almOff, almOn;  // salir / entrar en ALARMA
  bool enabled;
};

#define ALARM_SRC_BME 0
#define ALARM_SRC_BMV 1
#define ALARM_SRC_CO 2
#define ALARM_SRC_CO2 3
#define ALARM_SRC_WIND 4

static const AlarmRule kAlarmRules[] = {
    // nombre      valor         fuente          sube    preOff    preOn   almOff    almOn  activa
    {"temp",      &g_temp,      ALARM_SRC_BME,  true,    45.0f,   50.0f,   55.0f,   60.0f, true},
    {"rh",        &g_hum,       ALARM_SRC_BME,  false,   25.0f,   20.0f,   15.0f,   10.0f, true},
    {"pm1",       &g_pm1,       ALARM_SRC_BMV,  true,    90.0f,  100.0f,  225.0f,  250.0f, true},
    {"pm25",      &g_pm25,      ALARM_SRC_BMV,  true,    90.0f,  100.0f,  225.0f,  250.0f, true},
    {"pm10",      &g_pm10,      ALARM_SRC_BMV,  true,    90.0f,  100.0f,  225.0f,  250.0f, true},
    {"co",        &g_co,        ALARM_SRC_CO,   true,     6.0f,    8.0f,   15.0f,   20.0f, true},
    {"windSpeed", &g_windSpeed, ALARM_SRC_WIND, true,     2.0f,    5.0f,    8.0f,   10.0f, true},

    // ---- DESACTIVADAS, cada una por un motivo medido y no por descuido ----
    //
    // gas: la hoja da 15000/7000 ohmios, pero la resistencia de gas del BME690 sube
    // de ~6k en frio a ~45k cuando el calefactor se estabiliza, y con el ciclo de
    // sueno CADA medida empieza en frio. Los nodos C2 y C3 leen 5685 ohmios en aire
    // limpio - por debajo de la cifra de alarma de la hoja - mientras C1 lee 47033.
    // El mismo sensor en la misma sala abarca las dos, asi que un umbral absoluto no
    // puede separar un calefactor frio del humo. Hace falta una lectura con el
    // calefactor estabilizado antes de que un numero aqui signifique algo.
    {"gas",  &g_gas,  ALARM_SRC_BME, false, 18000.0f, 15000.0f, 9000.0f, 7000.0f, false},

    // co2: la hoja da 800/1000 ppm, pero los tres nodos leen 1104-1517 ppm en aire
    // limpio de interior, ya por encima de la alarma; la propia referencia de la hoja
    // es 555. Ademas la lectura del CM1106 se CONGELA dentro de un arranque y solo
    // cambia con ciclos de alimentacion, asi que un disparo basado en ella estaria
    // ciego a un incendio real y podria quedarse enganchado en un valor viejo.
    {"co2",  &g_co2,  ALARM_SRC_CO2, true,   750.0f,  800.0f,  900.0f, 1000.0f, false},

    // pres: las cuatro celdas de la hoja son 1031 hPa - sin histeresis, sin
    // separacion entre niveles, sin direccion indicada. La referencia de la hoja es
    // 970 y los nodos leen 943-950. Tal cual esta, o no dispararia nunca o
    // dispararia siempre.
    {"pres", &g_pres, ALARM_SRC_BME, true,  1031.0f, 1031.0f, 1031.0f, 1031.0f, false},
};

#define ALARM_RULE_COUNT (sizeof(kAlarmRules) / sizeof(kAlarmRules[0]))

// ---- UMBRALES VIVOS ----
//
// kAlarmRules de arriba son ahora los valores POR DEFECTO. Los valores con los que
// de verdad se compara viven aqui, en memoria RTC, para que Modulo A pueda cambiar
// uno por radio sin reflashear - que es la razon de ser del comando de umbral.
//
// RTC_NOINIT para que un cambio sobreviva al sueno: un nodo que olvidara sus
// umbrales en cada despertar volveria en silencio a los compilados nada mas
// configurarlo, y el panel mostraria un valor que el nodo no usa.
//
// Mismo orden que kAlarmRules. [0]=preOff [1]=preOn [2]=almOff [3]=almOn.
#define THR_PRE_OFF 0
#define THR_PRE_ON 1
#define THR_ALM_OFF 2
#define THR_ALM_ON 3
RTC_NOINIT_ATTR float g_thr[ALARM_RULE_COUNT][4];

// Rango en el que un umbral de esta placa puede estar. Deliberadamente ancho: no
// es para detectar un valor mal elegido - eso es decision del operador - sino para
// detectar BASURA. Las magnitudes van de porcentajes de humedad a ohmios de gas,
// asi que el limite superior tiene que dar cabida a 18000.
#define THR_MIN_CREIBLE (-100.0f)
#define THR_MAX_CREIBLE (100000.0f)

// Comprueba los umbrales vivos y resiembra los que no sean creibles.
//
// g_thr vive en RTC_NOINIT: memoria que sobrevive al sueno y al reset porque NO se
// inicializa. Eso la hace util y fragil a la vez - un brownout, una perdida de
// alimentacion a medias o un cambio en la disposicion de las variables la dejan
// con lo que hubiera en esas direcciones.
//
// El 2026-10-12 g_thr[0] - los cuatro umbrales de la temperatura - aparecio asi:
//
//     pre 0.0/0.0   alarma 4.7e19 / -5.8e31
//
// Con preOn = 0 la regla dispara con cualquier temperatura positiva; con
// almOn = -5.8e31 dispara siempre. El nodo quedo en ALARMA DE INCENDIO sin poder
// salir, y ningun reset lo arreglaba porque el valor corrupto sobrevivia con el
// resto de la memoria RTC.
//
// Peor aun: la histeresis quedaba INVERTIDA. El umbral de salida era 4,7e19, asi
// que la misma regla estaba disparada segun el de entrada y limpia segun el de
// salida. Las dos mitades del codigo veian cosas distintas y ninguna se
// contradecia a si misma.
//
// Que un numero estropeado en RAM enclave un detector de incendios para siempre no
// es aceptable, asi que se comprueba en cada arranque.
static void seedThresholds();

static void validateThresholds() {
  // LA DERIVA TAMBIEN SE VALIDA, y por la misma razon que los umbrales: vive en
  // RTC y un valor malo se queda ahi. Estrechar la guarda del aprendizaje impide
  // que vuelva a entrar uno absurdo, pero no limpia el que ya esta guardado - y
  // ese divide todos los suenos.
  if (isnan(g_clockDrift) || g_clockDrift < CLOCK_DRIFT_MIN || g_clockDrift > CLOCK_DRIFT_MAX) {
    Serial.printf("*** DERIVA FUERA DE RANGO (%ld ppm) - se vuelve a 0 ***\n",
                  (long)((g_clockDrift - 1.0f) * 1000000.0f));
    g_clockDrift = 1.0f;
    Serial.flush();
  }

  uint8_t malos = 0;
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    for (uint8_t k = 0; k < 4; k++) {
      const float v = g_thr[i][k];
      // isnan() cubre el NaN, que no falla ninguna comparacion de rango.
      if (isnan(v) || v < THR_MIN_CREIBLE || v > THR_MAX_CREIBLE) {
        malos++;
      }
    }
  }
  // SE IMPRIMEN SIEMPRE, no solo cuando hay corrupcion.
  //
  // Antes solo se veian con la alarma levantada - o sea, cuando ya era tarde. El
  // 2026-10-12 el nodo paso horas en alarma de incendio por un umbral corrupto que
  // nadie podia ver en el log. Una linea por arranque es barata comparada con eso.
  Serial.print("Umbrales:");
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    if (!kAlarmRules[i].enabled) continue;
    Serial.printf(" %s %.0f/%.0f|%.0f/%.0f", kAlarmRules[i].name,
                  (double)g_thr[i][THR_PRE_OFF], (double)g_thr[i][THR_PRE_ON],
                  (double)g_thr[i][THR_ALM_OFF], (double)g_thr[i][THR_ALM_ON]);
  }
  Serial.println();
  Serial.flush();

  if (!malos) return;

  Serial.printf("\n*** %u UMBRAL(ES) CORRUPTO(S) EN RTC - resembrando desde la tabla ***\n",
                (unsigned)malos);
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    Serial.printf("    %-9s  pre %.1f/%.1f  alarma %.1f/%.1f\n", kAlarmRules[i].name,
                  (double)g_thr[i][THR_PRE_OFF], (double)g_thr[i][THR_PRE_ON],
                  (double)g_thr[i][THR_ALM_OFF], (double)g_thr[i][THR_ALM_ON]);
  }
  seedThresholds();

  // Y bajar la alarma: si estaba levantada, pudo levantarla el umbral corrupto, y
  // dejarla puesta con los umbrales ya buenos seria arrastrar una falsa alarma sin
  // motivo. Si la causa era real, el primer barrido la vuelve a levantar en
  // segundos - que es justo lo que debe pasar.
  if (g_alarmState != ALARM_NORMAL) {
    Serial.println("*** la alarma estaba levantada con umbrales corruptos - se baja ***");
    g_alarmState = ALARM_NORMAL;
    g_alarmClearRun = 0;
    g_prealarmMs = 0;
  }
  Serial.flush();
}

// Copia los valores por defecto de la tabla sobre los vivos. Se llama en la siembra
// del BOOTCOUNT_MAGIC y cuando validateThresholds() encuentra basura.
static void seedThresholds() {
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    g_thr[i][THR_PRE_OFF] = kAlarmRules[i].preOff;
    g_thr[i][THR_PRE_ON] = kAlarmRules[i].preOn;
    g_thr[i][THR_ALM_OFF] = kAlarmRules[i].almOff;
    g_thr[i][THR_ALM_ON] = kAlarmRules[i].almOn;
  }
}

// Traduce un nombre como "temp_pre_on" o "pm25_alarma_off" a la celda exacta que lee
// la logica de alarma, o nullptr si no corresponde a ninguna regla (asi una clave
// desconocida se informa en vez de aceptarse en silencio - un umbral que parece
// aplicado y no lo esta es lo peor que puede pasar aqui).
//
// Separada de setThresholdByName para que el ACK pueda RELEER lo que se guardo en
// vez de devolver el texto que llego - ver donde se llama. Una sola busqueda para
// las dos direcciones hace que el valor que se le dice a Modulo A no pueda
// diferir del valor con el que compara este nodo.
static float *thresholdSlot(const char *key) {
  static const struct { const char *suffix; uint8_t idx; } kSuffix[] = {
      {"_pre_off", THR_PRE_OFF},
      {"_pre_on", THR_PRE_ON},
      {"_alarma_off", THR_ALM_OFF},
      {"_alarma_on", THR_ALM_ON},
      // El documento del protocolo usa estos en vez de los nombres de la hoja. Se
      // aceptan los dos en vez de adivinar cual mandara Modulo A.
      {"_alarm_off", THR_ALM_OFF},
      {"_alarm_on", THR_ALM_ON},
  };

  const size_t klen = strlen(key);
  for (size_t sfx = 0; sfx < sizeof(kSuffix) / sizeof(kSuffix[0]); sfx++) {
    const size_t slen = strlen(kSuffix[sfx].suffix);
    if (klen <= slen) continue;
    if (strcmp(key + klen - slen, kSuffix[sfx].suffix) != 0) continue;

    const size_t namelen = klen - slen;
    for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
      if (strlen(kAlarmRules[i].name) != namelen) continue;
      if (strncmp(kAlarmRules[i].name, key, namelen) != 0) continue;
      return &g_thr[i][kSuffix[sfx].idx];
    }
  }
  return nullptr;
}

// Pone un umbral por su nombre. HISTORICO: ya no se llama desde ningun sitio (el CFG
// usa thresholdSlot() directamente).
static bool setThresholdByName(const char *key, float value) {
  float *slot = thresholdSlot(key);
  if (!slot) return false;
  *slot = value;
  return true;
}

// El sensor de esta regla ha dado una lectura EN ESTE BARRIDO?
//
// ESTA GUARDA ES ESENCIAL, no orden defensivo. Las variables de lectura valen 0.0
// cuando un sensor no ha leido, y la humedad dispara con valores BAJOS - asi que un
// BME690 sin leer presentaria rh = 0.0, por debajo incluso del umbral de alarma de
// 10, y todo nodo con el sensor de humedad muerto estaria en alarma de incendio
// permanente. Una regla cuyo sensor esta ausente, desactivado o sin dato fresco se
// SALTA, nunca se evalua contra un cero.
static bool ruleHasData(uint8_t source) {
  switch (source) {
    case ALARM_SRC_BME:  return bmeReady && g_bmeEnabled && g_bmeFresh;
    case ALARM_SRC_BMV:  return bmvReady && g_bmvEnabled && g_bmvFresh;
    case ALARM_SRC_CO:   return sen0466Ready && g_sen0466Enabled && g_coFresh;
    case ALARM_SRC_CO2:  return g_cm1106Enabled && g_co2Fresh;
    case ALARM_SRC_WIND: return g_calypsoEnabled && g_calFresh && g_windValid;
  }
  return false;
}

// Ha cruzado esta regla el umbral de ENTRADA del nivel dado? Tiene en cuenta si la
// magnitud sube o baja con el fuego.
static bool ruleTripped(const AlarmRule &r, float v, bool fireLevel, size_t idx) {
  const float on = g_thr[idx][fireLevel ? THR_ALM_ON : THR_PRE_ON];
  return r.higherIsWorse ? (v >= on) : (v <= on);
}

// Ha vuelto por debajo del umbral de SALIDA? La separacion entre SALIDA y ENTRADA
// es la histeresis: un valor entre los dos mantiene el estado en el que ya estaba.
static bool ruleCleared(const AlarmRule &r, float v, bool fireLevel, size_t idx) {
  const float off = g_thr[idx][fireLevel ? THR_ALM_OFF : THR_PRE_OFF];
  return r.higherIsWorse ? (v < off) : (v > off);
}

// El nivel mas alto que pide cualquier regla por si sola. O entre magnitudes: basta
// una. Devuelve ALARM_NORMAL si no hay nada disparado. Solo mira reglas activas y
// con dato fresco (ruleHasData).
static AlarmState evaluateRules(const char **whyOut) {
  AlarmState worst = ALARM_NORMAL;
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    const AlarmRule &r = kAlarmRules[i];
    if (!r.enabled || !ruleHasData(r.source)) continue;
    const float v = *r.value;
    if (ruleTripped(r, v, true, i)) {
      *whyOut = r.name;
      return ALARM_FIRE;  // no hay nada por encima de esto: dejar de buscar
    }
    if (worst == ALARM_NORMAL && ruleTripped(r, v, false, i)) {
      *whyOut = r.name;
      worst = ALARM_PREALARM;
    }
  }
  return worst;
}

// QUE REGLA MANTIENE LA ALARMA, y con que numeros.
//
// Sin esto el nodo se queda en alarma sin decir por que, que es exactamente lo
// que paso en C3 el 2026-10-09: nueve minutos en ALARMA DE INCENDIO y ni una
// linea en el log. Un detector que no sabe explicar su propia alarma no se puede
// distinguir de uno averiado.
//
// Imprime el valor vivo contra el umbral que lo retiene, y NOMBRA las magnitudes
// que se estan ignorando por falta de dato fresco - que es la otra mitad de la
// historia: una regla sin dato no retiene nada, pero tampoco confirma nada.
//
// Limitado a una linea cada ALARM_WHY_GAP_MS. En alarma el nodo no duerme y
// evalua en cada vuelta de loop(), asi que sin freno esto inundaria el puerto.
#define ALARM_WHY_GAP_MS 30000UL

// Definida mas abajo, junto al resto del nombrado de estados.
static const char *alarmStateName(AlarmState st);

static void reportWhyStillRaised(bool fireLevel, const char *rama, int want) {
  static uint32_t lastMs = 0;
  const uint32_t now = millis();
  if (lastMs != 0 && now - lastMs < ALARM_WHY_GAP_MS) return;
  lastMs = now;

  char retienen[160];
  size_t rp = 0;
  char sinDato[120];
  size_t sp = 0;

  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    const AlarmRule &r = kAlarmRules[i];
    if (!r.enabled) continue;

    if (!ruleHasData(r.source)) {
      if (sp < sizeof(sinDato) - 24) {
        sp += snprintf(sinDato + sp, sizeof(sinDato) - sp, "%s%s", sp ? " " : "", r.name);
      }
      continue;
    }
    if (ruleCleared(r, *r.value, fireLevel, i)) continue;

    const float off = g_thr[i][fireLevel ? THR_ALM_OFF : THR_PRE_OFF];
    if (rp < sizeof(retienen) - 40) {
      rp += snprintf(retienen + rp, sizeof(retienen) - rp, "%s%s=%.1f (sale %s %.1f)",
                     rp ? ", " : "", r.name, (double)*r.value,
                     r.higherIsWorse ? "bajo" : "sobre", (double)off);
    }
  }

  Serial.printf("[alarm] %s SIGUE por la rama '%s' (evaluateRules dice %s): %s\n",
                alarmStateName((AlarmState)g_alarmState), rama,
                alarmStateName((AlarmState)want),
                rp ? retienen : "ninguna regla con dato la retiene");
  if (sp) Serial.printf("[alarm] sin dato fresco, ignoradas: %s\n", sinDato);

  // LOS UMBRALES VIVOS, no los de la tabla compilada.
  //
  // g_thr esta en RTC y solo se siembra desde kAlarmRules cuando cambia el
  // BOOTCOUNT_MAGIC. Si el nodo arranca con un magic que ya coincidia, conserva
  // los que hubiera - que pueden venir de un build de medicion anterior, o de un
  // comando CFG de Modulo A. Imprimirlos es la unica forma de distinguir "la
  // logica falla" de "los umbrales no son los que creo".
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    const AlarmRule &r = kAlarmRules[i];
    if (!r.enabled || !ruleHasData(r.source)) continue;
    Serial.printf("[alarm]   %-9s = %8.2f   pre %.1f/%.1f   alarma %.1f/%.1f  (%s)\n",
                  r.name, (double)*r.value,
                  (double)g_thr[i][THR_PRE_OFF], (double)g_thr[i][THR_PRE_ON],
                  (double)g_thr[i][THR_ALM_OFF], (double)g_thr[i][THR_ALM_ON],
                  r.higherIsWorse ? "sube=peor" : "baja=peor");
  }
  Serial.flush();
}

// Han vuelto TODAS las reglas activas por debajo del umbral de salida del nivel en
// el que estamos? Todas tienen que estar de acuerdo para bajar: una sola todavia
// disparada lo mantiene. (Las reglas sin dato fresco se saltan.)
static bool allRulesCleared(bool fireLevel) {
  for (size_t i = 0; i < ALARM_RULE_COUNT; i++) {
    const AlarmRule &r = kAlarmRules[i];
    if (!r.enabled || !ruleHasData(r.source)) continue;
    if (!ruleCleared(r, *r.value, fireLevel, i)) return false;
  }
  return true;
}

static const char *alarmStateName(AlarmState st) {
  switch (st) {
    case ALARM_NORMAL:   return "NORMAL";
    case ALARM_PREALARM: return "PRE-ALARM";
    case ALARM_FIRE:     return "ALARM";
  }
  return "?";
}

// LA MAQUINA DE ESTADOS DE LA ALARMA. Se ejecuta tras cada lectura.
//
//   - SUBE al momento cuando una regla dispara (NORMAL -> PRE-ALARMA o -> ALARMA).
//   - Una PRE-ALARMA que sigue disparada al acabar PREALARM_CONFIRM_MS (60 min)
//     sube a ALARMA.
//   - BAJA solo tras ALARM_CLEAR_CONSECUTIVE (5) lecturas seguidas con todas las
//     reglas limpias, y de nivel en nivel (ALARMA -> PRE-ALARMA -> NORMAL), para
//     que un solo descenso a mitad de un incendio no lo termine.
//   - Modulo A puede bajarla a mano con CFG,ALARM=0.
//
// Subir es instantaneo y bajar es lento, a proposito: el coste de llegar tarde a un
// incendio no es simetrico con el de seguir alerta unas lecturas de mas.
static void updateAlarmStateInner() {
#if FORCE_ALARM_STATE
  // Congelado para una medida de consumo: ni sube, ni baja, ni se miran los
  // sensores. Ver FORCE_ALARM_STATE.
  return;
#endif
  const char *why = "";
  const AlarmState want = evaluateRules(&why);

  // QUE SE ACABE LA VENTANA DE CONFIRMACION ES EN SI UNA SUBIDA.
  //
  // Se comprueba antes de la comparacion normal, porque "sigue disparada a nivel de
  // pre-alarma despues de toda la ventana" no sube `want` por si solo - la lectura
  // no ha empeorado, simplemente no se ha ido, y esa es justo la cuestion. Sin esto
  // el nodo se quedaria en pre-alarma indefinidamente ante un incendio real que
  // nunca llegara del todo al umbral de alarma.
  if (g_alarmState == ALARM_PREALARM && want >= ALARM_PREALARM &&
      g_prealarmMs >= PREALARM_CONFIRM_MS) {
    g_alarmState = ALARM_FIRE;
    g_alarmClearRun = 0;
    Serial.printf("\n*** ALARM: %s still over threshold after %lu min - confirmed ***\n",
                  why, (unsigned long)(PREALARM_CONFIRM_MS / 60000UL));
    Serial.flush();
    return;
  }

  // Subir es inmediato en cualquier nivel.
  if (want > g_alarmState) {
    const AlarmState from = (AlarmState)g_alarmState;
    g_alarmState = want;
    g_alarmClearRun = 0;
    // La ventana empieza con la causa inicial, asi que solo se reinicia al entrar de
    // verdad en pre-alarma desde normal - no al subir a alarma.
    if (from == ALARM_NORMAL) g_prealarmMs = 0;
    Serial.printf("\n*** %s: %s *** (was %s) - staying awake, all sensors continuous\n",
                  alarmStateName(want), why, alarmStateName(from));
    Serial.flush();
    return;
  }

  if (g_alarmState == ALARM_NORMAL) return;

  // Sigue disparada en el nivel actual: la cuenta de lecturas limpias vuelve a cero.
  if (want >= g_alarmState) {
    g_alarmClearRun = 0;
    reportWhyStillRaised(g_alarmState == ALARM_FIRE, "una regla sigue disparada", (int)want);
    return;
  }

  // Bajar de nivel en nivel, y solo tras una serie de lecturas limpias.
  const bool leavingFire = (g_alarmState == ALARM_FIRE);
  if (!allRulesCleared(leavingFire)) {
    g_alarmClearRun = 0;
    reportWhyStillRaised(leavingFire, "no todas han bajado del umbral de salida", (int)want);
    return;
  }

  if (++g_alarmClearRun < ALARM_CLEAR_CONSECUTIVE) {
    Serial.printf("[alarm] %s, clean read %u/%d\n", alarmStateName((AlarmState)g_alarmState),
                  (unsigned)g_alarmClearRun, ALARM_CLEAR_CONSECUTIVE);
    return;
  }

  g_alarmClearRun = 0;
  g_alarmState = leavingFire ? ALARM_PREALARM : ALARM_NORMAL;
  // Al volver de alarma a pre-alarma se empieza una ventana nueva en vez de heredar
  // una ya agotada, que haria volver a subir en la siguiente lectura.
  g_prealarmMs = 0;
  Serial.printf("\n*** dropped to %s after %d clean reads ***\n",
                alarmStateName((AlarmState)g_alarmState), ALARM_CLEAR_CONSECUTIVE);
  Serial.flush();
}

// RE-FASEO AL CAMBIAR DE ESTADO.
//
// Cada estado tiene sus propios periodos - el T/H/P pasa de 5 min a 3 y a 1 - asi
// que el calendario calculado para el estado anterior deja de significar lo que
// decia en el instante del cambio.
//
// Es un ENVOLTORIO y no codigo al final de updateAlarmStateInner() porque esa
// funcion sale por media docena de `return` distintos; aqui se recoge cualquiera
// de ellos.
//
// El estado al que se faseo vive en RTC: en operacion normal cada despertar es un
// arranque nuevo, asi que una variable `static` normal valdria 0xFF siempre y
// re-fasearia en cada despertar, tirando el calendario entero cada vez.
static void updateAlarmState() {
  const uint8_t from = g_schedPhasedForState;
  updateAlarmStateInner();
  if (g_schedPhasedForState == g_alarmState) return;

  const uint8_t to = g_alarmState;
  g_schedPhasedForState = to;
  schedRephase(schedNowMs(), to, "cambio de estado");

  // SUBIR DE ESTADO: medir YA, decision del usuario. Entrar en prealarma significa
  // que algo cruzo un umbral, y ese es el momento de mirar, no tres minutos
  // despues. Vencimiento a cero = vence en el acto.
  //
  // BAJAR: no se fuerza nada. Volver a normal es precisamente cuando hay que
  // RE-ALINEARSE al reloj de pared - que es lo que acaba de hacer schedRephase -
  // para que el nodo vuelva a medir en los mismos instantes que sus hermanos.
  if (to > from) {
    for (uint8_t j = 0; j < JOB_TX; j++) g_nextDueMs[j] = 0;
    Serial.printf("[sched] %s -> %s: lectura inmediata de todo\n",
                  alarmStateName((AlarmState)from), alarmStateName((AlarmState)to));
  } else {
    Serial.printf("[sched] %s -> %s: re-alineado al reloj\n",
                  alarmStateName((AlarmState)from), alarmStateName((AlarmState)to));
  }
  Serial.flush();
}

// 10. TRANSMISION A MODULO B Y PAQUETE DE ESTADO (STAT)
// =============================================================================

// Envia todo lo que hay en el almacen y lo vacia.
//   - Una sola lectura sale con el formato simple de siempre (13 campos).
//   - Varias salen en LOTE: "B,<cuantas>,<lectura>;<lectura>;..." donde cada lectura
//     lleva ademas su antiguedad en segundos, para que Modulo B la feche.
//   - Un lote que no cabe en los 240 bytes del RYLR998 se reparte en varios
//     paquetes en vez de cortarse.
// Despues abre la ventana de escucha: el unico momento en que llega un CFG.
static void transmitStore() {
  if (!radioReady) {
    Serial.printf("TX skipped (radio not initialized): %u stored reading(s) held\n",
                  (unsigned)g_storeCount);
    return;
  }
  if (g_storeCount == 0) {
    Serial.println("[tx] nothing stored this cycle - not transmitting");
    return;
  }

  if (g_storeDropped > 0) {
    Serial.printf("[tx] WARNING %lu reading(s) were dropped - store too small for the\n"
                  "     configured LORA_TRANS/SENSOR_READ ratio\n",
                  (unsigned long)g_storeDropped);
  }

  // Todos los sensores en reposo antes de transmitir, y despues sus rails. El
  // RYLR998 pide ~120 mA al enviar y esta placa no puede dar eso encima del consumo
  // de reposo de los rails de sensores: se cae en radio.send() si se dejan encendidos.
  quiesceAll();
  delay(SLOT_GAP_MS);
  powerDownSensorRails();

  char packet[241];
  uint16_t sent = 0;
  while (sent < g_storeCount) {
    int len = 0;
    packet[0] = 0;
    uint16_t first = sent;

    if (g_storeCount == 1) {
      // Una sola lectura: formato simple de 13 campos.
      const StoredReading &r = g_store[0];
      len = snprintf(packet, sizeof(packet),
                     "%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%.2f,%d",
                     r.temp, r.hum, r.pres, r.gas, r.pm1, r.pm25, r.pm10, r.co2, r.co,
                     r.coTemp, r.windAngle, r.windSpeed, (int)r.windValid);
      sent = 1;
    } else {
      // Lote: "B,<cuantas>," y despues las lecturas separadas por ';', cada una
      // terminada en su antiguedad, para que el receptor las situe en el tiempo sin
      // que aqui haga falta un reloj.
      len = snprintf(packet, sizeof(packet), "B,%u,", (unsigned)(g_storeCount - first));
      while (sent < g_storeCount) {
        const StoredReading &r = g_store[sent];
        char one[176];
        // EDAD EN SEGUNDOS REALES, no en despertares. El nodo ya sabe cuanto
        // tiempo ha pasado de verdad - lo cuenta nodeNowMs() y lo corrige la
        // deriva aprendida - asi que Modulo B no tiene que adivinar nada.
        const uint32_t ahoraS = (uint32_t)(nodeNowMs() / 1000ULL);
        uint32_t age = (ahoraS >= r.takenS) ? (ahoraS - r.takenS) : 0;
        int oneLen = snprintf(one, sizeof(one),
                              "%.1f,%.1f,%.1f,%.0f,%.1f,%.1f,%.1f,%.0f,%.1f,%.1f,%.0f,%.1f,%d,%lu;",
                              r.temp, r.hum, r.pres, r.gas, r.pm1, r.pm25, r.pm10, r.co2, r.co,
                              r.coTemp, r.windAngle, r.windSpeed, (int)r.windValid,
                              (unsigned long)age);
        if (oneLen < 0) break;
        if (len + oneLen >= (int)sizeof(packet) - 1) break;  // este paquete esta lleno
        memcpy(packet + len, one, oneLen + 1);
        len += oneLen;
        sent++;
      }
      if (sent == first) {  // una sola lectura no cabia ni sola
        Serial.println("[tx] ERROR one reading exceeds the packet size - dropping it");
        sent++;
        continue;
      }
    }

    if (len < 0) len = 0;
    if (len >= (int)sizeof(packet)) len = (int)sizeof(packet) - 1;

    // LED apagado durante el propio envio. La rafaga de transmision es el momento
    // mas critico de todo el ciclo, y el indicador no debe ser una de las cargas que
    // compiten con ella - el mismo razonamiento que pone en reposo a los sensores
    // justo arriba. Se vuelve a encender en el acto, asi que a la vista sigue
    // pareciendo "trabajando".
    ledOff();
    bool ok = radio.send(LORA_RX_ADDR, packet, (uint8_t)len);
    ledWorking();
    Serial.printf("TX [%u..%u/%u] %d bytes: %s  (%s)\n", (unsigned)first, (unsigned)(sent - 1),
                  (unsigned)g_storeCount, len, packet, ok ? "sent" : "send FAILED");
  }

  // Enviado = hecho: vaciar el almacen y empezar a acumular otra vez.
  g_storeCount = 0;
  g_storeDropped = 0;

  // Ventana de escucha tras el envio: el unico momento en que un CFG de Modulo A
  // puede llegar a este nodo.
  LoRaMessage msg;
  uint32_t t0 = millis();
  while (millis() - t0 < LORA_POST_TX_LISTEN_MS) {
    if (radio.poll(msg)) {
      handleInboundMessage(msg);
      break;
    }
    delay(1);  // cede el procesador a FreeRTOS para que la tarea inactiva alimente al perro guardian
  }
}


// ---- Paquete de estado (STAT), transmitStatus() mas abajo ----
//
// Se manda un paquete de salud compacto junto con la telemetria.
//
// POR QUE EXISTE: la telemetria lleva solo VALORES. Un sensor que se muere puede
// seguir pareciendo sano en los datos - asi es exactamente como el co2=663 congelado
// del nodo C1 estuvo dias sin explicacion mientras su camino de lectura estaba
// muerto. Con un cable serie esa ambiguedad es molesta; con el nodo fuera y sin
// cable es fatal para cualquier prueba, porque no hay otra forma de enterarse.
//
// Es un paquete APARTE a proposito, y no campos extra en la telemetria: anadir un
// campo romperia el interprete de 13 campos de Modulo B, mientras que una trama
// desconocida se registra tal cual y todo lo demas la ignora sin problema.
//
//   STAT,<despertar>,<motivoReinicio>,<arranquesFallidos>,<bme><bmv><co2><co><viento>,
//        <antiguedad>,<mV en el pin>,<% de carga>,<estado de alarma>
//
// Los cinco indicadores valen 1 = leyo bien, 0 = no. Leerlos como la respuesta a
// "que sensores estaban vivos de verdad cuando salio este paquete".

// ---- Medida de la bateria ----
//
// Milivoltios en el pin de medida (GPIO14), promediados en varias muestras porque
// una sola lectura del ADC de esta pieza es ruidosa. Van en el STAT para poder
// seguir una descarga por radio en vez de quedarse al lado del nodo con un
// polimetro.
//
// ---- Carga, como porcentaje de ENERGIA UTILIZABLE ----
//
// CALIBRADO el 2026-09-23 a partir de las resistencias de la placa actualizada.
//
// El punto de medida es la union de un divisor de dos resistencias sobre el pack:
//
//     Vpack --[ 51.1k ]--+-- GPIO14
//                        |
//                     [ 322k ]
//                        |
//                       GND
//
// asi que el pin ve la rama INFERIOR y el factor hasta los bornes es
//
//     factor = (Rarriba + Rabajo) / Rabajo = (51.1 + 322) / 322 = 1.15870
//
// Comprobacion contra lecturas reales: 2304 mV en el pin - lo mas alto visto en un
// pack cargado - son 2670 mV en los bornes, una celda de supercondensador de 2,7 V
// cargada. Coincide.
//
// OJO: es la placa ACTUALIZADA. Una placa antigua no lleva divisor y su pin flota
// en ~100 mV haga lo que haga el pack; eso no es una bateria vacia ni un fallo.
// Mirar la revision antes de interpretar los milivoltios.
//
// LIMITACION CONOCIDA EN LA PARTE ALTA. Este divisor apenas divide - escala por
// 0,863 - asi que un pack lleno de 3,8 V pone 3280 mV en el pin. El ADC del ESP32-S3
// con la atenuacion por defecto de 11 dB esta especificado hasta unos 3100 mV y
// pierde linealidad bastante antes, asi que el ultimo tramo de la carga se lee
// comprimido y acaba saturando: por encima de ~3,59 V en bornes el pin satura y el
// indicador deja de subir, probablemente sobre el 90% en vez del 100%.
//
// OJO, IMPORTANTE: todo este bloque se calculo para el pack de
// SUPERCONDENSADORES. Desde el 2026-09-24 la fuente es una bateria de LITIO (hasta
// 4,2 V), para la que ni el divisor ni la formula de energia sirven tal cual. Hay que
// recalcularlo antes de fiarse del porcentaje.
#define SUPERCAP_DIVIDER_RATIO 1.15870f

// 1 cuando SUPERCAP_DIVIDER_RATIO es un numero MEDIDO. Con 0 el porcentaje se
// informa como -1 ("desconocido") en vez de calcularse: con un factor inventado la
// cuenta daba un 0% muy convincente, que es falso y alarmante. Los milivoltios en
// crudo se mandan igualmente.
#define SUPERCAP_CALIBRATED 1

// Tension de pack lleno: 3,8 V (dato del 2026-09-23, de la epoca de los
// supercondensadores). Se llego a tomar por error 2,7 V como "lleno" porque era lo
// mas alto visto en un dia de pruebas, cuando en realidad el pack nunca termino de
// cargarse: una lectura solo es el maximo si algo dice que la carga habia acabado.
#define SUPERCAP_V_FULL 3.8f

// Tension de pack VACIO = el umbral de hibernacion, a proposito.
//
// El nodo se para en HIB_ENTER_V para proteger las celdas, asi que la energia por
// debajo de eso no es energia utilizable - es energia que se ha decidido no usar. Con
// los dos iguales, 0% y "hibernando ya" son el mismo punto, y el indicador significa
// lo que un operador supone: cuanto queda antes de que se pare.
#define SUPERCAP_V_EMPTY HIB_ENTER_V





// Porcentaje de ENERGIA utilizable restante, no de tension.
//
// Un supercondensador almacena 0,5*C*V^2, asi que la energia cae con el CUADRADO de
// la tension mientras la tension cae de forma lineal. Un porcentaje de tension diria
// "50%" a 3,1 V cuando solo queda un ~35% de la energia utilizable - halagador, y
// enganoso justo cuando importa. Se usa la forma de energia:
//
//     (V^2 - Vvacio^2) / (Vlleno^2 - Vvacio^2)
//
// asi que 50% significa aproximadamente la mitad del TIEMPO restante.
// (Para una bateria de litio esta formula no aplica; ver el aviso de arriba.)
static int supercapPercent(float volts) {
  const float lo = SUPERCAP_V_EMPTY * SUPERCAP_V_EMPTY;
  const float hi = SUPERCAP_V_FULL * SUPERCAP_V_FULL;
  float e = (volts * volts - lo) / (hi - lo);
  if (e < 0.0f) e = 0.0f;
  if (e > 1.0f) e = 1.0f;
  return (int)(e * 100.0f + 0.5f);
}

// Media de 8 lecturas del ADC en el pin de medida, en milivoltios.
static int readSupercapMv() {
  const int kSamples = 8;
  uint32_t sum = 0;
  for (int i = 0; i < kSamples; i++) {
    sum += (uint32_t)analogReadMilliVolts(PIN_SUPERCAP_SENSE);
    delay(2);
  }
  return (int)(sum / kSamples);
}

// Tension en los bornes del pack, a traves del divisor calibrado.
static float supercapVolts() {
  return (readSupercapMv() / 1000.0f) * SUPERCAP_DIVIDER_RATIO;
}

// =============================================================================
// 11. BATERIA BAJA: HIBERNACION
// =============================================================================
//
// Poner en reposo, cortarlo todo y dormir mucho tiempo. Se llama cuando el pack ha
// bajado a HIB_ENTER_V. A los sensores se les dice que duerman ANTES de que
// desaparezca su rail, en vez de quitarles la corriente a media medida - la misma
// cortesia que el sueno normal tiene con el BMV080 y el CM1106.
//
// Los sleep() de los sensores no cuestan casi nada aqui porque el rail se va a ir
// de todos modos. Existen para que el apagado sea ordenado, y porque dos de ellos
// (BME690, SEN0466) eran funciones vacias que hacian que "dormir a los sensores" no
// significara nada.
static void enterHibernation(float volts, bool announce) {
  Serial.printf("\n[HIBERNATE] pack at %.2fV, below %.2fV - stopping to protect the cells\n",
                volts, (double)HIB_ENTER_V);
  Serial.flush();

  if (railsArePowered()) {
    if (bmeReady) bme690.sleep();
    if (sen0466Ready) sen0466.sleep();
    if (bmvReady) bmv080.stopMeasurement();
    cm1106.sleep();
  }

#if HIB_ANNOUNCE_ENTRY
  // La UNICA transmision que hace la hibernacion, y solo al entrar. El silencio a
  // partir de aqui es deliberado, pero un silencio sin explicacion no se distingue en
  // Modulo A de un nodo muerto - y distinguirlos de otro modo cuesta un viaje al
  // bosque.
  if (announce) {
    sendHibernationNotice(volts);
  }
#else
  (void)announce;
#endif

  g_hibernating = true;
  ledOff();
  enterDeepSleep("hibernating - low battery", HIB_CHECK_SECONDS);
}

// Manda el paquete STAT descrito arriba.
static void transmitStatus() {
  // Los indicadores vienen de la ultima LECTURA, no de este despertar de envio - ver
  // g_lastReadHealth. La antiguedad dice cuantos despertares hace de eso, para que
  // un dato viejo se vea en vez de darse por actual.
  uint8_t h = g_lastReadHealth;
  unsigned long ageTicks =
      (g_wakeCount >= g_lastReadWake) ? (unsigned long)(g_wakeCount - g_lastReadWake) : 0UL;

  int vmv = readSupercapMv();
  float vcap = (vmv / 1000.0f) * SUPERCAP_DIVIDER_RATIO;

  // SIN DIVISOR MONTADO -> DECIR "DESCONOCIDO", NUNCA "0%".
  //
  // Solo la placa actualizada lleva el divisor de 51,1k/322k. En una antigua GPIO14
  // flota y lee ~100 mV haga lo que haga el pack - y con el factor calibrado eso
  // pasa directo por la formula y sale como un 0% muy convincente.
  //
  // Un nodo que esta claramente vivo y transmitiendo mientras dice que su bateria
  // esta vacia es peor que uno que admite que no lo sabe: invita a cambiar un pack
  // sano, y dispararia una alarma de bateria baja en cada placa antigua.
  //
  // El nodo no podria funcionar con un pack tan bajo, asi que una lectura por debajo
  // de SUPERCAP_V_SANITY_FLOOR significa que falta la linea de medida, no la
  // energia. Asi un mismo firmware se comporta bien en las dos revisiones de placa.
  int pct;
  if (!SUPERCAP_CALIBRATED || vcap < SUPERCAP_V_SANITY_FLOOR) {
    pct = -1;
  } else {
    pct = supercapPercent(vcap);
  }

  char msg[96];
  // El estado de alarma se ANADE como noveno campo en vez de insertarse, para que un
  // Modulo B con firmware antiguo siga leyendo los ocho que conoce por posicion y
  // simplemente ignore el extra. Insertarlo habria desplazado todos los campos de
  // detras y corrompido en silencio las lecturas de carga - el mismo tipo de fallo
  // que el del interprete de lotes que desplazaba todos los valores dos posiciones.
  int len = snprintf(msg, sizeof(msg), "STAT,%lu,%d,%lu,%d%d%d%d%d,%lu,%d,%d,%d",
                     (unsigned long)g_wakeCount, (int)esp_reset_reason(),
                     (unsigned long)nvsBoots,
                     (h >> 4) & 1, (h >> 3) & 1, (h >> 2) & 1, (h >> 1) & 1, h & 1,
                     ageTicks, vmv, pct, (int)g_alarmState);
  if (len < 0) len = 0;
  if (len >= (int)sizeof(msg)) len = (int)sizeof(msg) - 1;

  bool ok = radio.send(LORA_RX_ADDR, msg, (uint8_t)len);
  if (!SUPERCAP_CALIBRATED) {
    Serial.printf("[stat] %s  (%d mV at the pin; %% UNCALIBRATED - set the divider ratio) (%s)\n",
                  msg, vmv, ok ? "sent" : "FAILED");
  } else if (pct < 0) {
    Serial.printf("[stat] %s  (%d mV at the pin = %.2fV, below the %.1fV sanity floor - "
                  "no divider fitted? older PCB) (%s)\n",
                  msg, vmv, vcap, (double)SUPERCAP_V_SANITY_FLOOR, ok ? "sent" : "FAILED");
  } else {
    Serial.printf("[stat] %s  (cap %.2fV = %d%% usable, flags from wake %lu, %lu tick(s) ago) (%s)\n",
                  msg, vcap, pct, (unsigned long)g_lastReadWake, ageTicks,
                  ok ? "sent" : "FAILED");
  }
  Serial.flush();
}

// True si un sensor que deberia funcionar no leyo bien en este barrido. Enciende el
// LED rojo. "Deberia" significa activado Y arrancado, para que un Calypso ausente o
// un SEN0466 saltado a proposito no den una falsa alarma.
static bool sensorsDegraded() {
  if (bmeReady && g_bmeEnabled && g_bmeSt != ReadingStatus::Ok) return true;
  if (bmvReady && g_bmvEnabled && g_bmvSt != ReadingStatus::Ok) return true;
  if (g_cm1106Enabled && g_co2St != ReadingStatus::Ok) return true;
  return false;
}

// RECUPERACION DEL RAIL DE 3V3 TRAS EL CALEFACTOR DEL BME690.
//
// Lo dificil de leer varios sensores en un despertar es el calefactor de gas del
// BME690 a 320 C. Fallaba de una forma muy concreta: el BME690 se leia primero, su
// propio gas salia bajo (15764 en vez de ~45000) cuando el rail caia, y los dos
// sensores siguientes ya no podian arrancar:
//
//   gas 52925 -> bmv=OK     co2=OK
//   gas 15764 -> bmv=NoAck  co2=Timeout
//
// Ampliar el hueco entre turnos de 250 ms a 1500 ms no sirvio, porque la caida pasa
// DENTRO del ciclo del calefactor y no despues. Lo que si sirvio fue un ciclo
// completo de alimentacion del rail entre el calefactor y el siguiente sensor. Asi
// que esa recuperacion se hace aqui, justo despues de que funcione el calefactor:
// rail abajo, espera, rail arriba.
//
// Cuesta la espera de arranque del BMV080 una vez por cada medida de gas.
static void recoverRailsAfterHeater() {
  // SOLO SE CICLA EL RAIL DE 3V3. Antes se bajaban los dos.
  //
  // La recuperacion existe por un unico motivo: el BMV080 no arranca despues del
  // calefactor del BME690 si no se le cicla de verdad la alimentacion. Las dos
  // piezas estan en el rail de 3V3 (PIN_PCB_EN_A / GPIO10). El rail de 5V lleva el
  // CM1106 y el Calypso, que no tienen nada que ver.
  //
  // Bajar tambien el de 5V tenia un coste invisible hasta medir los tiempos: el
  // SEN0466 necesitaba 210 s de alimentacion CONTINUA para estabilizarse, y un corte
  // de 800 ms en cada medida le impedia acumularlos. Ciclar solo el rail que lo
  // necesita es ademas sencillamente mas correcto: los otros sensores se ciclaban
  // sin motivo.
  Serial.println("[read] 3V3 rail recovery after the BME690 heater (5V stays up)");
  Serial.flush();

  digitalWrite(PIN_PCB_EN_A, (PIN_PCB_EN_A_ACTIVE == LOW) ? HIGH : LOW);
  delay(HEATER_RECOVERY_OFF_MS);

  g_railsUp = true;
  digitalWrite(PIN_PCB_EN_A, PIN_PCB_EN_A_ACTIVE);
  delay(PIN_PCB_EN_SETTLE_MS);

  // El BMV080 necesita su ventana de arranque completa despues de cualquier ciclo de
  // alimentacion o da NoAck - justo el fallo que esta recuperacion existe para
  // evitar, asi que saltarse esta espera anularia su sentido.
  delay(BMV080_STARTUP_DELAY_MS);
}


// Arranca el sensor de CO cuando el rail de 5V lleva alimentado de forma continua
// el tiempo suficiente para que su lectura signifique algo. Se llama en cada vuelta
// de un estado que no duerme.
//
// El reloj de calentamiento vuelve a cero cada vez que cae el rail, porque un
// sensor a medio calentar informa con seguridad y mal - el fallo que esta guarda
// existe para evitar.
//
// HOY NO HACE NADA: todo su cuerpo esta dentro de
// "#if SLEEP_SKIP_SEN0466 && SEN0466_RUN_WHEN_AWAKE", y SLEEP_SKIP_SEN0466 vale 0
// (el SEN0466 se lee como los demas desde que va alimentado de forma permanente).
static uint32_t g_coWarmStartMs = 0;
static bool g_coWarmRunning = false;

static void sen0466MaintainWarmup() {
#if SLEEP_SKIP_SEN0466 && SEN0466_RUN_WHEN_AWAKE
  if (sen0466Ready && g_sen0466Enabled) return;  // ya esta funcionando

  // g_railsUp sigue al rail de 3V3, que se corta en cada envio por diseno. Al sensor
  // de CO solo le importa el de 5V, asi que con TX_KEEP_5V_RAIL el reloj de
  // calentamiento tiene que seguir corriendo durante el envio - si no, se reiniciaria
  // en cada transmision y no terminaria nunca, que es justo lo que mostro la primera
  // prueba en banco.
#if !TX_KEEP_5V_RAIL
  if (!g_railsUp) {
    if (g_coWarmRunning) {
      Serial.println("[co] 5V rail dropped - SEN0466 warm-up restarts from zero");
      g_coWarmRunning = false;
    }
    return;
  }
#endif

  if (!g_coWarmRunning) {
    g_coWarmRunning = true;
    g_coWarmStartMs = millis();
    Serial.printf("[co] 5V rail up - warming the SEN0466, %lus before it is trusted\n",
                  (unsigned long)(SEN0466_WARMUP_MS / 1000));
    return;
  }

  const uint32_t waited = millis() - g_coWarmStartMs;
  if (waited < SEN0466_WARMUP_MS) return;

  sen0466Ready = sen0466.begin();
  g_sen0466Enabled = sen0466Ready;
  Serial.printf("[co] SEN0466 warmed %lus -> %s\n", (unsigned long)(waited / 1000),
                sen0466Ready ? "LIVE, CO now measured and alarm-capable"
                             : "begin() FAILED - still not measured");
  if (!sen0466Ready) g_coWarmRunning = false;  // volver a intentarlo
#endif
}

// Lee TODOS los sensores en una pasada, en turnos. SOLO se usa en el modo sin sueno
// (SLEEP_ENABLED=0, builds de banco); el firmware normal usa runDueSensorJobs().
static void readFastSensorsOnce() {
  uint32_t t0 = millis();
  char pending[64];

  for (int slot = 0; slot < SLOT_COUNT; slot++) {
    if (millis() - t0 >= SENSOR_READ_WINDOW_MS) {
      Serial.printf("[read] WINDOW EXPIRED after %lums - slots %d..%d skipped\n",
                    (unsigned long)(millis() - t0), slot, SLOT_COUNT - 1);
      break;
    }

    // TIEMPO DE CADA TURNO. El modelo de consumo necesita el tiempo que cada sensor
    // consume de verdad, y eso no es ninguno de los #define - esos son techos y
    // huecos, no lo que cuesta un turno en la practica. Se mide aqui, en la
    // secuencia real, porque el coste de un turno depende de lo que corrio antes: el
    // BMV080 va tras un ciclo de rail, el CM1106 tras un ciclo de alimentacion.
    //
    // "active" es el turno en si; "recovery" es la pausa obligada despues, que
    // tambien es tiempo con el nodo despierto y el ESP consumiendo sus ~35 mA. Las
    // dos van en el presupuesto, asi que se informan las dos.
    const uint32_t slotT0 = millis();
    runSlot(slot);
    const uint32_t slotMs = millis() - slotT0;

    const uint32_t recT0 = millis();
    quiesceAll();

    // El calefactor es la unica carga lo bastante grande como para impedir que
    // arranquen los vecinos, asi que se le da un ciclo de rail y no un simple hueco.
    if (slot == SLOT_BME690 && bmeReady && g_bmeEnabled) {
      recoverRailsAfterHeater();
    } else {
      delay(SLOT_GAP_MS);
    }
    Serial.printf("[t] %-8s active=%lums recovery=%lums\n", slotName(slot),
                  (unsigned long)slotMs, (unsigned long)(millis() - recT0));
    Serial.flush();
  }

  Serial.printf("[t] TOTAL burst=%lums\n", (unsigned long)(millis() - t0));
  Serial.flush();

  // Fotografiar la salud AHORA, mientras los estados se refieren a lecturas recien hechas.
  g_lastReadHealth = (uint8_t)(((g_bmeSt == ReadingStatus::Ok ? 1 : 0) << 4) |
                               ((g_bmvSt == ReadingStatus::Ok ? 1 : 0) << 3) |
                               ((g_co2St == ReadingStatus::Ok ? 1 : 0) << 2) |
                               ((g_coSt == ReadingStatus::Ok ? 1 : 0) << 1) |
                               ((g_windValid && g_calFresh) ? 1 : 0));
  g_lastReadWake = g_wakeCount;

  uint8_t stillPending = pendingSensors(pending, sizeof(pending));
  Serial.printf("[read] bme=%s(T%.1f H%.1f gas%.0f) bmv=%s(pm2.5=%.1f) co2=%s(%.0f) "
                "co=%s(%.2f) wind=%s(rx=%u)%s%s  (all sensors, %lums)\n",
                statusName(g_bmeSt), g_temp, g_hum, g_gas,
                statusName(g_bmvSt), g_pm25,
                statusName(g_co2St), g_co2,
                statusName(g_coSt), g_co,
                !g_calypsoEnabled ? "off"
                                 : ((g_windValid && g_calFresh) ? "OK" : "silent"),
                calypso.lastReadBytes(),
                stillPending ? "  STALE: " : "", stillPending ? pending : "",
                (unsigned long)(millis() - t0));
  if (g_windValid) {
    Serial.printf("[wind] angle=%.1f deg  speed=%.2f\n", g_windAngle, g_windSpeed);
  }
  Serial.flush();
}

// Lectura rapida del BME690: T/H/P con el calefactor APAGADO.
//
// T/H/P sale SIEMPRE de aqui, tambien en los minutos que coinciden con el VOC,
// aunque la lectura completa tambien los traiga. El calefactor a 320 C calienta el
// propio chip y sesga su temperatura y su humedad; reaprovecharlos dejaria una de
// cada cuatro muestras sesgada hacia arriba, y en un detector con el umbral en
// 50 C un sesgo periodico es peor que pagar 200 ms de lectura aparte.
//
// OJO (revision 2026-10-14): eso es lo que se pretende, pero no lo que pasa hoy. En
// runDueSensorJobs() el trabajo de T/H/P va ANTES que el de VOC, y el turno del VOC
// (runSlot SLOT_BME690) tambien escribe g_temp/g_hum/g_pres. Asi que en los minutos
// en que coinciden, el valor que se guarda es el de la lectura con calefactor.
static void readBmeFastOnly() {
  if (!bmeReady || !g_bmeEnabled) return;
  Reading env = bme690.readFast();
  g_bmeSt = env.status;
  if (env.ok()) {
    g_bmeFresh = true;
    g_temp = env.values[0];
    g_hum = env.values[1];
    g_pres = env.values[2];
    // g_gas NO se toca a proposito: la rapida no mide gas y deja values[3] a cero
    // (count es 3 justo para que no se confunda). Copiarlo borraria el ultimo VOC
    // bueno y lo publicaria como una medida de cero.
  }
}

// Que slot sirve a cada trabajo. El BME690 es el unico con DOS trabajos sobre un
// mismo sensor, y por eso se despacha aparte: el slot completo enciende el
// calefactor y es el VOC, la rapida es el T/H/P.
static void readJob(uint8_t job) {
  // ARRANQUE PEREZOSO. setup() solo inicializa los sensores que vencian en ESE
  // despertar, asi que uno que venza mas tarde - al escalar a alarma, donde todos
  // vencen de golpe y el nodo ya no vuelve a pasar por setup() - llegaria aqui sin
  // arrancar. Sin esto, un episodio de incendio se mediria con los dos o tres
  // sensores que tocaban por calendario en vez de con los cinco.
  switch (job) {
    case JOB_BME_THP:
    case JOB_BME_VOC:
      if (!bmeReady && g_bmeEnabled) {
        Serial.println("[sched] BME690 no estaba arrancado - arrancando ahora");
        bmeReady = bme690.begin();
      }
      break;
    case JOB_SEN0466:
      if (!sen0466Ready && g_sen0466Enabled) {
        Serial.println("[sched] SEN0466 no estaba arrancado - arrancando ahora");
        sen0466Ready = sen0466.begin();
      }
      break;
    case JOB_BMV080:
      if (!bmvReady && g_bmvEnabled) {
        Serial.println("[sched] BMV080 no estaba arrancado - arrancando ahora");
        // Su espera de arranque tambien se salto en setup(), asi que se paga aqui.
        delay(BMV080_STARTUP_DELAY_MS);
        bmvReady = bmv080.begin();
      }
      break;
    default:
      break;
  }

  switch (job) {
    case JOB_BME_THP: readBmeFastOnly(); break;
    case JOB_BME_VOC: runSlot(SLOT_BME690); break;
    case JOB_CM1106: runSlot(SLOT_CM1106); break;
    case JOB_SEN0466: runSlot(SLOT_SEN0466); break;
    case JOB_BMV080: runSlot(SLOT_BMV080); break;
    case JOB_CALYPSO: runSlot(SLOT_CALYPSO); break;
    default: break;
  }
}

// Ejecuta los trabajos de sensor que vencen en este despertar.
//
// EN SERIE Y NUNCA A LA VEZ. Leerlos en paralelo reinicia la placa - 31 reinicios
// contra 2 en la prueba que lo demostro - asi que el bucle es secuencial y cada
// trabajo deja su hueco de reposo antes del siguiente. Eso es tambien lo que hace
// imposible que varias medidas caigan en el mismo instante cuando coinciden: el
// barrido las reparte, y por eso cada lectura se fecha con su hora real en vez de
// fingir que todas ocurrieron a la vez.
static bool runDueSensorJobs(uint8_t state) {
  bool any = false;
  const uint32_t t0 = millis();

  // LAS BANDERAS DE FRESCURA SE LIMPIAN AQUI, Y ESTE ES EL UNICO SITIO.
  //
  // clearFreshFlags() existia desde hacia tiempo y NO LA LLAMABA NADIE. En
  // operacion normal no se notaba: cada despertar es un arranque nuevo y estas
  // banderas son variables normales, no RTC, asi que nacen a false solas.
  //
  // Pero EN ALARMA EL NODO NO DUERME. Sin esta llamada, una bandera puesta a true
  // al principio del episodio se quedaba asi durante todo el incendio, y a partir
  // de ahi:
  //
  //   * ruleHasData() daba por buena una magnitud que llevaba media hora sin
  //     leerse, asi que la alarma podia sostenerse - o confirmarse - sobre un
  //     valor viejo;
  //   * r.windValid publicaba valid=1 con un viento congelado, que es exactamente
  //     el fallo que ya se diagnostico en C3 el 2026-10-07 y que el comentario de
  //     storeCurrentReading describe como "el mismo patron que el co2=663 de C1".
  //
  // La proteccion estaba escrita y desconectada, que es la peor de las dos
  // posibilidades: parece que el problema esta cubierto.
  //
  // SOLO SI DE VERDAD SE VA A LEER ALGO, y esto no es un detalle. En alarma el
  // nodo no duerme y loop() entra aqui cada SAMPLE_GAP_MS, asi que la mayoria de
  // pasadas no tienen ningun trabajo vencido. Limpiando en todas, ninguna regla
  // tendria dato fresco, evaluateRules() devolveria NORMAL y allRulesCleared()
  // diria que si - porque se salta las reglas sin dato - de modo que el contador
  // de lecturas limpias subiria cada dos segundos y LA ALARMA SE BAJARIA SOLA EN
  // DIEZ SEGUNDOS, en vez de exigir cinco lecturas reales por debajo del umbral.
  //
  // Arreglar la frescura y romper la confirmacion de la alarma en el mismo cambio
  // habria sido un mal negocio.
  if (!schedAnySensorDue()) return false;
  clearFreshFlags();

  for (uint8_t j = 0; j < JOB_TX; j++) {
    if (!g_jobDue[j]) continue;

    // Techo del despertar. Si un sensor se atasca, los que falten se quedan sin
    // leer esta vuelta pero su vencimiento NO se mueve: volveran a vencer ya
    // mismo, en lugar de perderse hasta el siguiente multiplo.
    if (millis() - t0 >= SENSOR_READ_WINDOW_MS) {
      Serial.printf("[sched] ventana agotada tras %lums - quedan trabajos sin leer\n",
                    (unsigned long)(millis() - t0));
      break;
    }

    const uint32_t slotT0 = millis();
    readJob(j);
    const uint32_t slotMs = millis() - slotT0;

    quiesceAll();
    // El calefactor es la unica carga lo bastante grande como para impedir que
    // arranque el sensor siguiente, asi que se le da un ciclo de rail y no un
    // simple hueco.
    if (j == JOB_BME_VOC && bmeReady && g_bmeEnabled) {
      recoverRailsAfterHeater();
    } else {
      delay(SLOT_GAP_MS);
    }

    Serial.printf("[t] %-15s activo=%lums\n", kJobs[j].name, (unsigned long)slotMs);
    Serial.flush();
    schedMarkDone(j, state);
    any = true;
  }

  if (any) {
    g_lastReadHealth = (uint8_t)(((g_bmeSt == ReadingStatus::Ok ? 1 : 0) << 4) |
                                 ((g_bmvSt == ReadingStatus::Ok ? 1 : 0) << 3) |
                                 ((g_co2St == ReadingStatus::Ok ? 1 : 0) << 2) |
                                 ((g_coSt == ReadingStatus::Ok ? 1 : 0) << 1) |
                                 ((g_windValid && g_calFresh) ? 1 : 0));
    g_lastReadWake = g_wakeCount;
    Serial.printf("[sched] barrido %lums  T%.1f H%.1f gas%.0f pm2.5=%.1f co2=%.0f "
                  "co=%.2f viento %.0fdeg/%.2f\n",
                  (unsigned long)(millis() - t0), g_temp, g_hum, g_gas, g_pm25,
                  g_co2, g_co, g_windAngle, g_windSpeed);
    Serial.flush();
  }
  return any;
}

// =============================================================================
// 12. loop()
// =============================================================================
//
// En funcionamiento normal cada despertar entra aqui despues de setup(), hace UNA
// vez el trabajo que le toca y se duerme. Ramas, por orden:
//   - MODO SEGURO: sensores apagados, un paquete de socorro, y reinicio a los 60 s.
//   - PRE-ALARMA / ALARMA: no duerme; consulta el calendario en cada vuelta.
//   - SIN SUENO (SLEEP_ENABLED=0, banco): lee y transmite sin parar.
//   - NORMAL: el trabajo de este despertar, y a dormir hasta el proximo vencimiento.
void loop() {
  // Una vuelta que dura esto cuenta como sana, asi que se borra la cuenta de NVS y el
  // siguiente arranque empieza de cero. Sin esto cada reflasheo acabaria disparando
  // el modo seguro. 15 s supera de sobra lo que la placa aguantaba cuando fallaba.
  static bool markedGood = false;
  if (!markedGood && millis() > 15000) {
    markedGood = true;
    bootGuardMarkGood();
  }

  // ---- MODO SEGURO ----
  // Los sensores siguen apagados, pero el nodo NO se queda callado, y NO se queda
  // aqui para siempre.
  //
  // Antes hacia las dos cosas. g_safeMode se decide una vez en setup(); loop()
  // borraba la cuenta de NVS a los 15 s y luego volvia aqui en cada vuelta, sin
  // dormir y por tanto sin reiniciarse nunca para volver a evaluarlo. Un solo
  // arranque desafortunado enclavaba el nodo en modo seguro hasta que una persona lo
  // desenchufaba - y con la radio sin iniciar, el nodo no podia decirlo. El nodo C1
  // estuvo asi desde el 2026-09-11, encendido y despierto, pareciendo desde
  // cualquier angulo una placa muerta.
  if (g_safeMode) {
    static uint32_t lastSaid = 0;
    if (millis() - lastSaid > 5000) {
      lastSaid = millis();
      Serial.printf("[safe mode] sensors off, %lus until auto-recovery reboot\n",
                    (unsigned long)((SAFE_MODE_RECOVER_MS - millis()) / 1000));
      Serial.flush();
    }

    // Avisarlo por radio, una vez. Un nodo con problemas tiene que poder oirse: un
    // paquete de socorro es mucho mas util que el silencio, y cuesta una sola
    // transmision. Los sensores siguen apagados; solo se enciende el rail de LoRa.
    static bool distressSent = false;
    if (!distressSent && millis() > 3000) {
      distressSent = true;
      sendSafeModeDistress();
    }

    // Despues, reiniciar en modo normal. La cuenta de NVS se borro arriba a los 15 s,
    // asi que el siguiente arranque empieza limpio; si la averia sigue ahi, volvera a
    // entrar en modo seguro y a mandar otro paquete de socorro, que es un latido
    // visible en vez de un enclavamiento silencioso.
    if (millis() >= SAFE_MODE_RECOVER_MS) {
      Serial.println("[safe mode] recovery window elapsed - restarting into normal mode");
      Serial.flush();
      delay(50);
      esp_restart();
    }

    delay(200);
    return;
  }

  // ---- PRE-ALARMA / ALARMA: sin dormir ----
  //
  // El nodo se queda despierto y sigue leyendo hasta que las lecturas se calmen o
  // Modulo A lo baje, y transmite con los periodos de su estado (15 min en
  // pre-alarma, 3 min en alarma). La ventana de escucha de cada envio es tambien por
  // donde llega una orden de bajarla.
  //
  // El STAT sale junto a las lecturas: el STAT es el paquete que lleva el estado de
  // alarma, asi que sin el Modulo B veria a un nodo transmitiendo sin saber que esta
  // en alarma. Es todo el camino entre "este nodo ha detectado un incendio" y "a
  // alguien se le avisa".
  if (g_alarmState >= ALARM_PREALARM) {
#if FORCE_ALARM_STATE
    // Repetir el aviso con la placa en marcha: quien lea el puerto serie media hora
    // despues no deberia tener que haber visto el mensaje de arranque para saber que
    // esto es un build de prueba con una alarma falsa.
    static uint32_t lastForcedWarn = 0;
    if (millis() - lastForcedWarn > 30000) {
      lastForcedWarn = millis();
      Serial.printf("[TEST BUILD] alarm state FORCED to %s - not from the sensors\n",
                    (FORCE_ALARM_STATE >= 2) ? "ALARM" : "PRE-ALARM");
      Serial.flush();
    }
#endif
    // Acumular antes de evaluar, para que la ventana este al dia cuando
    // updateAlarmState la compruebe. Solo se cronometra la pre-alarma; la alarma
    // completa no tiene nada que confirmar.
    static uint32_t lastAccumMs = 0;
    const uint32_t nowMs = millis();
    if (lastAccumMs != 0 && g_alarmState == ALARM_PREALARM) {
      g_prealarmMs += (nowMs - lastAccumMs);
    }
    lastAccumMs = nowMs;

    // Sin dormir, el rail de 5V se mantiene y el sensor de CO podria calentarse.
    // (Hoy esta funcion no hace nada; ver su comentario.)
    sen0466MaintainWarmup();

    // MISMO PLANIFICADOR QUE EN NORMAL, otros periodos. Antes esta rama leia TODOS
    // los sensores en cada pasada, lo que ignoraba por completo la tabla por
    // estado: en alarma el CO2 debe seguir yendo cada 30 min aunque el T/H/P vaya
    // cada minuto - el CM1106 es el sensor menos fiable que hay y machacarlo no
    // aporta nada.
    //
    // No se duerme: aqui el nodo esta despierto y loop() gira cada SAMPLE_GAP_MS,
    // asi que el planificador se consulta en cada vuelta en vez de calcular un
    // sueno.
    schedEvaluate();
    const bool algoLeido = runDueSensorJobs(g_alarmState);
    updateAlarmState();
    if (algoLeido) storeCurrentReading();

    if (g_jobDue[JOB_TX]) {
      transmitStore();
      transmitStatus();
      schedMarkDone(JOB_TX, g_alarmState);
    }
    delay(SAMPLE_GAP_MS);
    return;  // en estado alto nunca se llega al camino del sueno
  }

#if !SLEEP_ENABLED
  // ---- CONTINUO (banco) ----
  // Leer en cada vuelta y transmitir con un temporizador. Sin dormir, el puerto USB
  // se mantiene y el LED se puede mirar el tiempo que haga falta.
  sen0466MaintainWarmup();
  readFastSensorsOnce();
  updateAlarmState();
  storeCurrentReading();

  static uint32_t lastTxMs = 0;
  if (millis() - lastTxMs >= NOSLEEP_TX_GAP_MS) {
    lastTxMs = millis();
    ledWorking();
    transmitStore();
    ledOff();
  }
  delay(SAMPLE_GAP_MS);
  return;
#endif

  // ---- NORMAL: el trabajo de este despertar, y a dormir ----
  // Lo que debe este despertar se decidio en setup(), antes de encender nada - ver
  // la salida temprana de "nada vence".
  bool readDue = g_readDue;
  bool txDue = g_txDue;

  // TODO EL TRABAJO DE ESTE DESPERTAR SE HACE UNA SOLA VEZ, entre aqui las veces
  // que se entre en loop().
  //
  // loop() se repite mientras el nodo esta despierto sin dormir: durante toda la
  // ventana de 30 s del arranque en frio, por ejemplo. Lo que vence se decide una
  // vez en setup() y no cambia dentro de un arranque, asi que sin esta guarda cada
  // vuelta repetiria todo el trabajo. Medido antes del arreglo: 17 lecturas y SEIS
  // TRANSMISIONES IDENTICAS dentro de una sola ventana de arranque en frio - energia
  // de radio tirada y filas duplicadas llegando a Modulo B.
  //
  // Un static normal es el ambito correcto: cada despertar es un arranque nuevo
  // desde el sueno, asi que esto empieza en false en cada uno por construccion.
  static bool tickWorkDone = false;
  if (!tickWorkDone) {
    tickWorkDone = true;

    // Solo lo que vence, cada sensor en su turno. Ya no hay "centinelas" ni
    // "burst": esa division existia para repartir el coste entre ticks, y con
    // periodos en minutos cada sensor lleva el suyo propio.
    const bool algoLeido = runDueSensorJobs(g_alarmState);

    updateAlarmState();

    // Se guarda una fila si se midio algo. A diferencia del esquema por ticks ya
    // no hay lecturas "parciales que no merecen guardarse": una fila con solo
    // T/H/P es legitima, porque el resto de sensores no VENCIA - no es que
    // fallaran.
    if (algoLeido) storeCurrentReading();

    if (txDue) {
      ledWorking();  // blanco durante la transmision
      transmitStore();
      transmitStatus();
      schedMarkDone(JOB_TX, g_alarmState);
    }

    // La salud solo es juzgable cuando han medido los sensores lentos; en un
    // despertar que solo traia T/H/P, el CO2 "ausente" es lo esperado y no una
    // averia.
    if (g_jobDue[JOB_BMV080] && g_jobDue[JOB_CM1106] && sensorsDegraded()) {
      Serial.println("[health] un sensor que se esperaba no leyo - LED rojo");
      Serial.flush();
      ledTrouble();
    }
  }

  if (g_alarmState >= ALARM_PREALARM) return;  // quedarse despierto - lo atiende la rama de arriba

  if (coldBootWindowOpen()) {
    delay(SAMPLE_GAP_MS);
    return;  // todavia dentro de la ventana de reflasheo - no dormir aun
  }

  // Duerme hasta el PROXIMO VENCIMIENTO menos su preparacion, no una cantidad
  // fija. Es el cambio que hace que "cada 5 minutos" signifique cinco minutos:
  // antes se dormian 10 s ADEMAS del trabajo, y el periodo real salia 18,8 s.
  enterDeepSleep(readDue || txDue ? "trabajo hecho" : "nada vencia",
                 schedSleepSeconds(g_alarmState));

  // Solo se llega aqui con SLEEP_ENABLED a 0.
  delay(SAMPLE_GAP_MS);
}

#pragma once

// =============================================================================
// MAPA DE PINES DE MODULO C (PLACA CHIP FOREST, ESP32-S3)
// =============================================================================
//
// Este archivo empezo describiendo un banco de pruebas sin placa fabricada (un
// rail de 3V3 siempre encendido, sin transistores de corte). Hoy describe la placa
// real. Cada valor de abajo se confirmo en la practica (escaneos I2C, lecturas que
// funcionan), no se dio por bueno a partir de un esquema.
//
// Resumen:
//   GPIO1/2   I2C (BME690, BMV080, SEN0466)      GPIO10  rail 3V3   (LOW  = on)
//   GPIO3     CM1106 EN                           GPIO11  rail 5V    (HIGH = on)
//   GPIO4/5   UART radio LoRa                     GPIO13  rail LoRa  (LOW  = on)
//   GPIO6/7   UART CM1106                         GPIO14  medida de bateria (analogica)
//   GPIO8/9   UART Calypso                        GPIO15  BMV080 EN  (LOW  = on)
//   GPIO12    CM1106 RDY (NO TOCAR)               GPIO16  SEN0466 EN (LOW  = on)
//                                                 GPIO21  LED de estado

// Bus I2C compartido: BME690, SEN0466 y BMV080 cuelgan todos de el.
#define PIN_I2C_SDA 1
#define PIN_I2C_SCL 2

// Sensor de CO2 Cubic CM1106SL-NS: UART2. Cuelga del rail de 5 V conmutado.
#define PIN_CM1106_TX 6  // TX del ESP -> RX del sensor
#define PIN_CM1106_RX 7  // RX del ESP <- TX del sensor
#define PIN_CM1106_EN 3  // activo a nivel alto, con una resistencia de 1k en serie en la placa

// Anemometro Calypso ULP PRO: UART1. Variante TTL-UART/NMEA confirmada (no RS485).
// 2026-09-01: RX paso de 9 a 8 y TX de 8 a 9. El TX del sensor (cable verde) esta
// fisicamente en GPIO8: un barrido en banco por GPIO5/6/8/9 y velocidades de 4800 a
// 115200 solo obtuvo una frase $IIMWV valida con RX=GPIO8 a 38400. Ademas esta
// unidad necesita 5 V, no 3,3 V: a 3,3 V estaba muda y empezo a emitir en cuanto se
// paso a 5 V.
#define PIN_CALYPSO_TX 9  // TX del ESP -> RX/amarillo del sensor (no se usa: el sensor emite solo)
#define PIN_CALYPSO_RX 8  // RX del ESP <- TX/verde del sensor

// GPIO12 va al CM1106 de CUBIC (el sensor de CO2; Cubic es el fabricante), que es
// lo que queria decir la nota original de "futura linea RDY del CM1106".
//
// NO CONDUCIRLO. Se puso a nivel bajo un momento mientras se buscaba la
// habilitacion del BMV080, antes de que se describiera la placa. El CM1106
// funciona, y forzar una linea contra un sensor que funciona es una buena manera
// de dejar de tenerlo.
#define PIN_CM1106_RDY 12

// Habilitacion del BMV080. Tiene que estar a nivel BAJO para que el sensor funcione.
//
// Sin conducir, el BMV080 abre perfectamente por I2C - contesta en su direccion y
// bmv080_open() tiene exito - pero se niega a encender el laser y devuelve el
// estado 114 del SDK (E_BMV080_ERROR_OPERATION_MODE_CHANNELS_OUT_OF_SYNC), cuya
// pista de Bosch es "comprobar que la alimentacion del BMV080 es suficiente y
// estable". Esa pista apunta a la alimentacion, que es exactamente lo que parece
// una habilitacion sin activar vista desde el sensor.
//
// Paso de 12 a 14 y a 15 conforme se describio la placa; el 12 resulto ser el
// CM1106 y el 14 la medida de bateria. Se activa junto con los rails, antes de abrir
// ningun sensor, para que este estable cuando el SDK hable con la pieza.
#define PIN_BMV080_EN 15
#define PIN_BMV080_EN_ACTIVE LOW

// Medida de la tension de la bateria - ENTRADA ANALOGICA, NUNCA UNA SALIDA.
//
// No se debe conducir nunca. Durante la busqueda de la habilitacion del BMV080 se
// confundio GPIO14 con otra candidata y se mantuvo a nivel bajo como salida: eso
// pelea contra el divisor que la alimenta y puede danarlo. Se deja escrito aqui
// para que nadie lo repita.
//
// GPIO14 es ADC2_CH3 en el ESP32-S3. El ADC2 no esta disponible con el WiFi activo,
// lo que no importa en Modulo C (sin WiFi) pero si en Modulo B.
//
// El nombre viene de la epoca de los supercondensadores. La conversion de la
// tension del pin a porcentaje de carga esta en chip_forest_lora_tx.cpp
// (SUPERCAP_*). OJO: desde el 2026-09-24 la fuente es una bateria de litio, y esa
// conversion se calculo para supercondensadores.
#define PIN_SUPERCAP_SENSE 14

// ---------- Habilitaciones de los rails de sensores (GPIO10/11) ----------
// La placa real conecta los DOS rails de alimentacion de sensores a traves de
// estos pines: cada uno maneja un transistor que cierra el circuito hacia el rail
// de 5 V y el de 3V3 respectivamente (confirmado 2026-09-04).
//
// Hay que activarlos al principio de setup(), antes de Wire.begin() y de cualquier
// begin() de sensor: desactivados, los sensores no tienen corriente, y un escaneo
// I2C o una prueba por UART no encontraria nada y pareceria hardware muerto. El
// Calypso y el CM1106 necesitan precisamente el rail de 5 V.
//
// LAS DOS PUERTAS TIENEN POLARIDAD OPUESTA. Se establecio el 2026-09-08 probando
// las cuatro combinaciones de GPIO10/11 y leyendo el CM1106 (el unico aparato de
// 5 V conectado) en cada una:
//
//     GPIO10=LOW  GPIO11=LOW   -> mudo
//     GPIO10=LOW  GPIO11=HIGH  -> contesta     <- rail de 5 V encendido
//     GPIO10=HIGH GPIO11=LOW   -> mudo
//     GPIO10=HIGH GPIO11=HIGH  -> contesta     <- rail de 5 V encendido
//
// Asi que GPIO11 maneja el rail de 5 V y es ACTIVO A NIVEL ALTO, mientras que GPIO10
// (3V3) es activo a nivel bajo, como se supuso al principio: los sensores de 3V3
// siempre han funcionado con el a LOW.
//
// Esto estuvo oculto durante dias por la averia de la masa flotante: sin una
// referencia firme, poner GPIO11 "a LOW" no era un nivel bajo real en el transistor,
// y el rail de 5 V seguia encendido por casualidad. Al reparar la masa, LOW paso a
// ser un nivel bajo de verdad y apago el rail de 5 V como debia, lo que parecio que
// el CM1106 se moria. No se moria: estaba sin corriente.
#define PIN_PCB_EN_A 10          // rail de sensores de 3V3
#define PIN_PCB_EN_A_ACTIVE LOW
#define PIN_PCB_EN_B 11          // rail de sensores de 5V
#define PIN_PCB_EN_B_ACTIVE HIGH
#define PIN_PCB_EN_SETTLE_MS 300

// Se conserva para que compilen programas antiguos, pero solo es correcto para la
// puerta A. Mejor usar las macros por puerta de arriba.
#define PIN_PCB_EN_ACTIVE PIN_PCB_EN_A_ACTIVE

// ---------- Puerta de alimentacion de la radio LoRa RYLR998 (GPIO13) ----------
// Tercera puerta, aparte de las dos anteriores: un transistor P-FET en la

#pragma once

#include <stdint.h>

// =============================================================================
// RESULTADO DE UNA LECTURA DE SENSOR
// =============================================================================
//
// Un sensor que FALLA y un sensor que MIDE CERO tienen que poder distinguirse en
// quien los llama. Por eso cada lectura lleva un estado explicito, en vez de un
// valor "especial" escondido en el numero.
enum class ReadingStatus : uint8_t {
  NotInitialized,  // begin() nunca tuvo exito
  NotReady,        // el sensor esta bien pero no tiene nada nuevo en este ciclo
  Timeout,         // se pidio el dato y no llego respuesta (o llego incompleta) a tiempo
  NoAck,           // fallo del bus: el dispositivo no contesto en absoluto
  InvalidFrame,    // llego respuesta pero no paso la comprobacion de checksum/formato
  Ok
};

// Que significa cada posicion de values[] y cuantas hay (count) depende del
// sensor; esta documentado en la cabecera de cada driver.
struct Reading {
  ReadingStatus status = ReadingStatus::NotInitialized;
  float values[4] = {0, 0, 0, 0};
  uint8_t count = 0;

  bool ok() const { return status == ReadingStatus::Ok; }
};

// =============================================================================
// INTERFAZ COMUN DE TODOS LOS SENSORES

// Todos los drivers implementan estas cuatro funciones, asi que quien los usa no

class ISensor {
public:
  virtual ~ISensor() = default;

  // Arranque, una sola vez. false = no detectado o no se pudo configurar.
  virtual bool begin() = 0;

  virtual Reading read() = 0;

  // Dejar el sensor en reposo. La alimentacion la sigue controlando el firmware
  // principal (los rails): esto solo le dice al sensor que deje de trabajar, no
  // le quita la corriente.
  virtual void sleep() = 0;

  virtual const char *name() const = 0;
};

// =============================================================================
// BME690 (Bosch) - TEMPERATURA, HUMEDAD, PRESION Y GAS (VOC)
// =============================================================================
//
// Bus I2C compartido (SDA/SCL de pins.h). En esta revision de placa el pin SDO
// queda al aire, asi que la direccion no esta fijada en hardware: se prueba
// primero primaryAddr y despues la otra de {0x76, 0x77}, porque las dos son
// estados legitimos.
//
// values[]: [0] = temperatura C, [1] = humedad %, [2] = presion hPa,
//           [3] = resistencia de gas en ohmios.
class Bme690Sensor : public ISensor {
public:
  explicit Bme690Sensor(uint8_t primaryAddr) : _addr(primaryAddr) {}

  bool begin() override;

  // Lectura COMPLETA: temperatura, humedad, presion Y resistencia de gas. Enciende
  // el calefactor de gas a 320 C. Para el trabajo de gas (VOC) del calendario, no
  // para lecturas frecuentes.
  // (Una hoja de calculo antigua daba ~10,8 s para esta lectura; medido en el banco
  // el 2026-10-06, la llamada vuelve en ~37 ms porque el calefactor trabaja dentro
  // del sensor y no bloquea al ESP32.)
  Reading read() override;

  // Lectura RAPIDA: solo temperatura, humedad y presion, con el calefactor de gas
  // APAGADO. Es la del trabajo de T/H/P, el mas frecuente del calendario. El
  // calefactor es, con diferencia, la parte cara de una medida del BME690: sin el,
  // el consumo baja de ~3,1 mA a casi nada.
  //
  // values[]: [0] = temperatura, [1] = humedad, [2] = presion, [3] = 0 (sin gas).
  // count vale 3, para que nadie confunda el gas ausente con uno real.
  Reading readFast();

  // En que direccion se quedo begin(). Prueba la configurada y luego la otra, asi
  // que quien llama no puede saber de otro modo cual esta viva - y eso importa
  // cuando un sensor arranca bien pero despues no devuelve nada.
  uint8_t address() const { return _addr; }
  // Antes era una funcion vacia, y "dormir el sensor" no hacia nada. El calefactor
  // ES el consumo del BME690 (el resto es despreciable), asi que dormirlo es
  // apagar el calefactor y dejarlo apagado.
  void sleep() override;
  const char *name() const override { return "BME690"; }

  // Valor crudo del registro de identificacion del chip (0xD0), leido directamente
  // cuando begin() falla, sin pasar por el driver de Bosch. begin() resume cinco
  // fallos distintos en un solo `false`; esto es lo que los distingue. En una pieza
  // sana vale 0x61. Solo tiene sentido si chipIdValid() es true.
  uint8_t lastChipId() const { return _chipId; }
  bool chipIdValid() const { return _chipIdValid; }
  // Que forma de acceso contesto: 0 = inicio repetido (la que usa el driver de
  // Bosch), 1 = separada por STOP, -1 = ninguna.
  int chipIdMode() const { return _chipIdMode; }

private:
  void primeAndMark();
  // Enciende o apaga el calefactor de gas, recordando el estado actual para que la
  // escritura de configuracion (una transaccion I2C) solo se haga cuando cambia de
  // verdad.
  bool setHeater(bool on);

  uint8_t _addr;
  bool _primed = false;  // historico: ver primeAndMark() en el .cpp
  uint8_t _chipId = 0;
  bool _chipIdValid = false;
  int _chipIdMode = -1;
  bool _heaterOn = true;  // begin() deja activo el perfil por defecto de la libreria
  BME69X_7Semi _sensor;
};

#include "Bme690Sensor.h"

#include <Wire.h>

// Lectura cruda del registro de identificacion del chip, saltandose por completo
// el driver de Bosch.
//
// Hace falta porque el begin() de la libreria resume varios fallos muy distintos
// en un solo `false`: la prueba de ACK en I2C, el reinicio por software de
// bme69x_init(), la comprobacion del ID, la lectura de variante y la carga de la
// calibracion. El nodo C2 mostro `ACK at 0x76` en el escaneo del bus y
// `BME690: NOT FOUND` en begin() en EL MISMO arranque: el aparato contesta a su
// direccion pero algo posterior falla, y sin el valor crudo no hay forma de
// distinguir una pieza equivocada de una lectura corrupta.
//
// El valor esperado es 0x61 (comun a BME680/688/690). En los foros hay casos de
// este registro devolviendo 0x00 o 0xF1 en buses que si dan ACK: es corrupcion del
// bus, no una pieza distinta. La prueba de direccion es un solo byte y sobrevive;
// la lectura de un registro es una transaccion mas larga y no.
//
// Dos formas de acceso, porque fallan por motivos distintos y un aparato que
// rechaza una puede contestar a la otra:
//
//   modo 0  escribir registro, INICIO REPETIDO, leer   - lo que hace el driver de Bosch
//   modo 1  escribir registro, STOP, y una lectura nueva aparte
//
// Si el modo 1 funciona donde falla el 0, la pieza esta viva y el problema es el
// inicio repetido; si fallan los dos pero la direccion sigue dando ACK, el aparato
// tiene corriente para tirar de la linea pero no para funcionar.
static bool readChipIdMode(uint8_t addr, uint8_t &out, int mode) {
  Wire.beginTransmission(addr);
  Wire.write(0xD0);  // BME69X_REG_CHIP_ID
  if (Wire.endTransmission(mode == 1) != 0) return false;
  if (mode == 1) delay(2);
  if (Wire.requestFrom((int)addr, 1) != 1) return false;
  out = (uint8_t)Wire.read();
  return true;
}

// Prueba los dos modos por orden y dice cual contesto (-1 si ninguno).
static bool readChipId(uint8_t addr, uint8_t &out, int &modeUsed) {
  for (int mode = 0; mode < 2; mode++) {
    if (readChipIdMode(addr, out, mode)) {
      modeUsed = mode;
      return true;
    }
    delay(5);
  }
  modeUsed = -1;
  return false;
}

// LA PRIMERA MEDIDA FORZADA DESPUES DE begin() SIEMPRE FALLA EN ESTA PIEZA.
//
// Medido en el nodo C2: lectura 1 -> Timeout, lectura 2 -> OK (gas 41451),
// lectura 3 -> OK, en todos los arranques. getData() de la libreria dispara una
// medida, espera la duracion calculada y trata cualquier estado distinto de OK de
// bme69x_get_data() como fallo - y los avisos de Bosch como NO_NEW_DATA son
// codigos POSITIVOS, asi que un aviso en la primera medida no se distingue de un
// error. El calefactor de gas tambien necesita una pasada antes de estabilizarse,
// y por eso es la primera medida la que avisa.
//
// Mientras el nodo leia sin parar, esto costaba una lectura de cada cien y paso
// desapercibido. Con el ciclo de sueno es fatal: cada despertar es un begin()
// nuevo seguido de UNA sola lectura, asi que la unica lectura que se hace es la que
// siempre falla, y el BME690 no aportaria nunca un valor. Por eso read() y
// readFast() reintentan una vez.

// Arranca el sensor, probando las dos direcciones y hasta tres veces.
bool Bme690Sensor::begin() {
  _chipId = 0;
  _chipIdValid = false;

  // Se repite el arranque entero en vez de rendirse al primer intento malo.
  //
  // bme69x_init() hace un REINICIO POR SOFTWARE y lee el ID justo despues. Una pieza
  // a la que se le acaba de encender el rail - que es cada despertar con el ciclo de
  // sueno - puede estar todavia estabilizandose cuando llega esa lectura, y el
  // driver devuelve el mismo `false` que para un sensor ausente de verdad. Tres
  // intentos con pausa cuestan ~150 ms si falla y nada en absoluto si va bien.
  for (int attempt = 0; attempt < 3; attempt++) {
    if (_sensor.begin(_addr)) {
      primeAndMark();
      return true;
    }

    uint8_t altAddr = (_addr == 0x76) ? 0x77 : 0x76;
    if (_sensor.begin(altAddr)) {
      _addr = altAddr;
      primeAndMark();
      return true;
    }

    delay(50);
  }

  // Ha fallado. Se lee el ID crudo para que quien llama pueda decir POR QUE y no
  // solo "NOT FOUND": 0x61 = la pieza esta bien y falla el driver; 0x00 o basura =
  // el bus corrompe las lecturas; ninguna respuesta = no esta de verdad.
  _chipIdValid = readChipId(_addr, _chipId, _chipIdMode);
  return false;
}

// Marca el sensor como arrancado. A PROPOSITO NO HACE NINGUNA MEDIDA.
//
// Antes lanzaba aqui una medida de cebado que se descartaba, lo que significaba
// encender el calefactor a 320 C dentro de setup(), justo antes de bmv080.begin().
// El efecto se midio: con el cebado aqui el BME690 arrancaba y el BMV080 daba
// NoAck y el CM1106 Timeout; sin el, arrancaban el BMV080 y el CM1106 y el BME690
// no. Los sensores se turnaban.
//
// setup() arranca todos los sensores seguidos, asi que era el unico sitio donde no
// se aplicaba la regla de "un sensor cada vez" (los turnos gobiernan solo las
// LECTURAS). Un calefactor encendido ahi es justo la carga simultanea que los
// turnos existen para evitar.
//
// Ademas el cebado no hace falta: read() ya reintenta, lo que cubre el "la primera
// medida siempre falla", y lo hace dentro del turno del propio BME690, donde el
// calefactor tiene la alimentacion para el solo.
void Bme690Sensor::primeAndMark() {
  _primed = true;
  _chipId = 0x61;  // si arranco, el ID coincidia por definicion
  _chipIdValid = true;
  _chipIdMode = 0;
}

// Dormir = calefactor apagado. El calefactor es todo el consumo de este sensor; el
// resto del chip consume microamperios. Se llama antes de cortar el rail de 3V3,
// para dejar la pieza en reposo en vez de quitarle la corriente a media medida.
// Cortar el rail ahorra mucho mas (un sensor sin alimentacion no consume nada), asi
// que lo que aporta esto es un apagado ordenado, no el ahorro.
void Bme690Sensor::sleep() {
  setHeater(false);
}

// El calefactor es todo el coste de una lectura del BME690. Cambiarlo es una
// escritura de configuracion por I2C, asi que se recuerda el estado y solo se
// escribe cuando cambia de verdad.
//
// 320 C / 150 ms son los valores por defecto de la propia libreria, y se restauran
// al reactivarlo para que las cifras de gas sigan siendo comparables con las
// tomadas antes de que se tocara nunca el calefactor.
bool Bme690Sensor::setHeater(bool on) {
  if (_heaterOn == on) return true;
  if (!_sensor.setHeater(on, 320, 150)) return false;
  _heaterOn = on;
  return true;
}

// Temperatura, humedad y presion con el calefactor APAGADO - ver la cabecera.
Reading Bme690Sensor::readFast() {
  Reading r;

  if (!setHeater(false)) {
    r.status = ReadingStatus::NotReady;
    return r;
  }

  float t, h, pr, g;
  for (int attempt = 0; attempt < 2; attempt++) {
    if (_sensor.getData(t, h, pr, g)) {
      r.status = ReadingStatus::Ok;
      r.values[0] = t;
      r.values[1] = h;
      r.values[2] = pr;
      r.values[3] = 0.0f;  // no se midio gas: no hacerlo pasar por real
      r.count = 3;
      return r;
    }
    delay(20);
  }

  r.status = ReadingStatus::Timeout;
  return r;
}

// Lectura completa, con el calefactor encendido (readFast() puede haberlo apagado).
Reading Bme690Sensor::read() {
  setHeater(true);

  Reading r;
  float temperature, humidity, pressure, gas;

  // Un reintento. Cubre la primera medida que siempre falla y una medida que avise
  // a mitad, que de otro modo tiraria los datos ambientales de todo un despertar.
  // Dos fallos seguidos son una averia real y se informan.
  for (int attempt = 0; attempt < 2; attempt++) {
    if (_sensor.getData(temperature, humidity, pressure, gas)) {
      r.status = ReadingStatus::Ok;
      r.values[0] = temperature;
      r.values[1] = humidity;
      r.values[2] = pressure;
      r.values[3] = gas;
      r.count = 4;
      return r;
    }
    delay(20);
  }

  r.status = ReadingStatus::Timeout;
  return r;
}

#pragma once

#include "ISensor.h"
#include <Wire.h>
#include <sfTkArdI2C.h>
#include <sfTk/bmv080.h>
#include <sfTk/bmv080_defs.h>

// =============================================================================
// BMV080 (Bosch) - PARTICULAS PM1 / PM2.5 / PM10
// =============================================================================
//
// Bus I2C compartido. Usa directamente el SDK de Bosch en vez del envoltorio de
// SparkFun para Arduino (SparkFunBMV080/sfDevBMV080): ese envoltorio reduce cada
// bmv080_status_code_t a un simple verdadero/falso, y eso hizo invisible el fallo
// de cableado original (bmv080_open fallando con E_BMV080_ERROR_HW_WRITE). Las
// funciones de lectura/escritura de abajo estan copiadas de la implementacion
// privada del propio envoltorio, no reinventadas: desplazar la cabecera de 16 bits
// un bit a la izquierda para I2C y delegar en sfTkIBus::readRegister/writeRegister,
// con el byte mas significativo primero.
//
// SET_LOOP_TASK_STACK_SIZE(60*1024) en el .cpp es obligatorio:
// bmv080_serve_interrupt necesita mas pila de la que el ESP32 da por defecto a la
// tarea de loop (segun el comentario de la propia libreria de SparkFun).
//
// Los ficheros del SDK de Bosch (bmv080.h, bmv080_defs.h, lib_bmv080.a,
// lib_postProcessor.a) NO vienen en la libreria publica: se copian a mano desde
// bosch_bmv080_sdk/ con restore.ps1 (ver platformio.ini).
//
// values[]: [0] = PM1, [1] = PM2.5, [2] = PM10 (ug/m3), [3] = obstruido (1/0).
// Necesita varios segundos funcionando antes de que una lectura sea valida
// (BMV080_STARTUP_DELAY_MS en Config.h cubre arranque, precalentamiento del laser y
// autotest). Pueden llegar lecturas antes de que el valor se estabilice del todo,
// porque is_outside_measurement_range es el unico indicador de validez que expone
// el SDK.
class Bmv080Sensor : public ISensor {
public:
  explicit Bmv080Sensor(uint8_t addr, TwoWire &wire = Wire) : _addr(addr), _wire(wire) {}

  bool begin() override;
  Reading read() override;
  void sleep() override;
  const char *name() const override { return "BMV080"; }

  // Encendido por turnos. Con el laser funcionando consume ~68 mA, con diferencia
  // la mayor carga del sistema, asi que solo se enciende durante su propia medida.
  // Arrancar y parar reutilizan el sensor ya abierto, asi que ninguna de las dos
  // vuelve a pagar el coste de abrir/reiniciar/autotest.
  bool startMeasurement();
  bool stopMeasurement();
  bool isMeasuring() const { return _measuring; }

  // Por que fallo begin(), y el codigo de estado del SDK detras. Sin esto, begin()
  // juntaria "no esta en el bus en ninguna direccion" y "esta, pero el laser no
  // arranca" en un solo verdadero/falso - dos averias con causas y arreglos
  // completamente distintos.
  enum class FailStage { None, Open };
  FailStage failStage() const { return _failStage; }
  int lastOpenStatus() const { return _lastOpenStatus; }
  // Estado del SDK en el ultimo intento de arrancar el laser.
  int lastStartStatus() const { return _lastStartStatus; }

  // En que pasada abrio. Sirve para saber si los reintentos hacen falta de verdad
  // o si el sensor abre siempre a la primera. 0 = no abrio.
  uint8_t openPasses() const { return _openPasses; }

private:
  // Cuantas veces se recorre el rango de direcciones antes de rendirse, y la pausa
  // entre pasadas. El fallo de apertura es INTERMITENTE - la misma placa abre en un
  // arranque y no en el siguiente - asi que repetir el recorrido entero es lo que
  // mas sube la probabilidad de exito por unidad de esfuerzo.
  static const uint8_t kOpenPasses = 3;
  static const uint16_t kOpenRetryMs = 100;

  uint8_t _addr;
  uint8_t _openPasses = 0;
  TwoWire &_wire;
  sfTkArdI2C _bus;
  bmv080_handle_t _handle = nullptr;
  FailStage _failStage = FailStage::None;
  int _lastOpenStatus = 0;

#define BMV080_POST_RESET_MS 100
// Pausa entre intentos de arrancar el laser.
#define BMV080_START_RETRY_MS 150

// ---- Funciones que el SDK de Bosch llama para hablar con el bus I2C ----

// Leer registros. En I2C la cabecera se desplaza un bit a la izquierda.
int8_t Bmv080Sensor::readCB(bmv080_sercom_handle_t handle, uint16_t header, uint16_t *payload,
                             uint16_t payloadLength) {
  if (handle == nullptr) return -5;
  sfTkIBus *bus = (sfTkIBus *)handle;
  if (bus->type() == ksfTkBusTypeI2C) header = header << 1;
  size_t nRead = 0;
  sfTkError_t rc = bus->readRegister(header, payload, payloadLength, nRead);
  return (rc == ksfTkErrOk && nRead == payloadLength) ? 0 : -2;
}

// Escribir registros.
int8_t Bmv080Sensor::writeCB(bmv080_sercom_handle_t handle, uint16_t header, const uint16_t *payload,
                              uint16_t payloadLength) {
  if (handle == nullptr) return -5;
  sfTkIBus *bus = (sfTkIBus *)handle;
  if (bus->type() == ksfTkBusTypeI2C) header = header << 1;
  sfTkError_t rc = bus->writeRegister(header, payload, payloadLength);
  return (rc == ksfTkErrOk) ? 0 : -3;
}

// Esperar el tiempo que pida el SDK.
int8_t Bmv080Sensor::delayCB(uint32_t durationMs) {
  delay(durationMs);
  return 0;
}

// El SDK llama aqui cuando tiene una medida nueva: se guarda para read().
void Bmv080Sensor::onDataReady(bmv080_output_t output, void *callbackParameters) {
  Bmv080Sensor *self = (Bmv080Sensor *)callbackParameters;
  self->_lastOutput = output;
  self->_dataReady = true;
}

// Abre el sensor probando todas las direcciones posibles, tres pasadas, y lo deja
// configurado pero SIN medir (laser apagado).
bool Bmv080Sensor::begin() {
  // La direccion I2C la fijan los pines CS/SDO, y de verdad cambia entre placas:
  // el nodo C1 contesta en 0x57 y el C2 en 0x56. En vez de exigir una compilacion
  // por placa - inviable en un despliegue de 200 nodos - se prueba primero la
  // direccion configurada y luego el resto del rango, igual que Bme690Sensor hace
  // con su pareja 0x76/0x77.
  //
  //   CS alto, SDO alto -> 0x57      CS bajo, SDO alto -> 0x55
  //   CS alto, SDO bajo -> 0x56      CS bajo, SDO bajo -> 0x54
  static const uint8_t kStrapAddrs[4] = {0x57, 0x56, 0x55, 0x54};

  bmv080_status_code_t rc = E_BMV080_ERROR_HW_WRITE;

  // TRES PASADAS POR EL RANGO, no una. El fallo es intermitente - la misma placa
  // abre en un arranque y no en el siguiente - asi que repetir es lo que mas
  // sube la probabilidad de exito por unidad de esfuerzo.
  for (uint8_t pasada = 0; pasada < kOpenPasses && rc != E_BMV080_OK; pasada++) {
    if (pasada > 0) {
      // Una pausa entre pasadas. El ciclo del pin de habilitacion seria mejor -
      // devolveria el chip a un estado conocido - pero ese pin lo conduce el
      // firmware principal, no este driver, y meterlo aqui le daria al driver una
      // dependencia de pins.h que no tiene.
      delay(kOpenRetryMs);
    }

    for (uint8_t attempt = 0; attempt < 5; attempt++) {
      // El intento 0 usa la direccion configurada; los siguientes recorren el
      // rango y se saltan la que ya se probo.
      uint8_t tryAddr = (attempt == 0) ? _addr : kStrapAddrs[attempt - 1];
      if (attempt > 0 && tryAddr == _addr) continue;

      _bus.init(_wire, tryAddr);
      _bus.setByteOrder(SFTK_MSBFIRST);

      rc = bmv080_open(&_handle, (bmv080_sercom_handle_t)&_bus, (bmv080_callback_read_t)readCB,
                       (bmv080_callback_write_t)writeCB, (bmv080_callback_delay_t)delayCB);
      if (rc == E_BMV080_OK) {
        _addr = tryAddr;  // recordarla, para que read() y cualquier reintento usen la buena
        _openPasses = (uint8_t)(pasada + 1);
        break;
      }
      // Si bmv080_open falla no deja nada abierto, asi que no hay que cerrar nada.
      _handle = nullptr;
    }
  }
  _lastOpenStatus = (int)rc;
  if (rc != E_BMV080_OK) {
    _failStage = FailStage::Open;
    return false;
  }

  bmv080_reset(_handle);
  // Tras un reinicio hay que esperar antes de que el sensor acepte una orden de
  // medida. Sin esta pausa, el primer bmv080_start_continuous_measurement() despues
  // de begin() falla.
  delay(BMV080_POST_RESET_MS);

  bool doObstructionDetection = true;
  bmv080_set_parameter(_handle, "do_obstruction_detection", (void *)&doObstructionDetection);

  // AQUI NO SE ARRANCA NINGUNA MEDIDA.
  //
  // Antes begin() terminaba con bmv080_start_continuous_measurement() y devolvia su
  // resultado, lo que juntaba dos fallos completamente distintos bajo un mismo "NOT
  // FOUND": el sensor ausente, y el sensor presente pero incapaz de ARRANCAR SU
  // LASER. El laser es la carga de 68 mA que ha tumbado esta placa repetidas veces,
  // asi que el segundo caso era el probable - y se informaba como sensor ausente,
  // lo que mandaba el diagnostico hacia el bus I2C y el hardware en vez de hacia
  // la alimentacion.
  //
  // Ademas era trabajo tirado: quien llama para la medida en el acto, porque el
  // laser solo se enciende dentro del turno del propio BMV080, con todo lo demas en
  // reposo. Abrir el sensor es el trabajo de begin(); medir es el del turno.
  //
  // Asi que begin() tiene exito cuando el sensor ABRE. Un laser que no arranca se
  // informa despues, en el turno, como fallo de lectura - que es lo que es.
  _measuring = false;
  _failStage = FailStage::None;
  return true;
}

// Enciende el laser y empieza a medir de forma continua.
bool Bmv080Sensor::startMeasurement() {
  if (_handle == nullptr) return false;
  if (_measuring) return true;

  // Se reintenta el arranque. Aqui es donde se enciende de verdad el laser - la
  // carga de 68 mA que ha tumbado esta placa varias veces - asi que una negativa no
  // prueba que el sensor este roto, y reintentar tras una pausa corta no cuesta nada
  // si va bien. Se guarda el estado del SDK para poder diagnosticar un fallo
  // persistente en vez de adivinarlo.
  bmv080_status_code_t rc = E_BMV080_ERROR_HW_WRITE;
  for (int attempt = 0; attempt < 3; attempt++) {
    rc = bmv080_start_continuous_measurement(_handle);
    if (rc == E_BMV080_OK) break;
    delay(BMV080_START_RETRY_MS);
  }

  _lastStartStatus = (int)rc;
  _measuring = (rc == E_BMV080_OK);
  _dataReady = false;
  return _measuring;
}

// Apaga el laser.
bool Bmv080Sensor::stopMeasurement() {
  if (_handle == nullptr || !_measuring) return true;
  bmv080_status_code_t rc = bmv080_stop_measurement(_handle);
  _measuring = false;  // se da por apagado pase lo que pase: nunca creer el laser encendido
  _dataReady = false;
  return rc == E_BMV080_OK;
}

// Atiende al SDK; si en esta llamada entrego una medida (onDataReady), la devuelve.
Reading Bmv080Sensor::read() {
  Reading r;
  _dataReady = false;
  bmv080_status_code_t rc = bmv080_serve_interrupt(_handle, onDataReady, this);

  if (rc != E_BMV080_OK) {
    r.status = (rc == E_BMV080_ERROR_HW_WRITE || rc == E_BMV080_ERROR_HW_READ) ? ReadingStatus::NoAck
                                                                                : ReadingStatus::InvalidFrame;
    return r;
  }

  if (!_dataReady) {
    r.status = ReadingStatus::NotReady;
    return r;
  }

  r.status = ReadingStatus::Ok;
  r.values[0] = _lastOutput.pm1_mass_concentration;
  r.values[1] = _lastOutput.pm2_5_mass_concentration;
  r.values[2] = _lastOutput.pm10_mass_concentration;
  r.values[3] = _lastOutput.is_obstructed ? 1.0f : 0.0f;
  r.count = 4;
  return r;
}

// Dormir = apagar el laser.
void Bmv080Sensor::sleep() {
  stopMeasurement();
}

// =============================================================================
// CM1106 (Cubic) - SENSOR DE CO2
// =============================================================================
//
// UART, protocolo pregunta-respuesta con checksum: la suma de los 8 bytes de la
// respuesta, incluido el propio byte de checksum, es 0 modulo 256. El pin EN se
// pone a nivel alto y se espera un tiempo de calentamiento antes del primer intento
// de lectura. Cuelga del rail de 5 V conmutado.
//
// MIDE SOLO AL ENCENDERSE: despues repite ese valor el resto del tiempo. Para
// obtener un valor NUEVO hay que hacerle un ciclo de alimentacion con EN
// (powerCycle()). No es un apano de un fallo: es como funciona la pieza.
//
// COMSEL (seleccion UART/I2C) no esta fijado en esta revision de placa, pero la
// pieza ha devuelto siempre tramas UART con checksum valido, asi que en esta
// unidad arranca en modo UART de forma fiable.
//
// values[]: [0] = CO2 en ppm.
class Cm1106Sensor : public ISensor {
public:
  // Algunas placas no sacan el pin EN; para esas, pasar kNoEnPin (y warmupMs = 0).
  static constexpr uint8_t kNoEnPin = 0xFF;

  Cm1106Sensor(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint8_t enPin, uint32_t warmupMs,
               uint32_t baud = 9600)
      : _serial(serial), _rxPin(rxPin), _txPin(txPin), _enPin(enPin), _warmupMs(warmupMs), _baud(baud) {}

  bool begin() override;
  Reading read() override;
  void sleep() override;
  const char *name() const override { return "CM1106"; }

  // Bytes crudos de la ultima respuesta, sea cual sea su veredicto. Un valor en
  // ppm que nunca cambia puede significar dos cosas muy distintas - un sensor que
  // devuelve una trama nueva pero congelada, o un camino de lectura que reutiliza
  // en silencio un valor viejo - y solo los bytes del cable las distinguen.
  // Longitud 0 si no llego nada.
  const uint8_t *lastRaw() const { return _lastRaw; }
  uint8_t lastRawLen() const { return _lastRawLen; }

  // Tramas validas seguidas con el MISMO valor de CO2 (bytes resp[3..4]).
  //
  // A proposito NO compara la trama entera: se vio el 2026-09-04 que esta unidad
  // lleva un contador en resp[6] que sube en cada lectura (y el checksum con el),
  // asi que dos tramas nunca son identicas aunque la medida este clavada. Comparar
  // tramas enteras no detectaria nunca nada. Que el aparato este vivo y que la
  // medida este viva son dos cosas distintas: esto vigila la segunda.
  uint16_t frozenValueRun() const { return _frozenRun; }

  // Los bytes del valor de CO2 de la ultima trama valida, para el registro.
  uint16_t lastValueRaw() const { return _lastValue; }

  // Ciclo de alimentacion con EN: apagar offMs, encender y esperar el
  // calentamiento. La hoja de datos no ofrece reinicio por software para este
  // caso, y la unidad solo se ha recuperado siempre con un corte completo.
  void powerCycle(uint32_t offMs = 500);

private:
  HardwareSerial &_serial;
  uint8_t _rxPin, _txPin, _enPin;
  uint32_t _warmupMs;
  uint32_t _baud;
  uint8_t _lastRaw[8] = {0};
  uint8_t _lastRawLen = 0;
  uint8_t _prevRaw[8] = {0};
  uint8_t _prevRawLen = 0;
  uint16_t _frozenRun = 0;
  uint16_t _lastValue = 0;
  bool _haveValue = false;
};

#include "Cm1106Sensor.h"

// Comando "leer CO2" del protocolo Cubic: 11 01 01 ED.
static const uint8_t READ_CO2_CMD[] = {0x11, 0x01, 0x01, 0xED};

// Enciende el sensor (EN alto), abre el UART y espera el calentamiento.
bool Cm1106Sensor::begin() {
  if (_enPin != kNoEnPin) {
    pinMode(_enPin, OUTPUT);
    digitalWrite(_enPin, HIGH);
  }
  _serial.begin(_baud, SERIAL_8N1, _rxPin, _txPin);
  if (_warmupMs > 0) delay(_warmupMs);
  return true;  // sin saludo inicial: la presencia se confirma con la primera lectura valida
}

// Manda el comando de lectura y espera hasta 1 s los 8 bytes de respuesta:
//   16 05 01 <CO2 alto> <CO2 bajo> <estado> <contador> <checksum>
Reading Cm1106Sensor::read() {
  Reading r;

  while (_serial.available()) _serial.read();
  _serial.write(READ_CO2_CMD, sizeof(READ_CO2_CMD));

  uint8_t resp[8];
  uint8_t len = 0;
  uint32_t start = millis();
  while (millis() - start < 1000 && len < sizeof(resp)) {
    if (_serial.available()) resp[len++] = _serial.read();
  }

  // Guardar los bytes del cable ANTES de decidir nada, para que quien llama pueda
  // distinguir una trama congelada pero nueva de un valor reciclado, acabe como
  // acabe esta lectura.
  memcpy(_prevRaw, _lastRaw, sizeof(_prevRaw));
  _prevRawLen = _lastRawLen;
  memcpy(_lastRaw, resp, sizeof(_lastRaw));
  _lastRawLen = len;

  if (len < 8) {
    r.status = ReadingStatus::Timeout;
    return r;
  }
  if (resp[0] != 0x16 || resp[1] != 0x05 || resp[2] != 0x01) {
    r.status = ReadingStatus::InvalidFrame;
    return r;
  }

  uint16_t sum = 0;
  for (uint8_t i = 0; i < 7; i++) sum += resp[i];
  uint8_t checksum = (uint8_t)(256 - (sum % 256));
  if (checksum != resp[7]) {
    r.status = ReadingStatus::InvalidFrame;
    return r;
  }

  r.status = ReadingStatus::Ok;
  const uint16_t value = (uint16_t)resp[3] * 256 + resp[4];

  // Vigilar si la MEDIDA se ha quedado clavada, no la trama - ver frozenValueRun().
  if (_haveValue && value == _lastValue) {
    if (_frozenRun < 0xFFFF) _frozenRun++;
  } else {
    _frozenRun = 0;
  }
  _lastValue = value;
  _haveValue = true;

  r.values[0] = value;
  r.count = 1;
  return r;
}

// Reposo = quitarle la alimentacion con EN.
void Cm1106Sensor::sleep() {
  if (_enPin != kNoEnPin) digitalWrite(_enPin, LOW);
}

void Cm1106Sensor::powerCycle(uint32_t offMs) {
  if (_enPin == kNoEnPin) return;
  digitalWrite(_enPin, LOW);
  delay(offMs);
  digitalWrite(_enPin, HIGH);
  if (_warmupMs > 0) delay(_warmupMs);
  // Las tramas de antes del ciclo no dicen nada del sensor despues de el.
  _frozenRun = 0;
  _lastRawLen = 0;
  _prevRawLen = 0;
  _haveValue = false;
  while (_serial.available()) _serial.read();
}

// CALYPSO ULP PRO - ANEMOMETRO ULTRASONICO (VIENTO)
// =============================================================================
//
// UART, frases NMEA 0183 (variante TTL-UART/NMEA confirmada, no RS485). Necesita
// 5 V y unos 20 s desde que recibe corriente hasta que empieza a emitir.
//
// El sensor emite SOLO, sin que se le pregunte. read() vacia lo que haya en el
// buffer y devuelve la ultima frase $..MWV completa y con checksum valido vista en
// esa llamada; NotReady si no llego nada nuevo desde la anterior. El hardware
// admite tambien modo pregunta-respuesta segun la hoja de datos, pero no esta
// implementado: no hace falta.
//
// values[]: [0] = angulo del viento (grados, relativo), [1] = velocidad (tal como
// la da el sensor; normalmente m/s en el campo NMEA por defecto de esta unidad),
// [2] = estado (1 = valido/A, 0 = no valido).
//
// La frase se guarda en un array de char de tamano fijo, no en un String: el
// sensor emite sin parar, y un String reconstruido en cada linea en un aparato que
// debe funcionar dias enteros acabaria fragmentando la memoria.
class CalypsoSensor : public ISensor {
public:
  CalypsoSensor(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint32_t baud = 38400)
      : _serial(serial), _rxPin(rxPin), _txPin(txPin), _baud(baud) {}

  bool begin() override;
  Reading read() override;
  void sleep() override {}
  const char *name() const override { return "Calypso"; }

  // Descarta los bytes del buffer y reinicia la linea a medio construir. Se llama
  // justo antes de escuchar: lo que hubiera en el buffer es un fragmento viejo y
  // seguramente cortado, y procesarlo solo desperdicia la primera lectura y puede
  // descolocar la siguiente.
  void flushInput();

  // Bytes leidos en la ultima llamada a read(). Distingue "sensor sin conectar o
  // mudo" (0 en todos los ciclos) de "llegan datos pero aun ninguna frase valida"
  // (>0).
  uint16_t lastReadBytes() const { return _lastReadBytes; }

  // La ultima trama MWV que paso el checksum, TAL CUAL llego.
  //
  // Hace falta porque el sensor puede emitir tramas perfectamente validas cuyo
  // contenido sea cero: eso no se distingue de un fallo de parseo mirando solo el
  // valor. Con la trama cruda delante se ve de inmediato si el cero lo dijo el
  // sensor o lo invento este driver leyendo el campo equivocado.
  const char *lastFrame() const { return _lastFrame; }

private:
  static constexpr uint8_t kLineMax = 120;

  HardwareSerial &_serial;
  uint8_t _rxPin, _txPin;
  uint32_t _baud;
  char _line[kLineMax + 1] = {0};
  uint8_t _lineLen = 0;
  uint16_t _lastReadBytes = 0;
  char _lastFrame[kLineMax + 1] = {0};

  static bool checksumOk(const char *sentence, uint8_t len, int starIdx);
};

#include "CalypsoSensor.h"
#include <stdlib.h>
#include <string.h>

// Solo abre el UART. No hay saludo inicial: la presencia del sensor se confirma
// con la primera frase valida que llegue.
bool CalypsoSensor::begin() {
  _serial.begin(_baud, SERIAL_8N1, _rxPin, _txPin);
  return true;  // sin saludo inicial: la presencia se confirma con la primera frase valida
}

// Checksum NMEA: XOR de todos los caracteres entre '$' y '*', comparado con los
// dos digitos hexadecimales que van detras del '*'.
bool CalypsoSensor::checksumOk(const char *sentence, uint8_t len, int starIdx) {
  if (starIdx < 0 || len < starIdx + 3) return false;

  uint8_t calc = 0;
  for (int i = 1; i < starIdx; i++) calc ^= sentence[i];

  char hex[3] = {sentence[starIdx + 1], sentence[starIdx + 2], '\0'};
  uint8_t received = (uint8_t)strtol(hex, nullptr, 16);
  return calc == received;
}

void CalypsoSensor::flushInput() {
  while (_serial.available()) _serial.read();
  _lineLen = 0;
  _lastReadBytes = 0;  // empezar la cuenta de bytes de la ventana de escucha que sigue
}

// Vacia el buffer del UART juntando caracteres en lineas. Cuando una linea es una
// frase MWV ($IIMWV,angulo,R,velocidad,M,A*checksum) con checksum valido, la
// trocea por comas y saca angulo, velocidad y validez.
Reading CalypsoSensor::read() {
  Reading r;
  r.status = ReadingStatus::NotReady;

  uint16_t bytesThisCall = 0;
  while (_serial.available()) {
    char c = _serial.read();
    bytesThisCall++;

    if (c == '\n') {
      if (_lineLen > 0 && _line[0] == '$' && _lineLen > 5 && memcmp(_line + 3, "MWV", 3) == 0) {
        int starIdx = -1;
        for (uint8_t i = 0; i < _lineLen; i++) {
          if (_line[i] == '*') {
            starIdx = i;
            break;
          }
        }

        if (checksumOk(_line, _lineLen, starIdx)) {
          // Copiarla ANTES de trocearla: el parseo de abajo pone un '\0' en cada
          // coma y destruye la linea original.
          memcpy(_lastFrame, _line, _lineLen);
          _lastFrame[_lineLen] = 0;

          char *fields[6] = {};
          uint8_t fieldIdx = 0;
          fields[fieldIdx++] = _line;
          for (uint8_t i = 0; i < _lineLen && fieldIdx < 6; i++) {
            if (_line[i] == ',' || _line[i] == '*') {
              _line[i] = '\0';
              fields[fieldIdx++] = _line + i + 1;
            }
          }

          // Campos: [1] angulo, [3] velocidad, [5] estado ('A' = valido).
          if (fieldIdx >= 6) {
            r.status = ReadingStatus::Ok;
            r.values[0] = atof(fields[1]);
            r.values[1] = atof(fields[3]);
            r.values[2] = (fields[5][0] == 'A') ? 1.0f : 0.0f;
            r.count = 3;
          } else {
            r.status = ReadingStatus::InvalidFrame;
          }
        }
      }
      _lineLen = 0;
    } else if (c != '\r') {
      if (_lineLen < kLineMax) {
        _line[_lineLen++] = c;
        _line[_lineLen] = '\0';
      } else {
        _lineLen = 0;  // linea demasiado larga: es basura, se descarta
      }
    }
  }

  _lastReadBytes += bytesThisCall;
  return r;
}

#pragma once

#include <Arduino.h>

// =============================================================================
// DRIVER DE LA RADIO LoRa REYAX RYLR998
// =============================================================================
//
// Lo comparten Modulo C (transmite) y Modulo B (recibe). Antes habia una copia en
// cada modulo que solo se diferenciaba en los comentarios; se unifico aqui porque
// el driver era identico. Ver radio/README.md para los parametros de enlace de
// partida y como cambian alcance por tiempo en el aire.
//
// La radio se maneja con comandos de texto AT por un puerto serie (UART): se le
// manda "AT+SEND=..." y contesta "+OK" o "+ERR=..."; cuando llega un paquete por
// el aire, la radio escribe sola una linea "+RCV=...".

// Un paquete recibido, interpretado a partir de una linea
// "+RCV=<direccion>,<longitud>,<datos>,<rssi>,<snr>".
// El limite de 240 bytes de carga es el que documenta habitualmente AT+SEND del
// RYLR998. No se ha sacado de una hoja de datos incluida en este repositorio (no
// hay ninguna), asi que es una suposicion de partida a verificar contra el
// firmware real del modulo.
struct LoRaMessage {
  uint16_t senderAddr = 0;    // direccion AT+ADDRESS de quien lo envio
  char payload[241] = {0};    // los datos, terminados en '\0'
  uint8_t length = 0;         // longitud de los datos
  int16_t rssi = 0;           // intensidad de la senal recibida (dBm)
  int8_t snr = 0;             // relacion senal/ruido (dB)
};

// AT+PARAMETER=<factorDeEnsanchamiento>,<anchoDeBanda>,<tasaDeCodificacion>,<preambulo>.
// El ancho de banda es el INDICE/codigo del modulo, no un valor en kHz - ver
// radio/README.md para los valores de partida y su efecto en alcance y tiempo.
struct RYLR998Params {
  uint8_t spreadingFactor;  // SF: mas alto = mas alcance y mas tiempo en el aire
  uint8_t bandwidth;        // codigo de ancho de banda (7 = 125 kHz)
  uint8_t codingRate;       // tasa de codificacion (1 = 4/5)
  uint16_t preamble;        // longitud del preambulo
};

// Driver minimo por comandos AT.
//
// begin() y send() son BLOQUEANTES: esperan la respuesta rapida +OK/+ERR del
// modulo. poll() NO lo es: solo vacia los bytes que ya estan en el buffer de
// recepcion del UART, asi que se puede llamar en cada vuelta de loop() sin que
// la radio detenga nunca el programa (en Modulo B en particular, nunca debe
// detener el servidor web de handleClient()).
class RYLR998 {
public:
  RYLR998(HardwareSerial &serial, uint8_t rxPin, uint8_t txPin, uint32_t baud = 115200)
      : _serial(serial), _rxPin(rxPin), _txPin(txPin), _baud(baud) {}

  // Bloqueante: ejecuta AT, AT+ADDRESS, AT+NETWORKID, AT+BAND y AT+PARAMETER por
  // orden, y abandona en el primero que no conteste +OK. Si devuelve false, revisa
  // primero el cableado y la velocidad (rylr998_bridge.cpp) antes de suponer que
  // un parametro es incorrecto.
  // Pasando un flujo `debug` (por ejemplo &Serial) imprime cada comando y su
  // respuesta en el momento, para confirmar que el modulo acepto cada ajuste en
  // vez de fiarse de un unico verdadero/falso al final.
  bool begin(uint16_t addr, uint16_t networkId, uint32_t bandHz, const RYLR998Params &params,
             Print *debug = nullptr);

  // Bloqueante: escribe AT+SEND=<direccion>,<longitud>,<datos> y espera +OK, con
  // un limite de kSendTimeoutMs. El tiempo real en el aire depende de la longitud
  // y del SF/BW/CR configurados, pero esta llamada nunca espera indefinidamente.
  bool send(uint16_t destAddr, const char *data, uint8_t len);

  // Bloqueante: AT+CRFOP=<dbm>, de 0 a 22. El modulo arranca a 22 dBm, y su rafaga
  // de transmision pide ~120 mA - suficiente para tumbar una alimentacion que no
  // aguanta un escalon de carga, que es exactamente lo que le pasa a la placa de
  // Modulo C (se reiniciaba en cada transmision). Menos potencia cambia alcance
  // por una rafaga mas pequena. NO se incluye en begin() a proposito: este archivo
  // lo comparte Modulo B, que tiene alimentacion de sobra y ningun motivo para
  // renunciar a alcance.
  bool setTxPower(uint8_t dbm, Print *debug = nullptr);

  // No bloqueante: vacia lo que haya en el buffer y devuelve true (con outMsg
  // relleno) en cuanto encuentra una linea +RCV= completa y bien formada. Todo lo
  // demas que vea por el camino (ecos +OK sueltos, basura) se descarta.
  bool poll(LoRaMessage &outMsg);

private:
  static constexpr uint16_t kLineMax = 260;  // "+RCV=" + dir + long + 240 B de datos + rssi + snr
  static constexpr uint32_t kSendTimeoutMs = 2000;

  HardwareSerial &_serial;
  uint8_t _rxPin, _txPin;
  uint32_t _baud;

  // Linea en construccion para poll(): los bytes llegan sueltos y se juntan aqui
  // hasta ver el fin de linea.
  char _line[kLineMax + 1] = {0};
  uint16_t _lineLen = 0;

  // Un paquete que llego mientras se esperaba el +OK de un comando. Se guarda
  // aqui para que el siguiente poll() lo entregue en vez de perderlo.
  LoRaMessage _pending;
  bool _hasPending = false;

  bool sendATCommand(const char *cmd, Print *debug, uint32_t timeoutMs = 1000);
  bool waitForOK(uint32_t timeoutMs, Print *debug);
  static bool tryParseLine(char *line, uint16_t len, LoRaMessage &outMsg);
};

#include "RYLR998.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Ajusta la potencia de transmision (0..22 dBm). Ver la explicacion en RYLR998.h.
bool RYLR998::setTxPower(uint8_t dbm, Print *debug) {
  if (dbm > 22) dbm = 22;
  char cmd[24];
  snprintf(cmd, sizeof(cmd), "AT+CRFOP=%u", dbm);
  return sendATCommand(cmd, debug);
}

// Abre el UART y configura la radio paso a paso. Cada comando tiene que contestar
// +OK; al primer fallo se abandona y se devuelve false.
bool RYLR998::begin(uint16_t addr, uint16_t networkId, uint32_t bandHz, const RYLR998Params &params,
                     Print *debug) {
  _serial.begin(_baud, SERIAL_8N1, _rxPin, _txPin);

  char cmd[48];

  // Contesta el modulo? Si esto falla, es cableado, velocidad o alimentacion.
  if (!sendATCommand("AT", debug)) return false;

  // Direccion propia de esta radio. Tiene que ser distinta en cada nodo.
  snprintf(cmd, sizeof(cmd), "AT+ADDRESS=%u", addr);
  if (!sendATCommand(cmd, debug)) return false;

  // Red: solo se oyen radios con el mismo NETWORKID.
  snprintf(cmd, sizeof(cmd), "AT+NETWORKID=%u", networkId);
  if (!sendATCommand(cmd, debug)) return false;

  // Frecuencia en Hz (868 MHz en Europa).
  snprintf(cmd, sizeof(cmd), "AT+BAND=%lu", (unsigned long)bandHz);
  if (!sendATCommand(cmd, debug)) return false;

  // Parametros de modulacion. Emisor y receptor tienen que coincidir.
  snprintf(cmd, sizeof(cmd), "AT+PARAMETER=%u,%u,%u,%u", params.spreadingFactor, params.bandwidth,
           params.codingRate, params.preamble);
  if (!sendATCommand(cmd, debug)) return false;

  return true;
}

// Envia un paquete: cabecera "AT+SEND=<destino>,<longitud>," + los datos tal cual
// + fin de linea, y espera el +OK del modulo.
bool RYLR998::send(uint16_t destAddr, const char *data, uint8_t len) {
  char header[24];
  snprintf(header, sizeof(header), "AT+SEND=%u,%u,", destAddr, len);
  _serial.print(header);
  _serial.write(reinterpret_cast<const uint8_t *>(data), len);
  _serial.print("\r\n");
  return waitForOK(kSendTimeoutMs, nullptr);
}

// Manda un comando AT (opcionalmente mostrandolo por `debug`) y espera +OK.
bool RYLR998::sendATCommand(const char *cmd, Print *debug, uint32_t timeoutMs) {
  if (debug) {
    debug->print("  -> ");
    debug->println(cmd);
  }
  _serial.print(cmd);
  _serial.print("\r\n");
  bool ok = waitForOK(timeoutMs, debug);
  if (debug && !ok) debug->println("     (no +OK - check wiring/baud or that value is accepted)");
  return ok;
}

// Lee lineas del modulo hasta ver +OK (true), +ERR (false) o agotar el tiempo
// (false).
bool RYLR998::waitForOK(uint32_t timeoutMs, Print *debug) {
  uint32_t start = millis();
  char line[kLineMax + 1];
  uint16_t lineLen = 0;

  while (millis() - start < timeoutMs) {
    while (_serial.available()) {
      char c = _serial.read();

      if (c == '\n') {
        if (lineLen > 0) {
          line[lineLen] = '\0';
          if (debug) {
            debug->print("  <- ");
            debug->println(line);
          }
          if (strncmp(line, "+OK", 3) == 0) return true;
          if (strncmp(line, "+ERR", 4) == 0) return false;
          // Puede ser un +RCV= de un paquete que llego justo mientras se esperaba
          // la respuesta a este comando. Se guarda para poll() en vez de tragarse
          // en silencio un paquete real.
          if (!_hasPending && tryParseLine(line, lineLen, _pending)) _hasPending = true;
        }
        lineLen = 0;
      } else if (c != '\r') {
        if (lineLen < kLineMax) {
          line[lineLen++] = c;
        } else {
          lineLen = 0;  // linea demasiado larga: es basura, se descarta
        }
      }
    }
  }
  return false;
}

// No bloqueante. Primero entrega el paquete guardado por waitForOK(), si lo hay;
// despues procesa solo los bytes que ya esten en el buffer del UART.
bool RYLR998::poll(LoRaMessage &outMsg) {
  if (_hasPending) {
    outMsg = _pending;
    _hasPending = false;
    return true;
  }

  while (_serial.available()) {
    char c = _serial.read();

    if (c == '\n') {
      bool matched = false;
      if (_lineLen > 0) {
        _line[_lineLen] = '\0';
        matched = tryParseLine(_line, _lineLen, outMsg);
      }
      _lineLen = 0;
      if (matched) return true;
    } else if (c != '\r') {
      if (_lineLen < kLineMax) {
        _line[_lineLen++] = c;
      } else {
        _lineLen = 0;
      }
    }
  }
  return false;
}

// Interpreta "+RCV=<direccion>,<longitud>,<datos>,<rssi>,<snr>".
//
// Se guia por la LONGITUD declarada en vez de partir ingenuamente por comas, para
// que una coma dentro de los datos (las tramas de telemetria estan llenas de ellas)
// no descoloque los campos que vienen detras.
//
// Lo que NO resiste: un salto de linea dentro de los datos, porque poll() y
// waitForOK() cortan las lineas por '\n' antes de llegar aqui. Ni la telemetria de
// Modulo C ni los comandos CFG/ACK llevan ninguno, pero no esta protegido.
bool RYLR998::tryParseLine(char *line, uint16_t len, LoRaMessage &outMsg) {
  if (len < 7 || memcmp(line, "+RCV=", 5) != 0) return false;

  // Direccion del emisor.
  char *p = line + 5;
  char *comma1 = strchr(p, ',');
  if (!comma1) return false;
  *comma1 = '\0';
  int addr = atoi(p);

  // Longitud de los datos.
  char *lenStart = comma1 + 1;
  char *comma2 = strchr(lenStart, ',');
  if (!comma2) return false;
  *comma2 = '\0';
  int dataLen = atoi(lenStart);
  if (dataLen < 0 || dataLen > (int)(sizeof(outMsg.payload) - 1)) return false;

  // Los datos: exactamente dataLen bytes, seguidos de una coma.
  char *dataStart = comma2 + 1;
  if ((dataStart - line) + dataLen + 1 > len) return false;
  if (dataStart[dataLen] != ',') return false;

  // RSSI y SNR, despues de los datos.
  char *rssiStart = dataStart + dataLen + 1;
  char *comma3 = strchr(rssiStart, ',');
  if (!comma3) return false;
  *comma3 = '\0';
  int rssi = atoi(rssiStart);

  char *snrStart = comma3 + 1;
  int snr = atoi(snrStart);

  memcpy(outMsg.payload, dataStart, dataLen);
  outMsg.payload[dataLen] = '\0';
  outMsg.length = (uint8_t)dataLen;
  outMsg.senderAddr = (uint16_t)addr;
  outMsg.rssi = (int16_t)rssi;
  outMsg.snr = (int8_t)snr;
  return true;
}
